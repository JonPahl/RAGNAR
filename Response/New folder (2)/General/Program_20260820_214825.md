static Serilog.ILogger CreateLogger()
{
    return new LoggerConfiguration()
        .ReadFrom.ConfigurationSection(GetAppConfigSection())
        .Enrich.FromLogContext()
        .Enrich.WithMachineName()
        .Enrich.WithThreadId()
        .CreateLogger();
}

static AppConfiguration LoadConfiguration(Serilog.ILogger logger)
{
    var config = new AppConfiguration();
    try
    {
        var builder = new ConfigurationBuilder()
            .AddJsonFile("appsettings.json", optional: false, reloadOnChange: true)
            .AddEnvironmentVariables();

        var configuration = builder.Build();
        configuration.Bind(config);

        // Validate critical settings
        ValidateConfiguration(config, logger);
    }
    catch (Exception ex)
    {
        logger.Fatal(ex, "Failed to load or validate configuration.");
        throw;
    }

    return config;
}

static void ValidateConfiguration(AppConfiguration config, Serilog.ILogger logger)
{
    if (string.IsNullOrWhiteSpace(config.ApplicationOptions?.SourceDirectory))
        throw new InvalidOperationException("SourceDirectory is required in ApplicationOptions.");

    if (string.IsNullOrWhiteSpace(config.EmbeddingOptions?.Host))
        throw new InvalidOperationException("EmbeddingOptions.Host is required.");

    if (config.EmbeddingOptions.Port is < 1 or > 65535)
        throw new ArgumentOutOfRangeException(nameof(config.EmbeddingOptions.Port), "Port must be between 1 and 65535.");

    logger.Information("Configuration validated successfully.");
}

static IConfigurationSection GetAppConfigSection()
{
    // Helper to get Serilog config section (assumes appsettings.json root)
    var builder = new ConfigurationBuilder().AddJsonFile("appsettings.json", optional: false);
    var config = builder.Build();
    return config.GetSection("Serilog");
}

static void ConfigureServices(IServiceCollection services, AppConfiguration config, Serilog.ILogger logger)
{
    // Core
    services.AddSingleton(config);
    services.AddSingleton(logger);

    // Configuration validation
    services.AddOptions<OllamaOptions>()
        .Bind(config.OllamaOptions)
        .Validate<OllamaOptionsValidator>();

    services.AddOptions<EmbeddingOptions>()
        .Bind(config.EmbeddingOptions)
        .Validate<EmbeddingOptionsValidator>();

    // Logging
    services.AddLogging(logging => logging.AddSerilog(logger));

    // CLI & Output
    services.AddSingleton<IOutputWriter, AnsiOutputWriter>();
    services.AddSingleton<IApplicationHeader, ApplicationHeader>();

    // Embedding
    services.AddSingleton<IQdrantClient>(sp =>
    {
        var endpoint = new Uri($"http://{config.EmbeddingOptions.Host}:{config.EmbeddingOptions.Port}");
        return new QdrantClient(endpoint);
    });
    services.AddSingleton<IEmbeddingService, OllamaEmbeddingService>();
    services.AddSingleton<IEmbedTextPipeline, EmbeddingPipeline>();

    // Questions
    services.AddSingleton<IQuestionFactory, DefaultQuestionFactory>();
    services.AddSingleton<IFileQuestionProvider, FileQuestionProvider>();
    services.AddSingleton<IFilterStrategy, XmlCommentFilterStrategy>();

    // LLM / AI
    services.AddSingleton<IOllamaClientFactory, OllamaClientFactory>();
    services.AddSingleton<IOllamaResponse, OllamaResponseProvider>();
    services.AddSingleton<ISystemPromptProvider, SystemPromptProvider>();

    // Agents
    services.AddKeyedTransient<ISystemPromptProvider, SummaryAgentPrompt>("Summary");
    services.AddTransient<SummaryAgent>();

    // File parsing
    services.AddTransient<IFileParseFactory, FileParseFactory>();

    // Utilities
    services.AddTransient<IResponseWriter, ResponseWriter>();
}

public sealed class MainCommand : AsyncCommand<MainCommand.Settings>
{
    public sealed class Settings : CommandSettings
    {
        [CommandOption("--source <DIR>")]
        [Description("Source directory to analyze")]
        public string? SourceDirectory { get; set; }

        [CommandOption("--vector-store <NAME>")]
        [Description("Vector store collection name")]
        public string? VectorStoreName { get; set; }
    }

    private readonly IApplicationHeader _header;
    private readonly IEmbeddingPipeline _embeddingPipeline;
    private readonly ILogger<MainCommand> _logger;

    public MainCommand(IApplicationHeader header, IEmbeddingPipeline embeddingPipeline, ILogger<MainCommand> logger)
    {
        _header = header;
        _embeddingPipeline = embeddingPipeline;
        _logger = logger;
    }

    public override async Task<int> ExecuteAsync(CommandContext context, Settings settings)
    {
        _header.RenderBranding();

        // Override config if CLI args provided
        if (!string.IsNullOrEmpty(settings.SourceDirectory))
            (context.Services.GetRequiredService<AppConfiguration>() as dynamic).ApplicationOptions.SourceDirectory = settings.SourceDirectory;

        if (!string.IsNullOrEmpty(settings.VectorStoreName))
            (context.Services.GetRequiredService<AppConfiguration>() as dynamic).ApplicationOptions.VectorStoreName = settings.VectorStoreName;

        try
        {
            _logger.Information("Initializing vector store...");
            await _embeddingPipeline.EnsureCollectionExistsAsync(context.CancellationToken);

            _logger.Information("Populating vector store...");
            await _embeddingPipeline.PopulateAsync(context.CancellationToken);

            _logger.Information("✅ Vector store ready.");
        }
        catch (Exception ex)
        {
            _logger.Fatal(ex, "Failed to initialize vector store.");
            return 1;
        }

        return 0;
    }
}
