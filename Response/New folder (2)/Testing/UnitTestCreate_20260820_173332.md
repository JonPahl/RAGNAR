```csharp
using System.Reflection;

public class AssemblyExtensionsTests
{
    [Fact]
    public void InformationalVersion_Returns_Attribute_Or_Default()
    {
        var asm = typeof(AssemblyExtensionsTests).Assembly;

        // Set attribute
        var attr = new AssemblyInformationalVersionAttribute("2.5.1-preview");
        var builder = new AssemblyBuilderProxy(asm);
        var modifiedAsm = builder.DefineVersion(attr);

        var version = modifiedAsm.InformationalVersion;
        version.Should().Be("2.5.1-preview");
    }

    private class AssemblyBuilderProxy
    {
        private readonly Assembly _original;
        public AssemblyBuilderProxy(Assembly asm) => _original = asm;

        public Assembly DefineVersion(AssemblyInformationalVersionAttribute attr)
        {
            var ab = AssemblyBuilder.DefineDynamicAssembly(new AssemblyName("Temp"), AssemblyBuilderAccess.Run);
            var mb = ab.DefineDynamicModule("MainModule");
            mb.SetCustomAttribute(attr.GetType(), new byte[0]); // simplified
            return ab;
        }
    }
}
```
