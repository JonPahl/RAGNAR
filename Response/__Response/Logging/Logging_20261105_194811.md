/// <summary>Fluent builder for constructing Qdrant filter expressions.</summary>
/// <example><![[CDATA[[var f = new QdrantFilterBuilder(log).AddRange("len", 50).Build();]]]]></example>
public sealed class QdrantFilterBuilder(Serilog.ILogger logger)
{
    private readonly List<Condition> _must = [[]];
    private readonly List<Condition> _mustNot = [[]];
    private readonly List<Condition> _should = [[]];

    /// <summary>Adds a keyword match-exclusion condition.</summary>
    /// <param name="field">The payload field to match.</param>
    /// <param name="value">The keyword value to exclude.</param>
    /// <returns>The builder for chaining.</returns>
    /// <example><![[CDATA[[builder.AddExclusion("category", "skip");]]]]></example>
    public QdrantFilterBuilder AddExclusion(string field, string value)
    {
        logger.Debug("Adding exclusion: {Field} = {Value}", field, value);
        _mustNot.Add(Conditions.MatchKeyword(field, value));
        return this;
    }

    /// <summary>Adds a keyword match-inclusion condition to the positive filter set.</summary>
    /// <param name="field">The payload field to match against.</param>
    /// <param name="value">The keyword value that must be present.</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.AddMatch("Category", "XML");]]]]></example>
    public QdrantFilterBuilder AddMatch(string field, string value)
    {
        logger.Debug("Adding match: {Field} = {Value}", field, value);
        _must.Add(Conditions.MatchKeyword(field, value));
        return this;
    }

    /// <summary>Adds a numeric range condition (gte and/or lte) to the positive filter set.</summary>
    /// <param name="field">The payload field name to constrain.</param>
    /// <param name="gte">Greater-than-or-equal lower bound (optional).</param>
    /// <param name="lte">Less-than-or-equal upper bound (optional).</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.AddRange("CommentLength", 50, 500);]]]]></example>
    public QdrantFilterBuilder AddRange(string field, double? gte = null, double? lte = null)
    {
        logger.Debug("Adding range on {Field}: gte={Gte}, lte={Lte}", field, gte, lte);
        var range = new Range();
        if (gte is null && lte is null)
            throw new ArgumentException("At least one of gte or lte must be provided.");
        if (gte is not null)
            range.Gte = gte.Value;
        if (lte is not null)
            range.Lte = lte.Value;
        _must.Add(new Condition { Field = new FieldCondition { Key = field, Range = range } });
        return this;
    }

    /// <summary>Adds an except-keywords condition excluding specified values from a field.</summary>
    /// <param name="field">The payload field containing keywords.</param>
    /// <param name="keywords">Keyword values to exclude from matches.</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.AddExceptKeywords("Category", "Test", "Tests");]]]]></example>
    public QdrantFilterBuilder AddExceptKeywords(string field, params string[[]] keywords)
    {
        logger.Debug("Adding except-keywords on {Field}: [[{Keywords}]]", field, string.Join(", ", keywords));
        _must.Add(new Condition { Field = new FieldCondition { Key = field, Match = new Match { ExceptKeywords = new RepeatedStrings { Strings = { keywords } } } } });
        return this;
    }

    /// <summary>Adds a numeric range must-not condition to the negative filter set.</summary>
    /// <param name="field">The payload field name to constrain.</param>
    /// <param name="gte">Greater-than-or-equal bound (optional).</param>
    /// <param name="lte">Less-than-or-equal bound (optional).</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.AddExclusionRange("CommentLength", null, 500);]]]]></example>
    public QdrantFilterBuilder AddExclusionRange(string field, double? gte = null, double? lte = null)
    {
        logger.Debug("Adding exclusion range on {Field}: gte={Gte}, lte={Lte}", field, gte, lte);
        if (gte is null && lte is null)
            throw new ArgumentException("At least one of gte or lte must be provided.");
        var range = new Range();
        if (gte is not null)
            range.Gte = gte.Value;
        if (lte is not null)
            range.Lte = lte.Value;
        _mustNot.Add(new Condition { Field = new FieldCondition { Key = field, Range = range } });
        return this;
    }

    /// <summary>Removes all accumulated must, mustNot, and should filter conditions.</summary>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.Clear();]]]]></example>
    public QdrantFilterBuilder Clear()
    {
        logger.Debug("Clearing all filter conditions ({Must}, {MustNot}, {Should})", _must.Count, _mustNot.Count, _should.Count);
        _must.Clear();
        _mustNot.Clear();
        _should.Clear();
        return this;
    }

    /// <summary>Indicates whether no filter conditions have been added yet.</summary>
    /// <example><![[CDATA[[bool empty = builder.IsEmpty;]]]]></example>
    public bool IsEmpty => _must.Count == 0 && _mustNot.Count == 0 && _should.Count == 0;

    private static Condition CloneCondition(Condition c) => c.Clone();

    /// <summary>Builds and returns the final Qdrant <see cref="Filter"/> instance.</summary>
    /// <returns>A composed filter with must and mustNot conditions.</returns>
    /// <example><![[CDATA[[Filter f = builder.Build();]]]]></example>
    public Filter Build()
    {
        logger.Debug("Building filter: {Must} must, {MustNot} mustNot, {Should} should", _must.Count, _mustNot.Count, _should.Count);
        if (IsEmpty)
        {
            throw new InvalidOperationException("QdrantFilterBuilder is empty. Add at least one condition before calling Build().");
        }

        return new Filter
        {
            Must = { _must },
            MustNot = { _mustNot },
            Should = { _should }
        };
    }
}

/// Middleware that writes progress indicators to the output writer for each pipeline stage.
/// </summary>
/// <example><![[CDATA[[runner.UseMiddleware(new ProgressMiddleware(writer, logger).Invoke);]]]]></example>
public sealed class ProgressMiddleware(IOutputWriter writer, Serilog.ILogger logger)
{
    /// <summary>
    /// Writes a progress label before stage execution and a completion marker after.
    /// </summary>
    /// <param name="context">The stage execution context.</param>
    /// <param name="next">The next middleware link or core stage delegate.</param>
    /// <param name="cancellationToken">Token to abort.</param>
    /// <returns>A task that completes when the wrapped stage finishes.</returns>
    /// <example><![[CDATA[[await mw.Invoke<TCtx>(ctx, () => Task.CompletedTask, ct);]]]]></example>
    public async Task Invoke<TContext>(StageExecutionContext<TContext> context, Func<Task> next, CancellationToken cancellationToken)
        where TContext : class
    {
        var label = $"[[{context.StageIndex + 1}/{context.StageTotal}]] {context.Stage.Name}";
        logger.Debug("ProgressMiddleware: starting stage {Label}", label);
        writer.Write($"{label} … ");
        var sw = Stopwatch.StartNew();
        await next().ConfigureAwait(false);
        sw.Stop();
        writer.WriteLine($"✓ {sw.Elapsed:hh\\:mm\\:ss}");
        writer.WriteLine("");
        logger.Debug("ProgressMiddleware: completed stage {Label} in {Elapsed}", label, sw.Elapsed);
    }
}

    {
        if (_numbered)
        {
            logger.Debug("NumberedRowDecorator.Render: already numbered, delegating to inner");

        _numbered = true;
        logger.Debug("NumberedRowDecorator.Render: prepending index column '{Header}' with {Rows} rows", columnHeader, inner.Table.Rows.Count);

        inner.Render();
        logger.Debug("NumberedRowDecorator.Render: completed");

{
    /// <summary>
    /// Determines whether the file has one of the allowed extensions (case-insensitive).
    /// </summary>
    /// <param name="entry">The file system entry.</param>
    /// <param name="allowedExtensions">The collection of allowed file extensions.</param>
    /// <param name="logger">Serilog logger for diagnostics.</param>
    /// <returns><c>true</c> if the extension is allowed; otherwise, <c>false</c>.</returns>
    /// <example><![[CDATA[[bool isAllowed = entry.HasAllowedExtension([[".pdf", ".docx"]], log);]]]]></example>
    public static bool HasAllowedExtension(this FileSystemInfo entry, IReadOnlyCollection<string> allowedExtensions, Serilog.ILogger? logger = null)
    {
        logger?.Debug("HasAllowedExtension: checking {Entry} against [[{Allowed}]]", entry.Name, string.Join(", ", allowedExtensions));
        Guard.Against.Null(allowedExtensions);
        if (entry is DirectoryInfo)
        {
            logger?.Debug("HasAllowedExtension: {Entry} is a directory, returning false", entry.Name);
            return false;
        }
        var extension = Path.GetExtension(entry.Name);
        var result = allowedExtensions.Contains(extension, StringComparer.OrdinalIgnoreCase);
        logger?.Debug("HasAllowedExtension: {Entry} (ext={Ext}) → {Result}", entry.Name, extension, result);
        return result;
    }

### Registration notes

When wiring up the new logger-dependent constructors, pass the `Serilog.ILogger` from your DI container:

```csharp
// Example: constructing a QdrantFilterBuilder
var filter = new QdrantFilterBuilder(Log.Logger)
    .AddRange("len", 50)
    .AddExceptKeywords("Category", "Test")
    .Build();

// Example: constructing a PipelineRunner context
var ctx = new QuestionPipelineContext(Log.Logger);
```

For the optional-logger static helpers (`OutputPathHelper`, `FileSystemEntryExtensions`), the `logger` parameter is nullable and defaults to `null`, so existing call sites compile without modification until you opt in.
