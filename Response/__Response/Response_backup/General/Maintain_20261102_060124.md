### Fix 1 — `NumberedRowDecorator`: replace rows, don't append

```csharp
// BEFORE (buggy – duplicates every row)
table.AddRow(row);

// AFTER (correct – rebuild the row list)
public void Render()
{
    if (_numbered)
    {
        inner.Render();
        return;
    }

    _numbered = true;
    var table = inner.Table;

    // 1. Ensure the index column exists
    if (!table.Columns.Contains(columnHeader, StringComparer.OrdinalIgnoreCase))
        table.AddColumns(columnHeader);

    // 2. Rebuild rows with a 1-based index prepended
    var existing = table.Rows.Select(r => r.ToArray()).ToList();
    table.Rows.Clear();                          // ← clear originals

    for (var i = 0; i < existing.Count; i++)
    {
        var indexed = new string[[table.Columns.Count]];
        indexed[[0]] = (i + 1).ToString();
        for (var c = 0; c < existing[[i]].Length; c++)
            indexed[[c + 1]] = existing[[i]][[c]];

        table.AddRow(new KeyValuePair<int, List<TableCell>>(
            i, indexed.Select(c => new TableCell(c)).ToList()));
    }

    // 3. Delegate final rendering to inner
    inner.Render();
}
```

> ```csharp
> // If the API only supports AddRow, recreate the table:
> var fresh = table.Reset();            // project-specific API
> // …then re-add columns and rows as above
> ```

### Fix 2 — `FileDiscoveryService`: use truly async enumeration

```csharp
// BEFORE (blocking sync call inside async iterator)
foreach (var path in Directory.EnumerateFiles(directory, "*.*", SearchOption.AllDirectories))
{
    // …
    yield return path;
    await Task.Yield();
}

// AFTER (asynchronous, non-blocking)
public async IAsyncEnumerable<string> GetFilesAsync(
    string directory,
    IOptions<RagnarConfig> options,
    [[EnumeratorCancellation]] CancellationToken cancellationToken)
{
    var filter = options.Value.FileLoadOptions;
    Guard.Against.Null(filter);
    Guard.Against.NullOrEmpty(directory);

    logger.Information("Discovering files in {Directory}", directory);

    if (!Directory.Exists(directory))
        throw new DirectoryNotFoundException(directory);

    var discoveredCount = 0;
    var skippedCount = 0;

    // Use the .NET 9+ async enumeration API (or a manual recursive approach)
    await using var enumerator =
        Directory.EnumerateFilesAsync(directory, "*.*", SearchOption.AllDirectories)
                 .WithCancellation(cancellationToken)
                 .GetAsyncEnumerator();

    while (await enumerator.MoveNextAsync().ConfigureAwait(false))
    {
        var fi = new FileInfo(enumerator.Current);
        if (!validator.IsValid(fi, filter))
        {
            skippedCount++;
            continue;
        }

        discoveredCount++;
        logger.Verbose("Eligible file: {Path}", fi.FullName);
        yield return fi.FullName;
    }

    logger.Information(
        "File discovery complete: {Discovered} eligible, {Skipped} excluded",
        discoveredCount, skippedCount);
}
```

> If you must target .NET 8 or earlier, wrap the sync enumeration in `await Task.Run(…)` per batch or switch to a manual recursive `Stack<DirectoryInfo>` that calls `await Task.Yield()` every N items.

### Fix 3 — `PipelineRunner`: lock the middleware snapshot correctly & remove dead code

```csharp
// BEFORE (partial lock + commented-out block)
private async Task ExecuteStageWithMiddlewareAsync(
    StageExecutionContext<TContext> ctx, Func<Task> coreExecution, CancellationToken ct)
{
    List<Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task>> snapshot;
    lock (_gate)
        snapshot = [[.._middlewares]];          // ← OK
    // …build onion…
}

public override async Task<TContext> ExecuteAsync(TContext context, CancellationToken ct)
{
    // …
    //try {                                    // ← dead code
    //if (snapshot[[i]].AsyncPolicy is null) {
    // await stage.ExecuteAsync(context, ct)
    // …
    //}
    // …
}

// AFTER (clean)
private async Task ExecuteStageWithMiddlewareAsync(
    StageExecutionContext<TContext> ctx, Func<Task> coreExecution, CancellationToken ct)
{
    // Atomic snapshot under the lock
    var snapshot = _middlewares.ToArray();    // ToArray is a copy; safe to read after

    Func<Task> pipeline = coreExecution;
    for (var i = snapshot.Length - 1; i >= 0; i--)
    {
        var mw = snapshot[[i]];
        var nextRef = pipeline;
        pipeline = () => mw(ctx, nextRef, ct);
    }

    await pipeline().ConfigureAwait(false);
}

public override async Task<TContext> ExecuteAsync(TContext context, CancellationToken ct)
{
    var totalSw = Stopwatch.StartNew();
    var snapshot = _stages.ToArray();         // single atomic copy

    for (var i = 0; i < snapshot.Length; i++)
    {
        var entry  = snapshot[[i]];
        var stage  = entry.PipelineStage;

        if (!stage.ShouldRun(context))
        {
            logger.Information("Skipping stage [[{Index}]]: {Name}", i, stage.Name);
            continue;
        }

        logger.Information("Starting stage [[{Index}/{Total}]]: {Name}", i + 1, snapshot.Length, stage.Name);

        var execCtx = new StageExecutionContext<TContext>(stage, context, i, snapshot.Length);

        Func<Task> core = entry.AsyncPolicy is null
            ? async () => await stage.ExecuteAsync(context, ct).ConfigureAwait(false)
            : async () => await entry.AsyncPolicy
                         .ExecuteAsync(() => stage.ExecuteAsync(context, ct))
                         .ConfigureAwait(false);

        try
        {
            await ExecuteStageWithMiddlewareAsync(execCtx, core, ct).ConfigureAwait(false);
        }
        catch (OperationCanceledException) when (ct.IsCancellationRequested) { throw; }
        catch (Exception ex)
        {
            logger.Fatal(ex, "Stage [[{Index}]] {Name} failed", i, stage.Name);
            throw new PipelineStageException(stage.Name, ex);
        }
    }

    logger.Information("Pipeline finished in {Total}", totalSw.Elapsed);
    return context;
}
```

> **Key change:** `ToArray()` creates a value copy of the list contents *before* the lock is released, so no further lock is needed during iteration. All commented-out code is deleted.

### Fix 5 — `QuestionFilterSelector`: don't silently override user strategies

```csharp
public sealed class QuestionFilterSelector(Serilog.ILogger logger, IEnumerable<IFilterStrategy> strategies)
{
    private readonly Serilog.ILogger _logger = logger;
    private readonly Dictionary<QuestionCategory, IFilterStrategy> _strategies;

    public QuestionFilterSelector(Serilog.ILogger logger, IEnumerable<IFilterStrategy> strategies)
    {
        _logger     = logger;
        _strategies = strategies.ToDictionary(s => s.SupportedCategory);

        // Register built-in fallbacks ONLY when the user hasn't supplied one.
        if (!_strategies.ContainsKey(QuestionCategory.XML))
        {
            _strategies[[QuestionCategory.XML]] = new XmlCommentFilterStrategy(logger);
            logger.Debug("Auto-registered built-in XML filter strategy.");
        }
        // …add other built-ins the same way…
    }

    public Filter FindFilter(QuestionCategory category, int commentLengthThreshold = 0)
    {
        _logger.Debug("Finding filter for {Category}, threshold {Threshold}",
                      category, commentLengthThreshold);

        var filter = _strategies.TryGetValue(category, out var strategy)
            ? strategy.CreateFilter(commentLengthThreshold)
            : new Filter();

        _logger.Debug("Filter resolved for {Category}: {Must} must, {MustNot} mustNot",
                      category, filter.Must?.Count ?? 0, filter.MustNot?.Count ?? 0);
        return filter;
    }
}
```

> The guard `if (!_strategies.ContainsKey(…))` already prevents the overwrite — the fix is purely to **log** the auto-registration so it's visible in diagnostics.
