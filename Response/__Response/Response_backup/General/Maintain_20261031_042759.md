```csharp
public sealed class QuestionFilterSelector(
    Serilog.ILogger logger,
    IEnumerable<IFilterStrategy> strategies)
{
    private readonly Dictionary<QuestionCategory, IFilterStrategy> _strategies;

    public QuestionFilterSelector(Serilog.ILogger logger, IEnumerable<IFilterStrategy> strategies)
    {
        _strategies = strategies.ToDictionary(s => s.SupportedCategory);
        // Register defaults once, in the constructor:
        if (!_strategies.ContainsKey(QuestionCategory.XML))
            _strategies[[QuestionCategory.XML]] = new XmlCommentFilterStrategy(logger);
    }

    public Filter FindFilter(QuestionCategory category, int commentLengthThreshold = 0)
    {
        logger.Debug("Finding filter for category {Category}, threshold {Threshold}",
                     category, commentLengthThreshold);

        var filter = _strategies.TryGetValue(category, out var strategy)
            ? strategy.CreateFilter(commentLengthThreshold)
            : new Filter();

        logger.Debug("Filter resolved for {Category}: {HasConditions} conditions",
                     category, filter.Must?.Count > 0 || filter.MustNot?.Count > 0);
        return filter;
    }
```

```csharp
// ✅ FIX — use the async directory API (available in .NET 7+)
public async IAsyncEnumerable<string> GetFilesAsync(
    string directory, IOptions<RagnarConfig> options,
    [[EnumeratorCancellation]] CancellationToken cancellationToken)
{
    var filter = options.Value.FileLoadOptions;
    Guard.Against.Null(filter);
    Guard.Against.NullOrEmpty(directory);

    logger.Information("Discovering files in {Directory} with {Ext} allowed extensions",
                      directory, string.Join(", ", filter.AllowedExtensions));

    if (!Directory.Exists(directory))
        throw new DirectoryNotFoundException(directory);

    var discovered = 0;
    var skipped    = 0;

    var enumerator = Directory.EnumerateFilesAsync(directory, "*.*", SearchOption.AllDirectories)
        .GetAsyncEnumerator(cancellationToken);

    try
    {
        while (await enumerator.MoveNextAsync().ConfigureAwait(false))
        {
            cancellationToken.ThrowIfCancellationRequested();
            var fi = new FileInfo(enumerator.Current);

            if (!validator.IsValid(fi, filter))
            {
                skipped++;
                continue;
            }

            discovered++;
            logger.Verbose("Eligible file: {Path}", enumerator.Current);
            yield return enumerator.Current;
        }
    }
    finally
    {
        await enumerator.DisposeAsync().ConfigureAwait(false);
    }

    logger.Information("File discovery complete: {Discovered} eligible, {Skipped} excluded",
                      discovered, skipped);
}
```

```csharp
// ✅ FIX — registry-based dispatch, extensible without editing the factory
public class CodeParserFactory(
    Serilog.ILogger logger,
    IOptions<RagnarConfig> options,
    IEnumerable<IFileParser> parsers) : IFileParseFactory
{
    private readonly Dictionary<string, IFileParser> _registry;

    public CodeParserFactory(Serilog.ILogger logger,
        IOptions<RagnarConfig> options,
        IEnumerable<IFileParser> parsers)
    {
        _registry = parsers
            .GroupBy(p => p.SupportedExtensions)          // each parser declares its own extensions
            .SelectMany(g => g.Select(p => (ext: g.Key, parser: p)))
            .ToDictionary(e => e.ext, e => e.parser);
    }

    public async Task<IEnumerable<CodeDocument>> ParseAsync(string file, CancellationToken ct)
    {
        var ext = new FileInfo(file).Extension.ToLowerInvariant();
        var parser = _registry.GetValueOrDefault(ext) ?? _registry[["*"]]; // "*" = fallback
        logger.Debug("Dispatching {Ext} → {Parser}", ext, parser.GetType().Name);
        var docs = await parser.ParseAsync(file, ct).ConfigureAwait(false);
        logger.Debug("Produced {Count} documents", docs.Count());
        return docs;
    }
}
```
