## 2. `StageTimingMiddleware` – Retry Count Never Populated

**Problem:** The comment `// populate via a separate retry middleware if needed` indicates `RetryCount` is always `0`. A dedicated retry middleware should increment it.

```csharp
// --- NEW FILE: StageRetryMiddleware.cs ---
/// <summary>
/// Middleware that wraps a stage with Polly-based retry logic and
/// records the number of attempts in <see cref="StageMetric.RetryCount"/>.
/// </summary>
/// <remarks>
/// Must be registered *after* <see cref="StageTimingMiddleware"/> in the
/// middleware chain so that timing captures the full (retried) duration.
/// </remarks>
/// <example><![[CDATA[[runner.UseMiddleware(new StageRetryMiddleware(logger, 3).Invoke);]]]]></example>
public sealed class StageRetryMiddleware(Serilog.ILogger logger, int maxRetries)
{
    /// <summary>
    /// Invokes the stage up to <paramref name="maxRetries"/> additional times
    /// on transient faults, incrementing the retry counter in the context metric.
    /// </summary>
    public async Task Invoke<TContext>(
        StageExecutionContext<TContext> context,
        Func<Task> next,
        CancellationToken cancellationToken) where TContext : class
    {
        cancellationToken.ThrowIfCancellationRequested();
        ArgumentNullException.ThrowIfNull(context);

        var attempt = 0;
        Exception? lastException = null;

        while (true)
        {
            attempt++;
            try
            {
                if (attempt > 1)
                    logger.Warning(
                        "Stage {Name} – retry attempt {Attempt}/{Max}",
                        context.Stage.Name, attempt, maxRetries);

                await next().ConfigureAwait(false);
                break; // success
            }
            catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
            {
                throw; // user cancel – do not retry
            }
            catch (Exception ex) when (attempt <= maxRetries)
            {
                lastException = ex;
                logger.Error(ex,
                    "Stage {Name} attempt {Attempt} failed; will retry",
                    context.Stage.Name, attempt);
                await Task.Delay(TimeSpan.FromSeconds(attempt), cancellationToken)
                    .ConfigureAwait(false); // simple linear back-off
            }
            catch (Exception ex)
            {
                // Final attempt failed – record and rethrow
                if (context.Metric is null)
                    context.Metric = new StageMetric { StageName = context.Stage.Name };
                context.Metric.RetryCount = attempt - 1;
                context.Metric.Success = false;
                context.Metric.Error = ex.Message;
                throw;
            }
        }

        // Record successful retry count (0 if first attempt succeeded)
        if (context.Metric is null)
            context.Metric = new StageMetric { StageName = context.Stage.Name, Success = true };
        context.Metric.RetryCount = attempt - 1;
        context.Metric.Success = true;
    }
}
```

**Wiring it into the pipeline:**

```csharp
var timing   = new StageTimingMiddleware(logger);
var retry    = new StageRetryMiddleware(logger, maxRetries: 3);

runner
    .UseMiddleware((ctx, next, ct) => timing.Invoke(ctx, next, ct))   // outer: timing
    .UseMiddleware((ctx, next, ct) => retry.Invoke(ctx, next, ct));   // inner: retry
```

Because the middleware chain is an **onion**, `StageTimingMiddleware` wraps `StageRetryMiddleware`, so the stopwatch captures *all* retries, and `RetryCount` is set by the inner middleware before the outer one reads it in its `finally` block.

```csharp
        throw;
```

```csharp
// --- AFTER (timeout is explicit) ---
private async Task<T> WithTimeoutAsync<T>(
    Func<CancellationToken, Task<T>> operation,
    CancellationToken outerToken)
{
    using var linkedCts = CancellationTokenSource.CreateLinkedTokenSource(outerToken);
    linkedCts.CancelAfter(_timeout);
    try
    {
        return await operation(linkedCts.Token).ConfigureAwait(false);
    }
    catch (OperationCanceledException) when (outerToken.IsCancellationRequested)
    {
        // User pressed Ctrl+C – propagate as-is
        logger.Warning("Embedding generation cancelled by user.");
        throw;

            $"Embedding generation exceeded the { _timeout.TotalSeconds:F0 }s timeout.",
            null);
```

```csharp
// --- AFTER (strategies fully injected) ---
public sealed class QuestionFilterSelector(
    Serilog.ILogger logger,
    IEnumerable<IFilterStrategy> strategies)
{
    // Immutable snapshot – no runtime mutation
    private readonly IReadOnlyDictionary<QuestionCategory, IFilterStrategy> _strategies
        = strategies.ToDictionary(s => s.SupportedCategory, StringComparer.Ordinal);

    public Filter FindFilter(QuestionCategory category, int commentLengthThreshold = 0)
    {
        logger.Debug("Finding filter for {Category}, threshold {Threshold}",
            category, commentLengthThreshold);

        if (!_strategies.TryGetValue(category, out var strategy))
        {
            logger.Warning("No filter strategy registered for {Category}; returning empty filter.",
                category);
            return new Filter();
        }

        var filter = strategy.CreateFilter(commentLengthThreshold);
        logger.Debug("Filter resolved for {Category}: {Must} must, {MustNot} mustNot",
            category,
            filter.Must?.Count ?? 0,
            filter.MustNot?.Count ?? 0);
        return filter;
    }
}
```

```csharp
    {

    // 3. Delegate final rendering to inner
```
