## PowerShell Reorganization Script

```powershell
<#
.SYNOPSIS
    Reorganizes Ragnar source files into a structured folder/namespace hierarchy.

.DESCRIPTION
    1. Creates the target directory tree.
    2. Moves each .cs file to its new location.
    3. Rewrites the `namespace` declaration (or adds one if missing).
    4. Adds `using` directives where cross-namespace references are needed.
    5. Preserves git history (uses git mv when available).

.PARAMETER Root
    Path to the Ragnar project root (the folder containing Ragnar.csproj).

.EXAMPLE
    .\Reorganize-Ragnar.ps1 -Root "C:\Users\jp325\source\Ragnar\Src\Ragnar"
#>
[[CmdletBinding()]]
param(
    [[Parameter(Mandatory)]]
    [[ValidateScript({ Test-Path "$_.Ragnar.csproj" })]]
    [[string]]$Root
)

$ErrorActionPreference = "Stop"
Push-Location $Root
try {
    # ─────────────────────────────────────────────
    # 1. Define the mapping:  fileName → (targetFolder, namespace)
    # ─────────────────────────────────────────────
    $map = [[ordered]]@{

        # Core / Pipeline infrastructure
        "PipelineRunner.cs"            = @{ Folder = "Core/Pipeline";              Ns = "Ragnar.Core.Pipeline" }
        "StageEntry.cs"                = @{ Folder = "Core/Pipeline";              Ns = "Ragnar.Core.Pipeline" }
        "PipelineMiddleware.cs"        = @{ Folder = "Core/Pipeline";              Ns = "Ragnar.Core.Pipeline" }
        "StageMetric.cs"               = @{ Folder = "Core/Pipeline";              Ns = "Ragnar.Core.Pipeline" }
        "StageTimingMiddleware.cs"     = @{ Folder = "Core/Pipeline.Middleware";   Ns = "Ragnar.Core.Pipeline.Middleware" }
        "ProgressMiddleware.cs"        = @{ Folder = "Core/Pipeline.Middleware";   Ns = "Ragnar.Core.Pipeline.Middleware" }

        # Core / Abstractions (interfaces)
        "IFileParser.cs"               = @{ Folder = "Core/Abstractions";          Ns = "Ragnar.Core.Abstractions" }
        "IFileEligibilityValidator.cs" = @{ Folder = "Core/Abstractions";          Ns = "Ragnar.Core.Abstractions" }
        "IFileDiscoveryService.cs"     = @{ Folder = "Core/Abstractions";          Ns = "Ragnar.Core.Abstractions" }
        "INotificationService.cs"      = @{ Folder = "Core/Abstractions";          Ns = "Ragnar.Core.Abstractions" }

        # Configuration
        "FileLoadOptions.cs"           = @{ Folder = "Configuration";              Ns = "Ragnar.Configuration" }
        "EmbeddingOptions.cs"          = @{ Folder = "Configuration";              Ns = "Ragnar.Configuration" }
        "GenerationOptions.cs"         = @{ Folder = "Configuration";              Ns = "Ragnar.Configuration" }

        # Services / File Discovery
        "FileDiscoveryService.cs"      = @{ Folder = "Services/FileDiscovery";     Ns = "Ragnar.Services.FileDiscovery" }
        "FileSystemEntryExtensions.cs" = @{ Folder = "Services/FileDiscovery";     Ns = "Ragnar.Services.FileDiscovery" }

        # Services / Parsing
        "CodeParserFactory.cs"         = @{ Folder = "Services/Parsing";           Ns = "Ragnar.Services.Parsing" }

        # Services / Vector Store
        "QdrantFilterBuilder.cs"       = @{ Folder = "Services/VectorStore";       Ns = "Ragnar.Services.VectorStore" }
        "CollectionInfo.cs"            = @{ Folder = "Services/VectorStore";       Ns = "Ragnar.Services.VectorStore" }

        # Services / Embedding
        "EmbeddingSetupExtension.cs"   = @{ Folder = "Services/Embedding";         Ns = "Ragnar.Services.Embedding" }

        # Services / LLM
        "IOllamaClientFactory.cs"      = @{ Folder = "Services/Llm";               Ns = "Ragnar.Services.Llm" }
        "OllamaChatClientBuilder.cs"   = @{ Folder = "Services/Llm";               Ns = "Ragnar.Services.Llm" }

        # Services / Fingerprinting
        "JsonFileFingerprintStore.cs"  = @{ Folder = "Services/Fingerprinting";    Ns = "Ragnar.Services.Fingerprinting" }

        # Questions domain
        "QuestionPipelineContext.cs"   = @{ Folder = "Questions";                  Ns = "Ragnar.Questions" }
        "QuestionsExtensions.cs"       = @{ Folder = "Questions";                  Ns = "Ragnar.Questions" }
        "QuestionFilterSelector.cs"    = @{ Folder = "Questions";                  Ns = "Ragnar.Questions" }
        "XmlCommentFilterStrategy.cs"  = @{ Folder = "Questions/Filtering";        Ns = "Ragnar.Questions.Filtering" }

        # Pipeline / Stages
        "BrandingStage.cs"             = @{ Folder = "Pipeline/Stages";            Ns = "Ragnar.Pipeline.Stages" }
        "DeltaEmbedStage.cs"           = @{ Folder = "Pipeline/Stages";            Ns = "Ragnar.Pipeline.Stages" }

        # UI
        "ViewLocator.cs"               = @{ Folder = "Ui";                         Ns = "Ragnar.Ui" }
        "NumberedRowDecorator.cs"      = @{ Folder = "Ui";                         Ns = "Ragnar.Ui" }
        "OutputPathHelper.cs"          = @{ Folder = "Ui";                         Ns = "Ragnar.Ui" }

        # Data / CSV
        "EnumConverter.cs"             = @{ Folder = "Data/Csv";                   Ns = "Ragnar.Data.Csv" }
    }

    # ─────────────────────────────────────────────
    # 2. Create all target directories
    # ─────────────────────────────────────────────
    $folders = $map.Values | ForEach-Object { $_.Folder } | Sort-Object -Unique
    foreach ($f in $folders) {
        $full = Join-Path $Root $f
        if (-not (Test-Path $full)) {
            Write-Host "  [[DIR]] $f" -ForegroundColor DarkGray
            New-Item -ItemType Directory -Path $full -Force | Out-Null
        }
    }

    # ─────────────────────────────────────────────
    # 3. Move files + rewrite namespace
    # ─────────────────────────────────────────────
    $gitRepo = (& git rev-parse --is-inside-work-tree 2>$null) -eq "true"

    foreach ($entry in $map.GetEnumerator()) {
        $srcName  = $entry.Key
        $srcPath  = Join-Path $Root $srcName
        $destDir  = $entry.Value.Folder
        $destNs   = $entry.Value.Ns
        $destPath = Join-Path (Join-Path $Root $destDir) $srcName

        if (-not (Test-Path $srcPath)) {
            Write-Warning "  [[SKIP]] $srcName not found at root."
            continue
        }

        # ── Move (prefer git mv to preserve history) ──
        if ($gitRepo) {
            & git mv --quiet $srcPath $destPath 2>$null
        } else {
            Move-Item -Path $srcPath -Destination $destPath -Force
        }
        Write-Host "  [[MOVE]] $srcName → $destDir/$srcName" -ForegroundColor Cyan

        # ── Rewrite the namespace declaration ──
        $content = [[System.IO.File]]::ReadAllText($destPath)

        # If a namespace line exists, replace it; otherwise insert before the first type
        $pattern = '^\s*(?:global\s+using|using)\s';  # don't touch usings
        $nsRegex = '(?m)^\s*namespace\s+[[\w.]]+\s*\{'
        if ($content -match $nsRegex) {
            $content = [[regex]]::Replace($content, $nsRegex, "namespace $destNs {")
        } else {
            # No namespace found — wrap the type declarations.
            # Find the first type-level keyword (public/internal/record/class/interface/enum)
            $typeRegex = '(?m)^(?=(?:\[[(?:System\.|Serilog\.|Polly\.)?[[\w.]]+\]]|public |internal |private |record |class |interface |enum ))'
            $match = [[regex]]::Match($content, $typeRegex)
            if ($match.Success) {
                $before = $content.Substring(0, $match.Index)
                $after  = $content.Substring($match.Index)
                $content = "$before namespace $destNs
{
    $after"
                # Add closing brace
                $content = $content.TrimEnd() + "`r`n}"
            }
        }

        [[System.IO.File]]::WriteAllText($destPath, $content)
        Write-Host "  [[NS]]   namespace $destNs" -ForegroundColor DarkCyan
    }

    # ─────────────────────────────────────────────
    # 4. Update .csproj to exclude Properties subfolder from compilation
    #    (launchSettings.json should already be excluded by default SDK)
    # ─────────────────────────────────────────────
    Write-Host "`n[[OK]] Reorganization complete." -ForegroundColor Green
    Write-Host "     Run  dotnet build  to verify all references resolve.`n"
}
finally {
    Pop-Location
}
```

## Design Rationale (why this grouping)

| Principle | How it's applied |
|---|---|
| **Single Responsibility per folder** | Pipeline *infrastructure* (runner, stages, middleware) lives in `Core/Pipeline` so it stays decoupled from *domain* stages in `Pipeline/Stages`. |
| **Abstractions in one place** | All `I*` interfaces that cross feature boundaries live in `Core/Abstractions`, making dependency direction clear: **Services → Core, Stages → Core + Configuration**. |
| **Configuration is read-only data** | `EmbeddingOptions`, `GenerationOptions`, `FileLoadOptions` are pure POCOs/records with no behaviour → `Configuration`. |
| **Domain model isolated** | `Question`, `QuestionCategory`, `QuestionPipelineContext`, filter strategies, and extensions sit in `Questions/` so the LLM/vector-store plumbing never leaks into question logic. |
| **Infrastructure swappable** | `JsonFileFingerprintStore` can be replaced with a DB-backed store without touching pipeline code, because both depend on the `IFileFingerprintStore` abstraction in `Core/Abstractions`. |
| **UI is a leaf** | `ViewLocator`, `NumberedRowDecorator`, and `OutputPathHelper` depend on everything above but nothing depends on them → `Ui` folder at the periphery. |
| **Data / IO concerns separate** | `EnumConverter` (CSV) and file I/O helpers are not part of the pipeline or the question domain → `Data/Csv` and `Services/FileDiscovery`. |

This structure makes the dependency flow **unidirectional** (arrows point left → right):

```
Ui  →  Pipeline/Stages  →  Services/*  →  Core/*  →  Configuration
         ↓
      Questions
```

No circular references, easy to test in isolation, and each folder maps 1-to-1 with a namespace so the compiler and IDE resolve everything predictably.
