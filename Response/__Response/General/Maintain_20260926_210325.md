### 2c. Flawed `CharacterCount` in `StringExtensions.cs`

```csharp
// ❌ Problems:
//   • Counts attribute names/values inside tags (e.g. <span style="x"> → "stylex" counted)
//   • The `c != '/'` exclusion is arbitrary and wrong for self-closing tags
//   • No handling of CDATA, comments <!-- -->, or nested tags

public int CharacterCount()
{
    var count = 0;
    var insideTag = false;
    for (var i = 0; i < xmlComment.Length; i++)
    {
        var c = xmlComment[[i]];
        if (c == '<') { insideTag = true; continue; }
        if (c == '>') { insideTag = false; continue; }
        if (!insideTag && c != '/')   // ← why exclude '/'?
            count++;
    }
    return count;
}
```

```csharp
// ✅ FIX: strip ALL XML tags (including attributes) and whitespace-only lines,
//    then count remaining visible characters.
public int CharacterCount()
{
    // Remove CDATA sections, comments, then all tags (with attributes).
    var noCdata   = Regex.Replace(xmlComment, @"<!\[[CDATA\[[.*?\]]\]]>", "", RegexOptions.Singleline);
    var noComment = Regex.Replace(noCdata,    @"<!--.*?-->",        "", RegexOptions.Singleline);
    var noTags    = Regex.Replace(noComment,  @"<[[^>]]*>",           "");

    return noTags.Trim().Length;
}
```
