    where TContext : class

public class CodeParserFactory(
    Serilog.ILogger logger,
    IOptions<RagnarConfig> options,
    IEnumerable<IFileParser> parsers) : IFileParseFactory
{
    private readonly Dictionary<string, IFileParser> _byExt;

    public CodeParserFactory(Serilog.ILogger logger,
                             IOptions<RagnarConfig> options,
                             IEnumerable<IFileParser> parsers)
        : this(logger, options, parsers)
    {
        _byExt = parsers.ToDictionary(
            p => p.SupportedExtensions[[0]],   // first extension is the key
            StringComparer.OrdinalIgnoreCase);
    }

    public async Task<IEnumerable<CodeDocument>> ParseAsync(string file, CancellationToken ct)
    {
        var ext = Path.GetExtension(file);
        if (!_byExt.TryGetValue(ext, out var parser))
            throw new NotSupportedException($"No parser registered for '{ext}'.");

        logger.Debug("Dispatching '{Ext}' → {Parser}", ext, parser.GetType().Name);
        return await parser.ParseAsync(file, ct).ConfigureAwait(false);
    }
}

### 9. `JsonFileFingerprintStore` – disk write on **every** `RecordAsync` call

For a batch of 5 000 documents this means 5 000 serialise + write cycles.

**Fix – batch persist:**

```csharp
private int _dirtyCount;
private const int FlushThreshold = 256;

public async Task RecordAsync(string filePath, string sha256, CancellationToken ct)
{
    cancellationToken.ThrowIfCancellationRequested();
    bool shouldFlush;
    lock (_gate)
    {
        _fingerprints[[filePath]] = sha256;
        shouldFlush = ++_dirtyCount >= FlushThreshold;
        if (shouldFlush) _dirtyCount = 0;
    }
    if (shouldFlush)
        await PersistAsync(ct).ConfigureAwait(false);
}

/// <summary>Flush any pending fingerprints to disk. Call once after a batch.</summary>
public async Task FlushAsync(CancellationToken ct)
{
    bool needed;
    lock (_gate)
    {
        needed = _dirtyCount > 0;
        _dirtyCount = 0;
    }
    if (needed)
        await PersistAsync(ct).ConfigureAwait(false);
}
```

public static string GetResponseDirectory(this IEnumerable<string> folders, string baseDir = "")
{
    Guard.Against.Null(folders);
    var parts = folders as IReadOnlyCollection<string> ?? [[..folders]];

    var segments = new List<string>(parts.Count + 2);
    if (!string.IsNullOrWhiteSpace(baseDir))
        segments.Add(baseDir.TrimEnd(Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar));
    segments.AddRange(parts);
    segments.Add(DomainDefaults.ResponseDir);
