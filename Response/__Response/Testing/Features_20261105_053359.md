## 1. Incremental (Delta) Embedding

**Problem:** Every run re-embeds *all* files. On a 50 k-file repo this wastes minutes and API credits.

**Idea:** Persist a SHA-256 hash of each file's content alongside its Qdrant payload. On subsequent runs, skip files whose hash is unchanged.

### `IFileFingerprintStore.cs`

```csharp
/// <summary>Persists file-content fingerprints so unchanged files are skipped on re-embed.</summary>
/// <example><![[CDATA[[if (await store.IsUnchangedAsync(path, hash)) skip;]]]]></example>
public interface IFileFingerprintStore
{
    /// <summary>Returns true when the file's hash matches the last stored value.</summary>
    /// <param name="filePath">Absolute or relative file path used as the key.</param>
    /// <param name="sha256">Current SHA-256 hex digest of the file content.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <returns><c>true</c> if the file has not changed since the last successful embed.</returns>
    /// <example><![[CDATA[[bool skip = await store.IsUnchangedAsync("src/App.cs", "ab3f…");]]]]></example>
    Task<bool> IsUnchangedAsync(string filePath, string sha256, CancellationToken cancellationToken);

    /// <summary>Records a new fingerprint after a successful upsert.</summary>
    /// <param name="filePath">File path key.</param>
    /// <param name="sha256">Hash to store.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <example><![[CDATA[[await store.RecordAsync("src/App.cs", "ab3f…", ct);]]]]></example>
    Task RecordAsync(string filePath, string sha256, CancellationToken cancellationToken);

    /// <summary>Removes the fingerprint (e.g. after a file is deleted from source).</summary>
    /// <param name="filePath">File path key.</param>
```

### `JsonFileFingerprintStore.cs`

```csharp
/// <summary>JSON-file-backed fingerprint store that survives across runs.</summary>
/// <param name="logger">Serilog logger for diagnostics.</param>
/// <param name="basePath">Directory where the fingerprint file lives (defaults to <c>.ragnar</c> beside the source root).</param>
/// <example><![[CDATA[[var store = new JsonFileFingerprintStore(log, Path.Combine(root, ".ragnar"));]]]]></example>
public sealed class JsonFileFingerprintStore(Serilog.ILogger logger, string basePath)
    : IFileFingerprintStore
{
    private readonly Lock _gate = new();
    private readonly Dictionary<string, string> _fingerprints = [[]];
    private string _filePath = string.Empty;

    /// <summary>Loads (or creates) the fingerprint file from disk.</summary>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <example><![[CDATA[[await store.InitializeAsync(ct);]]]]></example>
    public async Task InitializeAsync(CancellationToken cancellationToken)
    {
        Directory.CreateDirectory(basePath);
        _filePath = Path.Combine(basePath, "fingerprints.json");
        if (File.Exists(_filePath))
        {
            var json = await File.ReadAllTextAsync(_filePath, cancellationToken).ConfigureAwait(false);
            _fingerprints = JsonSerializer.Deserialize<Dictionary<string, string>>(json) ?? [[]];
            logger.Information("Loaded {Count} existing fingerprints from {Path}", _fingerprints.Count, _filePath);
        }
        else
        {
            logger.Information("No existing fingerprint file at {Path}; starting fresh", _filePath);
        }
    }

    public async Task<bool> IsUnchangedAsync(string filePath, string sha256, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        lock (_gate)
            return _fingerprints.TryGetValue(filePath, out var stored) && stored == sha256;
        // (await Task.Yield(); — keep async signature)
        await Task.Yield();
    }

    public async Task RecordAsync(string filePath, string sha256, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        lock (_gate)
            _fingerprints[[filePath]] = sha256;
        await PersistAsync(cancellationToken).ConfigureAwait(false);
    }

    public async Task RemoveAsync(string filePath, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        lock (_gate)
            _fingerprints.Remove(filePath);
        await PersistAsync(cancellationToken).ConfigureAwait(false);
    }

    private async Task PersistAsync(CancellationToken ct)
    {
        string snapshot;
        lock (_gate)
            snapshot = JsonSerializer.Serialize(_fingerprints, new JsonSerializerOptions { WriteIndented = true });
        await File.WriteAllTextAsync(_filePath, snapshot, ct).ConfigureAwait(false);
    }
}
```

### `DeltaEmbedStage.cs` — a new pipeline stage that wraps the existing writer

```csharp
/// <summary>
/// Wraps the standard embed-and-upsert flow, skipping files whose SHA-256 fingerprint
/// matches the last successful embed.
/// </summary>
/// <param name="logger">Serilog logger.</param>
/// <param name="fingerprintStore">Fingerprint persistence backend.</param>
/// <param name="writer">The existing Qdrant upsert writer.</param>
/// <param name="options">Application configuration (vector store name, etc.).</param>
/// <example><![[CDATA[[runner.AddStage(new DeltaEmbedStage(log, store, writer, opts));]]]]></example>
public sealed class DeltaEmbedStage(
    Serilog.ILogger logger,
    IFileFingerprintStore fingerprintStore,
    VectorStoreWriter writer,
    IOptions<RagnarConfig> options)
    : IPipelineStage<EmbeddingContext>
{
    /// <inheritdoc />
    /// <returns>The string "Delta Embed".</returns>
    public string Name => "Delta Embed";

    /// <inheritdoc />
    /// <returns>A Polly retry policy (3 attempts, 2 s base delay).</returns>
    public AsyncRetryPolicy? RetryPolicy =>
        Polly.Policy
            .Handle<HttpRequestException>()
            .Or<TimeoutException>()
            .WaitAndRetryAsync(
                [[TimeSpan.FromSeconds(2), TimeSpan.FromSeconds(5), TimeSpan.FromSeconds(10)]],
                (attempt, ts, ex, _) =>
                    logger.Warning(ex, "Delta embed attempt {Attempt} failed; retrying in {Delay}", attempt, ts));

    /// <inheritdoc />
    public bool ShouldRun(EmbeddingContext context) => context.Documents is { Count: > 0 };

    /// <summary>Executes incremental upsert, skipping unchanged files.</summary>
    /// <param name="context">Shared embedding context carrying the parsed documents.</param>
    /// <param name="cancellationToken">Token to abort.</param>
    /// <returns>Completed task once all eligible documents are upserted.</returns>
    /// <example><![[CDATA[[await stage.ExecuteAsync(ctx, ct);]]]]></example>
    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        Guard.Against.Null(context.Documents);

        var eligible = new List<CodeDocument>();
        var skipped = 0;

        foreach (var doc in context.Documents)
        {
            cancellationToken.ThrowIfCancellationRequested();
            var hash = ComputeSha256(doc.Code);
            if (await fingerprintStore.IsUnchangedAsync(doc.FilePath, hash, cancellationToken))
            {
                skipped++;
                continue;
            }

            eligible.Add(doc);
            await fingerprintStore.RecordAsync(doc.FilePath, hash, cancellationToken);
        }

        logger.Information("Delta embed: {Total} docs, {Eligible} changed, {Skipped} unchanged",
            context.Documents.Count, eligible.Count, skipped);

        if (eligible.Count > 0)
        {
            var result = await writer.UpsertBatchAsync(eligible, cancellationToken).ConfigureAwait(false);
            logger.Information("Delta embed upsert: {Status}, {Upserted} vectors", result.Status, result.Upserted);
        }
    }

    private static string ComputeSha256(string content)
    {
        using var sha = SHA256.Create();
        var bytes = sha.ComputeHash(Encoding.UTF8.GetBytes(content));
        return Convert.ToHexString(bytes).ToLowerInvariant();
    }
}
```

```csharp
/// <summary>
/// Embeds the user snippet, issues a Qdrant <c>SearchAsync</c> call,
/// and stores ranked results in the context.
/// </summary>
/// <param name="logger">Serilog logger.</param>
/// <param name="embeddingService">Service that produces the query vector.</param>
/// <param name="qdrant">Qdrant client for the vector search.</param>
/// <param name="options">Application options (collection name, dimension).</param>
/// <example><![[CDATA[[runner.AddStage(new SimilaritySearchStage(log, embed, qdrant, opts));]]]]></example>
public sealed class SimilaritySearchStage(
    Serilog.ILogger logger,
    IEmbeddingService embeddingService,
    IQdrantClient qdrant,
    IOptions<RagnarConfig> options)
    : IPipelineStage<SimilaritySearchContext>
{
    public string Name => "Similarity Search";
    public AsyncRetryPolicy? RetryPolicy => null;
    public bool ShouldRun(SimilaritySearchContext context) => !string.IsNullOrWhiteSpace(context.Snippet);

    /// <summary>Executes the vector similarity query against Qdrant.</summary>
    /// <param name="context">Search context carrying the snippet and TopK.</param>
    /// <param name="cancellationToken">Token to abort.</param>
    /// <returns>Completed task once results are stored in <see cref="SimilaritySearchContext.Results"/>.</returns>
    /// <example><![[CDATA[[await stage.ExecuteAsync(ctx, ct);]]]]></example>
    public async Task ExecuteAsync(SimilaritySearchContext context, CancellationToken cancellationToken)
    {
        Guard.Against.NullOrWhiteSpace(context.Snippet);

        logger.Information("Embedding query snippet ({Length} chars)…", context.Snippet.Length);
        var queryVector = await embeddingService.EmbedAsync(context.Snippet, cancellationToken).ConfigureAwait(false);

        var searchParams = new SearchPointsBuilder(
            options.Value.ApplicationOptions.VectorStoreName,
            queryVector)
        .WithLimit(context.TopK)
        .WithWithPayload(true)
        .WithWithVectors(false);

        if (context.Category.HasValue)
        {
            searchParams = searchParams.WithFilter(
                new QdrantFilterBuilder()
                    .AddMatch("Category", context.Category.Value.ToString())
                    .Build());
        }

        var hits = await qdrant.SearchAsync(searchParams, cancellationToken).ConfigureAwait(false);

        context.Results = hits
            .Select(h => new SimilarityResult
            {
                Score = h.Score,
                FilePath = h.Payload[["FilePath"]]?.ToString() ?? "unknown",
                Snippet = h.Payload[["Code"]]?.ToString() ?? string.Empty,
                PointId = h.Id.ToString(),
            })
            .ToList();

        logger.Information("Similarity search returned {Count} results (top score {Top})",
            context.Results.Count, context.Results[[0]]?.Score?.ToString("F4"));
    }
}
```

```csharp
/// <summary>
/// Ollama-backed streaming client using the <c>/api/chat</c> endpoint with
/// <c>"stream": true</c>.
/// </summary>
/// <param name="logger">Serilog logger.</param>
/// <param name="http">Configured <see cref="IHttpClientHandler"/> pointing at Ollama.</param>
/// <param name="options">Ollama connection options (host, model, timeout).</param>
/// <example><![[CDATA[[
/// var client = new OllamaStreamingClient(log, http, ollamaOpts);
/// await foreach (var tok in client.StreamAsync(prompt, genOpts, ct))
///     Console.Write(tok);
/// ]]]]></example>
public sealed class OllamaStreamingClient(
    Serilog.ILogger logger,
    IHttpClientHandler http,
    IOptions<OllamaOptions> options)
    : IStreamingLlmClient
{
    /// <summary>Streams chat-completion tokens from the Ollama server.</summary>
    /// <param name="prompt">Full prompt string.</param>
    /// <param name="generationOptions">Token / temperature parameters.</param>
    /// <param name="cancellationToken">Aborts the stream immediately.</param>
    /// <returns>Async sequence of text chunks.</returns>
    /// <example><![[CDATA[[await foreach (var t in client.StreamAsync("Hello", opts, ct)) { … }]]]]></example>
    public async IAsyncEnumerable<string> StreamAsync(
        string prompt,
        in GenerationOptions generationOptions,
        [[EnumeratorCancellation]] CancellationToken cancellationToken)
    {
        var url = $"{options.Value.Host}/api/chat";
        var payload = new
        {
            model = options.Value.LlmModel,
            messages = new[[]] { new { role = "user", content = prompt } },
            stream = true,
            options = new
            {
                temperature = generationOptions.Temperature,
                repeat_penalty = generationOptions.RepeatPenalty,
                num_predict = generationOptions.MaxTokens,
            },
        };

        using var request = new HttpRequestMessage(HttpMethod.Post, url)
        {
            Content = new StringContent(
                JsonSerializer.Serialize(payload),
                Encoding.UTF8,
                "application/json"),
        };

        logger.Debug("Streaming chat request to {Url}, model {Model}", url, options.Value.LlmModel);

        using var response = await http.SendAsync(request, HttpCompletionOption.ResponseHeadersRead, cancellationToken)
            .ConfigureAwait(false);
        response.EnsureSuccessStatusCode();

        using var stream = await response.Content.ReadAsStreamAsync(cancellationToken).ConfigureAwait(false);
        using var reader = new StreamReader(stream);

        while (await reader.ReadLineAsync(cancellationToken).ConfigureAwait(false) is { } line)
        {
            cancellationToken.ThrowIfCancellationRequested();
            if (string.IsNullOrWhiteSpace(line)) continue;

            var chunk = JsonSerializer.Deserialize<OllamaStreamChunk>(line);
            if (chunk?.Message?.Content is { } text && !string.IsNullOrEmpty(text))
                yield return text;

            if (chunk?.Done == true)
                break;
        }

        logger.Debug("Stream complete");
    }

    /// <summary>Deserialised shape of a single Ollama SSE/JSONL stream chunk.</summary>
    private sealed record OllamaStreamChunk
    {
        public OllamaChunkMessage? Message { get; init; }
        public bool? Done { get; init; }
    }

    private sealed record OllamaChunkMessage
    {
        public string? Content { get; init; }
    }
}
```

```csharp
        Console.WriteLine($"  │  Status     : {info.Status,-37}│");
        Console.WriteLine("  └──────────────────────────────────────────────────┘");

        if (info.VectorsCount < 10)
        {

            logger.Warning("Collection {C} has only {N} vectors", collection, info.VectorsCount);

            Console.WriteLine($"  ⚠  Qdrant reports status '{info.Status}' — investigate shard health.");
            logger.Warning("Collection {C} status: {S}", collection, info.Status);

        else
        {
            Console.WriteLine("  ✓  Collection is healthy.");
        }
    }
```
