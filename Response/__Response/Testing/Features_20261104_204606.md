/// <summary>
/// Stores fingerprints in a local JSON file (e.g. .ragnar-fingerprints.json)
/// so that subsequent runs can skip unchanged files.
/// </summary>
public sealed class JsonFileFingerprintStore(Serilog.ILogger logger, IOptions<RagnarConfig> options)
    : IFingerprintStore
{
    private static readonly JsonSerializerOptions _jsonOpts = new()
    {
        WriteIndented = true,
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase
    };

    private string StorePath =>
        Path.Combine(options.Value.ApplicationOptions.SourceDirectory, ".ragnar-fingerprints.json");

    public async Task<Dictionary<string, FileFingerprint>> LoadAsync(CancellationToken ct)
    {
        if (!File.Exists(StorePath))
            return [[]];

        var json = await File.ReadAllTextAsync(StorePath, ct).ConfigureAwait(false);
        var map = JsonSerializer.Deserialize<Dictionary<string, FileFingerprint>>(json, _jsonOpts)
                  ?? [[]];
        logger.Information("Loaded {Count} existing fingerprints", map.Count);
        return map;
    }

    public Task SaveAsync(IReadOnlyCollection<FileFingerprint> fingerprints, CancellationToken ct)
    {
        var map = fingerprints.ToDictionary(f => f.Path);
        var json = JsonSerializer.Serialize(map, _jsonOpts);
        return File.WriteAllTextAsync(StorePath, json, ct);
    }

    public IReadOnlyDictionary<string, FileFingerprint> GetExisting(IReadOnlyCollection<string> paths)
    {
        if (!File.Exists(StorePath))
            return [[]];

        var map = LoadAsync(CancellationToken.None).GetAwaiter().GetResult();
        return paths
            .Where(p => map.ContainsKey(p))
            .ToDictionary(p => p, p => map[[p]]);
    }
}

```csharp
/// <summary>
/// Compares discovered files against stored fingerprints and removes
/// unchanged files from the context before parsing/embedding.
/// </summary>
public sealed class IncrementalFilterStage(
    Serilog.ILogger logger,
    IFingerprintStore store) : IPipelineStage<EmbeddingContext>
{
    public string Name => "Incremental Filter";
    public AsyncRetryPolicy? RetryPolicy => null;
    public bool ShouldRun(EmbeddingContext context) => true;

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken ct)
    {
        ct.ThrowIfCancellationRequested();
        var allFiles = context.DiscoveredFiles ?? throw new InvalidOperationException("No files discovered.");

        var existing = await store.LoadAsync(ct).ConfigureAwait(false);
        var toProcess = new List<string>();
        var toSkip = new List<string>();

        foreach (var path in allFiles)
        {
            var fi = new FileInfo(path);
            if (fi.Length == 0) continue;

            var hash = ComputeSha256(fi);
            var normalized = fi.FullName;

            if (existing.TryGetValue(normalized, out var prev) &&
                prev.ContentHash == hash &&
                prev.Length == fi.Length)
            {
                toSkip.Add(normalized);
            }
            else
            {
                toProcess.Add(normalized);
            }
        }

        logger.Information(
            "Incremental filter: {Total} files, {Skip} unchanged, {Process} to process",
            allFiles.Count, toSkip.Count, toProcess.Count);

        context.DiscoveredFiles = toProcess;

        // Persist updated fingerprints for processed files (called after embedding stage)
        context.PendingFingerprints = allFiles.ToDictionary(
            p => new FileInfo(p).FullName,
            p => new FileFingerprint
            {
                Path = new FileInfo(p).FullName,
                ContentHash = ComputeSha256(new FileInfo(p)),
                LastWriteUtc = new FileInfo(p).LastWriteTimeUtc,
                Length = new FileInfo(p).Length
            });
    }

    private static string ComputeSha256(FileInfo fi)
    {
        using var stream = fi.OpenRead();
        return Convert.ToHexString(SHA256.HashData(stream)).ToLowerInvariant();
    }
}

/// <summary>
/// Implements <see cref="IStreamingLlmClient"/> by calling Ollama's
/// <c>/api/chat</c> endpoint with <c>stream: true</c> and reading NDJSON chunks.
/// </summary>
public sealed class OllamaStreamingLlmClient(
    IHttpClientFactory httpClientFactory,
    IOptions<RagnarConfig> config,
    Serilog.ILogger logger) : IStreamingLlmClient
{
    private sealed record ChatRequest(
        string model,
        ChatMessage[[]] messages,
        StreamOptions options);

    private sealed record ChatMessage(string role, string content);
    private sealed record StreamOptions(bool stream, int? options_stream_options);

    public async IAsyncEnumerable<string> StreamChatAsync(
        string prompt,
        string? systemPrompt,
        GenerationOptions options,
        [[EnumeratorCancellation]] CancellationToken cancellationToken)
    {
        var ollamaOpts = config.Value.OllamaOptions;
        using var http = httpClientFactory.CreateClient("ollama");
        http.Timeout = ollamaOpts.Timeout;

        var payload = new ChatRequest(
            model: ollamaOpts.LlmModel,
            messages: [[
                .. (systemPrompt is not null ? [[new ChatMessage("system", systemPrompt)]] : [[]]),
                new ChatMessage("user", prompt)
            ]],
            options: new StreamOptions(stream: true, options_stream_options: null));

        using var content = new StringContent(
            System.Text.Json.JsonSerializer.Serialize(payload),
            Encoding.UTF8, "application/json");

        using var response = await http
            .PostAsync("/api/chat", content, cancellationToken)
            .ConfigureAwait(false);

        response.EnsureSuccessStatusCode();
        using var stream = await response.Content.ReadAsStreamAsync(cancellationToken).ConfigureAwait(false);
        using var reader = new StreamReader(stream);

        logger.Debug("Streaming response from {Model}", ollamaOpts.LlmModel);

        while (!reader.EndOfStream && !cancellationToken.IsCancellationRequested)
        {
            var line = await reader.ReadLineAsync(cancellationToken).ConfigureAwait(false);
            if (string.IsNullOrWhiteSpace(line)) continue;

            // Ollama streams NDJSON: {"message":{"content":"chunk"},"done":false}
            using var doc = JsonDocument.Parse(line);
            if (doc.RootElement.TryGetProperty("message", out var msg) &&
                msg.TryGetProperty("content", out var chunk))
            {
                var text = chunk.GetString();
                if (!string.IsNullOrEmpty(text))
                {
                    yield return text;
                    await Task.Yield(); // cooperative yield
                }
            }

            if (doc.RootElement.TryGetProperty("done", out var done) && done.GetBoolean())
                break;
        }

        logger.Debug("Stream complete");
    }
}

/// <summary>
/// Queries the vector store for context, then streams the LLM answer
/// token-by-token to the console for immediate feedback.
/// </summary>
public sealed class StreamingAnswerStage(
    IStreamingLlmClient llm,
    IQdrantClient qdrant,
    IOptions<RagnarConfig> config,
    Serilog.ILogger logger) : IPipelineStage<QuestionPipelineContext>
{
    public string Name => "Stream Answer";
    public AsyncRetryPolicy? RetryPolicy => null;
    public bool ShouldRun(QuestionPipelineContext context) => context.Questions is { Count: > 0 };

    public async Task ExecuteAsync(QuestionPipelineContext context, CancellationToken ct)
    {
        ct.ThrowIfCancellationRequested();

        foreach (var question in context.Questions!)
        {
            ct.ThrowIfCancellationRequested();
            logger.Information("Streaming answer for: {Question}", question.Text);

            // 1. Retrieve relevant code chunks from Qdrant
            var topK = config.Value.GenerationOptions.ContextWindow / 4; // heuristic
            var chunks = await qdrant.SearchAsync(
                collection: config.Value.ApplicationOptions.VectorStoreName,
                vector: question.Embedding!,
                limit: topK,
                cancellationToken: ct).ConfigureAwait(false);

            // 2. Build the prompt
            var systemPrompt = "You are a code assistant. Answer using the provided context.";
            var userPrompt = BuildPrompt(question, chunks);

            // 3. Stream the answer
            Console.Write($"{question.Text}\n{new string('─', 60)}\n  ");
            var fullResponse = new StringBuilder();

            await foreach (var token in llm.StreamChatAsync(
                userPrompt, systemPrompt, config.Value.GenerationOptions, ct).ConfigureAwait(false))
            {
                Console.Write(token);
                fullResponse.Append(token);
            }

            Console.WriteLine(); // trailing newline
            context.Responses[[question.Id]] = fullResponse.ToString();
            logger.Information("Answer complete for {Id} ({Len} chars)", question.Id, fullResponse.Length);
        }
    }

    private static string BuildPrompt(Question question, IEnumerable<CodeDocument> chunks)
        => $"""
             Context:
             {string.Join("\n\n", chunks.Select(c => $"--- {c.FileName} ---\n{c.Code}"))}

             Question: {question.Text}

             Answer:
             """;
}

/// <summary>
/// Uses a single LLM call to rank candidate passages against a query.
/// The model is asked to return a JSON array of {index, score} pairs.
/// </summary>
public sealed class LlmReranker(
    IHttpClientFactory httpClientFactory,
    IOptions<RagnarConfig> config,
    Serilog.ILogger logger) : IReranker
{
    private const string SystemPrompt = """
        You are a relevance scorer. Given a question and a list of code passages,
        assign each passage a relevance score from 0.0 to 1.0.
        Return ONLY a JSON array: [[{"index":0,"score":0.9},{"index":1,"score":0.3},...]]
        Do not include any other text.
        """;

    public async Task<IReadOnlyList<RerankedResult>> RerankAsync(
        string query,
        IReadOnlyList<CodeDocument> candidates,
        int topN,
        CancellationToken ct)
    {
        ct.ThrowIfCancellationRequested();
        if (candidates.Count == 0) return [[]];

        // Build numbered candidate list
        var numbered = string.Join("\n\n",
            candidates.Select((c, i) => $"[[{i}]] {c.Code[[..Math.Min(c.Code.Length, 500)]]}"));

        var userPrompt = $"Question: {query}\n\nPassages:\n{numbered}\n\nScores:";

        var ollamaOpts = config.Value.OllamaOptions;
        using var http = httpClientFactory.CreateClient("ollama");
        http.Timeout = ollamaOpts.Timeout;

        var payload = new
        {
            model = ollamaOpts.LlmModel,
            messages = new[[]]
            {
                new { role = "system", content = SystemPrompt },
                new { role = "user", content = userPrompt }
            },
            stream = false,
            options = new { temperature = 0.0f }
        };

        using var content = new StringContent(
            System.Text.Json.JsonSerializer.Serialize(payload),
            Encoding.UTF8, "application/json");

        using var response = await http
            .PostAsync("/api/chat", content, ct)
            .ConfigureAwait(false);
        response.EnsureSuccessStatusCode();

        var body = await response.Content.ReadAsStringAsync(ct).ConfigureAwait(false);
        using var doc = JsonDocument.Parse(body);
        var scoredLine = doc.RootElement
            .GetProperty("message").GetProperty("content").GetString()!;

        // Parse the JSON array (strip any markdown fences the model might add)
        var json = scoredLine
            .Replace("```json", "")
            .Replace("```", "")
            .Trim();

        var pairs = JsonSerializer.Deserialize<List<(int index, double score)>>(
            // Custom deserialization for the tuple shape
            null!, new JsonSerializerContext { }); // see note below

        // Fallback: manual parse
        var results = new List<RerankedResult>();
        using var arr = JsonDocument.Parse(json).RootElement.EnumerateArray();
        var list = new List<(int, double)>();
        foreach (var el in arr)
        {
            var idx = el.GetProperty("index").GetInt32();
            var sc = el.GetProperty("score").GetDouble();
            list.Add((idx, sc));
        }

        list
            .OrderByDescending(x => x.score)
            .Take(topN)
            .ForEach(x => results.Add(new RerankedResult(candidates[[x.Item1]], x.Item2)));

        logger.Information("Reranked {Count} candidates, kept top {TopN}", candidates.Count, results.Count);
        return results;
    }
}

/// <summary>
/// Stores query history as an append-only JSON-lines file (.ragnar-history.jsonl).
/// Each line is one <see cref="QueryRecord"/> serialised to JSON.
/// </summary>
public sealed class JsonFileQueryHistoryStore(
    Serilog.ILogger logger,
    IOptions<RagnarConfig> options) : IQueryHistoryStore
{
    private readonly SemaphoreSlim _writeLock = new(1, 1);

    private string HistoryPath =>
        Path.Combine(options.Value.ApplicationOptions.SourceDirectory, ".ragnar-history.jsonl");

    private static readonly JsonSerializerOptions _opts = new()
    {
        WriteIndented = false, // one record per line
        PropertyNamingPolicy = JsonNamingPolicy.CamelCase
    };

    public async Task AppendAsync(QueryRecord record, CancellationToken ct)
    {
        await _writeLock.WaitAsync(ct).ConfigureAwait(false);
        try
        {
            var line = JsonSerializer.Serialize(record, _opts);
            await using var fs = new FileStream(HistoryPath,
                FileMode.Append, FileAccess.Write, FileShare.Read);
            await fs.WriteAsync(Encoding.UTF8.GetBytes(line + "\n"), ct).ConfigureAwait(false);
            logger.Verbose("Appended query {Id} to history", record.Id);
        }
        finally
        {
            _writeLock.Release();
        }
    }

    public async Task<IReadOnlyList<QueryRecord>> GetRecentAsync(int count, CancellationToken ct)
    {
        if (!File.Exists(HistoryPath)) return [[]];

        var lines = await File.ReadAllLinesAsync(HistoryPath, ct).ConfigureAwait(false);
        return lines
            .Where(l => !string.IsNullOrWhiteSpace(l))
            .OrderByDescending(l => l)           // newest last → reverse
            .Take(count)
            .Reverse()
            .Select(l => JsonSerializer.Deserialize<QueryRecord>(l, _opts)!)
            .Where(r => r is not null)
            .ToList();
    }

    public async Task<QueryStats> GetStatsAsync(CancellationToken ct)
    {
        var all = await GetRecentAsync(int.MaxValue, ct).ConfigureAwait(false);
        if (all.Count == 0) return new QueryStats();

        var byCat = new Dictionary<QuestionCategory, int>();
        var topFiles = new Dictionary<string, int>();

        foreach (var r in all)
        {
            byCat[[r.Category]] = byCat.GetValueOrDefault(r.Category) + 1;
            foreach (var f in r.TopFiles)
                topFiles[[f]] = topFiles.GetValueOrDefault(f) + 1;
        }

        return new QueryStats
        {
            TotalQueries = all.Count,
            AvgDuration = TimeSpan.FromTicks(all.Sum(r => r.Duration.Ticks) / all.Count),
            HitRate = (double)all.Count(r => r.RawHitCount > 0) / all.Count,
            ByCategory = byCat,
            TopFiles = topFiles
                .OrderByDescending(x => x.Value)
                .Take(10)
                .ToDictionary(x => x.Key, x => x.Value),
            FirstQuery = all.Min(r => r.TimestampUtc),
            LastQuery = all.Max(r => r.TimestampUtc)
        };
    }
}

            var answer = context.Responses.GetValueOrDefault(q.Id) ?? "";
            var stages = context.StageMetrics?.ToList() ?? [[]];
            var topFiles = context.RerankedResults
                .GetValueOrDefault(q.Id, [[]])
                .Take(5)
                .Select(r => r.Document.FileName)
                .ToList();

            var record = new QueryRecord
            {
                Question = q.Text,
                Category = q.Category,
                RawHitCount = context.RawSearchResults.GetValueOrDefault(q.Id, [[]]).Count,
                RerankedCount = context.RerankedResults.GetValueOrDefault(q.Id, [[]]).Count,
                Answer = answer.Length > 2000 ? answer[[..2000]] + "…" : answer,
                Duration = context.Timings.GetValueOrDefault(q.Id, TimeSpan.Zero),
                Stages = stages,
                TopFiles = topFiles
            };

            await store.AppendAsync(record, ct).ConfigureAwait(false);
        }

        logger.Information("Query history recorded for {Count} question(s)", context.Questions!.Count);
    }
}
```
