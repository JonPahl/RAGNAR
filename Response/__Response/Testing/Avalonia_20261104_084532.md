```csharp
using Ragnar.Avalonia.Views;

using Serilog;

namespace Ragnar.Avalonia;

public partial class App : Application
{
    public static IServiceProvider Services { get; private set; } = null!;

    public override void Initialize()
    {
        AvaloniaXamlLoader.Load(this);
    }

    public override void OnFrameworkInitializationCompleted()
    {
        if (ApplicationLifetime is IClassicDesktopStyleApplicationLifetime desktop)
        {
            Services = BuildServiceProvider();

            desktop.MainWindow = new MainWindow
            {
                DataContext = Services.GetRequiredService<MainViewModel>()
            };

            base.OnFrameworkInitializationCompleted();
        }
    }

    private ServiceProvider BuildServiceProvider()
    {
        var services = new ServiceCollection();

        // ─── Serilog ───────────────────────────────────────────────────
        services.AddLogging(b =>
        {
            var inMemorySink = new Serilog.Sinks.InMemory.InMemorySink();
            b.AddSerilog(Logger.Create(c => c
                .WriteTo.InMemorySink(inMemorySink, shared: true)
                .WriteTo.File("logs/ragnar-.log", rollingInterval: RollingInterval.Day), logShared: true));
        });

        // ─── Existing Ragnar domain services ───────────────────────────
        // (mirrors what Program.cs in the CLI registers)
        services.AddSingleton<INotificationService, NotificationService>();
        services.AddSingleton<ConsoleBridge>();
        // Register IFileDiscoveryService, IFileParseFactory, IQdrantClient, etc.
        // as they were in the CLI's DI container.

        // ─── ViewModels ────────────────────────────────────────────────
        services.AddTransient<MainViewModel>();
        services.AddTransient<PipelineProgressViewModel>();
        services.AddTransient<FileBrowserViewModel>();
        services.AddTransient<QuestionAnswerViewModel>();
        services.AddTransient<SettingsViewModel>();
        services.AddTransient<LogViewerViewModel>();

        return services.BuildServiceProvider();
    }
}
```

```csharp
    {
```

```csharp
        Stages.Add(new StageProgressItem(name, index, total));

    }

    {

            item.Error = error;

    }

    {
        var item = Stages.FirstOrDefault(s => s.Name == name);
        if (item is not null) item.IsRunning = true;
        ActiveStageName = name;
    }
```

```csharp
using System.Collections.ObjectModel;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using Ragnar.Core;
using Serilog;

namespace Ragnar.Avalonia.ViewModels;

public partial class FileBrowserViewModel : ViewModelBase
{
    private readonly ILogger _logger;
    private readonly IFileDiscoveryService _discovery;
    private readonly IFileEligibilityValidator _validator;
    private readonly IOptions<RagnarConfig> _config;

    public FileBrowserViewModel(
        ILogger logger,
        IFileDiscoveryService discovery,
        IFileEligibilityValidator validator,
        IOptions<RagnarConfig> config)
    {
        _logger = logger;
        _discovery = discovery;
        _validator = validator;
        _config = config;
        Files = new ObservableCollection<FileEntryItem>();
    }

    [ObservableProperty]
    private string _sourceDirectory = _config.Value.ApplicationOptions.SourceDirectory;

    [ObservableProperty]
    private string _searchFilter = string.Empty;

    [ObservableProperty]
    private int _eligibleCount;

    [ObservableProperty]
    private int _skippedCount;

    public ObservableCollection<FileEntryItem> Files { get; }

    [RelayCommand]
    private async Task BrowseAsync()
    {
        // Use Avalonia's OS folder picker
        var lifetime = App.Instance.ApplicationLifetime;
        if (lifetime is not Avalonia.Controls.ApplicationLifetimes.IClassicDesktopStyleApplicationLifetime desktop)
            return;

        var folder = await desktop.MainWindow!.StorageProvider
            .OpenFolderPickerAsync(new Avalonia.Controls.StorageFolderPickerOpenOptions { AllowMultiple = false });

        if (folder is null) return;

        SourceDirectory = folder.Path.LocalPath;
        _logger.Information("User selected directory: {Dir}", SourceDirectory);
        await DiscoverFilesAsync();
    }

    [RelayCommand]
    private async Task DiscoverFilesAsync()
    {
        Files.Clear();
        EligibleCount = 0;
        SkippedCount = 0;
        IsBusy = true;
        StatusMessage = "Discovering files…";

        try
        {
            var eligible = 0;
            var skipped = 0;

            await foreach (var path in _discovery.GetFilesAsync(SourceDirectory, _config, CancellationToken.None))
            {
                var fi = new FileInfo(path);
                var options = _config.Value.FileLoadOptions;
                var valid = _validator.IsValid(fi, options);

                Files.Add(new FileEntryItem(fi.FullName, fi.Extension, fi.Length, valid));
                valid ? eligible++ : skipped++;
            }

            EligibleCount = eligible;
            SkippedCount = skipped;
            StatusMessage = $"Found {eligible} eligible, {skipped} skipped.";
        }
        catch (Exception ex)
        {
            _logger.Error(ex, "File discovery failed.");
            StatusMessage = $"Error: {ex.Message}";
        }
        finally
        {
            IsBusy = false;
        }
    }

    [RelayCommand]
    private void Refresh() => _ = DiscoverFilesAsync();
}

public partial class FileEntryItem : ObservableObject
{
    public FileEntryItem(string path, string extension, long size, bool eligible)
    {
        Path = path;
        Extension = extension;
        Size = size;
        Eligible = eligible;
    }

    public string Path { get; }
    public string Extension { get; }
    public long Size { get; }
    public bool Eligible { get; }

    public string FormattedSize => FormatSize(Size);

    private static string FormatSize(long bytes) => bytes switch
    {
        < 1024 => $"{bytes} B",
        < 1024 * 1024 => $"{bytes / 1024.0:F1} KB",
        < 1024L * 1024 * 1024 => $"{bytes / (1024.0 * 1024):F1} MB",
        _ => $"{bytes / (1024.0 * 1024 * 1024):F2} GB"
    };
}
```

```csharp
using Serilog;

namespace Ragnar.Avalonia.Services;

/// <summary>
/// Bridges existing Console.Write* calls (if any remain in domain code)
/// into the Avalonia log viewer so nothing is silently lost.
/// </summary>
public sealed class ConsoleBridge(ILogger logger) : IDisposable
{
    private TextWriter? _originalOut;
    private TextWriter? _originalErr;

    public void Redirect()
    {
        _originalOut = Console.Out;
        _originalErr = Console.Error;

        Console.SetOut(new LogTextWriter(logger, "stdout"));
        Console.SetError(new LogTextWriter(logger, "stderr"));
    }

    public void Restore()
    {
        if (_originalOut is not null) Console.SetOut(_originalOut);
        if (_originalErr is not null) Console.SetError(_originalErr);
    }

    public void Dispose() => Restore();

    private sealed class LogTextWriter(ILogger logger, string source) : TextWriter
    {
        public override void Write(char value) { }
        public override void Write(string? value)
        {
            if (string.IsNullOrWhiteSpace(value)) return;
            logger.Information("{Source}: {Text}", source, value);
```
