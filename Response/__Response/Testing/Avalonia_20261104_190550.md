using System.Collections.ObjectModel;
using System.Reactive.Linq;
using System.Reactive.Subjects;
using Microsoft.Extensions.Options;
using Reactive;
using ReactiveUI;
using Serilog;

namespace Ragnar.ViewModels;

public class PipelineViewModel : ViewModelBase
{
    private readonly Serilog.ILogger _logger;
    private readonly PipelineRunner<EmbeddingContext> _runner;
    private readonly BehaviorSubject<ProgressInfo> _progress = new(new());

    private string _sourceDirectory = "";
    private string _collectionName = "ragnar";
    private string _embeddingModel = "nomic-embed-text";
    private string _llmModel = "llama3.1";
    private int _batchSize = 16;
    private float _temperature = 0.2f;
    private bool _isRunning;
    private string _status = "Ready";
    private string _lastError = "";

    public PipelineViewModel(
        Serilog.ILogger logger,
        IOptions<RagnarConfig> config,
        VectorStoreViewModel vectorStore)
    {
        _logger = logger;
        Title = "Embedding Pipeline";

        // Pre-fill from config
        SourceDirectory = config.Value.ApplicationOptions.SourceDirectory;
        CollectionName = config.Value.ApplicationOptions.VectorStoreName;
        EmbeddingModel = config.Value.EmbeddingOptions.EmbeddingModel;
        LlmModel = config.Value.OllamaOptions.LlmModel;
        BatchSize = config.Value.EmbeddingOptions.BatchSize;
        Temperature = config.Value.GenerationOptions.Temperature;

        // Build the actual pipeline runner (reuse existing domain code)
        _runner = new PipelineRunner<EmbeddingContext>(logger)
            .AddStage(new BrandingStage(logger, new ConsoleApplicationHeader()))
            .AddStage(new FileDiscoveryStage(logger, new FileDiscoveryService(logger, new DefaultFileEligibilityValidator()), config))
            .AddStage(new FileParseStage(logger, new CodeParserFactory(logger, config, new CSharpSyntaxTreeChunker())))
            .AddStage(new EmbeddingStage(logger, new EmbeddingService(config), config))
            .AddStage(new VectorStoreStage(logger, new VectorStoreWriter(logger, new EmbeddingService(config), new QdrantClientWrapper(), new QdrantPointFactory(), config), config))
            .UseTiming(new StageTimingMiddleware(logger));

        // Expose stage list for the UI
        Stages = new ObservableCollection<StageInfo>(
            _runner.StageNames.Select(n => new StageInfo(n, StageState.Pending)));

        // ── Commands ──
        StartCommand = ReactiveCommand.CreateFromTask(
            async () => await StartEmbeddingPipelineAsync(),
            CanExecute: () => !IsRunning);

        StopCommand = ReactiveCommand.Create(() =>
        {
            CancellationTokenSource?.Cancel();
            Status = "Cancelled";
            IsRunning = false;
            _logger.Information("Pipeline cancelled by user");
        }, CanExecute: () => IsRunning);

        BrowseFolderCommand = ReactiveCommand.Create(() =>
        {
            var dlg = new Avalonia.Controls.FileDialog
            {
                Title = "Select Source Directory",
                CanPickFolders = true,
                AllowMultiple = false
            };
            dlg.ShowAsync(Avalonia.Application.Current.MainWindow!).ContinueWith(t =>
            {
                if (t.Result is [[var path, ..]])
                {
                    SourceDirectory = path;
                    _logger.Information("Source directory set to {Dir}", path);
                }
            });
        });

        // Progress
        ProgressObservable = _progress.AsObservable();
    }

    // ── Bindable properties ──
    public string SourceDirectory
    {
        get => _sourceDirectory;
        set => this.RaiseAndSetIfChanged(ref _sourceDirectory, value);
    }

    public string CollectionName
    {
        get => _collectionName;
        set => this.RaiseAndSetIfChanged(ref _collectionName, value);
    }

    public string EmbeddingModel
    {
        get => _embeddingModel;
        set => this.RaiseAndSetIfChanged(ref _embeddingModel, value);
    }

    public string LlmModel
    {
        get => _llmModel;
        set => this.RaiseAndSetIfChanged(ref _llmModel, value);
    }

    public int BatchSize
    {
        get => _batchSize;
        set => this.RaiseAndSetIfChanged(ref _batchSize, value);
    }

    public float Temperature
    {
        get => _temperature;
        set => this.RaiseAndSetIfChanged(ref _temperature, value);
    }

    public bool IsRunning
    {
        get => _isRunning;
        private set
        {
            this.RaiseAndSetIfChanged(ref _isRunning, value);
            StartCommand?.NotifyCanExecuteChanged();
            StopCommand?.NotifyCanExecuteChanged();
        }
    }

    public string Status
    {
        get => _status;
        private set => this.RaiseAndSetIfChanged(ref _status, value);
    }

    public string LastError
    {
        get => _lastError;
        private set => this.RaiseAndSetIfChanged(ref _lastError, value);
    }

    public ObservableCollection<StageInfo> Stages { get; }

    public IObservable<ProgressInfo> ProgressObservable { get; }

    // ── Commands ──
    public ReactiveCommand<Unit, Unit> StartCommand { get; }
    public ReactiveCommand<Unit, Unit> StopCommand { get; }
    public ReactiveCommand<Unit, Unit> BrowseFolderCommand { get; }

    private CancellationTokenSource? CancellationTokenSource { get; set; }

    // ── Execution ──
    public async Task StartEmbeddingPipelineAsync()
    {
        CancellationTokenSource = new();
        var ct = CancellationTokenSource.Token;
        IsRunning = true;
        Status = "Running…";
        LastError = "";

        // Reset stage states
        foreach (var s in Stages) s.State = StageState.Pending;

        _logger.Information("=== Pipeline start ===");

        try
        {
            var ctx = new EmbeddingContext
            {
                SourceDirectory = SourceDirectory,
                CollectionName = CollectionName
            };

            var total = Stages.Count;
            var completed = 0;

            // Hook into stage completion via middleware / manual tracking
            _progress.OnNext(new ProgressInfo(completed, total));

            await _runner.ExecuteAsync(ctx, ct);

            // Mark all as done
            foreach (var s in Stages) s.State = StageState.Completed;
            _progress.OnNext(new ProgressInfo(total, total));

            Status = "Complete";
            _logger.Information("=== Pipeline complete ===");
        }
        catch (OperationCanceledException)
        {
            Status = "Cancelled";
            _logger.Information("=== Pipeline cancelled ===");
        }
        catch (Exception ex)
        {
            Status = "Failed";
            LastError = ex.Message;
            _logger.Fatal(ex, "Pipeline failed");
            throw;
        }
        finally
        {
            IsRunning = false;
        }
    }
}

    {
        _logger = logger;
        Title = "Logs";

        Entries = new ObservableCollection<LogEntry>();

        ClearCommand = ReactiveCommand.Create(() =>
        {
            Entries.Clear();
            _logger.Debug("Log cleared");
        });

        PauseCommand = ReactiveCommand.Create(() =>
        {
            IsPaused = !IsPaused;
            _logger.Debug("Log {State}", IsPaused ? "paused" : "resumed");
        });

        // In production, hook a Serilog sink that pushes to Entries.
        // For this example we expose the method for the caller to invoke.
    }

    public ObservableCollection<LogEntry> Entries { get; }

    public string Filter
    {
        get => _filter;
        set => this.RaiseAndSetIfChanged(ref _filter, value);
    }

    public bool AutoScroll
    {
        get => _autoScroll;
        set => this.RaiseAndSetIfChanged(ref _autoScroll, value);
    }

    public bool IsPaused
    {
        get => _isPaused;
        private set => this.RaiseAndSetIfChanged(ref _isPaused, value);
    }

    public ReactiveCommand<Unit, Unit> ClearCommand { get; }
    public ReactiveCommand<Unit, Unit> PauseCommand { get; }

    // Called by a custom Serilog sink
    public void Append(string level, string message, DateTimeOffset ts)
    {
        if (IsPaused) return;
        if (Filter != "All" && !level.Contains(Filter, StringComparison.OrdinalIgnoreCase)) return;

        Entries.Add(new LogEntry(ts, level, message));
        // Auto-trim to last 5 000
        while (Entries.Count > 5_000) Entries.RemoveAt(0);
    }

        };
