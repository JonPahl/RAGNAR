```csharp
public static class StringExtensions
{
    extension(ReadOnlySpan<char> value)
    {
        /// <summary>Counts non-tag, non-comment characters in XML-like markup with zero allocations.</summary>
        public int CountNonTagContent()
        {
            var count = 0;
            var inTag = false;

            for (var i = 0; i < value.Length; i++)
            {
                switch (value[[i]])
                {
                    case '<': inTag = true; break;
                    case '>': inTag = false; break;
                    default:
                        if (!inTag && value[[i]] != '/') count++;
                        break;
                }
            }

            return count;
        }

        /// <summary>Retrieves the last folder segment via span trimming.</summary>
        public ReadOnlySpan<char> FolderName => value.TrimEnd('/', '\\');

        /// <summary>Checks exclusion membership using cached ordinal comparison.</summary>
        public bool IsExcluded(in IReadOnlyCollection<string> exclusions) => 
            exclusions.Contains(value.ToString(), StringComparer.OrdinalIgnoreCase);
    }

    extension(string value)
    {
        /// <summary>Safety wrapper for span-based counting with a single allocation fallback.</summary>
        public int CharacterCount() => value.AsSpan().CountNonTagContent();
    }
```

#### 3. Hot-Path Allocation Elimination (`VectorStoreWriter.cs` Pattern)
Replaces `.Select().ToList()` LINQ projection with a capacity-hinted `foreach` loop. Demonstrates `in ReadOnlyMemory<char>` usage and optimized parallelism.

```csharp
public sealed class VectorStoreWriter(ILogger logger, IEmbeddingService embeddingService, IQdrantClient qdrantClient, IQdrantPointFactory generatorService, IOptions<RagnarConfig> config) : IVectorStoreWriter
{
    private readonly ApplicationOptions _appOptions = config.Value.ApplicationOptions;

    public async Task<UpdateResult> UpsertBatchAsync(IEnumerable<CodeDocument> codeDocuments, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        
        // Avoid double enumeration & unnecessary List<> allocation
        var docs = codeDocuments as IReadOnlyList<CodeDocument> ?? [[..codeDocuments]];
        var batchSize = config?.Value?.EmbeddingOptions?.BatchSize ?? 32;
        var batches = docs.Chunk(batchSize);
        var succeeded = 0;

        await Parallel.ForEachAsync(batches, new ParallelOptions { CancellationToken = cancellationToken }, async (batch, ct) =>
        {
            // Pre-allocate buffer for prompt construction
            var texts = new List<string>(batch.Length);
            
            foreach (var doc in batch)
            {
                // Use span-backed formatting to avoid intermediate string allocations
                var prompt = FormatPrompt(doc);
                texts.Add(prompt);
            }

            if (texts.Count == 0) return;

            var embeddings = await embeddingService.GenerateBatchAsync(texts, ct).ConfigureAwait(false);
            
            // Batch upsert avoids per-document network round-trips
            var points = new List<PointStruct>(batch.Length * 2);
            for (var i = 0; i < batch.Length; i++)
            {
                var doc = batch[[i]];
                points.AddRange(generatorService.BuildPointStructs(doc.AsPoint(), embeddings[[i]].Vector.ToArray(), doc));
            }

            await qdrantClient.UpsertAsync(_appOptions.VectorStoreName, points, ct).ConfigureAwait(false);
            Interlocked.Add(ref succeeded, batch.Length);
        }).ConfigureAwait(false);

        return new UpdateResult { Status = succeeded == docs.Count ? UpdateStatus.Completed : UpdateStatus.UnknownUpdateStatus };
    }

    // Demonstrates 'in' modifier for large structs/spans in lambda-like contexts
    private static string FormatPrompt(in CodeDocument doc) => 
        $"Context: {doc.ElementName}\nCode:\n{doc.Code}";
}
```

#### 4. Optimized File Discovery (`FileDiscoveryService.cs` Pattern)
Eliminates `FileInfo` allocation, uses span-based extension methods, and leverages .NET 10's improved directory enumeration.

```csharp
public class FileDiscoveryService(ILogger logger) : IFileDiscoveryService
{
    private static readonly char[[]] _pathSeparators = [[Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar]];

    public async IAsyncEnumerable<string> EnumerateFilesAsync(string directory, IOptions<RagnarConfig> options, [[EnumeratorCancellation]] CancellationToken cancellationToken)
    {
        var filter = options.Value.FileLoadOptions;
        Guard.Against.NullOrEmpty(directory);

        if (!Directory.Exists(directory))
            throw new DirectoryNotFoundException(directory);

        // .NET 10: Path.GetExtension(ReadOnlySpan<char>) is allocation-free
        foreach (var path in Directory.EnumerateFiles(directory, "*.*", SearchOption.AllDirectories))
        {
            cancellationToken.ThrowIfCancellationRequested();
            
            var ext = path.AsSpan().GetFileNameAfterLastSeparator()!.GetExtension();
            if (!filter.AllowedExtensions.Contains(ext.ToString(), StringComparer.OrdinalIgnoreCase))
                continue;

            // Avoid new FileInfo(); use span-based name extraction
            var fileName = path.AsSpan().GetFileNameAfterLastSeparator();
            if (filter.ExcludedFiles.Contains(fileName.ToString(), StringComparer.OrdinalIgnoreCase))
                continue;

            // Check ancestor directories without allocation
            if (IsExcludedDirectory(path, filter.ExcludedDirectories))
                continue;

            yield return path;
        }
    }

    private static bool IsExcludedDirectory(string path, HashSet<string> exclusions)
    {
        var span = path.AsSpan();
        while (true)
        {
            var part = span.GetPathRoot() ?? span.Slice(0, span.IndexOfAny(_pathSeparators) + 1);
            if (exclusions.Contains(part.ToString(), StringComparer.OrdinalIgnoreCase))
                return true;
            
            var separatorIndex = span.IndexOfAny(_pathSeparators);
            if (separatorIndex == -1) break;
            
            span = span.Slice(separatorIndex + 1);
        }
        return false;
    }
}

// Helper extension for span manipulation
public static class PathSpanExtensions
{
    public static ReadOnlySpan<char>? GetFileNameAfterLastSeparator(this ReadOnlySpan<char> path)
    {
        var index = path.LastIndexOfAny('/', '\\');
        return index == -1 ? (ReadOnlySpan<char>?)path : path[[(index + 1)..]];
    }
}
```
