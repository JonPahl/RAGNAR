using Ragnar.Core;
using Xunit;

public class EmbeddingOptionsTests
{
    [Fact]
    public void ShouldStoreAllProperties()
    {
        var opts = new EmbeddingOptions
        {
            Host = new Uri("http://localhost:11434"),
            EmbeddingModel = "nomic-embed-text",
            Timeout = TimeSpan.FromSeconds(30),
            Dimension = 768,
            BatchSize = 16
        };

        Assert.Equal(new Uri("http://localhost:11434"), opts.Host);
        Assert.Equal("nomic-embed-text", opts.EmbeddingModel);
        Assert.Equal(TimeSpan.FromSeconds(30), opts.Timeout);
        Assert.Equal(768UL, opts.Dimension);
        Assert.Equal(16, opts.BatchSize);
    }

    [Fact]
    public void DimensionShouldAcceptBoundaryValues()
    {
        var low = new EmbeddingOptions
        {
            Host = new Uri("http://h"),
            EmbeddingModel = "m",
            Timeout = TimeSpan.FromSeconds(10),
            Dimension = 1,
            BatchSize = 1
        };
        Assert.Equal(1UL, low.Dimension);

        var high = new EmbeddingOptions
        {
            Host = new Uri("http://h"),
            EmbeddingModel = "m",
            Timeout = TimeSpan.FromSeconds(10),
            Dimension = 65536,
            BatchSize = 1
        };
        Assert.Equal(65536UL, high.Dimension);
    }

    [Fact]
    public void BatchSizeShouldAcceptBoundaryValues()
    {
        var low = new EmbeddingOptions
        {
            Host = new Uri("http://h"),
            EmbeddingModel = "m",
            Timeout = TimeSpan.FromSeconds(10),
            Dimension = 1,
            BatchSize = 1
        };
        Assert.Equal(1, low.BatchSize);

        var high = new EmbeddingOptions
        {
            Host = new Uri("http://h"),
            EmbeddingModel = "m",
            Timeout = TimeSpan.FromSeconds(10),
            Dimension = 1,
            BatchSize = 512
        };
        Assert.Equal(512, high.BatchSize);
    }

    [Fact]
    public void ShouldSupportEquality()
    {
        var a = new EmbeddingOptions
        {
            Host = new Uri("http://x"),
            EmbeddingModel = "model",
            Timeout = TimeSpan.FromSeconds(30),
            Dimension = 128,
            BatchSize = 8
        };
        var b = new EmbeddingOptions
        {
            Host = new Uri("http://x"),
            EmbeddingModel = "model",
            Timeout = TimeSpan.FromSeconds(30),
            Dimension = 128,
            BatchSize = 8
        };

        Assert.Equal(a, b);
    }

    [Fact]
    public void ShouldDifferWhenModelDiffers()
    {
        var a = new EmbeddingOptions
        {
            Host = new Uri("http://x"),
            EmbeddingModel = "model-a",
            Timeout = TimeSpan.FromSeconds(30),
            Dimension = 128,
            BatchSize = 8
        };
        var b = new EmbeddingOptions
        {
            Host = new Uri("http://x"),
            EmbeddingModel = "model-b",
            Timeout = TimeSpan.FromSeconds(30),
            Dimension = 128,
            BatchSize = 8
        };

        Assert.NotEqual(a, b);
    }

    {
        var original = new EmbeddingOptions
        {
            Host = new Uri("http://x"),
            EmbeddingModel = "model",
            Timeout = TimeSpan.FromSeconds(30),
            Dimension = 128,
            BatchSize = 8
        };

        var copy = original with { BatchSize = 32 };

        Assert.Equal(8, original.BatchSize);
        Assert.Equal(32, copy.BatchSize);
    }
}
