using FluentAssertions;
using System.Linq;
using Xunit;

public class ChunkBySyntaxTreeTests
{
    [Fact]
    public void ChunkSourceFile_WithValidClass_ReturnsCodeDocument()
    {
        // Arrange
        var chunker = new ChunkBySyntaxTree();
        const string code = @"
            /// <summary>
            /// Represents a test class for parsing.
            /// </summary>
            public class TestClass 
            {
                public void DoWork() { }
            }";

        // Act
        var result = chunker.ChunkSourceFile("Core/TestClass.cs", code);

        // Assert
        result.Should().NotBeNull();
        result.Should().HaveCount(1);
        
        var doc = result.First();
        doc.ElementType.Should().Be("ClassDeclaration");
        doc.ElementName.Should().Be("TestClass");
        doc.Comment.Should().Contain("Represents a test class for parsing.");
        doc.Category.Should().Be("Core"); // Inferred from path
    }

    [Fact]
    public void ChunkSourceFile_WithInvalidSyntax_ReturnsNull()
    {
        // Arrange
        var chunker = new ChunkBySyntaxTree();
        const string code = "public class Invalid {{{";

        // Act
        var result = chunker.ChunkSourceFile("test.cs", code);

        // Assert
        result.Should().BeNull();
    }

    [Fact]
    public void ChunkSourceFile_WithMultipleClasses_ReturnsAllChunks()
    {
        // Arrange
        var chunker = new ChunkBySyntaxTree();
        const string code = @"
            public class FirstClass { }
            /// <summary>Second</summary>
            public class SecondClass { }";

        // Act
        var result = chunker.ChunkSourceFile("test.cs", code);

        // Assert
        result.Should().HaveCount(2);
        result.Select(d => d.ElementName).Should().ContainInOrder("FirstClass", "SecondClass");
    }
}

        var orchestrator = new RagOrchestrator(
            writerMock.Object,
            configMock.Object,
            systemPromptMock.Object,
            saveServiceMock.Object,
            clientFactoryMock.Object,
            ollamaProviderMock.Object);

using FluentAssertions;
using FluentValidation;

// Example Validator matching your OllamaOptions/EmbeddingOptions structure
public class OllamaOptionsValidator : AbstractValidator<OllamaOptions>

    public OllamaOptionsValidator()

    [Fact]

    {
        var options = new OllamaOptions

            Timeout = TimeSpan.FromMinutes(20)

    public void Validate_InvalidOptions_ReturnsErrors()

### 🔍 Testability Notes & Recommendations

1. **`FileParseFactory` Instantiation**: Currently, `FileParseFactory` creates `ParseCSharpFile` and `FileParser` directly in its constructor (`new ParseCSharpFile(Logger)`). This makes it impossible to mock without refactoring. 
   ✅ **Fix**: Inject interfaces:
   ```csharp
   public class FileParseFactory(IOptions<ApplicationConfiguration> config, ILogger logger, IParseCSharpFile cSharpParser, IFileParser fileParser) : IFileParseFactory { ... }
   ```

2. **`ResponseWriter` File I/O**: Uses real `Directory.CreateDirectory` and `File.WriteAllTextAsync`. For strict unit tests, consider injecting `IFileSystem` (via `System.IO.Abstractions`) or move to integration tests using a temporary directory (`Path.GetTempPath()`).

3. **FluentValidation vs DataAnnotations**: Your builder uses `.ValidateDataAnnotations()`. If you want to keep that, the `OllamaOptionsValidator` above shows how FluentValidation integrates cleanly if you ever migrate validation logic out of attributes for better testability and composition.

4. **Static Dependencies**: Classes like `AnsiConsoleOutputWriter` wrap static libraries. They are best tested via integration tests or by extracting an interface that delegates to a mockable abstraction in production code.

This suite covers all public, logically testable methods while adhering to xUnit/Moq/FluentAssertions best practices. Let me know if you need virtual file system mocking for `ResponseWriter` or DI wiring helpers!
