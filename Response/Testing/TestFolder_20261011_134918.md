# ============================================================
# Reorganize-Tests.ps1
# Reorganizes all *Tests.cs / *Test.cs files into a domain-based
# folder layout under a "Tests" root directory.
#
# Usage:
#   pwsh ./Reorganize-Tests.ps1 [[-ProjectRoot <path>]] [[-DryRun]]
#
# Layout produced:
#   Tests/
#   ├── Pipeline/
#   │   └── Stages/
#   ├── Services/
#   ├── Infrastructure/
#   ├── Extensions/
#   ├── Parsing/
#   ├── Embedding/
#   └── Shared/
# ============================================================

[[CmdletBinding()]]
param(
    [[string]]$ProjectRoot = (Get-Location).Path,
    [[switch]]$DryRun
)

$ErrorActionPreference = 'Stop'

# ─── 1. Discover all test files ────────────────────────────
$testFiles = Get-ChildItem -Path $ProjectRoot -Recurse -File |
    Where-Object {
        $_.Name -match '^(.+?)(Tests?\.cs)$' -and
        $_.FullName -notmatch '[[\\/]](bin|obj|\.git|node_modules)[[\\/]]'
    } |
    Select-Object -ExpandProperty FullName

Write-Host "`nDiscovered $($testFiles.Count) test file(s).`n" -ForegroundColor Cyan

# ─── 2. Classification rules (ordered – first match wins) ──
# Each rule:  pattern on file/class name  →  target subfolder
$rules = @(
    # Pipeline stage tests
    @{ Pattern = '(?i)StageTests?\.cs$';                    Folder = 'Pipeline/Stages' }
    @{ Pattern = '(?i)Pipeline(Runner|Context)Tests?\.cs$'; Folder = 'Pipeline' }

    # Embedding / vector-store
    @{ Pattern = '(?i)(Embedding|VectorStore|Qdrant)Tests?\.cs$'; Folder = 'Embedding' }

    # Parsing / syntax
    @{ Pattern = '(?i)(Parser|Parse|Chunker|Syntax)Tests?\.cs$'; Folder = 'Parsing' }

    # Services
    @{ Pattern = '(?i)(Service|Generator|Discovery)Tests?\.cs$'; Folder = 'Services' }

    # Infrastructure (filters, repositories, factories)
    @{ Pattern = '(?i)(Filter|Builder|Repository|Factory|Strategy)Tests?\.cs$'; Folder = 'Infrastructure' }

    # Extension-method tests
    @{ Pattern = '(?i)(Extension|Helper|Extensions)Tests?\.cs$'; Folder = 'Extensions' }

    # Question / RAG orchestration
    @{ Pattern = '(?i)(Question|RagOrchestrat|Combine)Tests?\.cs$'; Folder = 'Pipeline/Stages' }

    # Configuration / options
    @{ Pattern = '(?i)(Options|Config|Defaults)Tests?\.cs$'; Folder = 'Infrastructure' }

    # Fallback
    @{ Pattern = '';                                          Folder = 'Shared' }
)

# ─── 3. Classify each file ─────────────────────────────────
$testRoot = Join-Path $ProjectRoot 'Tests'
$assignments = @()

foreach ($file in $testFiles) {
    $name = [[IO.Path]]::GetFileName($file)

    # Already in a sensible place?
    $relative = $file.Substring($ProjectRoot.Length).TrimStart('\','/')
    if ($relative -match '^Tests[[/\\]]') { continue }

    $target = $null
    foreach ($rule in $rules) {
        if ($rule.Pattern -and $name -match $rule.Pattern) {
            $target = Join-Path $testRoot $rule.Folder
            break
        }
        if (-not $rule.Pattern) {
            $target = Join-Path $testRoot $rule.Folder
        }
    }

    $assignments += [[PSCustomObject]]@{
        Source  = $file
        Target  = Join-Path $target $name
        Folder  = $target
    }
}

# ─── 4. Create directories & move files ────────────────────
Write-Host "Target layout:" -ForegroundColor Cyan
$assignments |
    Sort-Object Folder, Target |
    ForEach-Object {
        Write-Host ("  {0,-12}  {1}  →  {2}" -f
            (if ($DryRun) { '[[DRY-RUN]]' } else { 'MOVE' }),
            $_.Source.Substring($ProjectRoot.Length).TrimStart('\','/'),
            $_.Target.Substring($ProjectRoot.Length).TrimStart('\','/'))

        if (-not $DryRun) {
            $dir = Split-Path $_.Target -Parent
            if (-not (Test-Path $dir)) {
                New-Item -ItemType Directory -Path $dir -Force | Out-Null
            }
            Move-Item -Path $_.Source -Destination $_.Target -Force
        }
    }

# ─── 5. Create .gitkeep for empty dirs ─────────────────────
if (-not $DryRun) {
    Get-ChildItem -Path $testRoot -Recurse -Directory |
        ForEach-Object {
            $keep = Join-Path $_.FullName '.gitkeep'
            if (-not (Test-Path $keep) -and
                -not (Get-ChildItem -Path $_.FullName -File | Where-Object Name -ne '.gitkeep')) {
                New-Item -Path $keep -ItemType File -Force | Out-Null
            }
        }
}

# ─── 6. Suggest namespace updates ──────────────────────────
if (-not $DryRun) {
    Write-Host "`nTip: update each file's namespace to match the new folder path, e.g.:" -ForegroundColor Yellow
    Write-Host '  namespace Ragnar.Tests.Pipeline.Stages;' -ForegroundColor Gray
    Write-Host '  namespace Ragnar.Tests.Extensions;'    -ForegroundColor Gray
    Write-Host ""
    Write-Host "Run `Update-Namespaces.ps1` (below) to auto-patch them." -ForegroundColor Yellow
}

Write-Host "`nDone. $($assignments.Count) file(s) processed." -ForegroundColor Green

# ============================================================
# Update-Namespaces.ps1
# Rewrites the top-level `namespace` declaration in each test
# file to mirror its new folder path.
# ============================================================
[[CmdletBinding()]]
param(
    [[string]]$ProjectRoot = (Get-Location).Path,
    [[string]]$RootNamespace = 'Ragnar.Tests',
    [[switch]]$DryRun
)

$ErrorActionPreference = 'Stop'
$testRoot = Join-Path $ProjectRoot 'Tests'

Get-ChildItem -Path $testRoot -Recurse -File -Filter '*.cs' | ForEach-Object {
    $rel = $_.FullName.Substring($testRoot.Length).TrimStart('\','/')
    $rel = $rel.Replace('\','.') -replace '\.cs$',''
    $newNs = "$RootNamespace.$rel"

    $content = Get-Content -Path $_.FullName -Raw
    $pattern = '(?m)^\s*namespace\s+[[\w.]]+\s*;?\s*$'

    if ($content -match $pattern) {
        $updated = [[regex]]::Replace($content, $pattern, "namespace $newNs;")
        if ($updated -ne $content) {
            Write-Host "  {0}  →  namespace {1};" -f $_.Name, $newNs
            if (-not $DryRun) { Set-Content -Path $_.FullName -Value $updated -NoNewline }
        }
    }
    else {
        Write-Warning "  {0}: no namespace line found – add one manually." -f $_.Name
    }
}
Write-Host "Namespaces updated." -ForegroundColor Green
