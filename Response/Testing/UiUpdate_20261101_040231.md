# Ragnar Console UI — Improvement & Replacement Sketch

Ragnar is a **console (terminal) application**, so "UI" means the text-rendering layer: branding, tables, progress indicators, and prompt formatting. Below is a concrete, layered plan that either **improves** what exists (the `NumberedRowDecorator` / `ConsoleTableBuilder` / `BrandingStage` / `ShowPrompt` pipeline) or **replaces** it with a richer terminal-UI library.

---

## 1. Current State (What You Have)

| Concern | Where | Limitation |
|---|---|---|
| Banner / branding | `BrandingStage` → `IApplicationHeader.RenderBranding()` | Static text block, no color |
| Tables (questions, files) | `NumberedRowDecorator` + `ITableRenderer` + `ConsoleTableBuilder` | Custom, no ANSI colours, no column auto-sizing |
| Stage progress | `StageTimingMiddleware` → `logger.Information(...)` | One-line log after the fact; no live spinner |
| Prompt display | `OutputPathHelper.ShowPrompt()` | Hard-coded fence markers; no syntax highlighting |
| Logging | Serilog console sink | Plain text; no structured colour |
| CLI parsing | (presumably `Microsoft.Extensions.Hosting`) | No `--help` tree, no sub-commands visible |

---

## 2. Recommended Replacement: **Spectre.Console**

Spectre.Console is a single NuGet package that gives you:

- Rich **tables** with auto-sizing, borders, column alignment, and colour
- **Progress bars** (multiple, animated) with spinners
- **Panels / rules** for section headers
- **Prompts** (confirm, text, choice) for interactive flows
- **Syntax highlighting** for code snippets
- A **CLI framework** (`Spectre.Console.Cli`) with auto-generated `--help`
- Theme-aware colours (respects the user's terminal theme)

```csharp
// ── Branding ──────────────────────────────────────────────────────
public sealed class SpectreBrandingRenderer(Serilog.ILogger logger) : IBrandingRenderer
{
    private static readonly AnsiConsole Console = AnsiConsole.Create(
        new AnsiConsoleSettings
        {
            Profile = AnsiConsoleProfile.Full,
            ColorSystem = ColorSystemSupport.Detect,
        });

    public void RenderBanner()
    {
        Console.Write(new FigletText("Ragnar")
            .Style(Style.Parse("bold cyan")));

        Console.Write(new Rule($"v{AssemblyExtensions.InformationalVersion}" +
                               "  ·  Code-aware RAG for .NET")
            .Style(Style.Parse("dim")));
    }

    public void RenderStageHeader(string name, int index, int total)
        => Console.Write(new Rule($"[[{index + 1}/{total}]] {name}")
            .Style(Style.Parse("bold green")));

    public void RenderStageFooter(StageMetric m)
    {
        var style = m.Success ? "green" : "red";
        Console.WriteLine(new Text(
            $"  {m.Success ? "✓" : "✗"}  {m.StageName}  —  {m.Duration:hh\\:mm\\:ss\\.fff}  " +
            (m.RetryCount > 0 ? $"(retried ×{m.RetryCount}) " : "") +
            (m.Error is null ? "" : $"error: {m.Error}"))
            .Style(Style.Parse(style)));
    }
}
```

```csharp
// ── Tables ────────────────────────────────────────────────────────
public sealed class SpectreTableRenderer : IRichTableRenderer
{
    private static readonly AnsiConsole Console = AnsiConsole.Create(
        new AnsiConsoleSettings { ColorSystem = ColorSystemSupport.Detect });

    public void RenderQuestions(IReadOnlyList<Question> qs)
    {
        var table = new Table()
            .Border(TableBorder.Rounded)
            .AddColumn("#").Centered()
            .AddColumn("Category").MinWidth(10)
            .AddColumn("Prompt").MinWidth(40).MaxWidth(80)
            .AddColumn("Length");

        for (var i = 0; i < qs.Count; i++)
            table.AddRow(
                (i + 1).ToString(),
                qs[[i]].Category.ToString(),
                Truncate(qs[[i]].Prompt, 80),
                qs[[i]].Prompt.Length.ToString());

        Console.Write(table);
    }

    public void RenderFiles(IReadOnlyList<string> paths)
    {
        var table = new Table()
            .Border(TableBorder.Rounded)
            .AddColumn("#")
            .AddColumn("File");

        for (var i = 0; i < paths.Count; i++)
            table.AddRow((i + 1).ToString(), paths[[i]]);

        Console.Write(table);
    }

    public void RenderCollectionInfo(CollectionInfo info)
        => Console.Write(new Panel(
            new Text($"  {info.Name}  ·  {info.Dimension} dims  ·  " +
                     $"{info.VectorsCount:N0} vectors  ·  {info.Status}")
            .Style(info.Status == "green" ? "green" : "yellow"))
            .Border(PanelBorder.Rounded)
            .Title(new Text(" Qdrant Collection ").Style("bold cyan")));

    private static string Truncate(string s, int max)
        => s.Length <= max ? s : s[[..(max - 1)]] + "…";
}
```

```csharp
// ── Progress (used by FileDiscovery / Embedding stages) ──────────
public sealed class SpectreStageProgress : IStageProgress
{
    private readonly List<Task> _tasks = [[]];
    private readonly Progress _progress;

    public IAsyncDisposable Begin(string label)
    {
        _progress = new Progress();
        var task = new Task(() => _progress.Start(
            new ProgressTask(label)
                .AutoName(false)
                .AutoClear(true)
                .Spinner(Spinner.Known.Dots)));
        task.Start();
        _tasks.Add(task);
        return this;
    }

    public void Report(int current, int total, string? note = null)
    {
        var t = _tasks[[0]];
        t.Description = note ?? t.Description;
        t.Value = (double)current;
        t.MaxValue = total;
    }

    // IAsyncDisposable – waits for the progress spinner task to finish
    public async ValueTask DisposeAsync()
    {
        foreach (var t in _tasks)
            await t;
    }
}
```
