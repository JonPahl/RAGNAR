### Fix 1 – `QuestionFilterSelector`: remove per-call mutation, make thread-safe

```csharp
// QuestionFilterSelector.cs  (REPLACED)
public sealed class QuestionFilterSelector
{
    private readonly Dictionary<QuestionCategory, IFilterStrategy> _strategies;
    private readonly Serilog.ILogger _logger;

    public QuestionFilterSelector(Serilog.ILogger logger, IEnumerable<IFilterStrategy> strategies)
    {
        _logger = logger;
        _strategies = strategies.ToDictionary(s => s.SupportedCategory);

        // Register built-in strategies ONCE, at construction time.
        if (!_strategies.ContainsKey(QuestionCategory.XML))
            _strategies[[QuestionCategory.XML]] = new XmlCommentFilterStrategy(logger);
    }

    public Filter FindFilter(QuestionCategory category, int commentLengthThreshold = 0)
    {
        _logger.Debug("Finding filter for {Category}, threshold {Threshold}",
                      category, commentLengthThreshold);

        var filter = _strategies.TryGetValue(category, out var strategy)
            ? strategy.CreateFilter(commentLengthThreshold)
            : new Filter();

        _logger.Debug("Filter resolved for {Category}: {MustCount} must, {MustNotCount} mustNot",
                      category, filter.Must?.Count ?? 0, filter.MustNot?.Count ?? 0);
        return filter;
    }
}
```

**Why:** The old code called `_strategies.TryAdd(...)` on *every* invocation of `FindFilter`, which (a) is a data race under concurrent callers and (b) defeats the purpose of constructor injection.

### Fix 2 – `NumberedRowDecorator`: don't duplicate rows

```csharp
// NumberedRowDecorator.cs  (REPLACED)
public sealed class NumberedRowDecorator(ITableRenderer inner, string columnHeader = "#") : ITableRenderer
{
    private readonly ITableRenderer _inner = inner;
    private readonly string _columnHeader;
    private bool _numbered;

    public NumberedRowDecorator(ITableRenderer inner, string columnHeader = "#")
    {
        _inner = inner ?? throw new ArgumentNullException(nameof(inner));
        _columnHeader = columnHeader;
    }

    public ConsoleTableBuilder Table => _inner.Table;

    public void Render()
    {
        if (_numbered)
        {
            _inner.Render();
            return;
        }
        _numbered = true;

        var table = _inner.Table;

        // 1. Ensure the index column exists
        if (!table.Columns.Contains(_columnHeader, StringComparer.OrdinalIgnoreCase))
            table.AddColumns(_columnHeader);

        // 2. Prepend a 1-based index to every existing row IN PLACE
        var existing = table.Rows.ToList();
        for (var i = 0; i < existing.Count; i++)
        {
            var cells = existing[[i]].ToArray();
            var indexed = new string[[table.Columns.Count]];
            indexed[[0]] = (i + 1).ToString();
            for (var c = 0; c < cells.Length; c++)
                indexed[[c + 1]] = cells[[c]];

            table.Rows[[i]] = indexed.Select(c => new TableCell(c)).ToList(); // ← replace, don't add
        }

        // 3. Delegate final rendering to inner
        _inner.Render();
    }
}
```

**Why:** The original called `table.AddRow(...)` for each existing row, producing *2×N* rows after the first render.

    {

                        .ConfigureAwait(false);

            catch (Exception ex)

                throw new PipelineStageException(stage.Name, ex);

    /// Builds a response-directory path.

    /// <param name="baseDir">
    /// Optional root prefix.  Applied **only** when <paramref name="folders"/> is empty,
    /// to avoid silently discarding an explicit path.

        return Path.Join(parts.Append(DomainDefaults.ResponseDir).ToArray());

### Fix 5 – `StageTimingMiddleware`: expose a retry-count hook

```csharp
// StageMetric.cs – add an optional init-friendly property
public sealed record StageMetric
{
    public string  StageName   { get; init; } = string.Empty;
    public TimeSpan Duration   { get; init; }
    public bool     Success    { get; init; }
    [[Range(0, 10)]] public int  RetryCount { get; init; }
    public string?  Error      { get; init; }
}

// StageExecutionContext – make Metric init-only
public sealed record StageExecutionContext<TContext>(
    IPipelineStage<TContext> Stage,
    TContext                  Context,
    int                       StageIndex,
    int                       StageTotal)
    where TContext : class
{
    public StageMetric? Metric { get; init; }   // ← was { get; set; }
}

// StageTimingMiddleware – accept an optional Func to override RetryCount
public sealed class StageTimingMiddleware(Serilog.ILogger logger, Func<StageMetric, StageMetric>? retryEnricher = null)
{
    public async Task Invoke<TContext>(
        StageExecutionContext<TContext> context,
        Func<Task> next,
        CancellationToken cancellationToken)
        where TContext : class
    {
        cancellationToken.ThrowIfCancellationRequested();
        ArgumentNullException.ThrowIfNull(context);
        ArgumentNullException.ThrowIfNull(next);

        var sw = Stopwatch.StartNew();
        var succeeded = false;
        string? error = null;

        try
        {
            await next().ConfigureAwait(false);
            succeeded = true;
        }
        catch (Exception ex)
        {
            error = ex.Message;
            throw;
        }
        finally
        {
            sw.Stop();
            var metric = new StageMetric
            {
                StageName  = context.Stage.Name,
                Duration   = sw.Elapsed,
                Success    = succeeded,
                RetryCount = 0,
                Error      = error
            };
            context.Metric = retryEnricher?.Invoke(metric) ?? metric;   // ← hook

            logger.Information(
                "Stage [[{Index}/{Total}]] {Name} – {Elapsed} (success={Success}, retries={Retries})",
                context.StageIndex + 1, context.StageTotal,
                context.Stage.Name, sw.Elapsed, succeeded, metric.RetryCount);
        }
    }
}
```

Now a separate retry middleware can supply the true retry count via the `retryEnricher` delegate without a hard-coded `0`.
