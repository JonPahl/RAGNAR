            FileName = "test.cs",
            Directory = "/src"
        };

        };

        _qdrantMock
            .Setup(q => q.UpsertAsync("test_store", It.IsAny<List<Qdrant.Client.Models.PointStruct>>(), It.IsAny<CancellationToken>()))
            .ReturnsAsync(new Qdrant.Client.Models.UpdateResult { Status = Qdrant.Client.Models.UpdateStatus.Completed });

        var result = await _sut.UpsertBatchAsync([[doc]], CancellationToken.None);

        result.Status.Should().Be(Qdrant.Client.Models.UpdateStatus.Completed);
    }
