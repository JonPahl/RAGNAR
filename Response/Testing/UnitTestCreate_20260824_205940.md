### 1. `OllamaOptionsValidatorTests` (FluentValidation)
Validates configuration rules for Ollama API connections.
```csharp
using FluentValidation.TestHelper;
using Xunit;

public class OllamaOptionsValidatorTests
{
    private readonly OllamaOptionsValidator _validator = new();

    [Theory]
    [InlineData("localhost", 11434, TimeSpan.FromMinutes(20), "qwen3.6")] // Valid
    public void Validate_Should_Pass_When_All_Fields_Are_Valid(string host, int port, TimeSpan timeout, string model)
    {
        var options = new OllamaOptions { Host = host, Port = port, Timeout = timeout, LlmModel = model };
        _validator.ShouldNotHaveAnyValidationErrors(options);
    }

    [Theory]
    [InlineData("", 11434, TimeSpan.FromMinutes(20), "qwen3.6")] // Empty Host
    [InlineData("host", 0, TimeSpan.FromMinutes(20), "qwen3.6")] // Port < 1
    [InlineData("host", 70000, TimeSpan.FromMinutes(20), "qwen3.6")] // Port > 65535
    [InlineData("host", 11434, TimeSpan.Zero, "qwen3.6")] // Timeout <= 0
    [InlineData("host", 11434, TimeSpan.FromMinutes(20), "")] // Empty Model
    public void Validate_Should_Fail_When_Invalid_Field(string host, int port, TimeSpan timeout, string model)
    {
        var options = new OllamaOptions { Host = host, Port = port, Timeout = timeout, LlmModel = model };
        
        if (string.IsNullOrEmpty(host)) _validator.ShouldHaveValidationErrorFor(x => x.Host)(options);
        else if (port < 1 || port > 65535) _validator.ShouldHaveValidationErrorFor(x => x.Port)(options);
        else if (timeout <= TimeSpan.Zero) _validator.ShouldHaveValidationErrorFor(x => x.Timeout)(options);
        else _validator.ShouldHaveValidationErrorFor(x => x.LlmModel)(options);
    }
}
```

```csharp
        var config = new RagnarConfig
        {
            ApplicationOptions = new ApplicationOptions
            {
                SourceDirectory = _tempDir,
                VectorStoreName = "test_store"
            }
        };
        _configMock = new Mock<IOptions<RagnarConfig>>();
        _configMock.Setup(c => c.Value).Returns(config);
    }

    [Fact]
    public async Task WriteResponseAsync_CreatesFileWithCorrectStructure()
    {
        var writer = new ResponseWriter(_configMock.Object);
        var question = new Question(true, "Test Query", "test_file.cs", QuestionCategory.Refactor);
        var details = new SaveDetails(question, "Generated Answer", "00:15");

        var path = await writer.WriteResponseAsync(details, CancellationToken.None);

        Assert.NotNull(path);
        Assert.True(File.Exists(path));
        Assert.EndsWith(".md", path);

        var content = await File.ReadAllTextAsync(path, TestContext.Current.CancellationToken);
        
        Assert.Contains(question.MarkdownHeader, content);
        Assert.Contains("## Question: ", content);
        Assert.Contains("Test Query", content);
        Assert.Contains("**Method Call Duration**: 00:15", content);
        Assert.Contains("## Response: ", content);
        Assert.Contains("Generated Answer", content);

        Cleanup();
    }

    [Fact]
    public void BuildDirectory_CreatesCategorySubFolder()
    {
        var expected = Path.Join(_tempDir, "Response", "Refactor");
        // Verified implicitly via WriteResponseAsync above. Explicit check:
        Assert.True(Directory.Exists(expected));
    }

    private void Cleanup()
    {
        if (Directory.Exists(_tempDir)) Directory.Delete(_tempDir, true);
    }
```

### 🔍 Implementation Notes & Best Practices

1. **Static/Internal Instantiation**: `EmbeddingPipeline` directly instantiates `new Core.VectorStoreBuilder(...)`. For full isolation in production, extract an `IVectorStoreBuilder` interface and inject it via constructor. The current test verifies the *contract* (logs/writes) assuming the builder behaves correctly.
2. **DateTime.Now Dependency**: `ResponseWriter` uses `DateTime.Now` for filenames. Tests verify structural content rather than exact timestamps to remain deterministic across runs.
3. **FluentValidation v12+ Syntax**: Uses `.ShouldNotHaveAnyValidationErrors()` and `.ShouldHaveValidationErrorFor()` which require `using FluentValidation.TestHelper;`.
4. **Mocking Interfaces**: All tests assume standard .NET interfaces (`IOptions<T>`, `ILogger`, `IOutputWriter`, etc.) exist in your codebase. If they are custom, ensure their method signatures match the mock setups.
5. **Cleanup**: Temp directories are created per test and cleaned up to prevent CI/CD pollution. In xUnit v3+, you can also use `[Collection("TempDir")]` for shared cleanup if needed.
