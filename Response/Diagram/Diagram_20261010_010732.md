    Brand-->>Runner: branding rendered
    Runner->>QShow: ShouldRun(ctx)
    QShow-->>Runner: true

    QShow->>Console: Render questions table
    QShow-->>Runner: done

    %% Question loading
    QExec->>QBuilder: GetCategories().GetFileConfig().GetCsvFilesAsync(dir, ct)
    QBuilder->>Disc: GetFilesAsync(dir, filter, ct)
    Disc-->>QBuilder: async file paths
    QBuilder->>Parser: ParseAsync(file, ct)
    Parser-->>QBuilder: CodeDocuments
    QBuilder-->>QExec: IReadOnlyList<Question>

    %% Per-question loop
    loop For each Question
        QExec->>Console: MarkupLine "Question: …"
        QExec->>Qdrant: RetrieveContextAsync(collection, ct, filter)
        Qdrant-->>QExec: contextText (relevant code snippets)

        QExec->>Orch: ExecuteAsync(question, contextText, ct)

        Orch->>Ollama_E: (resolve client)
        Orch->>Ollama_L: GenerateResponse(request, ct)
        Ollama_L-->>Orch: generated answer text
        Orch->>Orch: append prompt if IncludeOriginalPrompt

        Orch->>Saver: WriteResponseAsync(ResponseRecord, ct)
        Saver->>Saver: create Response/ dir
        Saver->>Saver: File.WriteAllTextAsync(summary.md)
        Saver-->>Orch: full file path

        Orch-->>QExec: done
        QExec->>Console: MarkupLine "{n} of {total}"
    end

    Runner->>Runner: log "Pipeline finished in {time}"
    Runner-->>Dev: completed
