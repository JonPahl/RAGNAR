```csharp
/// <summary>Provides extension methods for DI service registration.</summary>
/// <remarks>Encapsulates embedding and plugin discovery registrations.</remarks>
/// <example><![[CDATA[[services.RegisterEmbeddingServices().LoadQuestionPlugins();]]]]></example>
public static class ApplicationConfigurationExtensions
{
    extension(IServiceCollection services)
    {
        /// <summary>Discovers and registers IQuestionSource implementations from plugin DLLs.</summary>
        /// <remarks>Loads assemblies from the base Questions/Plugins directory.</remarks>
        /// <example><![[CDATA[[services.LoadQuestionPlugins();]]]]></example>
        /// <returns>The service collection for chaining.</returns>
        public IServiceCollection LoadQuestionPlugins()
        {
            var pluginDir = Path.Join(AppContext.BaseDirectory, "Questions", "Plugins");
            Serilog.Log.Debug("Scanning plugin directory: {PluginDir}", pluginDir);

            if (!Directory.Exists(pluginDir))
            {
                Serilog.Log.Information("Plugin directory not found, skipping: {PluginDir}", pluginDir);
                return services;
            }

            var dlls = Directory.EnumerateFiles(pluginDir, "*.dll").ToList();
            Serilog.Log.Debug("Found {Count} plugin DLL(s) in {PluginDir}", dlls.Count, pluginDir);

            foreach (var dll in dlls)
            {
                try
                {
                    var assembly = AssemblyLoadContext.Default.LoadFromAssemblyPath(dll);
                    var providers = assembly.GetTypes()
                        .Where(t => typeof(IQuestionSource).IsAssignableFrom(t)
                            && t.IsClass
                            && !t.IsAbstract
                            && t.GetConstructor(Type.EmptyTypes) != null);

                    foreach (var type in providers)
                    {
                        Serilog.Log.Information("Registering question plugin: {PluginType} from {Assembly}",
                            type.FullName, assembly.GetName().Name);
                        services.AddTransient(typeof(IQuestionSource), type);
                    }
                }
                catch (Exception ex)
                {
                    Serilog.Log.Error(ex, "Failed to load question plugin: {DllPath}", dll);
                    AnsiConsole.MarkupLine($"[[red]]Failed to load plugin: {dll}[[/]]");
                    AnsiConsole.MarkupLine($"[[dim]]{ex.Message}[[/]]");
                }
            }

            Serilog.Log.Information("Plugin loading complete. Registered {Count} DLL(s).", dlls.Count);
            return services;
        }
    }
}
```

```csharp
/// <summary>
/// Fluent builder that aggregates questions from multiple sources into a final list.
/// </summary>
/// <example><![[CDATA[[var qs = builder.GetCategories().GetFileConfig().Build();]]]]></example>
public sealed class QuestionCombineBuilder(
    IQuestionSource questionProvider,
    IQuestionCatalogLoader questionLoader,
    IQuestionBuilder questionBuilder,
    IOptions<RagnarConfig> options,
    Serilog.ILogger logger) : IQuestionSourceBuilder
{
    private List<Question> Questions { get; } = [[]];

    /// <summary>Loads enabled questions for configured categories into the builder.</summary>
    /// <remarks>Appends to the internal question list; order is preserved.</remarks>
    /// <example><![[CDATA[[builder.GetCategories();]]]]></example>
    /// <returns>The builder instance for chaining.</returns>
    public IQuestionSourceBuilder GetCategories()
    {
        var categories = options.Value.ApplicationOptions.CategoriesToProcess.ToHashSet();
        logger.Debug("Loading questions for categories: {Categories}", string.Join(", ", categories));

        var loaded = questionLoader.LoadQuestions(true, categories);
        Questions.AddRange(loaded);
        logger.Information("GetCategories loaded {Count} question(s). Total accumulated: {Total}", loaded.Count(), Questions.Count);
        return this;
    }

    /// <summary>Loads questions defined in file-based configuration entries.</summary>
    /// <remarks>Uses FileConfigLoader to read per-file question settings.</remarks>
    /// <example><![[CDATA[[builder.GetFileConfig();]]]]></example>
    /// <returns>The builder instance for chaining.</returns>
    public IQuestionSourceBuilder GetFileConfig()
    {
        logger.Debug("Loading questions from file-based configuration.");
        var fcl = new FileConfigLoader();
        var configs = fcl.LoadQuestions().ToList();

        foreach (var config in configs)
        {
            var item = questionBuilder
                .WithText(config.Text)
                .WithFileName(config.Filename)
                .SetCategory(config.Category)
                .SetActive(config.IsEnabled);
            Questions.Add(item.Build());
        }

        logger.Information("GetFileConfig loaded {Count} question(s). Total accumulated: {Total}", configs.Count, Questions.Count);
        return this;
    }

    /// <summary>Asynchronously loads questions from CSV files in the plugin directory.</summary>
    /// <param name="pluginDir">Directory containing *.csv question files.</param>
    /// <param name="cancellationToken">Token to cancel the async load.</param>
    /// <remarks>Skips silently if the directory does not exist.</remarks>
    /// <example><![[CDATA[[await builder.GetCsvFilesAsync(dir, ct);]]]]></example>
    /// <returns>The builder instance for chaining.</returns>
    public async Task<IQuestionSourceBuilder> GetCsvFilesAsync(string pluginDir, CancellationToken cancellationToken)
    {
        if (!Directory.Exists(pluginDir))
        {
            logger.Debug("CSV plugin directory not found, skipping: {Dir}", pluginDir);
            return this;
        }

        logger.Debug("Loading questions from CSV files in: {Dir}", pluginDir);
        var count = 0;

        foreach (var csvFile in Directory.EnumerateFiles(pluginDir, "*.csv", SearchOption.AllDirectories))
        {
            cancellationToken.ThrowIfCancellationRequested();

            if (csvFile.Contains("Summary.csv", StringComparison.OrdinalIgnoreCase))
            {
                logger.Debug("Skipping summary file: {File}", csvFile);
                continue;
            }

            var csvConfigs = await questionProvider.LoadQuestionsAsync(csvFile, cancellationToken).ConfigureAwait(false);
            foreach (var csv in csvConfigs)
            {
                questionBuilder.WithText(csv.Text).WithFileName(csv.Filename).SetCategory(csv.Category).SetActive(csv.IsEnabled);
                Questions.Add(questionBuilder.Build());
                count++;
            }
        }

        logger.Information("GetCsvFilesAsync loaded {Count} question(s) from {Dir}. Total accumulated: {Total}", count, pluginDir, Questions.Count);
        return this;
    }

    /// <summary>Asynchronously loads questions from a single CSV file.</summary>
    /// <param name="csvFile">Path to the *.csv file containing question definitions.</param>
    /// <param name="cancellationToken">Token to cancel the async load.</param>
    /// <returns>The builder instance for chaining.</returns>
    /// <example><![[CDATA[[await builder.GetCsvFileAsync("data.csv", ct);]]]]></example>
    public async Task<IQuestionSourceBuilder> GetCsvFileAsync(string csvFile, CancellationToken cancellationToken)
    {
        if (!File.Exists(csvFile))
        {
            logger.Debug("CSV file not found, skipping: {File}", csvFile);
            return this;
        }

        logger.Debug("Loading questions from CSV file: {File}", csvFile);
        var csvConfigs = await questionProvider.LoadQuestionsAsync(csvFile, cancellationToken).ConfigureAwait(false);

        foreach (var csv in csvConfigs)
        {
            questionBuilder.WithText(csv.Text).WithFileName(csv.Filename).SetCategory(csv.Category).SetActive(csv.IsEnabled);
            Questions.Add(questionBuilder.Build());
        }

        logger.Information("GetCsvFileAsync loaded {Count} question(s) from {File}. Total accumulated: {Total}", csvConfigs.Count(), csvFile, Questions.Count);
        return this;
    }

    /// <summary>Returns the final, sorted, enabled question collection.</summary>
    /// <remarks>Orders by category then filename; filters disabled entries.</remarks>
    /// <example><![[CDATA[[var list = builder.Build();]]]]></example>
    /// <returns>Read-only list of enabled questions.</returns>
    public IReadOnlyList<Question> Build()
    {
        logger.Debug("Building final question list from {Total} accumulated question(s).", Questions.Count);

        var question = Questions
            .Where(q => q.IsEnabled)
            .OrderBy(q => q.Category.ToString())
            .ThenBy(q => q.Filename)
            .ToList()
            .AsReadOnly();

        if (question.Count == 0)
        {
            logger.Error("QuestionCombineBuilder produced zero questions. Check categories, file paths, and CSV sources.");
            throw new InvalidOperationException(
                "QuestionCombineBuilder produced zero questions. " +
                "Check categories, file paths, and CSV sources.");
        }

        logger.Information("Build complete: {Enabled} enabled question(s) from {Total} total.", question.Count, Questions.Count);
        return question;
    }

    /// <summary>Filters the question list to a specific set of categories.</summary>
    /// <remarks>Clears prior entries and replaces with the filtered set.</remarks>
    /// <example><![[CDATA[[builder.WithCategoryFilter();]]]]></example>
    /// <returns>The builder instance for chaining.</returns>
    public IQuestionSourceBuilder WithCategoryFilter()
    {
        var categories = options.Value.ApplicationOptions.CategoriesToProcess;

        if (categories.Length == 0)
        {
            logger.Debug("No category filter configured; keeping all questions.");
            return this;
        }

        var hash = categories.ToHashSet();
        var filtered = Questions.Where(q => hash.Contains(q.Category)).ToList();
        Questions.Clear();
        Questions.AddRange(filtered);
        logger.Information("WithCategoryFilter applied: kept {Kept} of {Original} question(s).", filtered.Count, Questions.Count + (Questions.Count - filtered.Count));
        return this;
    }

    /// <summary>Removes all accumulated questions and reports whether any remain.</summary>
    /// <returns>True if the list is still non-empty after clearing; false if empty.</returns>
    public bool Clear()
    {
        var hadItems = Questions.Count > 0;
        Questions.Clear();
        logger.Debug("Clear() invoked. Had {Count} item(s) before clear.", hadItems);
        return hadItems;
    }

    /// <summary>Adds a single question to the internal accumulation list.</summary>
    /// <param name="question">The question to add.</param>
    public void AddQuestion(Question question)
    {
        Questions.Add(question);
        logger.Debug("AddQuestion called. Total accumulated: {Total}", Questions.Count);
    }
}
```

```csharp
/// <summary>Selects and delegates file parsing to the appropriate syntax parser.</summary>
/// <example><![[CDATA[[var docs = await factory.ParseAsync("app.cs", ct);]]]]></example>
public class CodeParserFactory(Serilog.ILogger logger, ISyntaxTreeChunker cSharpFile) : IFileParseFactory
{
    private readonly ParseCSharpFile _codeParser = new(logger, cSharpFile);

    /// <summary>Parses file using correct parser based on extension.</summary>
    /// <param name="file">File path.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <returns>Parsed documents.</returns>
    /// <example><![[CDATA[[var docs = await factory.ParseAsync("Program.cs", ct);]]]]></example>
    public async Task<IEnumerable<CodeDocument>> ParseAsync(string file, CancellationToken cancellationToken)
    {
        logger.Debug("CodeParserFactory.ParseAsync: dispatching {File}", file);
        var fileInfo = new FileInfo(file);

        var docs = fileInfo.Extension switch
        {
            ".cs" => await GetCodeDocumentsAsync(file, cancellationToken).ConfigureAwait(false),
            _ => await ParseFile(file, cancellationToken).ConfigureAwait(false)
        };

        var count = docs.Count();
        logger.Information("CodeParserFactory.ParseAsync: parsed {File} into {Count} document(s).", file, count);
        return docs;
    }

    /// <param name="file">File path to parse.</param>

        logger.Debug("CodeParserFactory.ParseFile (generic): {File}", file);
```

```csharp
/// <summary>Selects appropriate question filter based on input criteria.</summary>
/// <example><![[CDATA[[var f = QuestionFilterSelector.FindFilter(QuestionCategory.XML, 100);]]]]></example>
public static class QuestionFilterSelector
{
    private static readonly ImmutableDictionary<QuestionCategory, IFilterStrategy> _strategies =
        ImmutableDictionary<QuestionCategory, IFilterStrategy>.Empty
            .Add(QuestionCategory.XML, new XmlCommentFilterStrategy());

    /// <summary>Selects appropriate filter strategy for a question category and optional size.</summary>
    /// <param name="category">The question category (e.g., XML).</param>
    /// <param name="commentLengthThreshold">Optional comment length threshold.</param>
    /// <returns>A configured <see cref="Filter"/> instance.</returns>
    /// <example><![[CDATA[[var filter = QuestionFilterSelector.FindFilter(QuestionCategory.XML, 100);]]]]></example>
    public static Filter FindFilter(QuestionCategory category, int commentLengthThreshold = 0)
    {
        Serilog.Log.Debug("QuestionFilterSelector.FindFilter: category={Category}, threshold={Threshold}", category, commentLengthThreshold);

        if (!_strategies.TryGetValue(category, out var strategy))
        {
            Serilog.Log.Debug("No strategy found for {Category}; returning empty filter.", category);
            return new Filter();
        }

        var filter = strategy.CreateFilter(commentLengthThreshold);
        Serilog.Log.Debug("Selected filter strategy {Strategy} for {Category}.", strategy.GetType().Name, category);
        return filter;
    }
}
```

```csharp
            cnt++;
        }

        AnsiConsole.Write(table);
        logger.Debug("ShowFileStage: rendered table with {Count} row(s).", files.Count);
    }

    /// <summary>Scrolls all points in the vector store and collects their file names.</summary>
    /// <param name="client">The IQdrantClient used for the scroll query.</param>
    /// <param name="pageSize">Page size for scroll pagination.</param>
    /// <param name="cancellationToken">Token to cancel the scroll.</param>
    /// <returns>A list of file name strings found in the collection.</returns>
    /// <example><![[CDATA[[var files = await stage.EnumerateStoredFiles(client);]]]]></example>
    private async Task<List<string>> EnumerateStoredFiles(IQdrantClient client, uint pageSize, CancellationToken cancellationToken)
    {
        var items = new List<string>();
        var collectionName = options.Value.ApplicationOptions.VectorStoreName;
        PointId? nextOffset = null;
        var pageCount = 0;

        do
        {
            cancellationToken.ThrowIfCancellationRequested();
            var response = await client.ScrollAsync(collectionName, limit: pageSize, offset: nextOffset, cancellationToken: cancellationToken).ConfigureAwait(false);
            pageCount++;

            foreach (var item in response.Result)
            {
                var fileName = item.Payload[["FileName"]].StringValue;
                items.Add(fileName);
            }

            nextOffset = response.NextPageOffset;
        }
        while (nextOffset is not null);

        logger.Debug("EnumerateStoredFiles: completed {Pages} page(s), {Count} file(s) total.", pageCount, items.Count);
        return items;
    }
}
```

```csharp
/// <summary>Orchestrates sequential execution of pipeline stages with retry and progress tracking.</summary>
/// <remarks>Thread-safe stage registration; supports reorder, remove, and per-stage retry policies.</remarks>
/// <example><![[CDATA[[await runner.AddStage(s).ExecuteAsync(ctx, ct);]]]]></example>
public sealed class PipelineRunner<TContext>(Serilog.ILogger logger) : BasePipeline<TContext> where TContext : class
{
    private readonly Lock _gate = new();
    private readonly List<StageEntry<TContext>> _stages = [[]];

    public sealed record StageEntry<TContext>(IPipelineStage<TContext> Stage, IAsyncPolicy? Policy = null);

    /// <summary>Registers a pipeline stage, enforcing unique names.</summary>
    /// <param name="stage">The stage to add to the execution sequence.</param>
    /// <param name="policy">Optional retry policy for the stage.</param>
    /// <returns>The runner instance for fluent chaining.</returns>
    /// <example><![[CDATA[[runner.AddStage(parsingStage);]]]]></example>
    public override IPipeline<TContext> AddStage(IPipelineStage<TContext> stage, IAsyncPolicy? policy = null)
    {
        Guard.Against.Null(stage);
        lock (_gate)
        {
            if (_stages.Any(x => x.Stage.Name == stage.Name))
                throw new ArgumentException($"Stage '{stage.Name}' is already registered.", nameof(stage));
            _stages.Add(new StageEntry<TContext>(stage, policy));
        }

        logger.Debug("PipelineRunner.AddStage: registered '{StageName}' (total: {Total}).", stage.Name, _stages.Count);
        return this;
    }

    public IReadOnlyList<string> StageNames
    {
        get
        {
            lock (_gate)
                return _stages.ConvertAll(s => s.Stage.Name).AsReadOnly();
        }
    }

    /// <summary>Removes all registered stages.</summary>
    /// <returns>The runner instance for fluent chaining.</returns>
    public PipelineRunner<TContext> ClearStages()
    {
        lock (_gate)
            _stages.Clear();
        logger.Debug("PipelineRunner.ClearStages: all stages removed.");
        return this;
    }

    /// <summary>Replaces a stage by name with a new implementation.</summary>
    /// <param name="oldName">Name of the stage to replace.</param>
    /// <param name="newStage">The replacement stage.</param>
    /// <param name="policy">Optional retry policy.</param>
    /// <returns>The runner instance for fluent chaining.</returns>
    public IPipeline<TContext> ReplaceStage(string oldName, IPipelineStage<TContext> newStage, IAsyncPolicy? policy = null)
    {
        logger.Debug("PipelineRunner.ReplaceStage: replacing '{Old}' with '{New}'.", oldName, newStage.Name);
        RemoveStage(oldName);
        return AddStage(newStage, policy);
    }

    /// <summary>Removes a previously registered stage by name.</summary>
    /// <param name="name">Display name of the stage to remove.</param>
    /// <returns>The runner instance for fluent chaining.</returns>
    /// <example><![[CDATA[[runner.RemoveStage("Parsing");]]]]></example>
    public override PipelineRunner<TContext> RemoveStage(string name)
    {
        lock (_gate)
            _stages.RemoveAll(s => s.Stage.Name == name);
        logger.Debug("PipelineRunner.RemoveStage: removed '{Name}'. Remaining: {Total}.", name, _stages.Count);
        return this;
    }

    /// <summary>Moves a registered stage to a new position in the sequence.</summary>
    /// <param name="fromName">Name of the stage to relocate.</param>
    /// <param name="newIndex">Zero-based target index; clamped to valid range.</param>
    /// <returns>The runner instance for fluent chaining.</returns>
    /// <example><![[CDATA[[runner.Reorder("Parsing", 0);]]]]></example>
    public override PipelineRunner<TContext> Reorder(string fromName, int newIndex)
    {
        lock (_gate)
        {
            var entry = _stages.FirstOrDefault(s => s.Stage.Name == fromName)
                ?? throw new KeyNotFoundException(fromName);
            _stages.Remove(entry);
            _stages.Insert(Math.Clamp(newIndex, 0, _stages.Count), entry);
        }

        logger.Debug("PipelineRunner.Reorder: moved '{Name}' to index {Index}.", fromName, Math.Clamp(newIndex, 0, _stages.Count));
        return this;
    }

    /// <summary>Executes all registered stages sequentially with retry and logging.</summary>
    /// <param name="context">Shared embedding context passed through each stage.</param>
    /// <param name="ct">Token to cancel pipeline execution.</param>
    /// <returns>The enriched context after all stages complete.</returns>
    /// <example><![[CDATA[[await runner.ExecuteAsync(ctx, ct);]]]]></example>
    public override async Task<TContext> ExecuteAsync(TContext context, CancellationToken ct)
    {
        var sw = Stopwatch.StartNew();
        List<StageEntry<TContext>> snapshot;
        lock (_gate)
            snapshot = [[.._stages]];

        logger.Information("Pipeline starting with {Total} stage(s).", snapshot.Count);

        for (var i = 0; i < snapshot.Count; i++)
        {
            var stage = snapshot[[i]].Stage;
            if (!stage.ShouldRun(context))
            {
                logger.Information("Skipping stage [[{Index}]]: {Name}", i, stage.Name);
                continue;
            }

            logger.Information("Starting stage [[{Index}/{Total}]]: {Name}", i + 1, snapshot.Count, stage.Name);
            var stageSw = Stopwatch.StartNew();
            try
            {
                if (snapshot[[i]].Policy is null)
                {
                    await stage.ExecuteAsync(context, ct).ConfigureAwait(false);
                }
                else
                {
                    await snapshot[[i]].Policy.ExecuteAsync(
                        async () => await stage.ExecuteAsync(context, ct).ConfigureAwait(false)
                    ).ConfigureAwait(false);
                }
            }
            catch (OperationCanceledException) when (ct.IsCancellationRequested)
            {
                logger.Warning("Pipeline cancelled at stage [[{Index}]]: {Name}", i, stage.Name);
                throw;
            }
            catch (Exception ex)
            {
                logger.Fatal(ex, "Stage [[{Index}]] {Name} failed after {Elapsed}", i, stage.Name, stageSw.Elapsed);
                throw new PipelineStageException(stage.Name, ex);
            }

            logger.Information("Completed stage [[{Index}]] {Name} in {Time}", i, stage.Name, stageSw.FormatElapsedTime());
        }

        logger.Information("Pipeline finished in {Total}", sw.FormatElapsedTime());
        return context;
    }
}
```

```csharp
/// <summary>Fluent builder for constructing Qdrant filter expressions.</summary>
/// <example><![[CDATA[[var f = new FilterBuilder().AddRange("len", 50).Build();]]]]></example>
public sealed class FilterBuilder
{
    private readonly List<Condition> _must = [[]];
    private readonly List<Condition> _mustNot = [[]];
    private static readonly Serilog.ILogger _log = Serilog.Log;

    /// <summary>Adds a keyword match-exclusion condition.</summary>
    /// <param name="field">The payload field to match.</param>
    /// <param name="value">The keyword value to exclude.</param>
    /// <returns>The builder for chaining.</returns>
    /// <example><![[CDATA[[builder.AddExclusion("category", "skip");]]]]></example>
    public FilterBuilder AddExclusion(string field, string value)
    {
        _mustNot.Add(Conditions.MatchKeyword(field, value));
        _log.Debug("FilterBuilder.AddExclusion: {Field} != '{Value}'. mustNot count: {Count}", field, value, _mustNot.Count);
        return this;
    }

    /// <summary>Adds a keyword match-inclusion condition to the positive filter set.</summary>
    /// <param name="field">The payload field to match against.</param>
    /// <param name="value">The keyword value that must be present.</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.AddMatch("Category", "XML");]]]]></example>
    public FilterBuilder AddMatch(string field, string value)
    {
        _must.Add(Conditions.MatchKeyword(field, value));
        _log.Debug("FilterBuilder.AddMatch: {Field} = '{Value}'. must count: {Count}", field, value, _must.Count);
        return this;
    }

    /// <summary>Adds a range condition with lower and/or upper bounds.</summary>
    /// <param name="field">Payload field name.</param>
    /// <param name="gte">Greater-than-or-equal bound (optional).</param>
    /// <param name="lte">Less-than-or-equal bound (optional).</param>
    /// <returns>The builder for chaining.</returns>
    /// <example><![[CDATA[[builder.AddRange("length", 100);]]]]></example>
    public FilterBuilder AddRange(string field, double? gte = null, double? lte = null)
    {
        if (gte is null && lte is null)
            throw new ArgumentException("At least one of gte or lte must be provided.");

        var range = new Range();
        if (gte is not null) range.Gte = gte.Value;
        if (lte is not null) range.Lte = lte.Value;
        _must.Add(new Condition { Field = new FieldCondition { Key = field, Range = range } });

        _log.Debug("FilterBuilder.AddRange: {Field} [[{Gte}, {Lte}]]. must count: {Count}", field, gte?.ToString() ?? "null", lte?.ToString() ?? "null", _must.Count);
        return this;
    }

    /// <summary>Adds an except-keywords condition excluding specified values from a field.</summary>
    /// <param name="field">The payload field containing keywords.</param>
    /// <param name="keywords">Keyword values to exclude from matches.</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.AddExceptKeywords("Category", "Test", "Tests");]]]]></example>
    public FilterBuilder AddExceptKeywords(string field, params string[[]] keywords)
    {
        _must.Add(new Condition { Field = new FieldCondition { Key = field, Match = new Match { ExceptKeywords = new RepeatedStrings { Strings = { keywords } } } } });
        _log.Debug("FilterBuilder.AddExceptKeywords: {Field} NOT IN [[{Keywords}]]. must count: {Count}", field, string.Join(", ", keywords), _must.Count);
        return this;
    }

    /// <summary>Excludes a range of values (must-not).</summary>
    /// <param name="field">Payload field name.</param>
    /// <param name="gte">Greater-than-or-equal bound (optional).</param>
    /// <param name="lte">Less-than-or-equal bound (optional).</param>
    /// <returns>The builder for chaining.</returns>
    public FilterBuilder AddExclusionRange(string field, double? gte = null, double? lte = null)
    {
        if (gte is null && lte is null)
            throw new ArgumentException("At least one of gte or lte must be provided.");

        var range = new Range();
        if (gte is not null) range.Gte = gte.Value;
        if (lte is not null) range.Lte = lte.Value;
        _mustNot.Add(new Condition { Field = new FieldCondition { Key = field, Range = range } });

        _log.Debug("FilterBuilder.AddExclusionRange: {Field} NOT IN [[{Gte}, {Lte}]]. mustNot count: {Count}", field, gte?.ToString() ?? "null", lte?.ToString() ?? "null", _mustNot.Count);
        return this;
    }

    public bool IsEmpty => _must.Count == 0 && _mustNot.Count == 0;

    /// <summary>Builds and returns the final Qdrant <see cref="Filter"/> instance.</summary>
    /// <returns>A composed filter with must and mustNot conditions.</returns>
    /// <example><![[CDATA[[Filter f = builder.Build();]]]]></example>
    public Filter Build()
    {
        if (IsEmpty)
        {
            _log.Warning("FilterBuilder.Build(): called with no conditions.");
            throw new InvalidOperationException("FilterBuilder is empty. Add at least one condition before calling Build().");
        }

        var filter = new Filter
        {
            Must = { _must },
            MustNot = { _mustNot }
        };

        _log.Information("FilterBuilder.Build(): compiled filter with {Must} must and {MustNot} mustNot condition(s).", _must.Count, _mustNot.Count);
        return filter;
    }
}
```

```csharp
    /// <param name="filter">Filter options.</param>
    /// <param name="cancellationToken">Cancellation cancellationToken.</param>
    /// <returns>Async sequence of file paths.</returns>
    /// <example><![[CDATA[[await foreach(var f in FileDiscoveryService.GetFilesAsync(opt, ".", new())){...}]]]]></example>
    /// <exception cref="DirectoryNotFoundException">Thrown when provided directory path is not found.</exception>
    /// <exception cref="ArgumentNullException">Thrown when <paramref name="filter"/> is null.</exception>
    /// <exception cref="ArgumentException">Thrown when no file options are provided.</exception>
    public IAsyncEnumerable<string> GetFilesAsync(string directory, FileLoadOptions filter, CancellationToken cancellationToken)
    {
        Guard.Against.Null(filter);
        Guard.Against.NullOrEmpty(directory);

        if (!Directory.Exists(directory))
        {
            _log.Error("Directory not found: {Directory}", directory);
            throw new DirectoryNotFoundException($"Directory not found: {directory}");
        }

        _log.Debug("FileDiscoveryService.GetFilesAsync: scanning {Directory} with {ExtCount} allowed extension(s).",
            directory, filter.AllowedExtensions.Count);
        return GetValuesAsync(directory, filter, cancellationToken);
    }

    /// <summary>Yields filtered file paths asynchronously with cancellation support.</summary>
    /// <param name="directory">Root search directory.</param>
    /// <param name="filter">File-load filter options (extensions, exclusions).</param>
    /// <param name="cancellationToken">Token to abort enumeration.</param>
    /// <returns>Async sequence of matching file paths.</returns>
    /// <example><![[CDATA[[await foreach (var f in GetValuesAsync(".", opts, ct)) { }]]]]></example>
    private async IAsyncEnumerable<string> GetValuesAsync(string directory, FileLoadOptions filter, [[EnumeratorCancellation]] CancellationToken cancellationToken = default)
    {
        var count = 0;
        foreach (var path in ListFiles(directory, filter))
        {
            cancellationToken.ThrowIfCancellationRequested();
            count++;
            yield return path;
        }

        _log.Information("FileDiscoveryService.GetValuesAsync: yielded {Count} file(s) from {Directory}.", count, directory);
    }

    /// <summary>Recursively lists files, skipping excluded directories and disallowed extensions.</summary>
    /// <param name="rootPath">Directory to enumerate recursively.</param>
    /// <param name="filter">Filter containing allowed extensions and excluded dirs.</param>
    /// <returns>Enumerable of full file paths passing all filters.</returns>
    /// <example><![[CDATA[[foreach (var f in ListFiles(".", opts)) { }]]]]></example>
    private IEnumerable<string> ListFiles(string rootPath, FileLoadOptions filter)
    {
        var allowedExtensions = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        foreach (var ext in filter.AllowedExtensions)
            allowedExtensions.Add(ext);

        var skippedDirectories = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
        foreach (var dir in filter.ExcludedDirectories)
            skippedDirectories.Add(dir);

        _log.Debug("FileDiscoveryService.ListFiles: root={Root}, allowed=[[{Extensions}]], excludedDirs=[[{Dirs}]]",
            rootPath,
            string.Join(", ", allowedExtensions),
            string.Join(", ", skippedDirectories));

        var results = Directory.EnumerateFiles(rootPath, "*.*", SearchOption.AllDirectories)
            .Where(filePath =>
            {
                var dirName = Path.GetDirectoryName(filePath);
                if (dirName != null)
                {
                    var segments = dirName.Split(Path.DirectorySeparatorChar, Path.AltDirectorySeparatorChar);
                    if (segments.Any(seg => skippedDirectories.Contains(seg)))
                        return false;
                }

                var ext = Path.GetExtension(filePath);
                return allowedExtensions.Contains(ext);
            })
            .ToList();

        _log.Debug("FileDiscoveryService.ListFiles: {Count} file(s) matched in {Root}.", results.Count, rootPath);
        return results;
    }
}
```

```csharp
            writer.Write(new Rule());

            logger.Information("Processing question [[{QuestionId}]] from user [[{UserId}]] with {ContextLength} chars",
                question.Text, Environment.UserName, contextText.Length.ToString("N0"));

            await ragOrchestrator.ExecuteAsync(question, contextText, cancellationToken).ConfigureAwait(false);
            processedCount++;

            writer.MarkupLine($"{processedCount} of {total}", Styles.Cyan);
            logger.Debug("Question {Count}/{Total} completed.", processedCount, total);
        }

        logger.Information("QuestionExecutionStage: all {Total} question(s) processed.", total);
    }
}
```
