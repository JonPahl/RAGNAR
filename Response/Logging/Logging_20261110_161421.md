{
    /// <summary>Initializes validation rules for file-loading options.</summary>
    /// <example><![[CDATA[[var v = new FileLoadOptionsValidator();]]]]></example>
    public FileLoadOptionsValidator()
    {
        logger.LogDebug("Initializing FileLoadOptionsValidator rules.");

        RuleFor(x => x.AllowedExtensions)
            .NotEmpty().WithMessage("At least one allowed extension is required.")
            .Must(exts => exts.All(e => !string.IsNullOrWhiteSpace(e) && e.Length > 1 && e[[0]] == '.'))
            .WithMessage("Every entry in AllowedExtensions must start with '.' and be longer than 1 char (e.g. \".cs\").");

        RuleFor(x => x.ExcludedFiles)
            .Must(files => files.All(f => !string.IsNullOrWhiteSpace(f)))
            .WithMessage("ExcludedFiles entries must be non-empty file names.");

        RuleFor(x => x.ExcludedDirectories)
            .Must(dirs => dirs.All(d => !string.IsNullOrWhiteSpace(d) && !d.Contains('\\', StringComparison.InvariantCulture) && !d.Contains('/', StringComparison.InvariantCulture)))

        logger.LogDebug("FileLoadOptionsValidator rules registered: AllowedExtensions, ExcludedFiles, ExcludedDirectories.");

{
    /// <summary>Positive (must-match) filter conditions.</summary>
    private readonly List<Condition> _must = [[]];
    /// <summary>Negative (must-not-match) filter conditions.</summary>
    private readonly List<Condition> _mustNot = [[]];
    private readonly List<Condition> _should = [[]];
    private readonly List<Condition> _shouldMust = [[]];

    private static void GuardField(string field) => Guard.Against.NullOrWhiteSpace(field, nameof(field));

    /// <summary>Adds a keyword match-exclusion condition.</summary>
    /// <param name="field">The payload field to match.</param>
    /// <param name="value">The keyword value to exclude.</param>
    /// <returns>The builder for chaining.</returns>
    /// <example><![[CDATA[[builder.AddExclusion("category", "skip");]]]]></example>
    public QdrantFilterBuilder AddExclusion(string field, string value)
    {
        GuardField(field);
        Guard.Against.Null(value, nameof(value));
        _mustNot.Add(Conditions.MatchKeyword(field, value));
        logger.LogDebug("AddExclusion: field={Field}, value={Value}", field, value);
        return this;
    }

    /// <summary>Adds a keyword match-inclusion condition to the positive filter set.</summary>
    /// <param name="field">The payload field to match against.</param>
    /// <param name="value">The keyword value that must be present.</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.AddMatch("Category", "XML");]]]]></example>
    public QdrantFilterBuilder AddMatch(string field, string value)
    {
        GuardField(field);
        Guard.Against.Null(value, nameof(value));
        _must.Add(Conditions.MatchKeyword(field, value));
        logger.LogDebug("AddMatch: field={Field}, value={Value}", field, value);
        return this;
    }

    /// <summary>Adds a numeric range condition (gte and/or lte) to the positive filter set.</summary>
    /// <param name="field">The payload field name to constrain.</param>
    /// <param name="gte">Greater-than-or-equal lower bound (optional).</param>
    /// <param name="lte">Less-than-or-equal upper bound (optional).</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.AddRange("CommentLength", 50, 500);]]]]></example>
    public QdrantFilterBuilder AddRange(string field, double? gte = null, double? lte = null)
    {
        GuardField(field);
        var range = new Range();
        if (gte is null && lte is null)
            throw new ArgumentException("At least one of gte or lte must be provided.");
        if (gte is not null) range.Gte = gte.Value;
        if (lte is not null) range.Lte = lte.Value;
        _must.Add(new Condition { Field = new FieldCondition { Key = field, Range = range } });
        logger.LogDebug("AddRange: field={Field}, gte={Gte}, lte={Lte}", field, gte, lte);
        return this;
    }

    /// <param name="field">The payload field containing keywords.</param>
    /// <param name="keywords">Keyword values to exclude from matches.</param>
    /// <returns>The builder instance for fluent chaining.</returns>

    public QdrantFilterBuilder AddExceptKeywords(string field, params string[[]] keywords)
    {
        _must.Add(new Condition { Field = new FieldCondition { Key = field, Match = new Match { ExceptKeywords = new RepeatedStrings { Strings = { keywords } } } } });
        logger.LogDebug("AddExceptKeywords: field={Field}, keywords=[[{Keywords}]]", field, string.Join(", ", keywords));
        return this;
    }

            throw new ArgumentException("At least one of gte or lte must be provided.");
        var range = new Range();
        if (gte is not null) range.Gte = gte.Value;
        if (lte is not null) range.Lte = lte.Value;
        _mustNot.Add(new Condition { Field = new FieldCondition { Key = field, Range = range } });
        logger.LogDebug("AddExclusionRange: field={Field}, gte={Gte}, lte={Lte}", field, gte, lte);
        return this;
    }

    public QdrantFilterBuilder AddShouldMatch(string field, string value)

        GuardField(field);
        _should.Add(Conditions.MatchKeyword(field, value));
        logger.LogDebug("AddShouldMatch: field={Field}, value={Value}", field, value);
        return this;

        _must.Add(new Condition { IsNull = new IsNullCondition { Key = field } });
        logger.LogDebug("AddIsNull: field={Field}", field);
        return this;

    /// <param name="field">The payload field to check for non-null.</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    public QdrantFilterBuilder AddIsNotNull(string field)

        GuardField(field);
        _mustNot.Add(new Condition { IsNull = new IsNullCondition { Key = field } });
        logger.LogDebug("AddIsNotNull: field={Field}", field);
        return this;
    }

    /// <summary>Adds an except-keywords condition from an enumerable collection.</summary>
    /// <param name="field">The payload field containing keywords.</param>
    /// <param name="keywords">Keyword values to exclude from matches.</param>
    /// <returns>The builder instance for fluent chaining.</returns>
    public QdrantFilterBuilder AddExceptKeywords(string field, IEnumerable<string> keywords)
    {
        GuardField(field);
        var list = keywords as IReadOnlyCollection<string> ?? keywords.ToList();
        if (list.Count == 0)
        {
            logger.LogDebug("AddExceptKeywords (IEnumerable): empty keyword list for field={Field}, skipping.", field);
            return this;
        }
        _must.Add(new Condition { Field = new FieldCondition { Key = field, Match = new Match { ExceptKeywords = new RepeatedStrings { Strings = { list } } } } });
        logger.LogDebug("AddExceptKeywords (IEnumerable): field={Field}, count={Count}", field, list.Count);
        return this;
    }

    /// <summary>Removes all accumulated must, mustNot, and should filter conditions.</summary>
    /// <returns>The builder instance for fluent chaining.</returns>
    /// <example><![[CDATA[[builder.Clear();]]]]></example>
    public QdrantFilterBuilder Clear()
    {
        var mustCount = _must.Count;
        var mustNotCount = _mustNot.Count;
        var shouldCount = _should.Count;
        _must.Clear();
        _mustNot.Clear();
        _should.Clear();
        logger.LogDebug("Clear: removed must={Must}, mustNot={MustNot}, should={Should}", mustCount, mustNotCount, shouldCount);
        return this;
    }

    /// <summary>Indicates whether no filter conditions have been added yet.</summary>
    /// <example><![[CDATA[[bool empty = builder.IsEmpty;]]]]></example>

    /// <summary>Builds and returns the final Qdrant <see cref="Filter"/> instance.</summary>
    /// <returns>A composed filter with must and mustNot conditions.</returns>
    /// <example><![[CDATA[[Filter f = builder.Build();]]]]></example>

        return new Filter
        {
            Must = { _must },
            MustNot = { _mustNot },
            Should = { _should }
        };
    }

    /// <summary>Non-throwing variant; returns <c>false</c> if the builder is empty.</summary>
    public bool TryBuild(out Filter filter)
    {
        if (IsEmpty)
        {
            logger.LogWarning("TryBuild: builder is empty, returning false.");
            filter = null!;
            return false;
        }

        filter = Build();
        logger.LogDebug("TryBuild: succeeded with must={Must}, mustNot={MustNot}, should={Should}", _must.Count, _mustNot.Count, _should.Count);
        return true;
    }

    /// <summary>Returns a diagnostic string describing the builder state.</summary>

    {
        var result = $"QdrantFilterBuilder[[must:{_must.Count} mustNot:{_mustNot.Count} should:{_should.Count}]]";
        logger.LogTrace("ToString: {State}", result);
        return result;
    }

/// <summary>Middleware that writes stage progress indicators to the output writer.</summary>
/// <param name="writer">The output writer to render progress to.</param>
/// <param name="logger">Optional Serilog logger for diagnostics.</param>
public sealed class ProgressMiddleware(IOutputWriter writer, ILogger? logger = null)
{
    /// <summary>Writes a progress label before and a completion/cancellation indicator after the stage runs.</summary>
    /// <param name="context">The stage execution context.</param>
    /// <param name="next">The next middleware link or core stage delegate.</param>
    /// <param name="cancellationToken">Token to abort.</param>
    /// <returns>A task that completes when the wrapped stage finishes.</returns>
    public async Task Invoke<TContext>(StageExecutionContext<TContext> context, Func<Task> next, CancellationToken cancellationToken)
        where TContext : class
    {
        var label = $"[[{context.StageIndex + 1}/{context.StageTotal}]] {context.Stage.Name}";
        logger?.LogDebug("ProgressMiddleware: starting stage '{Label}'", label);
        writer.WriteLine($"{label} … ");
        var sw = Stopwatch.StartNew();
        try
        {
            await next().ConfigureAwait(false);
            sw.Stop();
            logger?.LogDebug("ProgressMiddleware: stage '{Label}' completed in {Elapsed}", label, sw.Elapsed);
            writer.WriteLine($"✓ {sw.Elapsed:hh\\:mm\\:ss}");
        }
        catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
        {
            sw.Stop();
            logger?.LogWarning("ProgressMiddleware: stage '{Label}' cancelled after {Elapsed}", label, sw.Elapsed);
            writer.WriteLine($"✗ cancelled after {sw.Elapsed:hh\\:mm\\:ss}");
            throw;
        }
        catch (Exception ex)
        {
            sw.Stop();
            logger?.LogError(ex, "ProgressMiddleware: stage '{Label}' failed after {Elapsed}", label, sw.Elapsed);
            writer.WriteLine($"✗ {ex.GetType().Name} after {sw.Elapsed:hh\\:mm\\:ss}");
            throw;
        }

        writer.WriteLine();
    }
}

            category, filter.Must?.Count ?? 0, filter.MustNot?.Count ?? 0);

    {
        get
        {

## DeltaEmbedStage.cs (enhanced)

```csharp
/// <summary>
/// Wraps the standard embed-and-upsert flow, skipping files whose SHA-256 fingerprint
/// matches the last successful embed.
/// </summary>
/// <param name="logger">Serilog logger.</param>
/// <param name="fingerprintStore">Fingerprint persistence backend.</param>
/// <param name="writer">The existing Qdrant upsert writer.</param>
/// <param name="options">Application configuration (vector store name, etc.).</param>
/// <example><![[CDATA[[runner.AddStage(new DeltaEmbedStage(log, store, writer, opts));]]]]></example>
public sealed class DeltaEmbedStage(ILogger logger, IFileFingerprintStore fingerprintStore, IVectorStoreWriter writer, IOptions<RagnarConfig> options) : IPipelineStage<EmbeddingContext>
{
    /// <inheritdoc/>
    /// <returns>The string "Delta Embed".</returns>
    public string Name => "Delta Embed";

    /// <inheritdoc/>
    /// <returns>A Polly retry policy (3 attempts, 2 s base delay).</returns>
    public AsyncRetryPolicy? RetryPolicy => Polly.Policy
        .Handle<HttpRequestException>().Or<TimeoutException>()
        .WaitAndRetryAsync(
            [[TimeSpan.FromSeconds(2), TimeSpan.FromSeconds(5), TimeSpan.FromSeconds(10)]],
            (attempt, ts, ex, _) => logger.LogWarning(ex, "Delta embed attempt {Attempt} failed; retrying in {Delay}", attempt, ts));

    /// <inheritdoc/>
    public bool ShouldRun(EmbeddingContext context)
    {
        var result = context.Documents is { Count: > 0 };
        logger.LogDebug("DeltaEmbedStage.ShouldRun: Documents={Count} → ShouldRun={Run}", context.Documents?.Count ?? 0, result);
        return result;
    }

    /// <summary>Executes incremental upsert, skipping unchanged files.</summary>
    /// <param name="context">Shared embedding context carrying the parsed documents.</param>
    /// <param name="cancellationToken">Token to abort.</param>
    /// <returns>Completed task once all eligible documents are upserted.</returns>
    /// <example><![[CDATA[[await stage.ExecuteAsync(ctx, ct);]]]]></example>
    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        Guard.Against.Null(context.Documents);
        logger.LogInformation("DeltaEmbedStage.ExecuteAsync: starting with {Total} documents.", context.Documents.Count);

        await fingerprintStore.InitializeAsync(cancellationToken).ConfigureAwait(false);

        var eligible = new List<CodeDocument>();
        var skipped = 0;
        foreach (var doc in context.Documents)
        {
            cancellationToken.ThrowIfCancellationRequested();
            var hash = ComputeSha256(doc.Code);
            if (await fingerprintStore.IsUnchangedAsync(doc.FilePath, hash, cancellationToken).ConfigureAwait(false))
            {
                skipped++;
                logger.LogTrace("DeltaEmbedStage: skipping unchanged file '{File}'", doc.FilePath);
                continue;
            }

            eligible.Add(doc);
            logger.LogDebug("DeltaEmbedStage: file changed, marking eligible: '{File}'", doc.FilePath);
            await fingerprintStore.RecordAsync(doc.FilePath, hash, cancellationToken).ConfigureAwait(false);
        }

        logger.LogInformation("Delta embed: {Total} docs, {Eligible} changed, {Skipped} unchanged", context.Documents.Count, eligible.Count, skipped);

        if (eligible.Count > 0)
        {
            await fingerprintStore.FlushAsync(cancellationToken).ConfigureAwait(false);
            logger.LogDebug("DeltaEmbedStage: flushing fingerprints and upserting {Count} vectors.", eligible.Count);
            var result = await writer.UpsertBatchAsync(eligible, cancellationToken).ConfigureAwait(false);
            logger.LogInformation("Delta embed upsert: {Status}, {Upserted} vectors", result.Status, result.CalculateSize);
        }
        else
        {
            logger.LogInformation("Delta embed: no changes detected, skipping upsert.");
        }
    }

    private static string ComputeSha256(string content)
    {
        var bytes = SHA256.HashData(Encoding.UTF8.GetBytes(content));
        return Convert.ToHexString(bytes).ToLowerInvariant();
    }
}

/// Static class to call Custom system enumerable.

public class FileDiscoveryService(ILogger logger, IFileEligibilityValidator validator) : IFileDiscoveryService
{
    /// <summary>
    /// Yields filtered file paths asynchronously.
    /// </summary>
    /// <param name="directory">Search root.</param>
    /// <param name="cancellationToken">Cancellation cancellationToken.</param>
    /// <returns>Async sequence of file paths.</returns>
    /// <example><![[CDATA[[await foreach(var f in FileDiscoveryService.GetFilesAsync(opt, ".", new())){...}]]]]>
    /// </example>
    /// <exception cref="DirectoryNotFoundException">Thrown when provided directory path is not found.</exception>
    /// <exception cref="ArgumentNullException">Thrown when <paramref name="filter"/> is null.</exception>
    /// <exception cref="ArgumentException">Thrown when no file options are provided.</exception>
    public async IAsyncEnumerable<string> GetFilesAsync(string directory, IOptions<RagnarConfig> options, [[EnumeratorCancellation]] CancellationToken cancellationToken)
    {
        var filter = options.Value.FileLoadOptions;
        Guard.Against.Null(filter);
        Guard.Against.NullOrEmpty(directory);

        logger.LogInformation("Discovering files in {Directory} with {AllowedExtensions} allowed extensions",
            directory, string.Join(", ", filter.AllowedExtensions));

        if (!Directory.Exists(directory))
        {
            logger.LogError("FileDiscoveryService: directory '{Directory}' not found.", directory);
            throw new DirectoryNotFoundException(directory);
        }

        var discoveredCount = 0;
        var skippedCount = 0;
        var sw = Stopwatch.StartNew();

        foreach (var path in Directory.EnumerateFiles(directory, "*.*", SearchOption.AllDirectories))
        {
            cancellationToken.ThrowIfCancellationRequested();
            var fi = new FileInfo(path);
            if (!validator.IsValid(fi, filter))
            {
                skippedCount++;
                logger.LogTrace("FileDiscoveryService: skipped '{Path}'", path);
                continue;
            }

            discoveredCount++;
            logger.LogDebug("Eligible file: {Path}", path);
            yield return path;
            await Task.Yield();
        }

        sw.Stop();
        logger.LogInformation("File discovery complete in {Directory}: {Discovered} eligible, {Skipped} excluded ({Elapsed})",
            directory, discoveredCount, skippedCount, sw.Elapsed);
    }
}

## JsonFileFingerprintStore.cs (enhanced)

```csharp
/// <summary>
/// A JSON-file-backed fingerprint store that survives across runs.
/// </summary>
/// <param name="logger">Serilog logger for diagnostics.</param>
/// <param name="option">Application options.</param>
/// <example><![[CDATA[[var store = new JsonFileFingerprintStore(log, Path.Combine(root, ".ragnar"));]]]]></example>
public sealed class JsonFileFingerprintStore(ILogger logger, IOptions<RagnarConfig> option) : IFileFingerprintStore
{
    private readonly Lock _gate = new();
    private Dictionary<string, string> _fingerprints = [[]];
    private string _filePath = string.Empty;
    private int _dirtyCount;
    private const int _flushThreshold = 256;
    private readonly JsonSerializerOptions _jsonOption = new() { WriteIndented = true };

    /// <summary>Loads or creates the fingerprint file from disk.</summary>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <example><![[CDATA[[await store.InitializeAsync(ct);]]]]></example>
    public async Task InitializeAsync(CancellationToken cancellationToken)
    {
        var basePath = option.Value.ApplicationOptions.SourceDirectory;
        Directory.CreateDirectory(basePath);
        _filePath = Path.Combine(basePath, "fingerprints.json");

        if (File.Exists(_filePath))
        {
            var json = await File.ReadAllTextAsync(_filePath, cancellationToken).ConfigureAwait(false);
            _fingerprints = JsonSerializer.Deserialize<Dictionary<string, string>>(json) ?? [[]];
            logger.LogInformation("Loaded {Count} existing fingerprints from {Path}", _fingerprints.Count, _filePath);
        }
        else
        {
            logger.LogInformation("No existing fingerprint file at {Path}; starting fresh", _filePath);
        }
    }

    /// <summary>Checks if a file's fingerprint has changed.</summary>
    /// <param name="filePath">The path to the file.</param>
    /// <param name="sha256">The expected SHA-256 hash of the file.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    public async Task<bool> IsUnchangedAsync(string filePath, string sha256, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        bool result;
        lock (_gate)
            result = _fingerprints.TryGetValue(filePath, out var stored) && stored == sha256;
        logger.LogTrace("IsUnchangedAsync: '{File}' → {Unchanged}", filePath, result);
        await Task.Yield();
        return result;
    }

    /// <summary>Records a file's fingerprint.</summary>
    /// <param name="filePath">The path to the file.</param>
    /// <param name="sha256">The SHA-256 hash to store.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    public async Task RecordAsync(string filePath, string sha256, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        bool shouldFlush;
        lock (_gate)
        {
            _fingerprints[[filePath]] = sha256;
            shouldFlush = ++_dirtyCount >= _flushThreshold;
            if (shouldFlush) _dirtyCount = 0;
        }

        logger.LogTrace("RecordAsync: recorded fingerprint for '{File}' (dirty={Dirty})", filePath, _dirtyCount);

        if (shouldFlush)
            await PersistAsync(cancellationToken).ConfigureAwait(false);
    }

    /// <summary>Flushes any pending fingerprints to disk.</summary>
    /// <param name="cancellationToken">Cancellation token.</param>
    public async Task FlushAsync(CancellationToken cancellationToken)
    {
        bool needed;
        lock (_gate)
        {
            needed = _dirtyCount > 0;
            _dirtyCount = 0;
        }

        if (needed)
        {
            logger.LogDebug("FlushAsync: flushing pending fingerprints to '{Path}'.", _filePath);
            await PersistAsync(cancellationToken).ConfigureAwait(false);
        }
        else
        {
            logger.LogTrace("FlushAsync: no pending changes to flush.");
        }
    }

    /// <summary>Removes a file's fingerprint.</summary>
    /// <param name="filePath">The path to the file.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    public async Task RemoveAsync(string filePath, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        bool removed;
        lock (_gate)
            removed = _fingerprints.Remove(filePath);
        logger.LogDebug("RemoveAsync: removed fingerprint for '{File}' → {Removed}", filePath, removed);
        await PersistAsync(cancellationToken).ConfigureAwait(false);
    }

    /// <summary>Returns the total number of stored fingerprints.</summary>
    public int Count
    {
        get
        {
            lock (_gate)
                return _fingerprints.Count;
        }
    }

    private async Task PersistAsync(CancellationToken cancellationToken)
    {
        string snapshot;
        lock (_gate)
            snapshot = JsonSerializer.Serialize(_fingerprints, _jsonOption);
        logger.LogTrace("PersistAsync: writing {Length} bytes to '{Path}'.", snapshot.Length, _filePath);
        await File.WriteAllTextAsync(_filePath, snapshot, cancellationToken).ConfigureAwait(false);
    }
}

## PipelineRunner.cs (enhanced with additional debug logging)

```csharp
/// <summary>Orchestrates sequential execution of pipeline stages with retry and progress tracking.</summary>
/// <param name="logger">The logger used to record pipeline and stage execution events.</param>
/// <remarks>Thread-safe stage registration; supports reorder, remove, and per-stage retry policies.</remarks>
/// <example><![[CDATA[[await runner.AddStage(s).ExecuteAsync(ctx, ct);]]]]></example>
public sealed class PipelineRunner<TContext>(Microsoft.Extensions.Logging.ILogger logger) : BasePipeline<TContext> where TContext : class
{
    private readonly List<Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task>> _middlewares = [[]];

    /// <summary>Registers middleware to wrap the execution of each pipeline stage.</summary>
    /// <param name="middleware">The delegate that runs before and/or after the next middleware or stage.</param>
    /// <returns>This runner for fluent chaining.</returns>
    public override IPipeline<TContext> UseMiddleware(Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task> middleware)
    {
        Guard.Against.Null(middleware);
        lock (_gate)
            _middlewares.Add(middleware);
        logger.LogDebug("UseMiddleware: registered middleware (total={Count}).", _middlewares.Count);
        return this;
    }

    /// <summary>Registers timing middleware to measure stage execution duration.</summary>
    /// <param name="middleware">The timing middleware instance to apply to each stage.</param>
    /// <returns>This runner for fluent chaining.</returns>
    /// <example><![[CDATA[[runner.UseTiming(timingMiddleware);]]]]></example>
    public IPipeline<TContext> UseTiming(StageTimingMiddleware middleware)
    {
        logger.LogDebug("UseTiming: registering StageTimingMiddleware.");
        return UseMiddleware((ctx, next, ct) => middleware.Invoke(ctx, next, ct));
    }

    /// <summary>Executes a single stage with all registered middleware applied.</summary>
    /// <param name="ctx">The stage execution context for the current stage.</param>
    /// <param name="coreExecution">The innermost delegate that runs the stage logic.</param>
    /// <param name="ct">Cancellation token propagated through the middleware chain.</param>
    /// <returns>A task that completes when all middleware and the stage finish.</returns>
    /// <example><![[CDATA[[await runner.ExecuteStageWithMiddlewareAsync(ctx, core, ct);]]]]></example>
    private async Task ExecuteStageWithMiddlewareAsync(StageExecutionContext<TContext> ctx, Func<Task> coreExecution, CancellationToken ct)
    {
        List<Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task>> snapshot;
        lock (_gate)
            snapshot = [[.._middlewares]];

        logger.LogDebug("ExecuteStageWithMiddlewareAsync: applying {Count} middleware layers to stage '{Stage}'.", snapshot.Count, ctx.Stage.Name);

        var pipeline = coreExecution;
        for (var i = snapshot.Count - 1; i >= 0; i--)
        {
            var mw = snapshot[[i]];
            var nextRef = pipeline;
            pipeline = () => mw(ctx, nextRef, ct);
        }

        await pipeline().ConfigureAwait(false);
    }

    private readonly Lock _gate = new();
    private readonly List<StageEntry<TContext>> _stages = [[]];

    /// <summary>Registers a pipeline stage, enforcing unique names.</summary>
    /// <param name="stage">The stage to add to the execution sequence.</param>
    /// <param name="policy">Optional asynchronous policy for retrying the stage when execution fails.</param>
    /// <returns>The runner instance for fluent chaining.</returns>
    /// <example><![[CDATA[[runner.AddStage(parsingStage);]]]]></example>
    public override IPipeline<TContext> AddStage(IPipelineStage<TContext> stage, IAsyncPolicy? policy = null)
    {
        Guard.Against.Null(stage);
        lock (_gate)
        {
            if (_stages.Any(x => x.PipelineStage.Name == stage.Name))
            {
                logger.LogError("AddStage: stage '{Name}' is already registered.", stage.Name);
                throw new ArgumentException($"Stage '{stage.Name}' is already registered.", nameof(stage));
            }
            _stages.Add(new StageEntry<TContext>(stage, policy));
        }
        logger.LogDebug("AddStage: registered '{Name}' (position={Index}, total={Count}).", stage.Name, _stages.Count - 1, _stages.Count);
        return this;
    }

    /// <summary>Gets the ordered list of registered pipeline stage display names.</summary>
    /// <returns>A read-only list of registered stage display names in order.</returns>
    /// <example><![[CDATA[[var names = runner.StageNames;]]]]></example>
    public IReadOnlyList<string> StageNames
    {
        get
        {
            lock (_gate)
            {
                var names = _stages.ConvertAll(s => s.PipelineStage.Name).AsReadOnly();
                logger.LogTrace("StageNames: [[{Names}]]", string.Join(" → ", names));
                return names;
            }
        }
    }

    /// <summary>Removes all registered pipeline stages and returns the runner for chaining.</summary>
    /// <returns>The runner instance with an empty stage list.</returns>
    /// <example><![[CDATA[[runner.ClearStages();]]]]></example>
    public IPipeline<TContext> ClearStages()
    {
        int count;
        lock (_gate)
        {
            count = _stages.Count;
            _stages.Clear();
        }
        logger.LogDebug("ClearStages: removed {Count} stages.", count);
        return this;
    }

    /// <summary>Replaces an existing stage (matched by name) with a new stage and optional policy.</summary>
    /// <param name="oldName">Display name of the stage to replace.</param>
    /// <param name="newStage">The replacement stage instance.</param>
    /// <param name="policy">Optional retry policy for the
