public sealed class DeltaEmbedStage(ILogger logger, IFileFingerprintStore fingerprintStore, 
    IVectorStoreWriter writer) : IPipelineStage<EmbeddingContext>
{
    public string Name => "Delta Embed";
    public AsyncRetryPolicy? RetryPolicy => Polly.Policy.Handle<HttpRequestException>()
        .Or<TimeoutException>().WaitAndRetryAsync([[TimeSpan.FromSeconds(2), TimeSpan.FromSeconds(5), TimeSpan.FromSeconds(10)]]);

    public bool ShouldRun(EmbeddingContext context) => context.Documents is { Count: > 0 };

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        Guard.Against.Null(context.Documents);
        await fingerprintStore.InitializeAsync(cancellationToken).ConfigureAwait(false);
        var eligible = new List<CodeDocument>();
        var skipped = 0;

        foreach (var doc in context.Documents)
        {
            cancellationToken.ThrowIfCancellationRequested();
            var hash = ComputeSha256(doc.Code);
            if (await fingerprintStore.IsUnchangedAsync(doc.FilePath, hash, cancellationToken).ConfigureAwait(false))
            { skipped++; continue; }
            
            eligible.Add(doc);
            await fingerprintStore.RecordAsync(doc.FilePath, hash, cancellationToken).ConfigureAwait(false);
        }

        logger.LogInformation("Delta embed: {Total} docs, {Eligible} changed, {Skipped} unchanged", 
            context.Documents.Count, eligible.Count, skipped);
        
        if (eligible.Count > 0)
        {
            await fingerprintStore.FlushAsync(cancellationToken).ConfigureAwait(false);
            var result = await writer.UpsertBatchAsync(eligible, cancellationToken).ConfigureAwait(false);
            logger.LogInformation("Delta embed upsert: {Status}, {Upserted} vectors", 
                result.Status, result.CalculateSize());
        }
    }

    private static string ComputeSha256(string content)
    {
        var bytes = SHA256.HashData(Encoding.UTF8.GetBytes(content));
        return Convert.ToHexString(bytes).ToLowerInvariant();
    }
}

public sealed class UpsertStage(IVectorStoreWriter repository, ILogger logger, 
    IOptions<RagnarConfig> options, IOutputWriter writer) : IPipelineStage<EmbeddingContext>
{
    private readonly int _batchSize = options.Value.EmbeddingOptions.BatchSize;

    public string Name => "Embedding & upserting…";

        .Or<TimeoutException>().WaitAndRetryAsync(3, a => TimeSpan.FromSeconds(Math.Pow(2, a)));

    public bool ShouldRun(EmbeddingContext context)
    {
        Guard.Against.Null(context);
        return context.Documents.Any();
    }

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        var documents = context.Documents;
        if (documents.Count == 0)
        {

            return;
        }

        UpdateResult lastResult = new() { Status = UpdateStatus.Completed };
        await AnsiConsole.Progress().AutoClear(true).StartAsync(async ctx =>
        {

                lastResult = await repository.UpsertBatchAsync(batch, cancellationToken).ConfigureAwait(false);
                if (lastResult.Status != UpdateStatus.Completed)
                    logger.LogError("Upsert failed for batch: {Status}", lastResult.Status);
                task.Increment(batch.Length);
                ctx.Refresh();
            }
        }).ConfigureAwait(false);

        context.UpsertResult = lastResult;
        writer.MarkupLine(lastResult.Status == UpdateStatus.Completed 
            ? "[[green]]Upsert complete.[[/]]" : "[[red]]Upsert finished with errors.[[/]]");
    }

// RetryMiddleware.cs
public sealed class RetryMiddleware(ILogger logger)
{
    public async Task Invoke<TContext>(StageExecutionContext<TContext> context, Func<Task> next, 
        CancellationToken cancellationToken) where TContext : class
    {
        cancellationToken.ThrowIfCancellationRequested();
        ArgumentNullException.ThrowIfNull(context);
        ArgumentNullException.ThrowIfNull(next);

        var policy = context.Stage.RetryPolicy;
        if (policy is null) await next().ConfigureAwait(false);

        var attempt = 0;
        await policy.ExecuteAsync(async _ =>
        {
            attempt++;
            logger.LogDebug("Stage {Name} – retry attempt {Attempt}", 
                context.Stage.Name, attempt, General.Culture);
            await next().ConfigureAwait(false);
        }, cancellationToken).ConfigureAwait(false);

        context.Metric = context.Metric is not null 
            ? (context.Metric with { RetryCount = attempt })
            : new StageMetric { StageName = context.Stage.Name, RetryCount = attempt };
    }
}

public sealed class VectorStoreWriter(ILogger logger, IEmbeddingService embeddingService, 
    IQdrantClient qdrantClient, IQdrantPointFactory generatorService, IOptions<RagnarConfig> config) 
    : IVectorStoreWriter
{
    private readonly ApplicationOptions _appOptions = config.Value.ApplicationOptions;

    public async Task<UpdateResult> UpsertBatchAsync(IEnumerable<CodeDocument> codeDocuments, 
        CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        var docs = codeDocuments as List<CodeDocument> ?? [[..codeDocuments]];
        var batchSize = config?.Value?.EmbeddingOptions?.BatchSize ?? 32;
        var failed = new List<(int Index, Exception Ex)>();
        var succeeded = 0;

        foreach (var batch in docs.Chunk(batchSize))
        {
            try
            {
                var texts = batch.Select(d => $"Context: {d.ElementName}\nCode:\n{d.Code}").ToList();
                var generated = await embeddingService.GenerateBatchAsync(texts, cancellationToken)
                    .ConfigureAwait(false);
                
                var points = batch.Select((d, idx) => generatorService.BuildPointStructs(
                    d.AsPoint(), generated[[idx]].Vector.ToArray(), d)).SelectMany(p => p).ToList();
                
                await qdrantClient.UpsertAsync(_appOptions.VectorStoreName, points, 
                    cancellationToken: cancellationToken).ConfigureAwait(false);
                
                succeeded += batch.Length;
            }
            catch (OperationCanceledException) { throw; }
            catch (Exception ex)
            {
                logger.LogError(ex, "Batch upsert failed.", ex.Message);
                failed.AddRange(batch.Select((d, idx) => (succeeded + idx, ex)));
            }
        }

        if (failed.Count > 0)
            logger.LogError("Upsert incomplete: {Succeeded} ok, {Failed} failed. First: {Msg}", 
                succeeded, failed.Count, failed[[0]].Ex.Message);

        return new UpdateResult { Status = failed.Count == 0 ? UpdateStatus.Completed : UpdateStatus.UnknownUpdateStatus };
    }

    private string? _responseDir;

    public async Task<string> WriteResponseAsync(ResponseRecord record, CancellationToken cancellationToken)
    {
        var sourceDir = Path.GetDirectoryName(record?.Question?.Filename) ?? Directory.GetCurrentDirectory();
        _responseDir ??= Directory.CreateDirectory(Path.Join(sourceDir, "Response")).FullName;
        
        var fileName = $"{Path.GetFileNameWithoutExtension(record?.Question?.Filename)}_summary.md";
        var summaryPath = Path.Join(_responseDir, fileName);
        var content = string.Format(_template, record?.Answer, DateTime.UtcNow.ToString("O"));

        return summaryPath;
    }

// ResponseWriter.cs
public sealed class ResponseWriter(IOutputFormatter formatter, IPathResolver pathResolver, 
    IWriter fileWriter, IClock clock) : IResponseWriter
{
    public async Task<string> WriteResponseAsync(ResponseRecord record, CancellationToken cancellationToken)
    {
        var directory = pathResolver.ResolveResponseDirectory(record.Question.Category);
        pathResolver.EnsureDirectoryExists(directory);
        
        var stamp = clock.UtcNow.ToString("yyyyMMdd_HHmmss", General.Culture);
        var name = record.Question.Filename.Trim('"');
        var fileName = $"{name}_{stamp}.{formatter.FileExtension}";
        var fullPath = Path.Join(directory, fileName);
        
        if (!IsValidFilePath(fullPath))
            fullPath = string.Format(@"C:\Test\{0}.{1}", stamp, formatter.FileExtension);

        var content = formatter.FormatResponse(record);
        await fileWriter.WriteAsync(fullPath, content, cancellationToken).ConfigureAwait(false);
        return fullPath;
    }

    private static bool IsValidFilePath(string path)
    {
        if (string.IsNullOrWhiteSpace(path)) return false;
        try { Path.GetFullPath(path); return true; }
        catch when (ExceptionUtility.IsPathInvalid()) { return false; }
    }
}

// PathExtensions.cs
public static class PathExtensions
{
    /// <summary>Expands env vars in path and validates directory existence.</summary>
    /// <param name="path">Config-based folder path potentially containing env vars.</param>
    /// <returns>Fully expanded absolute directory path string.</returns>
    /// <exception cref="DirectoryNotFoundException">Resolved path does not exist.</exception>
    public static string ExpandDirectory(this string path)
    {
        Guard.Against.NullOrWhiteSpace(path);
        var expanded = Environment.ExpandEnvironmentVariables(path);
        var fullPath = System.IO.Path.GetFullPath(expanded);
        
        if (!Directory.Exists(fullPath))
            throw new DirectoryNotFoundException($"Resolved directory does not exist: '{fullPath}'. Original config value: '{path}'");
            
        return fullPath;
    }
}

// PipelineBuilder.cs
public sealed class PipelineBuilder<TContext>(ILogger logger) where TContext : class
{
    private readonly List<StageEntry<TContext>> _stages = [[]];
    private readonly List<Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task>> _middlewares = [[]];
    private readonly Lock _gate = new();

    /// <summary>Adds a stage to the pipeline with an optional retry policy.</summary>
    /// <param name="stage">The stage instance to register.</param>
    /// <param name="policy">Optional retry policy wrapping execution.</param>
    /// <returns>The current builder for fluent chaining.</returns>
    public PipelineBuilder<TContext> AddStage(IPipelineStage<TContext> stage, IAsyncPolicy? policy = null)
    {
        Guard.Against.Null(stage);
        lock (_gate)
        {
            if (_stages.Any(s => s.Stage.Name == stage.Name))
                throw new ArgumentException($"Stage '{stage.Name}' already registered.");
            _stages.Add(new StageEntry<TContext>(stage, policy));
        }
        return this;
    }

    /// <summary>Removes the stage with the supplied display name.</summary>
    /// <param name="name">Stage name to remove.</param>
    /// <returns>The current builder for fluent chaining.</returns>
    public PipelineBuilder<TContext> RemoveStage(string name)
    {
        lock (_gate) _stages.RemoveAll(s => s.Stage.Name == name);
        return this;
    }

    /// <summary>Moves a registered stage to a different execution index.</summary>
    /// <param name="name">Stage name to reorder.</param>
    /// <param name="newIndex">Zero-based target position.</param>
    /// <returns>The current builder for fluent chaining.</returns>
    public PipelineBuilder<TContext> Reorder(string name, int newIndex)
    {
        lock (_gate)
        {
            var e = _stages.FirstOrDefault(s => s.Stage.Name == name) ?? throw new KeyNotFoundException(name);
            _stages.Remove(e);
            _stages.Insert(Math.Clamp(newIndex, 0, _stages.Count), e);
        }
        return this;
    }

    /// <summary>Clears all registered stages from the builder.</summary>
    /// <returns>The current builder for fluent chaining.</returns>
    public PipelineBuilder<TContext> ClearStages() { lock (_gate) _stages.Clear(); return this; }

    /// <summary>Registers middleware to wrap each stage execution.</summary>
    /// <param name="mw">The middleware delegate to run around execution.</param>
    /// <returns>The current builder for fluent chaining.</returns>
    public PipelineBuilder<TContext> UseMiddleware(Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task> mw)
    {
        Guard.Against.Null(mw);
        lock (_gate) _middlewares.Add(mw);
        return this;
    }

    /// <summary>Builds the concrete pipeline instance from configuration.</summary>
    /// <returns>A pipeline ready to execute.</returns>
    public Pipeline<TContext> Build() => new(logger, _stages, _middlewares);

    /// <summary>Gets ordered list of registered stage names.</summary>
    public IReadOnlyList<string> StageNames { get => lock (_gate) _stages.ConvertAll(s => s.Stage.Name).AsReadOnly(); }
}
