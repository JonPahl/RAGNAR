/// <summary>
/// Defines async CRUD operations for a single aggregate entity.
/// </summary>
/// <typeparam name="T">The entity type persisted by this repository.</typeparam>
public interface IRepository<T> where T : class
{
    /// <summary>
    /// Gets the unique identifier of the entity.
    /// </summary>
    Guid EntityId { get; }

    /// <summary>
    /// Retrieves a single entity by its primary key.
    /// </summary>
    /// <param name="id">The unique identifier of the entity.</param>
    /// <param name="cancellationToken">Token to observe for cancellation.</param>
    /// <returns>A value task containing the entity, or <c>null</c>.</returns>
    /// <exception cref="ArgumentException">Thrown when <paramref name="id"/> is <see cref="Guid.Empty"/>.</exception>
    /// <example>
    /// <code>var user = await repo.FindByIdAsync(userId, ct);</code>
    /// </example>
    ValueTask<T?> FindByIdAsync(Guid id, CancellationToken cancellationToken = default);

    /// <summary>
    /// Executes a filtered projection query against the data store.
    /// </summary>
    /// <param name="filter">Lambda expression defining the filter predicate.</param>
    /// <param name="cancellationToken">Token to observe for cancellation.</param>
    /// <returns>A value task containing the filtered result set.</returns>
    /// <exception cref="ArgumentNullException">Thrown when <paramref name="filter"/> is <c>null</c>.</exception>
    /// <example>
    /// <code>var active = await repo.QueryAsync(e => e.IsActive, ct);</code>
    /// </example>
    ValueTask<IReadOnlyList<T>> QueryAsync(
        in Func<T, bool> filter,
        CancellationToken cancellationToken = default);

    /// <summary>
    /// Inserts a new entity or updates an existing one by key.
    /// </summary>
    /// <param name="entity">The entity instance to persist.</param>
    /// <param name="cancellationToken">Token to observe for cancellation.</param>
    /// <returns>A value task that signals completion of the upsert.</returns>
    /// <exception cref="ArgumentNullException">Thrown when <paramref name="entity"/> is <c>null</c>.</exception>
    ValueTask UpsertAsync(T entity, CancellationToken cancellationToken = default);
}

/// <summary>
/// Coordinates commit or rollback of all tracked repository changes.
/// </summary>
public interface IUnitOfWork : IAsyncDisposable
{
    /// <summary>
    /// Persists all pending changes to the backing store.
    /// </summary>
    /// <param name="cancellationToken">Token to observe for cancellation.</param>
    /// <returns>A value task that completes when the transaction is committed.</returns>
    /// <exception cref="InvalidOperationException">Thrown when no session is active.</exception>
    /// <exception cref="OperationCanceledException">Thrown when the operation is canceled.</exception>
    ValueTask CommitAsync(CancellationToken cancellationToken = default);

    /// <summary>
    /// Reverts all uncommitted changes for the current session.
    /// </summary>
    /// <param name="cancellationToken">Token to observe for cancellation.</param>
    /// <exception cref="InvalidOperationException">Thrown when no session is active.</exception>
    /// <exception cref="OperationCanceledException">Thrown when the operation is canceled.</exception>
    ValueTask RollbackAsync(CancellationToken cancellationToken = default);
}

/// <summary>
/// Provides in-process, fire-and-forget event dispatching.
/// </summary>
public interface IEventBus
{
    /// <summary>
    /// Publishes a domain event to all registered subscribers.
    /// </summary>
    /// <param name="eventName">The logical name of the event.</param>
    /// <param name="payload">The serialized event data to deliver.</param>
    /// <param name="cancellationToken">Token to observe for cancellation.</param>
    /// <returns>A value task that completes after all handlers finish.</returns>
    /// <exception cref="ArgumentNullException">Thrown when <paramref name="eventName"/> is <c>null</c> or empty.</exception>
    /// <exception cref="OperationCanceledException">Thrown when the operation is canceled.</exception>
    /// <example>
    /// <code>await bus.PublishAsync("OrderPlaced", json, ct);</code>
    /// </example>
    ValueTask PublishAsync(
        string eventName,
        in ReadOnlySpan<byte> payload,
        CancellationToken cancellationToken = default);
}
