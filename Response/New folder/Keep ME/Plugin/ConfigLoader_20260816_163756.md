### 2. Embedded Resource Implementation
Leverages `Assembly.GetManifestResourceStream` with proper naming conventions:
```csharp
public class EmbeddedResourceQuestionLoader : IQuestionConfigLoader
{
    private readonly string _resourceName;
    
    public EmbeddedResourceQuestionLoader(string resourceName) => _resourceName = resourceName;

    public async Task<IList<Question>> LoadAsync(CancellationToken ct = default)
    {
        var assembly = Assembly.GetExecutingAssembly();
        using var stream = assembly.GetManifestResourceStream(_resourceName);
        
        if (stream == null)
            throw new InvalidOperationException($"Embedded resource '{_resourceName}' not found. Ensure Build Action is 'Embedded Resource'.");

        using var reader = new StreamReader(stream, ct);
        var json = await reader.ReadToEndAsync(ct);
        
        return JsonSerializer.Deserialize<IList<Question>>(json, new JsonSerializerOptions { PropertyNameCaseInsensitive = true })!;
    }
}
```
> 💡 **Note:** Embedded resources must be named `<DefaultNamespace>.<FolderStructure>.<FileName>.json` (e.g., `MyApp.Config.questions.json`).

### 3. Azure App Configuration Implementation
Uses Microsoft's official provider and reads key-value pairs:
```csharp
public class AzureAppConfigQuestionLoader : IQuestionConfigLoader
{
    private readonly IConfiguration _config;
    private readonly string _keyPrefix;

    public AzureAppConfigQuestionLoader(IConfiguration config, string keyPrefix = "Questions:") => (_config, _keyPrefix) = (config, keyPrefix);

    public async Task<IList<Question>> LoadAsync(CancellationToken ct = default)
    {
        var questions = new List<Question>();
        
        // Reads all keys matching the prefix (e.g., Questions:Refactor:01, Questions:Logging:02)
        foreach (var kv in _config.GetSection(_keyPrefix).GetChildren())
        {
            if (!string.IsNullOrEmpty(kv.Value))
                questions.Add(JsonSerializer.Deserialize<Question>(kv.Value!)!);
        }

        return questions;
    }
}
```

public class QuestionConfigLoaderAdapter : IQuestionConfigLoader
{
    private readonly IQuestionConfigLoader _inner;
    
    public QuestionConfigLoaderAdapter(IQuestionConfigLoader inner) => _inner = inner;

    public async Task<IList<Question>> LoadAsync(CancellationToken ct = default)
    {
        var raw = await _inner.LoadAsync(ct);
        
        return raw.Select(q => 
            q.IsEnabled 
                ? Question.IsActive(q.Text, q.Key, q.Category) 
                : Question.IsDisabled(q.Text, q.Key, q.Category)).ToList();
    }
}
