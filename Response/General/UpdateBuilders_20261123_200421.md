using System;
using System.Collections.Generic;

namespace Ragnar.Domain.Models;

/// <summary>Fluent builder for constructing <see cref="CodeDocument"/> instances.</summary>
public sealed class CodeDocumentBuilder
{
    /// <summary>Immutable snapshot of accumulated build state.</summary>
    private readonly RecordSnapshot _state = new();

    /// <summary>Sets the file name on the document.</summary>
    public CodeDocumentBuilder FileName(in ReadOnlySpan<char> path)
    {
        _state.FileName = path.ToString();
        return this;
    }

    /// <summary>Sets the element type and name on the document.</summary>
    public CodeDocumentBuilder Element(string type, in string name)
    {
        _state.ElementType = type;
        _state.ElementName = name.Trim();
        return this;
    }

    /// <summary>Sets the source code text on the document.</summary>
    public CodeDocumentBuilder Code(in ReadOnlyMemory<char> text)
    {
        _state.Code = text.ToString();
        return this;
    }

    /// <summary>Sets the comment text and auto-computes its length.</summary>
    public CodeDocumentBuilder Comment(in string text)
    {
        _state.Comment = text;
        _state.CommentLength = text.Length;
        return this;
    }

    /// <summary>Finalizes and returns the constructed <see cref="CodeDocument"/>.</summary>
    public CodeDocument Build() => new()
    {
        FileName = _state.FileName,
        FilePath = _state.FilePath,
        ElementType = _state.ElementType,
        ElementName = _state.ElementName,
        Comment = _state.Comment,
        CommentLength = _state.CommentLength,
        Code = _state.Code ?? string.Empty,
    };

    /// <summary>Internal snapshot record to avoid repeated allocations during fluent chaining.</summary>
    private record RecordSnapshot(
        string FileName = default!,
        string FilePath = default!,
        string ElementType = default!,
        string ElementName = string.Empty,
        string Comment = string.Empty,
        int CommentLength = 0,
        string? Code = null);
}

using Qdrant.Client;
using System;
using System.Collections.Generic;
using System.Threading;
using VectorParams = Qdrant.Client.Models.VectorParams;
using PayloadSchemaType = Qdrant.Client.PayloadSchemaType;

namespace Ragnar.Infrastructure.VectorStores;

/// <summary>Builds and manages a Qdrant vector store collection with payload indices.</summary>
public sealed class VectorStoreBuilder : IVectorStoreBuilder
{
    private readonly ILogger _logger;
    private readonly IQdrantClient _client;
    private bool _isExisting;

    public string VectorStoreName { get; init; }
    public ulong Dimension { get; init; }

    public VectorStoreBuilder(ILogger logger, IOptions<RagnarConfig> options, IQdrantClient client)
    {
        _logger = Guard.Against.Null(logger);
        _client = Guard.Against.Null(client);
        
        var appOpts = options.Value.ApplicationOptions ?? throw new ArgumentNullException(nameof(options));
        VectorStoreName = Guard.Against.NullOrEmpty(appOpts.VectorStoreName);
        Dimension = options.Value.EmbeddingOptions.Dimension;
    }

    /// <summary>Ensures the collection exists, creating it if absent.</summary>
    public async Task<IVectorStoreBuilder> EnsureExistsAsync(CancellationToken ct)
    {
        _isExisting = await _client.CollectionExistsAsync(VectorStoreName, ct).ConfigureAwait(false);
        if (!_isExisting)
            await CreateAsync(ct).ConfigureAwait(false);

        return this;
    }

    /// <summary>Creates the Qdrant collection with cosine distance metric.</summary>
    public async Task<IVectorStoreBuilder> CreateAsync(CancellationToken cancellationToken)
    {
        await _client.CreateCollectionAsync(VectorStoreName, new VectorParams { Size = Dimension, Distance = Distance.Cosine }, cancellationToken).ConfigureAwait(false);
        _isExisting = true;
        return this;
    }

    /// <summary>Creates a payload index on a specified collection field.</summary>
    public async Task<IVectorStoreBuilder> MakeIndexAsync(string fieldName, PayloadSchemaType schemaType, CancellationToken ct)
    {
        await _client.CreatePayloadIndexAsync(VectorStoreName, fieldName, schemaType, ct).ConfigureAwait(false);
        return this;
    }

    /// <summary>Creates multiple payload indices in parallel-ready fashion.</summary>
    public async Task<IVectorStoreBuilder> MakeIndexesAsync(IEnumerable<(string Field, PayloadSchemaType Schema)> indexes, CancellationToken ct)
    {
        if (indexes is null) return this;

        await foreach (var (field, schema) in indexes.ToAsyncEnumerable(ct))
            await MakeIndexAsync(field, schema, ct).ConfigureAwait(false);

        return this;
    }

    /// <summary>Builds or validates the collection and applies indexes.</summary>
    public async Task<bool> BuildAsync(IEnumerable<(string Field, PayloadSchemaType Schema)>? indexes = null, CancellationToken cancellationToken = default)
    {
        var exists = await _client.CollectionExistsAsync(VectorStoreName, cancellationToken).ConfigureAwait(false);
        if (!exists)
        {
            await CreateAsync(cancellationToken).ConfigureAwait(false);
            _logger.LogInformation("Collection '{VectorStoreName}' created.", VectorStoreName);
        }

        if (indexes is not null)
            await MakeIndexesAsync(indexes, cancellationToken).ConfigureAwait(false);

        return exists;
    }

    public async Task<CollectionInfo> DescribeAsync(CancellationToken ct)
    {
        var info = await _client.GetCollectionInfoAsync(VectorStoreName, ct).ConfigureAwait(false);
        return new CollectionInfo
        {
            Name = VectorStoreName,
            Dimension = (int)Dimension,
            VectorsCount = (int)info.PointsCount,
            Status = info.Status.ToString(),
        };
    }

    public Task<IVectorStoreBuilder> ExistsAsync(CancellationToken cancellationToken) => throw new NotImplementedException();
}
