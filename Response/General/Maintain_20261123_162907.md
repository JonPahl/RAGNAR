*Replaces fragmented/static extensions with a single, self-contained type extension that adds members directly to the target type.*

```csharp
public static class StringExtensions
{
    /// <summary>
    /// C# 14 Extension Members: Adds utility methods directly to string without inheritance or wrapper types.
    /// </summary>
    extension(string value)
    {
        /// <summary>Counts non-tag, non-comment characters in an XML-style comment.</summary>
        public int CharacterCount()
        {
            var count = 0;
            var inTag = false;
            ReadOnlySpan<char> span = value;
            for (var i = 0; i < span.Length; i++)
            {
                if (span[[i]] == '<') inTag = true;
                else if (span[[i]] == '>') inTag = false;
                else if (!inTag && span[[i]] != '/') count++;
            }
            return count;
        }

        /// <summary>Extracts the final directory segment from a file path.</summary>
        public string FolderName => Path.GetFileName(value.TrimEnd('/', '\\'));

        /// <summary>Checks exclusion membership using 'in' to avoid stack copy for spans/collections.</summary>
        public bool IsExcluded(in IReadOnlyCollection<string> exclusions) 
            => exclusions.Contains(value, StringComparer.OrdinalIgnoreCase);
    }
}
```

```csharp
public static class MetricsMiddlewareExtensions
{
    /// <summary>
    /// Measures stage execution time and attaches metrics using C# 14 lambda modifiers.
    /// 'in' signals intent that context should not be mutated during timing capture.
    /// </summary>
    public static async Task InvokeAsync<TContext>(
        this Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task> middleware,
        in StageExecutionContext<TContext> ctx, 
        Func<Task> next, 
        CancellationToken ct) where TContext : class
    {
        ct.ThrowIfCancellationRequested();
        ArgumentNullException.ThrowIfNull(ctx);
        ArgumentNullException.ThrowIfNull(next);

        var sw = Stopwatch.StartNew();
        bool success;

        try
        {
            await next().ConfigureAwait(false);
            success = true;
        }
        catch (Exception ex) when (ct.IsCancellationRequested == false)
        {
            success = false;
            throw new PipelineStageException(ctx.Stage.Name, ex);
        }
        finally
        {
            sw.Stop();
            // C# 14 extension member for clean metric assignment without boilerplate
            ctx.RecordTiming(sw.Elapsed, success);
        }
    }
}
```

#### ✅ Example 3: Fixed Async Enumerator & Path Resolution
*Removes synchronous blocking calls from async contexts. Centralizes magic strings into configuration-driven constants. Uses `out` for safe directory creation.*

```csharp
public sealed class FileDiscoveryService(ILogger logger) : IFileDiscoveryService
{
    public async IAsyncEnumerable<string> EnumerateFilesAsync(
        string directory, 
        IOptions<RagnarConfig> options, 
        [[EnumeratorCancellation]] CancellationToken cancellationToken)
    {
        Guard.Against.NullOrEmpty(directory);
        var filter = options.Value.FileLoadOptions ?? throw new ArgumentNullException(nameof(options));

        if (!Directory.Exists(directory))
            throw new DirectoryNotFoundException($"Directory not found: {directory}");

        await Task.Yield(); // Establish proper async boundary for enumerator state machine

        foreach (var path in Directory.EnumerateFiles(directory, "*.*", SearchOption.AllDirectories))
        {
            cancellationToken.ThrowIfCancellationRequested();
            var fi = new FileInfo(path);
            
            if (!filter.IsEligible(fi)) continue;

            logger.LogDebug("Eligible: {Path}", path);
            yield return path;
        }
    }
}

public sealed class PathResolver(IOptions<RagnarConfig> options) : IPathResolver
{
    private readonly ApplicationOptions _appOpts = options.Value.ApplicationOptions ?? throw new ArgumentNullException(nameof(options));

    public string ResolveResponseDirectory(QuestionCategory? category)
    {
        var baseDir = category?.ToString() ?? "Uncategorized";
        // Centralized path composition; avoids magic strings
        return Path.Join(_appOpts.SourceDirectory, _appOpts.OutputFolder, baseDir);
    }

    public bool EnsureDirectoryExists(string path, out bool created)
    {
        if (string.IsNullOrWhiteSpace(path)) throw new ArgumentException("Path cannot be null or whitespace.", nameof(path));
        
        created = !Directory.Exists(path);
        if (created) Directory.CreateDirectory(path);
        return created;
    }
}
```

```csharp
public static class RagnarHostBuilder
{
    public static IHostBuilder CreateDefaultBuilder(string[[]] args) =>
        Host.CreateDefaultBuilder(args)
            .ConfigureAppConfiguration(cfg => cfg.AddJsonFile("appsettings.json", optional: true, reloadOnChange: true))
            .UseSerilog((ctx, cfg) => cfg.ReadFrom.Configuration(ctx.Configuration))
            .ConfigureServices((ctx, services) => RegisterInfrastructure(services, ctx))
            .ConfigureServices((ctx, services) => RegisterPipelines(services));

    private static void RegisterPipeline<TContext>(IHostBuilder host, PipelineBuilder<TContext> builder) where TContext : class =>
        host.Services.AddSingleton<Pipeline<TContext>>(sp => 
        {
            var log = sp.GetRequiredService<ILogger>();
            var middlewares = new List<Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task>>
            {
                new ExceptionHandlingMiddleware(log).InvokeAsync<TContext>,
                new StageTimingMiddleware(log).InvokeAsync<TContext>
            };
            
            // C# 14 lambda modifier: 'in' prevents unnecessary struct copies in delegate signatures
            return builder.UseMiddleware((in StageExecutionContext<TContext> ctx, Func<Task> next, CancellationToken ct) => 
                middlewares.Reverse().Aggregate(next, (nextFunc, mw) => 
                    async () => await mw(ctx, nextFunc, ct).ConfigureAwait(false))).Build();
        });
}
```
