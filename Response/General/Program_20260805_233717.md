```csharp
using Ragnar.Core.Factories;
using Ragnar.Core.Filters;
using Ragnar.Core.Validators;

var builder = WebApplication.CreateBuilder(args);

// Register services
builder.Services.AddHttpClient();
builder.Services.Configure<OllamaOptions>(builder.Configuration.GetSection("Ollama"));
builder.Services.Configure<EmbeddingOptions>(builder.Configuration.GetSection("Embedding"));

builder.Services.AddSingleton<IQuestionFactory, DefaultQuestionFactory>();
builder.Services.AddSingleton<IFilterStrategy, XmlCommentLengthFilterStrategy>();
builder.Services.AddSingleton<IFilterStrategy, XmlEmptyCommentFilterStrategy>();

builder.Services.AddSingleton<FilterInputValidator>();

var app = builder.Build();

// Configure pipeline
app.MapGet("/health", () => Results.Ok(new { Status = "Healthy" }));

app.Run();
```

2. **Use partial classes only when necessary**: Avoid splitting single classes across multiple files unless absolutely needed

   - Dependency Inversion: Use interfaces for dependencies

4. **Add documentation**: XML comments where appropriate
5. **Consistent naming**: Follow .NET naming conventions throughout
