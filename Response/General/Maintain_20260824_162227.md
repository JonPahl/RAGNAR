#### 4. Tight Coupling in `ResponseWriter` Path & Formatting
**Problem**: `BuildDirectory` and `FormatFile` are private static methods tightly coupled to hardcoded paths, date formats, and markdown structure. Hard to test or swap formatters.
**Fix**: Extract path strategy and formatting into interfaces. Use DI for flexibility.

```csharp
// ✅ Fixed: ResponseWriter.cs (Decoupled & Configurable)
public interface IPathStrategy { string BuildResponseDirectory(string baseDir, QuestionCategory category); }
public class DefaultPathStrategy : IPathStrategy 
{
    public string BuildResponseDirectory(string baseDir, QuestionCategory category) =>
        Path.Join(baseDir, "Response", category.ToString());
}

public sealed class ResponseWriter(IOptions<RagnarConfig> config, IPathStrategy pathStrategy) : IResponseWriter
{
    private readonly string _dateFormat = config.Value.ApplicationOptions.ResponseDateFormat ?? "yyyyMMdd_HHmmss";

    public async Task<string> WriteResponseAsync(SaveDetails details, CancellationToken ct)
    {
        var baseDir = config.Value.ApplicationOptions.SourceDirectory.ExpandDirectory();
        var dir = pathStrategy.BuildResponseDirectory(baseDir, details.Question.Category);
        Directory.CreateDirectory(dir);

        var fileName = $"{details.Question.Filename}_{DateTime.Now.ToString(_dateFormat)}.md";
        var filePath = Path.Join(dir, fileName);
        
        await File.WriteAllTextAsync(filePath, FormatMarkdown(details), ct);
        return filePath;
    }

    private static string FormatMarkdown(SaveDetails detail) => $"""
        {detail.Question.MarkdownHeader}
        > **Date Generated**: {DateTime.Now:G}
        > ## Question: 
        > {detail.Question.Text}
        > **Method Call Duration**: {detail.Duration}
        
        ## Response: 
        {detail.Response}
        """;
}
```
