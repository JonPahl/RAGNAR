public class RagnarConfigBuilder
{
    private readonly IConfiguration _config;
    public RagnarConfigBuilder(IConfiguration config) => _config = config;

    public RagnarConfig Build()
    {
        var config = new RagnarConfig();
        _config.GetSection("RagnarConfig").Bind(config);
        
        // Cross-section validation
        var results = FluentValidation.Validator.ValidateAsync(new OllamaOptionsValidator(), config.OllamaOptions).Result;
        if (!results.IsValid) throw new ConfigurationException(results.Errors);
        
        return config;
    }
}

// Program.cs / HostBuilder

public class VectorStoreBuilder
{
    private readonly IQdrantClient _client;
    private string _collectionName = "default";
    private int _dimension = 768;
    private Distance _distance = Distance.Cosine;

    public static VectorStoreBuilder Create(IQdrantClient client) => new(client);

    public VectorStoreBuilder WithCollection(string name) { _collectionName = name; return this; }
    public VectorStoreBuilder WithDimension(int dim) { _dimension = dim; return this; }
    
    public async Task<bool> BuildAsync(CancellationToken ct)
    {
        var exists = await _client.Collections.CollectionExistsAsync(_collectionName, ct);
        if (!exists)
            await _client.CreateCollectionAsync(new CreateCollection(_collectionName) 
            { VectorsConfig = new VectorParams { Size = _dimension, Distance = _distance } }, ct);
        return true;
    }
}

#### ✅ D. Leverage C# 14 / .NET 10 Features in Builders
You're already using `extension` members. Use them strategically for builder ergonomics without inheritance:
```csharp
// Instead of multiple static extension classes, group related builders
public static class BuilderExtensions
{
    public static T WithRequired<T>(this T obj, bool condition, string message) where T : notnull
    {
        if (!condition) throw new ArgumentException(message);
        return obj; // Returns self for chaining
    }

    public static IAsyncEnumerable<string> StreamFileLines(this FileInfo file) => 
        file.OpenText().ReadLinesAsync(); // .NET 10 async stream support
}
```
Also consider:
- `required` keyword for mandatory builder properties
- `global::using` to reduce namespace noise in builders
- `System.IO.Pipelines` instead of `StreamReader` for high-throughput CSV/file parsing
