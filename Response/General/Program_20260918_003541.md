```csharp
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;
using Ragnar.Abstractions;
using Ragnar.Branding;
using Ragnar.Core;
using Ragnar.FileQuestionProvider;
using Ragnar.Output;
using Ragnar.Services;
using Ragnar.Stages;
using Serilog;

namespace Ragnar;

/// <summary>
/// Entry point for the Ragnar application.
/// Bootstraps dependency injection, renders branding, and executes the embedding + Q&amp;A pipeline.
/// </summary>
/// <remarks>
/// All substantial logic lives in dedicated services, stages, and providers.
/// This class is intentionally thin: configure → build → run → report.
/// </remarks>
public static class Program
{
    /// <summary>
    /// Application entry point.
    /// </summary>
    /// <param name="args">Command-line arguments (reserved for future use).</param>
    public static async Task Main(string[[]] args)
    {
        IHost? host = null;

        try
        {
            host = await BuildHostAsync().ConfigureAwait(false);
            await host.StartAsync().ConfigureAwait(false);

            RenderBranding(host.Services);
            await ExecutePipelineAsync(host.Services).ConfigureAwait(false);
        }
        catch (OperationCanceledException)
        {
            AnsiConsole.MarkupLine("[[cyan]]Pipeline was cancelled.[[/cyan]]");
        }
        catch (Exception ex)
        {
            AnsiConsole.MarkupLine($"[[red]]Fatal error:[[/red]] {ex.Message}");
            Environment.ExitCode = 1;
        }
        finally
        {
            if (host is not null)
                await host.StopAsync().ConfigureAwait(false);
        }
    }

    // ─────────────────────────────────────────────────────────────────────────
    //  Host & Service Configuration
    // ─────────────────────────────────────────────────────────────────────────

    /// <summary>
    /// Creates and configures the <see cref="IHost"/> instance with all application services.
    /// </summary>
    private static IHost BuildHost()
    {
        return Host.CreateDefaultBuilder()
            .ConfigureAppConfiguration((ctx, config) =>
            {
                config.AddJsonFile("appsettings.json", optional: false, reloadOnChange: true);
                config.AddJsonFile("appsettings.Ragnar.json", optional: true, reloadOnChange: true);
            })
            .UseSerilog((ctx, cfg) => ctx.Configuration.GetSection("Serilog").ReadSerilog(cfg))
            .ConfigureServices(ConfigureServices)
            .Build();
    }

    /// <summary>
    /// Builds the host and returns it for use.
    /// </summary>
    private static Task<IHost> BuildHostAsync()
    {
        var host = BuildHost();
        return Task.FromResult(host);
    }

    // ─────────────────────────────────────────────────────────────────────────
    //  Runtime Steps
    // ─────────────────────────────────────────────────────────────────────────

    /// <summary>Renders the application banner and version to the console.</summary>
    private static void RenderBranding(IServiceProvider services)
    {
        services.GetRequiredService<IApplicationHeader>().RenderBranding();
    }

    /// <summary>
    /// Constructs the pipeline with its ordered stages and executes it.
    /// </summary>
    private static async Task ExecutePipelineAsync(IServiceProvider services)
    {
        var logger = services.GetRequiredService<ILogger<Program>>();
        var cts = services.GetRequiredService<CancellationTokenSource>();

        var context = new EmbeddingContext();
        var pipeline = BuildPipeline(services);

        var result = await pipeline.ExecuteAsync(context, cts.Token).ConfigureAwait(false);
        logger.Information("Pipeline completed. {Count} document(s) embedded.", result.Documents.Count);
    }

    /// <summary>
    /// Assembles the <see cref="IPipelineRunner"/> with stages in execution order.
    /// </summary>
    /// <remarks>
    /// Stage order matters:
    /// <list type="number">
    /// <item><see cref="FileDiscoveryStage"/> – scan source directory for eligible files</item>
    /// <item><see cref="ParsingStage"/> – parse discovered files into <c>CodeDocument</c> segments</item>
    /// <item><see cref="QuestionExecutionStage"/> – load questions, retrieve context, invoke RAG</item>
    /// </list>
    /// </remarks>
    private static IPipelineRunner BuildPipeline(IServiceProvider services)
    {
        var runner = services.GetRequiredService<IPipelineRunner>();

        runner
            .AddStage(services.GetRequiredService<FileDiscoveryStage>())
            .AddStage(services.GetRequiredService<ParsingStage>())
            .AddStage(services.GetRequiredService<QuestionExecutionStage>());

        return runner;
    }

    // ─────────────────────────────────────────────────────────────────────────
    //  Dependency Injection Registration
    // ─────────────────────────────────────────────────────────────────────────

    /// <summary>
    /// Registers all Ragnar-specific services into the container.
    /// </summary>
    /// <remarks>
    /// Grouped by responsibility for easy navigation:
    /// <list type="bullet">
    /// <item><b>Infrastructure</b> – clock, cancellation</item>
    /// <item><b>Output</b> – console writer, file writer, formatter, path resolver, response writer</item>
    /// <item><b>Branding</b> – application header</item>
    /// <item><b>Questions</b> – provider, embedding</item>
    /// <item><b>Core Services</b> – RAG orchestrator, summariser</item>
    /// <item><b>Pipeline Stages</b> – discovery, parsing, question execution</item>
    /// <item><b>Pipeline</b> – runner</item>
    /// </list>
    /// </remarks>
    private static void ConfigureServices(IServiceCollection services)
    {
        // Infrastructure
        services.AddSingleton<IClock, SystemClock>();
        services.AddSingleton(new CancellationTokenSource());

        // Output
        services.AddSingleton<IOutputWriter, OutputWriter>();
        services.AddSingleton<IOutputFormatter, MarkdownFormatter>();
        services.AddSingleton<IWriter, FileWriter>();
        services.AddSingleton<IPathResolver, PathResolver>();
        services.AddSingleton<IResponseWriter, ResponseWriter>();

        // Branding
        services.AddSingleton<IApplicationHeader, ApplicationHeader>();

        // Questions
        services.AddSingleton<IQuestionEmbedding, QuestionEmbedding>();
        services.AddSingleton<IQuestionProvider, CsvFileQuestionProvider>();

        // Core Services
        services.AddSingleton<IRagOrchestrator, RagOrchestrator>();
        services.AddSingleton<ISummaryService, SummaryService>();

        // Pipeline Stages
        services.AddSingleton<FileDiscoveryStage>();
        services.AddSingleton<ParsingStage>();
        services.AddSingleton<QuestionExecutionStage>();

        // Pipeline
        services.AddSingleton<IPipelineRunner, PipelineRunner>();
    }
}
```

## Key improvements

| Area | Before | After |
|------|--------|-------|
| **Class declaration** | `public class Program` (instantiable, no benefit) | `public static class Program` – signals "no state, no instances" |
| **Main body** | 30+ lines mixing bootstrap, branding, pipeline, and error handling | Three named steps: `RenderBranding` → `ExecutePipelineAsync` → error handling. Each step is a single, focused call. |
| **Pipeline construction** | Implicit (stages registered but never explicitly added to the runner) | `BuildPipeline` makes the **execution order** explicit and documented with a numbered `<remarks>` block |
| **Cancellation catch** | `when (args.Length == 0)` – a guard clause that made no logical sense | Plain `catch (OperationCanceledException)` – cancellation is cancellation |
| **Host lifecycle** | `scope.Start()` with no `Stop` – resources leaked on exit | `await host.StopAsync()` in a `finally` block guarantees cleanup |
| **Error output** | `Console.WriteLine` (unstyled, bypasses Spectre) | `AnsiConsole.MarkupLine` with colour, consistent with the rest of the app |
| **DI registration** | Grouped but buried in one 40-line method with a long XML doc | Same groups, but the method is split from the runtime code and the XML doc is a scannable bullet list |
| **`using` directives** | Included `Ragnar.Extensions` (unused) | Only the namespaces actually referenced |
| **`ConfigureAwait`** | Inconsistent | Applied uniformly on every `await` |

### Optional follow-up

If the stage list grows, replace `BuildPipeline` with a small configuration class or an `IStageProvider` that returns `IReadOnlyList<IPipelineStage<EmbeddingContext>>`, keeping the ordering rule in one place without editing `Program` every time a stage is added.
