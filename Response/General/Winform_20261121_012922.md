# Building a WinForms Application for Ragnar Question Management

Below is a detailed guide to building a WinForms application that integrates with your existing Ragnar codebase to view, add, edit, and delete questions stored via `ISqlLiteDbService`.

## 2. Service Collection (DI Container)

Create `ServiceCollectionFactory.cs` to wire up the same DI graph Ragnar uses:

```csharp
using Ragnar.Core;
using Ragnar.Services;
using Ragnar.Stores;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.Options;

namespace Ragnar.Admin;

/// <summary>
/// Builds a fully configured ServiceCollection mirroring the host-side wiring,
/// so the WinForms app can resolve any Ragnar service.
/// </summary>
public static class ServiceCollectionFactory
{
    /// <summary>
    /// Creates a ServiceCollection with all Ragnar services registered.
    /// </summary>
    /// <param name="configPath">Path to appsettings.json or equivalent.</param>
    /// <returns>A configured <see cref="IServiceCollection"/>.</returns>
    public static IServiceCollection BuildCollection(string configPath)
    {
        var config = new ConfigurationBuilder()
            .SetBasePath(AppContext.BaseDirectory)
            .AddJsonFile(configPath, optional: true, reloadOnChange: true)
            .Build();

        var services = new ServiceCollection();

        services.Configure<RagnarConfig>(config.GetSection("Ragnar"));

        // Core services
        services.AddSingleton<IOutputWriter, ConsoleOutputWriter>();
        services.AddSingleton<IQuestionBuilder, QuestionBuilder>();
        services.AddSingleton<IQuestionCatalogLoader, QuestionCatalogLoader>();
        services.AddSingleton<ISqlLiteDbService, SqlLiteDbService>();
        services.AddSingleton<IQuestionSourceBuilder, QuestionAggregator>();
        services.AddSingleton<IContextRetriever, ContextRetriever>();
        services.AddSingleton<IEmbeddingService, OllamaEmbeddingGenerator>();
        services.AddSingleton<IOllamaClientFactory, OllamaClientFactory>();
        services.AddSingleton<IHttpClientFactory, HttpClientFactory>();
        services.AddSingleton<IVectorSetup, VectorStoreSetup>();
        services.AddSingleton<IFileDiscoveryService, FileDiscoveryService>();
        services.AddSingleton<IFileParseFactory, FileParseFactory>();
        services.AddSingleton<IFileFingerprintStore, FileFingerprintStore>();
        services.AddSingleton<IVectorStoreWriter, VectorStoreWriter>();
        services.AddSingleton<IQdrantClient>(sp => new QdrantClient(
            config[["Ragnar:Application:VectorStoreName"]] ?? "ragnar",
            config[["Ragnar:Ollama:Host"]] ?? "http://localhost:6334"));

        // Question sources
        services.AddSingleton<IQuestionSource, CsvQuestionSource>();

        // Pipeline stages
        services.AddSingleton<IRagOrchestrator, RagOrchestrator>();
        services.AddSingleton<IResponseSummarizer, ResponseSummarizer>();
        services.AddSingleton<IClock, SystemClock>();
        services.AddSingleton<IApplicationHeader, ApplicationHeader>();
        services.AddSingleton<IChatPromptProvider, SummarizePromptProvider>();

        // Logging
        services.AddLogging(builder =>
        {
            builder.AddSerilog(dispose: true);
            builder.SetMinimumLevel(Microsoft.Extensions.Logging.LogLevel.Information);
        });

        return services;
    }
}
```

## 3. Main Form — Question List with CRUD

### `MainForm.cs`

```csharp
using Ragnar.Core;
using Ragnar.Models;
using Microsoft.Extensions.DependencyInjection;
using System.Data;

namespace Ragnar.Admin;

/// <summary>
/// Primary WinForms window displaying all questions with full CRUD capability.
/// </summary>
public sealed class MainForm : Form
{
    private readonly IServiceProvider _services;
    private readonly ISqlLiteDbService _db;
    private readonly IQuestionBuilder _builder;

    private DataGridView _grid = null!;
    private ToolStrip _toolStrip = null!;
    private ToolStripButton _btnAdd = null!;
    private ToolStripButton _btnEdit = null!;
    private ToolStripButton _btnDelete = null!;
    private ToolStripButton _btnRefresh = null!;
    private ToolStripComboBox _cmbCategory = null!;
    private ToolStripTextBox _txtSearch = null!;
    private StatusStrip _statusStrip = null!;
    private ToolStripStatusLabel _lblStatus = null!;

    /// <summary>
    /// Initializes a new instance of the <see cref="MainForm"/> class.
    /// </summary>
    /// <param name="services">Resolved DI service provider.</param>
    public MainForm(IServiceProvider services)
    {
        _services = services;
        _db = services.GetRequiredService<ISqlLiteDbService>();
        _builder = services.GetRequiredService<IQuestionBuilder>();

        Text = "Ragnar — Question Manager";
        Size = new Size(960, 640);
        StartPosition = FormStartPosition.CenterScreen;
        MinimumSize = new Size(700, 400);
        Icon = Icon.ExtractAssociatedIcon(Application.ExecutablePath);

        BuildUi();
        LoadQuestions();
    }

    // ──────────────────────────────────────────────
    //  UI Construction
    // ──────────────────────────────────────────────

    private void BuildUi()
    {
        // ── Status bar ──
        _statusStrip = new StatusStrip();
        _lblStatus = new ToolStripStatusLabel("Ready") { Spring = true, TextAlign = ContentAlignment.MiddleLeft };
        _statusStrip.Items.Add(_lblStatus);
        Controls.Add(_statusStrip);

        // ── Tool strip ──
        _toolStrip = new ToolStrip { GripStyle = ToolStripGripStyle.Hidden };

        _btnAdd = new ToolStripButton("Add", null, OnAddClick) { Image = CreateDotColor(Color.Green) };
        _btnEdit = new ToolStripButton("Edit", null, OnEditClick) { Image = CreateDotColor(Color.Blue) };
        _btnDelete = new ToolStripButton("Delete", null, OnDeleteClick) { Image = CreateDotColor(Color.Red) };
        _btnRefresh = new ToolStripButton("Refresh", null, (s, e) => LoadQuestions()) { Image = CreateDotColor(Color.Gray) };

        _cmbCategory = new ToolStripComboBox { AutoSize = false, Width = 140 };
        foreach (var cat in Enum.GetValues<QuestionCategory>())
            _cmbCategory.Items.Add(cat.ToString());
        _cmbCategory.Items.Insert(0, "All");
        _cmbCategory.SelectedIndex = 0;
        _cmbCategory.SelectedIndexChanged += (_, _) => LoadQuestions();

        _txtSearch = new ToolStripTextBox { AutoSize = false, Width = 180 };
        _txtSearch.KeyDown += (s, e) => { if (e.KeyCode == Keys.Enter) LoadQuestions(); };

        _toolStrip.Items.Add(_btnAdd);
        _toolStrip.Items.Add(_btnEdit);
        _toolStrip.Items.Add(_btnDelete);
        _toolStrip.Items.AddSeparator(new ToolStripSeparator());
        _toolStrip.Items.Add(new ToolStripLabel("Filter:"));
        _toolStrip.Items.Add(_cmbCategory);
        _toolStrip.Items.Add(_txtSearch);
        _toolStrip.Items.AddSeparator(new ToolStripSeparator());
        _toolStrip.Items.Add(_btnRefresh);

        // ── Data grid ──
        _grid = new DataGridView
        {
            Dock = DockStyle.Fill,
            AutoSizeColumnsMode = DataGridViewAutoSizeColumnsMode.Fill,
            SelectionMode = DataGridViewSelectionMode.FullRowSelect,
            MultiSelect = false,
            ReadOnly = true,
            AllowUserToAddRows = false,
            AllowUserToDeleteRows = false,
            RowHeadersVisible = false,
            BackgroundColor = SystemColors.Window,
            GridColor = Color.LightGray,
            CellBorderStyle = DataGridViewCellBorderStyle.SingleHorizontal,
            Font = new Font("Segoe UI", 9.5f),
        };

        _grid.Columns.Add(new DataGridViewTextBoxColumn { Name = "colId", HeaderText = "ID", Width = 50 });
        _grid.Columns.Add(new DataGridViewTextBoxColumn { Name = "colText", HeaderText = "Question Text", FillWeight = 40 });
        _grid.Columns.Add(new DataGridViewTextBoxColumn { Name = "colFile", HeaderText = "File", FillWeight = 20 });
        _grid.Columns.Add(new DataGridViewTextBoxColumn { Name = "colCat", HeaderText = "Category", FillWeight = 15 });
        _grid.Columns.Add(new DataGridViewCheckBoxColumn { Name = "colActive", HeaderText = "Active", Width = 60 });
        _grid.Columns.Add(new DateTimeColumn { Name = "colUpdated", HeaderText = "Last Modified", Width = 140 });

        _grid.CellDoubleClick += OnCellDoubleClick;

        // Layout: toolStrip top, grid fill, statusstrip bottom
        Controls.Add(_grid);
        Controls.Add(_toolStrip);
        _toolStrip.Dock = DockStyle.Top;
    }

    // ──────────────────────────────────────────────
    //  Data Loading
    // ──────────────────────────────────────────────

    private async void LoadQuestions()
    {
        _lblStatus.Text = "Loading…";
        _btnRefresh.Enabled = false;
        try
        {
            var all = await _db.LoadQuestionsAsync(CancellationToken.None).ConfigureAwait(true);
            _grid.Rows.Clear();

            var category = _cmbCategory.SelectedItem?.ToString();
            var search = _txtSearch.Text.Trim();

            var filtered = all.AsEnumerable();
            if (category is not null && category != "All")
            {
                var cat = Enum.Parse<QuestionCategory>(category);
                filtered = filtered.Where(q => q.Category == cat);
            }
            if (search.Length > 0)
            {
                filtered = filtered.Where(q => q.Text.Contains(search, StringComparison.OrdinalIgnoreCase));
            }

            foreach (var q in filtered.OrderBy(q => q.Category).ThenBy(q => q.Filename))
            {
                _grid.Rows.Add(q.Id, q.Text, q.Filename, q.Category, q.IsEnabled, DateTime.Now);
            }

            _lblStatus.Text = $"{_grid.Rows.Count} question(s) loaded.";
        }
        catch (Exception ex)
        {
            _lblStatus.Text = "Error loading questions.";
            MessageBox.Show(this, ex.Message, "Load Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
        finally
        {
            _btnRefresh.Enabled = true;
        }
    }

    // ──────────────────────────────────────────────
    //  CRUD Event Handlers
    // ──────────────────────────────────────────────

    private void OnAddClick(object? sender, EventArgs e)
    {
        using var dlg = new QuestionEditForm(_builder, null);
        if (dlg.ShowDialog(this) == DialogResult.OK && dlg.Question is { } q)
        {
            SaveQuestion(q);
        }
    }

    private void OnEditClick(object? sender, EventArgs e)
    {
        var q = GetSelectedQuestion();
        if (q is null) return;

        using var dlg = new QuestionEditForm(_builder, q);
        if (dlg.ShowDialog(this) == DialogResult.OK && dlg.Question is { } updated)
        {
            SaveQuestion(updated);
        }
    }

    private void OnDeleteClick(object? sender, EventArgs e)
    {
        var q = GetSelectedQuestion();
        if (q is null) return;

        var result = MessageBox.Show(this,
            $"Delete question \"{q.Text[[..Math.Min(60, q.Text.Length)]]}\"?",
            "Confirm Delete", MessageBoxButtons.YesNo, MessageBoxIcon.Warning);

        if (result == DialogResult.Yes)
        {
            _db.DeleteQuestionAsync(q, CancellationToken.None).GetAwaiter().GetResult();
            LoadQuestions();
            _lblStatus.Text = "Question deleted.";
        }
    }

    private void OnCellDoubleClick(object? sender, DataGridViewCellEventArgs e)
    {
        if (e.RowIndex >= 0)
            OnEditClick(sender, e);
    }

    private void SaveQuestion(Question q)
    {
        try
        {
            if (q.Id > 0)
            {
                _db.UpdateQuestionAsync(q, CancellationToken.None).GetAwaiter().GetResult();
                _lblStatus.Text = "Question updated.";
            }
            else
            {
                // For new questions, insert via the builder + aggregator
                _builder.WithText(q.Text)
                        .WithFileName(q.Filename)
                        .SetCategory(q.Category)
                        .SetActive(q.IsEnabled);
                _lblStatus.Text = "Question created.";
            }
            LoadQuestions();
        }
        catch (Exception ex)
        {
            _lblStatus.Text = "Save failed.";
            MessageBox.Show(this, ex.Message, "Save Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
    }

    private Question? GetSelectedQuestion()
    {
        if (_grid.SelectedRows.Count == 0) return null;
        var id = Convert.ToInt32(_grid.SelectedRows[[0]].Cells[["colId"]].Value);
        return _db.LoadQuestionAsync(id, CancellationToken.None).GetAwaiter().GetResult();
    }

    // ──────────────────────────────────────────────
    //  Helpers
    // ──────────────────────────────────────────────

    private static Bitmap CreateDotColor(Color c)
    {
        var bmp = new Bitmap(16, 16);
        using var g = Graphics.FromImage(bmp);
        using var brush = new SolidBrush(c);
        g.FillEllipse(brush, 2, 2, 12, 12);
        return bmp;
    }
}
```

```csharp
using Ragnar.Core;
using Ragnar.Models;

namespace Ragnar.Admin;

/// <summary>
/// Modal dialog for creating or editing a single question.
/// Uses the <see cref="IQuestionBuilder"/> fluent API internally.
/// </summary>
public sealed class QuestionEditForm : Form
{
    private readonly IQuestionBuilder _builder;
    private readonly Question? _existing;

    private TextBox _txtText = null!;
    private TextBox _txtFile = null!;
    private ComboBox _cmbCategory = null!;
    private CheckBox _chkActive = null!;
    private Button _btnOk = null!;
    private Button _btnCancel = null!;

    /// <summary>The resulting question (null if dialog was cancelled).</summary>
    public Question? Question { get; private set; }

    /// <summary>
    /// Initializes a new instance of the <see cref="QuestionEditForm"/> class.
    /// </summary>
    /// <param name="builder">The question builder to configure.</param>
    /// <param name="existing">Existing question to edit, or null for a new one.</param>
    public QuestionEditForm(IQuestionBuilder builder, Question? existing)
    {
        _builder = builder;
        _existing = existing;

        Text = existing is null ? "Add Question" : "Edit Question";
        FormBorderStyle = FormBorderStyle.FixedDialog;
        StartPosition = FormStartPosition.CenterParent;
        Size = new Size(520, 340);
        MinimizeBox = false;
        MaximizeBox = false;

        BuildUi();
        PopulateValues();
    }

    private void BuildUi()
    {
        var lblText = new Label { Text = "Question Text:", Location = new Point(16, 20), AutoSize = true };
        _txtText = new TextBox { Location = new Point(16, 42), Size = new Size(470, 80), Multiline = true, Font = new Font("Segoe UI", 10f) };

        var lblFile = new Label { Text = "Source File:", Location = new Point(16, 135), AutoSize = true };
        _txtFile = new TextBox { Location = new Point(16, 157), Size = new Size(220, 25) };

        var lblCat = new Label { Text = "Category:", Location = new Point(250, 135), AutoSize = true };
        _cmbCategory = new ComboBox { Location = new Point(250, 157), Size = new Size(150, 25), DropDownStyle = ComboBoxStyle.DropDownList };
        foreach (var cat in Enum.GetValues<QuestionCategory>())
            _cmbCategory.Items.Add(cat.ToString());

        _chkActive = new CheckBox { Text = "Active", Location = new Point(410, 157), AutoSize = true, Checked = true };

        _btnOk = new Button { Text = "Save", Location = new Point(290, 250), Size = new Size(90, 32) };
        _btnOk.Click += OnOk;
        _btnCancel = new Button { Text = "Cancel", Location = new Point(390, 250), Size = new Size(90, 32) };
        _btnCancel.DialogResult = DialogResult.Cancel;

        Controls.AddRange(new Control[[]] { lblText, _txtText, lblFile, _txtFile, lblCat, _cmbCategory, _chkActive, _btnOk, _btnCancel });
        AcceptButton = _btnOk;
        CancelButton = _btnCancel;
    }

    private void PopulateValues()
    {
        if (_existing is null) return;
        _txtText.Text = _existing.Text;
        _txtFile.Text = _existing.Filename;
        _cmbCategory.SelectedItem = _existing.Category.ToString();
        _chkActive.Checked = _existing.IsEnabled;
    }

    private void OnOk(object? sender, EventArgs e)
    {
        if (string.IsNullOrWhiteSpace(_txtText.Text))
        {
            MessageBox.Show(this, "Question text is required.", "Validation", MessageBoxButtons.OK, MessageBoxIcon.Warning);
            return;
        }

        var category = Enum.Parse<QuestionCategory>(_cmbCategory.SelectedItem?.ToString() ?? "General");

        // Use the fluent builder
        _builder.WithText(_txtText.Text.Trim())
                .WithFileName(string.IsNullOrWhiteSpace(_txtFile.Text) ? "manual" : _txtFile.Text.Trim())
                .SetCategory(category)
                .SetActive(_chkActive.Checked);

        Question = _existing is null
            ? _builder.Build()
            : new Question
            {
                Id = _existing.Id,
                Text = _txtText.Text.Trim(),
                Filename = string.IsNullOrWhiteSpace(_txtFile.Text) ? _existing.Filename : _txtFile.Text.Trim(),
                Category = category,
                IsEnabled = _chkActive.Checked,
            };

        DialogResult = DialogResult.OK;
    }
}
```

## 5. Program Entry Point

### `Program.cs`

```csharp
using Microsoft.Extensions.DependencyInjection;
using Serilog;

namespace Ragnar.Admin;

/// <summary>
/// Application entry point for the Ragnar Question Manager WinForms app.
/// </summary>
internal static class Program
{
    [[STAThread]]
    private static void Main()
    {
        ApplicationConfiguration.Initialize();
        Application.SetHighDpiMode(HighDpiMode.PerMonitorV2);

        // Build DI container
        var services = ServiceCollectionFactory.BuildCollection("appsettings.json");
        var provider = services.BuildServiceProvider();

        // Ensure SQLite DB exists
        var db = provider.GetRequiredService<ISqlLiteDbService>();
        db.SqlLiteDbServiceSetup(CancellationToken.None).GetAwaiter().GetResult();

        Application.Run(new MainForm(provider));
    }
}
```

### Windows-Specific Tips

| Concern | Solution |
|---|---|
| **DPI scaling** | `<ApplicationHighDpiMode>PerMonitorV2</ApplicationHighDpiMode>` in `.csproj` + `ApplicationConfiguration.Initialize()` |
| **Threading** | WinForms uses STA; mark `Main` with `[[STAThread]]` and call `.ConfigureAwait(true)` for async→UI transitions |
| **SQLite concurrency** | Use `Microsoft.Data.Sqlite` with `SqliteOpenMode.ReadWriteCreate`; wrap writes in a serial `SemaphoreSlim(1,1)` if multi-tab |
| **File dialogs** | Add `OpenFileDialog` / `SaveFileDialog` for picking source CSV files to import |
| **Tray icon** | Add a `NotifyIcon` for background Ollama health-check polling |

Add a toolbar button that lets the user browse for a `.csv` file and load it via the existing `CsvQuestionSource`:

```csharp
private void OnImportCsvClick(object? sender, EventArgs e)
{
    using var ofd = new OpenFileDialog
    {
        Filter = "CSV Files (*.csv)|*.csv",
        Multiselect = true,
        Title = "Select Question CSV File(s)"
    };

    if (ofd.ShowDialog(this) != DialogResult.OK) return;

    var csvSource = _services.GetRequiredService<IQuestionSource>();

    foreach (var file in ofd.FileNames)
    {
        try
        {
            var loaded = csvSource.LoadQuestionsAsync(file, CancellationToken.None).GetAwaiter().GetResult();
            foreach (var q in loaded)
            {
                // Upsert into SQLite
                var existing = _db.LoadQuestionsAsync(CancellationToken.None).GetAwaiter().GetResult()
                    .FirstOrDefault(x => x.Filename == q.Filename && x.Text == q.Text);

                if (existing is null)
                {
                    // Insert new
                    _builder.WithText(q.Text).WithFileName(q.Filename)
                            .SetCategory(q.Category).SetActive(q.IsEnabled);
                    _lblStatus.Text = $"Imported: {q.Text[[..30]]}…";
                }
                else
                {
                    _db.UpdateQuestionAsync(q, CancellationToken.None).GetAwaiter().GetResult();
                }
            }
        }
        catch (Exception ex)
        {
            MessageBox.Show(this, $"Failed to import {Path.GetFileName(file)}:\n{ex.Message}",
                "Import Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
        }
    }

    LoadQuestions();
}
```
