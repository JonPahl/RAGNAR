### 2. `PipelineRunner` – inconsistent fluent return types break chaining

**Problem:**
```csharp
public override IPipeline<TContext> AddStage(...)      // returns base interface
public override PipelineRunner<TContext> RemoveStage(...)  // returns concrete
public override PipelineRunner<TContext> Reorder(...)      // returns concrete
```
A caller writing `runner.AddStage(a).RemoveStage("b")` gets an `IPipeline<TContext>` which **doesn't expose `RemoveStage`**.

**Fix – unify all mutating methods to return the concrete type (or a sealed interface):**

```csharp
public sealed class PipelineRunner<TContext>(Serilog.ILogger logger) : IPipeline<TContext>
    where TContext : class
{
    private readonly Lock _gate = new();
    private readonly List<StageEntry<TContext>> _stages = [[]];

    public sealed record StageEntry<T>(IPipelineStage<T> Stage, IAsyncPolicy? Policy = null);

    /// <summary>Registers a stage. Returns this runner for chaining.</summary>
    public PipelineRunner<TContext> AddStage(IPipelineStage<TContext> stage, IAsyncPolicy? policy = null)
    {
        Guard.Against.Null(stage);
        lock (_gate)
        {
            if (_stages.Any(x => x.Stage.Name == stage.Name))
                throw new ArgumentException($"Stage '{stage.Name}' is already registered.", nameof(stage));
            _stages.Add(new StageEntry<TContext>(stage, policy));
        }
        return this;
    }

    /// <summary>Removes a stage by name. Returns this runner.</summary>
    public PipelineRunner<TContext> RemoveStage(string name)
    {
        lock (_gate)
            _stages.RemoveAll(s => s.Stage.Name == name);
        return this;
    }

    /// <summary>Reorders a stage. Returns this runner.</summary>
    public PipelineRunner<TContext> Reorder(string fromName, int newIndex)
    {
        lock (_gate)
        {
            var entry = _stages.FirstOrDefault(s => s.Stage.Name == fromName)
                ?? throw new KeyNotFoundException(fromName);
            _stages.Remove(entry);
            _stages.Insert(Math.Clamp(newIndex, 0, _stages.Count), entry);
        }
        return this;
    }

    public PipelineRunner<TContext> ReplaceStage(string oldName, IPipelineStage<TContext> newStage, IAsyncPolicy? policy = null)
        => RemoveStage(oldName).AddStage(newStage, policy);

    public PipelineRunner<TContext> ClearStages()
    {
        lock (_gate) _stages.Clear();
        return this;
    }

    public IReadOnlyList<string> StageNames
    {
        get { lock (_gate) return _stages.ConvertAll(s => s.Stage.Name).AsReadOnly(); }
    }

    /// <inheritdoc />
    public async Task<TContext> ExecuteAsync(TContext context, CancellationToken ct)
    {
        var sw = Stopwatch.StartNew();
        List<StageEntry<TContext>> snapshot;
        lock (_gate) snapshot = [[.. _stages]];

        for (var i = 0; i < snapshot.Count; i++)
        {
            var entry = snapshot[[i]];
            if (!entry.Stage.ShouldRun(context))
            {
                logger.Information("Skipping stage [[{Index}]]: {Name}", i, entry.Stage.Name);
                continue;
            }

            logger.Information("Starting stage [[{Index}/{Total}]]: {Name}", i + 1, snapshot.Count, entry.Stage.Name);
            var stageSw = Stopwatch.StartNew();
            try
            {
                if (entry.Policy is null)
                {
                    await entry.Stage.ExecuteAsync(context, ct).ConfigureAwait(false);
                }
                else
                {
                    await entry.Policy.ExecuteAsync(
                        async () => await entry.Stage.ExecuteAsync(context, ct).ConfigureAwait(false)
                    ).ConfigureAwait(false);
                }
            }
            catch (OperationCanceledException) when (ct.IsCancellationRequested)
            {
                throw; // let the caller handle cancellation
            }
            catch (Exception ex)
            {
                logger.Fatal(ex, "Stage [[{Index}]] {Name} failed after {Elapsed}", i, entry.Stage.Name, stageSw.Elapsed);
                throw new PipelineStageException(entry.Stage.Name, ex) from ex;
            }

            logger.Information("Completed stage [[{Index}]] {Name} in {Time}",
                i, entry.Stage.Name, stageSw.Elapsed);
        }

        logger.Information("Pipeline finished in {Total}", sw.Elapsed);
        return context;
    }
}
```

### 3. `CodeParserFactory` – dead code (both branches call the same thing)

**Problem:**
```csharp
private async Task<IEnumerable<CodeDocument>> ParseFile(...) 
    => await _codeParser.ParseAsync(file, cancellationToken);

private async Task<IEnumerable<CodeDocument>> GetCodeDocumentsAsync(...) 
    => await _codeParser.ParseAsync(file, cancellationToken);
```
Both methods are identical. The `switch` on extension is meaningless.

**Fix – collapse to a single path and make the factory extensible:**

```csharp
/// <summary>
/// Selects and delegates file parsing to the appropriate syntax parser.
/// </summary>
/// <example><![[CDATA[[var docs = await factory.ParseAsync("app.cs", ct);]]]]></example>
public class CodeParserFactory(
    Serilog.ILogger logger,
    ISyntaxTreeChunker cSharpFile,
    IEnumerable<IFileParser> parsers) : IFileParseFactory
{
    private readonly ParseCSharpFile _csharpParser = new(logger, cSharpFile);

    /// <summary>Parses a file using the correct parser based on extension.</summary>
    /// <param name="file">File path.</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <returns>Parsed code documents.</returns>
    /// <exception cref="NotSupportedException">No parser found for the file extension.</exception>
    public async Task<IEnumerable<CodeDocument>> ParseAsync(string file, CancellationToken cancellationToken)
    {
        var ext = Path.GetExtension(file);

        // 1. Try the dedicated C# parser first
        if (ext.Equals(".cs", StringComparison.OrdinalIgnoreCase))
            return await _csharpParser.ParseAsync(file, cancellationToken).ConfigureAwait(false);

        // 2. Fall back to any registered IFileParser that claims this extension
        var parser = parsers.FirstOrDefault(p => p.SupportedExtensions.Contains(ext, StringComparer.OrdinalIgnoreCase));
        if (parser is not null)
            return await parser.ParseAsync(file, cancellationToken).ConfigureAwait(false);

        // 3. Default: treat as plain text
        logger.Warning("No specific parser for extension '{Ext}'; falling back to text parser.", ext);
        return await _csharpParser.ParseAsync(file, cancellationToken).ConfigureAwait(false);
    }
}
```

> Update `IFileParser` to include `IReadOnlyCollection<string> SupportedExtensions { get; }`.

### 4. `QuestionCombineBuilder` – broken builder chain in CSV methods

**Problem:**
```csharp
questionBuilder.WithText(csv.Text).WithFileName(csv.Filename).SetCategory(csv.Category).SetActive(csv.IsEnabled);
Questions.Add(questionBuilder.Build());
```
The fluent calls mutate a shared `IQuestionBuilder` instance. If `questionBuilder` is a reference-type singleton from DI, concurrent calls or sequential calls will **overwrite** each other's state. The `SetActive` extension also just calls `AsActive()`/`AsInactive()` on the same instance.

**Fix – make `IQuestionBuilder` produce a fresh draft per call, or use a local builder:**

```csharp
public sealed class QuestionCombineBuilder(
    IQuestionSource questionProvider,
    IQuestionCatalogLoader questionLoader,
    IQuestionBuilderFactory questionBuilderFactory,   // ← factory, not a shared instance
    IOptions<RagnarConfig> options) : IQuestionSourceBuilder
{
    private List<Question> _questions = [[]];

    public IQuestionSourceBuilder GetCsvFilesAsync(string pluginDir, CancellationToken ct) =>
        GetCsvFileEnumerableAsync(Directory.Exists(pluginDir)
            ? Directory.EnumerateFiles(pluginDir, "*.csv", SearchOption.AllDirectories)
            : [[]], ct);

    private async IAsyncEnumerable<IQuestionSourceBuilder> GetCsvFileEnumerableAsync(
        IEnumerable<string> csvFiles, [[EnumeratorCancellation]] CancellationToken ct)
    {
        foreach (var csvFile in csvFiles)
        {
            ct.ThrowIfCancellationRequested();
            if (csvFile.Contains("Summary.csv", StringComparison.OrdinalIgnoreCase))
                continue;

            var configs = await questionProvider.LoadQuestionsAsync(csvFile, ct).ConfigureAwait(false);
            foreach (var cfg in configs)
            {
                // Each call gets a FRESH builder instance from the factory
                var q = questionBuilderFactory
                    .Create()
                    .WithText(cfg.Text)
                    .WithFileName(cfg.Filename)
                    .SetCategory(cfg.Category)
                    .SetActive(cfg.IsEnabled)
                    .Build();
                _questions.Add(q);
            }
            yield return this;
        }
    }

    public IReadOnlyList<Question> Build()
    {
        var result = _questions
            .Where(q => q.IsEnabled)
            .OrderBy(q => q.Category.ToString())
            .ThenBy(q => q.Filename)
            .ToList();

        return result.Count == 0
            ? throw new InvalidOperationException(
                "QuestionCombineBuilder produced zero questions. Check categories, file paths, and CSV sources.")
            : result.AsReadOnly();
    }
}
```

### 5. `FilterBuilder.AddRange` – XML doc mismatch & missing single-bounds overload

**Problem:** The doc shows `AddRange("len", 50)` (one value) but the signature is `AddRange(string, double?, double?)`. A caller passing `50` hits the `gte` parameter silently.

**Fix – add an explicit single-bound overload and fix docs:**

```csharp
/// <summary>Adds a less-than (upper-bound) range condition.</summary>
/// <param name="field">Payload field name.</param>
/// <param name="lte">Upper bound (less-than-or-equal).</param>
/// <returns>This builder.</returns>
/// <example><![[CDATA[[builder.AddRange("CommentLength", lte: 50);]]]]></example>
public FilterBuilder AddLte(string field, double lte) => AddRange(field, gte: null, lte: lte);

/// <summary>Adds a greater-than-or-equal (lower-bound) range condition.</summary>
/// <param name="field">Payload field name.</param>
/// <param name="gte">Lower bound.</param>
/// <returns>This builder.</returns>
/// <example><![[CDATA[[builder.AddRange("Score", gte: 7.0);]]]]></example>
public FilterBuilder AddGte(string field, double gte) => AddRange(field, gte: gte, lte: null);

/// <summary>Adds a two-sided range condition.</summary>
/// <param name="field">Payload field name.</param>
/// <param name="gte">Optional lower bound.</param>
/// <param name="lte">Optional upper bound.</param>
/// <returns>This builder.</returns>
/// <example><![[CDATA[[builder.AddRange("Length", gte: 10, lte: 500);]]]]></example>
public FilterBuilder AddRange(string field, double? gte = null, double? lte = null)
{
    if (gte is null && lte is null)
        throw new ArgumentException("At least one of gte or lte must be provided.", nameof(field));

    var range = new Range();
    if (gte is not null) range.Gte = gte.Value;
    if (lte is not null) range.Lte = lte.Value;
    _must.Add(new Condition { Field = new FieldCondition { Key = field, Range = range } });
    return this;
}
```

### 6. `NumberedRowDecorator.Render()` – not idempotent

**Problem:** Calling `Render()` twice prepends **another** `#` column and **duplicates** every row.

**Fix – guard with a flag:**

```csharp
public sealed class NumberedRowDecorator(ITableRenderer inner, string columnHeader = "#") : ITableRenderer
{
    private bool _numbered;

    public ConsoleTableBuilder Table => inner.Table;

    public void Render()
    {
        if (_numbered)
        {
            inner.Render();          // already numbered; just delegate
            return;
        }
        _numbered = true;

        var table = inner.Table;
        if (!table.Columns.Contains(columnHeader, StringComparer.OrdinalIgnoreCase))
            table.AddColumns(columnHeader);

        var existing = table.Rows.Select(r => r.ToArray()).ToList();
        for (var i = 0; i < existing.Count; i++)
        {
            var indexed = new string[[table.Columns.Count]];
            indexed[[0]] = (i + 1).ToString();
            Array.Copy(existing[[i]], 0, indexed, 1, existing[[i]].Length);
            table.AddRow(indexed.Select(c => new TableCell(c)).ToList());
        }

        inner.Render();
    }
}
```

### 10. `FileDiscoveryService` – not truly async; hard to test

**Problem:** `GetValuesAsync` wraps a synchronous `Directory.EnumerateFiles` in an `async IAsyncEnumerable`, giving a false impression of I/O parallelism. No DI seam for the directory listing.

**Fix – use `IAsyncEnumerable`-based enumeration and inject a listing delegate:**

```csharp
public sealed class FileDiscoveryService : IFileDiscoveryService
{
    private readonly IFileValidator _validator;

    public FileDiscoveryService(IFileValidator validator) => _validator = validator;

    public async IAsyncEnumerable<string> GetFilesAsync(
        string directory, FileLoadOptions filter,
        [[EnumeratorCancellation]] CancellationToken cancellationToken)
    {
        Guard.Against.Null(filter);
        Guard.Against.NullOrEmpty(directory);
        if (!Directory.Exists(directory))
            throw new DirectoryNotFoundException(directory);

        // Directory.EnumerateFiles is inherently sync; enumerate, filter, yield
        foreach (var path in Directory.EnumerateFiles(directory, "*.*", SearchOption.AllDirectories))
        {
            cancellationToken.ThrowIfCancellationRequested();

            var fi = new FileInfo(path);
            if (!_validator.IsValid(fi, filter))
                continue;

            yield return path;
            await Task.Yield(); // cooperative scheduling (optional)
        }
    }
}
```
