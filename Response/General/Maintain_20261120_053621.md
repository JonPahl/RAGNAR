{

    catch (Exception ex) { logger.LogCritical(ex, ex.Message); throw; }
}

// AFTER — single private helper; both public methods become 3 lines
private async Task<T> ExecuteWithTimeoutAsync<T>(CancellationToken ct, Func<CancellationToken, Task<T>> action)
{
    ct.ThrowIfCancellationRequested();
    try
    {
        using var timeoutCts = new CancellationTokenSource(_timeout);
        using var linked = CancellationTokenSource.CreateLinkedTokenSource(ct, timeoutCts.Token);
        return await action(linked.Token).ConfigureAwait(false);
    }
    catch (Exception) when (ct.IsCancellationRequested) { throw; }
    catch (Exception ex)
    {
        logger.LogCritical(ex, "Ollama embedding request failed");
        throw;
    }
}

public async Task<ReadOnlyMemory<float>> GenerateAsync(string input, CancellationToken ct)
{
    Guard.Against.NullOrWhiteSpace(input);
    return await ExecuteWithTimeoutAsync(ct, token =>
        _generator.GenerateAsync(input, token).ContinueWith(t => t.Result.Vector, TaskScheduler.Default)).ConfigureAwait(false);
}

public async Task<GeneratedEmbeddings<Embedding<float>>> GenerateBatchAsync(
    IReadOnlyCollection<string> inputs, CancellationToken ct)

## 6. Type-Safety and Null-Safety Gaps

### Problem
- `QuestionAggregator.GetCsvFilesAsync` calls `questionProvider.LoadQuestionsAsync(...)` **without a null check** on `questionProvider` (the `FirstOrDefault` can return `null`).
- `EmbeddingOptionsValidator` calls `.NotEmpty()` on `x.Dimension` (an `int`) — meaningless and confusing.
- `MarkdownFileParser.ToLines` does `content.ToString()` on a `ReadOnlyMemory<char>` just to call `.Split('\n')` — unnecessary allocation.

### Fix

```csharp

if (provider is null)
    throw new InvalidOperationException(
        $"No IQuestionSource registered for '{QuestionSource.Csv}'. " +
        $"Available: {string.Join(", ", questionProviders.Select(p => p.SourceName))}");

// 2. EmbeddingOptionsValidator — remove .NotEmpty() on int
public EmbeddingOptionsValidator()
{
    RuleFor(x => x.Host).NotEmpty().WithMessage("Host is required.");
    RuleFor(x => x.EmbeddingModel)
        .NotEmpty()
        .MaximumLength(256)
        .Matches(StringRegex)
        .WithMessage("Embedding model name must contain only alphanumeric, dot, hyphen, underscore.");
    RuleFor(x => x.Timeout)
        .GreaterThan(TimeSpan.Zero)
        .LessThan(TimeSpan.FromHours(1));
    RuleFor(x => x.Dimension)
        .GreaterThan(0).WithMessage("Dimension must be positive.");   // ← meaningful
    RuleFor(x => x.BatchSize)
        .InclusiveBetween(1, 512)
        .WithMessage("BatchSize must be between 1 and 512.");
}

// 3. MarkdownFileParser — work directly on the char span, avoid ToString allocation
private static List<string> ToLines(in ReadOnlyMemory<char> content)
{
    var lines = new List<string>();
    var chars = content.Span;
    int start = 0;

    for (int i = 0; i < chars.Length; i++)
    {
        if (chars[[i]] == '\n')
        {
            lines.Add(chars.Slice(start, i - start).ToString());
            start = i + 1;
        }
    }

    if (start < chars.Length)
        lines.Add(chars.Slice(start).ToString());

    return lines;
}
```

## 10. Pipeline Context: Mutability Without Encapsulation

### Problem
`EmbeddingContext` is a mutable POCO with `get; set;` on every property. Any stage can overwrite any field at any time, and there's no validation or invariants.

### Fix

```csharp
/// <summary>Immutable snapshot of pipeline state at a given stage boundary.</summary>
public sealed record EmbeddingContext
{
    public required string SourceDirectory { get; init; }
    public IReadOnlyList<string> DiscoveredFiles { get; init; } = [[]];
    public IReadOnlyList<CodeDocument> Documents { get; init; } = [[]];
    public UpdateResult? UpsertResult { get; init; }
    public TimeSpan TotalElapsed { get; init; }
    public QuestionCategory? CategoryFilter { get; init; }
    public int SkippedCount { get; init; }

    /// <summary>Creates a new context with one or more fields replaced.</summary>
    public EmbeddingContext With(IReadOnlyList<string> files = null,
                                 IReadOnlyList<CodeDocument> docs = null,
                                 UpdateResult? upsert = null,
                                 int? skipped = null) => this with
    {
        DiscoveredFiles = files ?? DiscoveredFiles,
        Documents = docs ?? Documents,
        UpsertResult = upsert ?? UpsertResult,
        SkippedCount = skipped ?? SkippedCount,
    };
}

// Stages now produce a NEW context instead of mutating:
public async Task<EmbeddingContext> ExecuteAsync(EmbeddingContext ctx, CancellationToken ct)
{
    var files = await DiscoverFilesAsync(ctx.SourceDirectory, ct).ConfigureAwait(false);
    return ctx.With(files: files);   // ← new immutable instance
}
```

> **Note:** This is a larger refactor. If the `with` pattern is too disruptive, at minimum add `Validate()` and `Freeze()` methods and call them between stages in `Pipeline.ExecuteAsync`.
