```csharp
/// <summary>
/// Wraps QdrantClient to perform scored vector similarity search
/// and project the results into <see cref="SearchHit"/> records.
/// </summary>
public sealed class QdrantVectorSearchService(
    IOllamaClientFactory clientFactory,
    IOptions<RagnarConfig> config) : IVectorSearchService
{
    private readonly ILogger _logger = // injected or created
        App.CreateBuilder().Build().Services.GetRequiredService<ILogger<QdrantVectorSearchService>>();

    public async Task<IReadOnlyList<SearchHit>> SearchAsync(
        float[[]] queryVector,
        int topK = 10,
        Filter? filter = null,
        double scoreThreshold = 0.0,
        CancellationToken cancellationToken = default)
    {
        Guard.Against.Null(queryVector);
        Guard.Against.PositiveOrZero(topK);

        var ollama = clientFactory.ResolveClient(OllamaServiceType.Ollama);
        var qdrant = new QdrantClient(
            host: config.Value.Embedding.Host.Host,
            port: (int)config.Value.Embedding.Host.Port,
            apiKey: null,
            useTls: config.Value.Embedding.Host.Scheme == UriScheme.Https);

        var collection = config.Value.Embedding.CollectionName;

        _logger.LogDebug(
            "Searching collection '{Collection}' with topK={TopK}, threshold={Threshold}",
            collection, topK, scoreThreshold);

        var points = await qdrant.SearchAsync(
            collection,
            new Vector(queryVector),
            top: topK,
            filter: filter,
            withPayload: true,
            scoreThreshold: scoreThreshold,
            cancellationToken: cancellationToken
        ).ConfigureAwait(false);

        var hits = points
            .Where(p => p.Score >= scoreThreshold)
            .Select(ToHit)
            .ToList();

        _logger.LogInformation(
            "Search complete: {Count} hits above threshold {Threshold}",
            hits.Count, scoreThreshold);

        return hits;
    }

    private static SearchHit ToHit(SearchResponse point) => new()
    {
        Id = Guid.Parse(point.Id.ToString()),
        Score = point.Score,
        FilePath = point.Payload?.TryGet("FilePath", out var fp) == true
            ? fp.GetString()
            : "unknown",
        Text = point.Payload?.TryGet("Text", out var txt) == true
            ? txt.GetString()
            : string.Empty,
        Payload = point.Payload?.Select(kv => new KeyValuePair<string, object?>(
            kv.Key,
            kv.Value.IsString() ? kv.Value.GetString() : kv.Value)).ToDictionary(kv => kv.Key, kv => kv.Value),
    };
}
```

```csharp
/// and sends the enriched prompt to the LLM for a grounded answer.

    IVectorSearchService searchService,
    IOllamaClientFactory clientFactory,
    IOptions<RagnarConfig> config) : IPipelineStage<RagQueryContext>

    public AsyncRetryPolicy? RetryPolicy => null;
    public bool ShouldRun(RagQueryContext ctx) => ctx.QueryText is not null;

    public async Task ExecuteAsync(RagQueryContext ctx, CancellationToken ct)
    {
        // 1. Embed the user's natural-language query
        var queryVector = (await embeddingService.GenerateAsync(ctx.QueryText, ct)).ToArray();

        // 2. Search the vector store
        var filter = new QdrantFilterBuilder()
            .AddExceptKeywords("Category", "Test", "Tests")
            .AddRange("CommentLength", gte: 20)
            .Build();

        var hits = await searchService.SearchAsync(
            queryVector, topK: config.Value.Generation.TopK, filter: filter, ct: ct);

        // 3. Build a grounded prompt
        var contextBlock = string.Join("\n\n---\n\n",
            hits.Select(h => $"// {h.FilePath} (score: {h.Score:F3})\n{h.Text}"));

        var prompt = $"""
            Using ONLY the code snippets below, answer the question.
            If the answer is not present, say "I don't have enough context."

            ## Retrieved Context
            {contextBlock}

            ## Question
            {ctx.QueryText}
            """;

        var client = clientFactory.ResolveClient(OllamaServiceType.Ollama);
        var response = await client.ChatAsync(
            config.Value.Generation.Model,
            [[new ChatMessage(ChatRole.User, prompt)]],
            new GenerationOptions
            {
                MaxTokens = config.Value.Generation.MaxTokens,
                Temperature = config.Value.Generation.Temperature,
            },
            ct);

        ctx.Citations = hits;
    }
```

## 2. Parallel File Processing with Backpressure

**Problem:** The current `DeltaEmbedStage` and parsing pipeline process documents **sequentially** (one at a time in a `foreach`). For a large monorepo (10 k+ files), this is a major bottleneck. There is no concurrency control, no backpressure, and no thread pool tuning beyond `ThreadCount` in `GenerationOptions`.

**Implementation:**

```csharp
// ── Channel-based bounded producer/consumer ─────────────────────────

/// <summary>
/// Provides a bounded async channel that enforces backpressure
/// between the file-discovery producer and the embed-consumer.
/// </summary>
public sealed class BackpressureChannel<T>(int maxUnbounded = 64) : IDisposable
{
    private readonly Channel<T> _channel = Channel.CreateBounded<T>(new BoundedChannelOptions(maxUnbounded)
    {
        SingleReader = true,
        SingleWriter = false,
        FullMode = BoundedChannelFullMode.Wait,
    });

    /// <summary>Writes an item; blocks if the buffer is full (backpressure).</summary>
    public ValueTask WriteAsync(T item, CancellationToken ct) => _channel.Writer.WriteAsync(item, ct);

    /// <summary>Signals the writer side that no more items will arrive.</summary>
    public void Complete() => _channel.Writer.Complete();

    /// <summary>Reads the next available item.</summary>
    public async ValueTask<(bool HasValue, T Item)> ReadAsync(CancellationToken ct)
    {
        if (await _channel.Reader.WaitToReadAsync(ct))
        {
            var item = _channel.Reader.Read();
            return (true, item);
        }
        return (false, default!);
    }

    public void Dispose() => _channel.Writer.Complete();
}
```

```csharp
// ── Parallel embedding stage ────────────────────────────────────────

/// <summary>
/// Replaces the sequential loop in DeltaEmbedStage with a
/// parallel consumer pool that respects <see cref="GenerationOptions.ThreadCount"/>.
/// </summary>
public sealed class ParallelEmbedStage(
    ILogger logger,
    IEmbeddingService embeddingService,
    IVectorStoreWriter writer,
    IFileFingerprintStore fingerprintStore,
    IOptions<RagnarConfig> options) : IPipelineStage<EmbeddingContext>
{
    public string Name => "Parallel Embed";
    public AsyncRetryPolicy? RetryPolicy => Polly.Policy
        .Handle<HttpRequestException>()
        .Or<TimeoutException>()
        .WaitAndRetryAsync(
            [[TimeSpan.FromSeconds(2), TimeSpan.FromSeconds(5), TimeSpan.FromSeconds(10)]],
            (attempt, delay, ex, _) => logger.LogWarning(
                "Embed attempt {Attempt} failed; retrying in {Delay}: {Ex}", attempt, delay, ex.Message));

    public bool ShouldRun(EmbeddingContext context) => context.Documents is { Count: > 0 };

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken ct)
    {
        var maxParallelism = Math.Clamp(options.Value.Generation.ThreadCount, 1, 64);
        var channel = new BackpressureChannel<CodeDocument>(maxUnbounded: maxParallelism * 2);

        // ── Producer: fingerprint-filter documents and feed the channel ──
        var producerTask = Task.Run(async () =>
        {
            await fingerprintStore.InitializeAsync(ct).ConfigureAwait(false);
            var skipped = 0;

            foreach (var doc in context.Documents)
            {
                ct.ThrowIfCancellationRequested();
                var hash = ComputeSha256(doc.Code);

                if (await fingerprintStore.IsUnchangedAsync(doc.FilePath, hash, ct).ConfigureAwait(false))
                {
                    Interlocked.Increment(ref skipped);
                    continue;
                }

                await fingerprintStore.RecordAsync(doc.FilePath, hash, ct).ConfigureAwait(false);
                await channel.WriteAsync(doc, ct).ConfigureAwait(false);
            }

            logger.LogInformation(
                "Producer finished: {Total} total, {Skipped} unchanged",
                context.Documents.Count, skipped);
        }, ct);

        // ── Consumer pool: embed + upsert in parallel ──
        var upserted = 0;
        var failed = 0;
        var results = new ConcurrentBag<CodeDocument>();

        async Task Consumer(int workerId)
        {
            try
            {
                while (await channel.ReadAsync(ct) is (true, var doc))
                {
                    ct.ThrowIfCancellationRequested();
                    var vectors = await embeddingService.GenerateAsync(doc.Code, ct).ConfigureAwait(false);
                    doc.Vector = vectors.ToArray();
                    results.Add(doc);
                    Interlocked.Increment(ref upserted);
                }
            }
            catch (OperationCanceledException) when (ct.IsCancellationRequested)
            {
                throw;
            }
            catch (Exception ex)
            {
                Interlocked.Increment(ref failed);
                logger.LogError(ex, "Worker {Id} failed: {Msg}", workerId, ex.Message);
            }
        }

        var workers = Enumerable.Range(0, maxParallelism)
            .Select(i => Task.Run(() => Consumer(i), ct))
            .ToArray();

        await producerTask.ConfigureAwait(false);
        channel.Complete(); // signal no more items

        await Task.WhenAll(workers).ConfigureAwait(false);
        await fingerprintStore.FlushAsync(ct).ConfigureAwait(false);

        if (results.Count > 0)
        {
            var batch = results.ToList();
            var result = await writer.UpsertBatchAsync(batch, ct).ConfigureAwait(false);
            logger.LogInformation(
                "Parallel embed complete: {Upserted} vectors upserted, {Failed} workers had errors, {Skipped} skipped",
                result.CalculateSize, failed,
                context.Documents.Count - results.Count);
        }
    }

    private static string ComputeSha256(string content) =>
        Convert.ToHexString(SHA256.HashData(Encoding.UTF8.GetBytes(content))).ToLowerInvariant();
}
```

## 3. Vector Drift Detection & Quality Scoring

**Problem:** Over time, as code evolves, stored embeddings may become semantically inconsistent (e.g., a refactored method whose meaning shifted but whose fingerprint wasn't re-embedded). There is no mechanism to **detect** staleness, **score** embedding quality, or **flag** vectors that should be regenerated.

**Implementation:**

```csharp

/// <summary>Result of a single drift-detection scan.</summary>
public sealed record DriftReport
{
    /// <summary>ISO-8601 timestamp of when the scan ran.</summary>
    public DateTime ScannedAt { get; init; } = DateTime.UtcNow;

    /// <summary>Total vectors inspected.</summary>

    public double AverageSimilarity { get; init; }

    public double MinSimilarity { get; init; }
```

```csharp
// ── Service ─────────────────────────────────────────────────────────

/// <summary>
/// Periodically re-embeds a sample of stored code and compares the new
/// vectors against the stored ones using cosine similarity.  Vectors
/// whose similarity drops below a threshold are flagged for re-embedding.
/// </summary>
public sealed class DriftDetectionService(
    ILogger logger,
    IVectorSearchService searchService,
    IEmbeddingService embeddingService,
    IOptions<RagnarConfig> config,
    IOllamaClientFactory clientFactory)
{
    /// <summary>
    /// Scans a random sample of <paramref name="sampleSize"/> vectors from the
    /// collection, re-embeds their source text, and computes cosine similarity
    /// against the stored vector.
    /// </summary>
    public async Task<DriftReport> ScanAsync(
        int sampleSize = 200,
        double threshold = 0.75,
        CancellationToken ct = default)
    {
        logger.LogInformation(
            "Drift scan starting: sample={Sample}, threshold={Threshold}",
            sampleSize, threshold);

        var qdrant = ResolveQdrant();
        var collection = config.Value.Embedding.CollectionName;

        // 1. Scroll a random sample of points with their payloads + vectors
        var sample = await ScrollSampleAsync(qdrant, collection, sampleSize, ct);

        var drifted = new List<DriftedVector>();
        var sum = 0.0;
        var min = double.MaxValue;

        foreach (var point in sample)
        {
            ct.ThrowIfCancellationRequested();
            var storedVec = point.Vector.ToArray();
            var text = point.Payload?.TryGet("Text", out var t) == true ? t.GetString() : null;
            if (text is null || storedVec.Length == 0)
                continue;

            // 2. Re-embed the current text
            var fresh = (await embeddingService.GenerateAsync(text, ct)).ToArray();

            // 3. Cosine similarity
            var sim = CosineSimilarity(storedVec, fresh);
            sum += sim;
            min = Math.Min(min, sim);

            if (sim < threshold)
            {
                drifted.Add(new DriftedVector
                {
                    PointId = Guid.Parse(point.Id.ToString()),
                    FilePath = point.Payload?.TryGet("FilePath", out var fp) == true ? fp.GetString() : "unknown",
                    StoredSimilarity = 1.0,
                    RecomputedSimilarity = sim,
                    Delta = 1.0 - sim,
                });
            }
        }

        drifted.SortBy(d => d.Delta).IsDescendingThenAscending();

        var report = new DriftReport
        {
            TotalVectors = sample.Count,
            DriftedVectors = drifted.Count,
            AverageSimilarity = sample.Count > 0 ? sum / sample.Count : 0,
            MinSimilarity = min == double.MaxValue ? 1.0 : min,
            Drifted = drifted,
            SimilarityThreshold = threshold,
        };

        logger.LogInformation(
            "Drift scan complete: {Total} inspected, {Drifted} below {Threshold}, avg={Avg:F4}",
            report.TotalVectors, report.DriftedVectors, threshold, report.AverageSimilarity);

        return report;
    }

    /// <summary>Deletes (or marks) the drifted vectors so they can be re-embedded.</summary>
    public async Task<int> PurgeDriftedAsync(DriftReport report, CancellationToken ct)
    {
        if (report.Drifted.Count == 0) return 0;

        var qdrant = ResolveQdrant();
        var ids = report.Drifted.Select(d => d.PointId.ToString()).ToList();
        await qdrant.DeleteAsync(
            config.Value.Embedding.CollectionName,
            new VectorIds { Ids = { ids.Select(Guid.Parse) } },
            cancellationToken: ct);

        logger.LogInformation("Purged {Count} drifted vectors for re-embedding", ids.Count);
        return ids.Count;
    }

    // ── helpers ──

    private static double CosineSimilarity(float[[]] a, float[[]] b)
    {
        if (a.Length != b.Length) throw new ArgumentException("Vector dimensions must match.");
        double dot = 0, magA = 0, magB = 0;
        for (var i = 0; i < a.Length; i++)
        {
            dot += a[[i]] * b[[i]];
            magA += a[[i]] * a[[i]];
            magB += b[[i]] * b[[i]];
        }
        var denom = Math.Sqrt(magA) * Math.Sqrt(magB);
        return denom == 0 ? 0 : dot / denom;
    }

    private QdrantClient ResolveQdrant() => new(
        host: config.Value.Embedding.Host.Host,
        port: (int)config.Value.Embedding.Host.Port,
        apiKey: null,
        useTls: config.Value.Embedding.Host.Scheme == UriScheme.Https);

    private static Task<List<PointStruct>> ScrollSampleAsync(
        QdrantClient qdrant, string collection, int limit, CancellationToken ct)
    {
        // Simplified: use Scroll with a random offset in production
        return Task.FromResult(
            qdrant.Scroll(collection, limit: limit, withPayload: true, withVectors: true,
                cancellationToken: ct).Points.ToList());
    }
}
```

## 4. Structured CLI with Subcommands

**Problem:** The application runs a single monolithic pipeline. There is no way to invoke discrete operations like `embed`, `search`, `status`, `drift`, or `questions` independently. A proper CLI with subcommands, `--dry-run`, `--verbose`, and `--watch` flags would dramatically improve developer ergonomics.

```csharp
                Console.WriteLine($"  {h.Score:F4}  {h.FilePath}");
                Console.WriteLine($"         {h.Text[[..Math.Min(120, h.Text.Length)]]}…");
            }
        }, query, topK, minScore, filter);

        return cmd;
    }

    // ── ragnar drift ────────────────────────────────────────────────
    private static Command DriftCommand()
    {
        var cmd = new Command("drift", "Detect and optionally purge stale vectors");
        var sample = new Option<int>("--sample", getDefaultValue: () => 200);
        var threshold = new Option<double>("--threshold", getDefaultValue: () => 0.75);
        var purge = new Option<bool>("--purge", "Delete drifted vectors after scanning");

        cmd.Add(sample);
        cmd.Add(threshold);
        cmd.Add(purge);

        cmd.SetHandler(async (int sample, double threshold, bool purge, CancellationToken ct) =>
        {
            var host = Host.CreateApplicationBuilder().Build();
            var svc = host.Services.GetRequiredService<DriftDetectionService>();

            var report = await svc.ScanAsync(sample, threshold, ct);

            Console.WriteLine($"Scanned: {report.TotalVectors} | Drifted: {report.DriftedVectors} | Avg: {report.AverageSimilarity:F4}");

            if (report.Drifted.Count > 0)
            {
                Console.WriteLine("\nWorst offenders:");
                foreach (var d in report.Drifted.Take(10))
                    Console.WriteLine($"  {d.RecomputedSimilarity:F4}  {d.FilePath}");
            }

            if (purge && report.Drifted.Count > 0)
            {
                var n = await svc.PurgeDriftedAsync(report, ct);
                Console.WriteLine($"Purged {n} vectors. Re-run 'ragnar embed' to regenerate.");
            }
        }, sample, threshold, purge);

        return cmd;
    }

    // ── ragnar status ───────────────────────────────────────────────
    private static Command StatusCommand()
    {
        var cmd = new Command("status", "Show collection metadata and pipeline health");
        cmd.SetHandler(async (CancellationToken ct) =>
        {
            var host = Host.CreateApplicationBuilder().Build();
            var qdrant = new QdrantClient("localhost", 6333, null, useTls: false);
            var info = await qdrant.CollectionsInfoAsync(ct);
            // print info
        });
        return cmd;
    }
}
```
