```csharp
public static class ValidationExtensions
{
    public static void ValidateExclusions(
        this AbstractValidator<IReadOnlyCollection<string>> validator,
        string propertyName = "Exclusions")
    {
        validator.RuleFor(x => x).NotNull()
            .WithMessage($"{propertyName} cannot be null.");
        
        validator.RuleForEach(x => x)
            .NotEmpty().WithMessage($"{propertyName} items must not be empty.")
            .MaximumLength(100).WithMessage($"{propertyName} items must be ≤100 chars.");
    }
}

// Usage:
public class FileLoadOptionsValidator : AbstractValidator<FileLoadOptions>
{
    public FileLoadOptionsValidator() => 
        RuleFor(x => x.ExcludedFiles).ValidateExclusions("ExcludedFiles");
}
```
