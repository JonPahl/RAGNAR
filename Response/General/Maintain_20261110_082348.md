### 1. `ReplaceStage` Race Condition (Thread Safety)

**Problem:** `ReplaceStage` reads `_stages.IndexOf(...)` *outside* the lock, then calls `RemoveStage` + `AddStage` + `Reorder` (each acquires the lock independently). Another thread can mutate `_stages` between calls.

```csharp
// ❌ BEFORE – three separate lock acquisitions, TOCTOU gap
public IPipeline<TContext> ReplaceStage(string oldName, IPipelineStage<TContext> newStage, IAsyncPolicy? policy = null)
{
    var oldIdx = _stages.IndexOf(_stages.Find(x => x.PipelineStage.Name == oldName)); // ← no lock!
    if (oldIdx == -1) throw new KeyNotFoundException();
    RemoveStage(oldName);   // lock #1
    AddStage(newStage, policy); // lock #2
    return Reorder(newStage.Name, oldIdx); // lock #3
}

// ✅ AFTER – single atomic transaction
public IPipeline<TContext> ReplaceStage(string oldName, IPipelineStage<TContext> newStage, IAsyncPolicy? policy = null)
{
    Guard.Against.Null(newStage, nameof(newStage));

    lock (_gate)
    {
        var oldIdx = _stages.FindIndex(x => x.PipelineStage.Name == oldName);
        if (oldIdx == -1)
            throw new KeyNotFoundException($"Stage '{oldName}' not found.");

        _stages.RemoveAt(oldIdx);

        if (_stages.Any(x => x.PipelineStage.Name == newStage.Name))
            throw new ArgumentException($"Stage '{newStage.Name}' is already registered.", nameof(newStage));

        _stages.Insert(oldIdx, new StageEntry<TContext>(newStage, policy));
    }

    return this;
}
```

### 2. `FileDiscoveryService` – Synchronous I/O in Async Path + `Task.Yield()`

**Problem:** `Directory.EnumerateFiles` blocks the thread; `await Task.Yield()` per iteration adds scheduler overhead for no benefit in a console app.

```csharp
// ❌ BEFORE
public async IAsyncEnumerable<string> GetFilesAsync(string directory, IOptions<RagnarConfig> options,
    [[EnumeratorCancellation]] CancellationToken cancellationToken)
{
    // …
    foreach (var path in Directory.EnumerateFiles(directory, "*.*", SearchOption.AllDirectories))
    {
        cancellationToken.ThrowIfCancellationRequested();
        var fi = new FileInfo(path);
        if (!validator.IsValid(fi, filter)) { skippedCount++; continue; }
        discoveredCount++;
        yield return path;
        await Task.Yield(); // ← unnecessary
    }
}

// ✅ AFTER – use the async enumeration API available in .NET 6+
public IAsyncEnumerable<string> GetFilesAsync(string directory, IOptions<RagnarConfig> options,
    CancellationToken cancellationToken)
{
    var filter = options.Value.FileLoadOptions;
    Guard.Against.Null(filter, nameof(filter));
    Guard.Against.NullOrEmpty(directory, nameof(directory));

    if (!Directory.Exists(directory))
        throw new DirectoryNotFoundException(directory);

    logger.LogInformation("Discovering files in {Directory}", directory);

    var discovered = 0;
    var skipped = 0;

    // Directory.EnumerateFilesAsync doesn't exist yet; wrap the sync call
    // but at least avoid the per-iteration Task.Yield.
    foreach (var path in Directory.EnumerateFiles(directory, "*", SearchOption.AllDirectories))
    {
        cancellationToken.ThrowIfCancellationRequested();
        var fi = new FileInfo(path);

        if (!validator.IsValid(fi, filter))
        {
            skipped++;
            continue;
        }

        discovered++;
        yield return path;
    }

    logger.LogInformation("Done: {Found} eligible, {Skipped} excluded", discovered, skipped);
}
```

> If targeting .NET 8+, prefer `Directory.EnumerateFilesAsync` (added in .NET 8) to truly offload I/O.

### 3. `CodeParserFactory` – Silent Fallback + Missing Multi-Extension Support

**Problem:** Unknown extensions silently fall back to `MarkdownFileParser`. Multi-extension parsers (e.g. `.cs` *and* `.csx`) are ignored because only `[[0]]` is indexed.

```csharp
// ❌ BEFORE
public class CodeParserFactory(ILogger logger, IOptions<RagnarConfig> options, IEnumerable<IFileParser> parsers)
    : IFileParseFactory
{
    private readonly Dictionary<string, IFileParser> _byExt =
        parsers.ToDictionary(p => p.SupportedExtensions[[0]], StringComparer.OrdinalIgnoreCase);

    public async Task<IEnumerable<CodeDocument>> ParseAsync(string file, CancellationToken ct)
    {
        var ext = Path.GetExtension(file);
        if (!_byExt.TryGetValue(ext, out var parser))
            parser = new MarkdownFileParser(options, logger); // ← silent fallback, new instance each call!
        // …
    }
}

// ✅ AFTER – explicit fallback with logging, full extension mapping, cached default
public sealed class CodeParserFactory(ILogger logger, IOptions<RagnarConfig> options,
    IEnumerable<IFileParser> parsers) : IFileParseFactory
{
    private readonly Dictionary<string, IFileParser> _byExt;
    private readonly IFileParser _fallback;

    public CodeParserFactory(ILogger logger, IOptions<RagnarConfig> options,
        IEnumerable<IFileParser> parsers)
    {
        _logger = logger;
        _fallback = new MarkdownFileParser(options, logger); // cached

        _byExt = new Dictionary<string, IFileParser>(StringComparer.OrdinalIgnoreCase);
        foreach (var parser in parsers)
        {
            foreach (var ext in parser.SupportedExtensions)
            {
                _byExt[[ext]] = parser; // all extensions, not just [[0]]
            }
        }
    }

    private readonly ILogger _logger;

    public async Task<IEnumerable<CodeDocument>> ParseAsync(string file, CancellationToken ct)
    {
        var ext = Path.GetExtension(file);
        var parser = _byExt.TryGetValue(ext, out var matched) ? matched : _fallback;

        if (parser == _fallback)
            _logger.LogWarning("No parser for '{Ext}'; falling back to {Fallback}",
                ext, _fallback.GetType().Name);
        else
            _logger.LogDebug("Dispatching '{Ext}' → {Parser}", ext, parser.GetType().Name);

        return await parser.ParseAsync(file, ct).ConfigureAwait(false);
    }
}
```

```csharp
// ✅ AFTER – remove dead code, keep a single method
public QdrantFilterBuilder AddExceptKeywords(string field, params string[[]] keywords)
{
    Guard.Against.NullOrWhiteSpace(field, nameof(field));
    if (keywords.Length == 0)
        return this;

    _must.Add(new Condition
    {
        Field = new FieldCondition
        {
            Key = field,
            Match = new Match
            {
                ExceptKeywords = new RepeatedStrings { Strings = { keywords } }
            }
        }
    });
    return this;
}
```

### 6. `StageTimingMiddleware` + `ProgressMiddleware` – Duplicate Timing

Both middlewares start their own `Stopwatch` and write elapsed time. Pick one as the source of truth:

```csharp
// ✅ Let ProgressMiddleware read the metric populated by StageTimingMiddleware
public sealed class ProgressMiddleware(IOutputWriter writer)
{
    public async Task Invoke<TContext>(
        StageExecutionContext<TContext> context, Func<Task> next,
        CancellationToken cancellationToken) where TContext : class
    {
        var label = $"[[{context.StageIndex + 1}/{context.StageTotal}]] {context.Stage.Name}";
        writer.WriteLine($"{label} …");

        try
        {
            await next().ConfigureAwait(false);
            // StageTimingMiddleware already populated context.Metric
            var elapsed = context.Metric?.Duration ?? TimeSpan.Zero;
            writer.WriteLine($"  ✓ {elapsed:hh\\:mm\\:ss\\.fff}");
        }
        catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
        {
            writer.WriteLine("  ✗ cancelled");
            throw;
        }
        catch (Exception ex)
        {
            writer.WriteLine($"  ✗ {ex.GetType().Name}: {ex.Message}");
            throw;
        }

        writer.WriteLine();
    }
}
```

### 7. Inconsistent Validation Strategy

The codebase mixes **FluentValidation** (`AbstractValidator<FileLoadOptions>`) with **`System.ComponentModel.DataAnnotations`** attributes (`[[Range]]`, `[[Length]]`) and **`Guard.Against`** (Guard.NET). Pick one primary strategy:

| Use-case | Recommended |
|----------|-------------|
| DTO / options records bound from config | `DataAnnotations` + `Validator.TryValidateObject` |
| Domain objects with cross-field rules | FluentValidation |
| Argument validation inside methods | `ArgumentNullException.ThrowIfNull` / `ArgumentOutOfRangeException` (built-in, zero-dependency) |

**Fix example – replace `Guard.Against` with built-ins for zero-dependency paths:**

```csharp
// ❌ BEFORE (requires Guard.NET package)
Guard.Against.Null(context);
Guard.Against.Null(next);
Guard.Against.NullOrWhiteSpace(field, nameof(field));

// ✅ AFTER (BCL only)
ArgumentNullException.ThrowIfNull(context);
ArgumentNullException.ThrowIfNull(next);
ArgumentException.ThrowIfNullOrWhiteSpace(field);
```
