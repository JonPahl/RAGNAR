using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Options;
using Serilog;
using Spectre.Console;

namespace Ragnar;

/// <summary>
/// Application entry point. Orchestrates configuration, DI, pipeline assembly, and execution.
/// </summary>
internal static class Program
{
    // ─────────────────────────────────────────────
    //  Entry Point
    // ─────────────────────────────────────────────

    static async Task<int> Main(string[] args)
    {
        try
        {
            await using var host = BuildHost(args).ConfigureAwait(false);
            await host.RunAsync().ConfigureAwait(false);
            return 0;
        }
        catch (ValidationException ex)
        {
            AnsiConsole.Write(RenderValidationErrors(ex));
            return 1;
        }
        catch (OperationCanceledException)
        {
            AnsiConsole.MarkupLine("[yellow]Operation cancelled by user.[/]");
            return 130; // SIGINT convention
        }
        catch (Exception ex)
        {
            AnsiConsole.Write(RenderFatalError(ex));
            return 1;
        }
    }

    // ─────────────────────────────────────────────
    //  Host Construction
    // ─────────────────────────────────────────────

    /// <summary>Builds and starts the generic host, wiring up logging, config, DI, and the pipeline.</summary>
    private static async Task<IHost> BuildHost(string[] args)
    {
        var builder = Host.CreateApplicationBuilder(args);

        ConfigureLogging(builder);
        ConfigureConfiguration(builder);
        ConfigureServices(builder.Services);

        await using var host = builder.Build().ConfigureAwait(false);
        await host.StartAsync().ConfigureAwait(false);

        await RunPipelineAsync(host, builder.Configuration).ConfigureAwait(false);

        return host;
    }

    // ─────────────────────────────────────────────
    //  Logging
    // ─────────────────────────────────────────────

    /// <summary>Initialises Serilog with console + file sinks.</summary>
    private static void ConfigureLogging(HostApplicationBuilder builder)
    {
        builder.Services.AddLogging(logging =>
        {
            logging.ClearProviders();
            logging.AddSerilog(CreateLoggerConfiguration(), dispose: true);
        });
    }

    private static LoggerConfiguration CreateLoggerConfiguration()
    {
        return new LoggerConfiguration()
            .MinimumLevel.Information()
            .Enrich.FromLogContext()
            .WriteTo.Console(outputTemplate: "[{Timestamp:HH:mm:ss} {Level:u3}] {Message:lj}{NewLine}{Exception}")
            .WriteTo.File(
                path: "logs/ragnar-.log",
                rollingInterval: RollingInterval.Day,
                retainedFileCountLimit: 14,
                outputTemplate: "[{Timestamp:yyyy-MM-dd HH:mm:ss} {Level:u3}] {Message:lj} @ {SourceContext}{NewLine}{Exception}");
    }

    // ─────────────────────────────────────────────
    //  Configuration & Validation
    // ─────────────────────────────────────────────

    /// <summary>Binds strongly-typed options and validates them before any work begins.</summary>
    private static void ConfigureConfiguration(HostApplicationBuilder builder)
    {
        builder.Services.AddOptions<RagnarConfig>()
            .Bind(builder.Configuration.GetSection("Ragnar"))
            .ValidateDataAnnotations()
            .ValidateOnStart();

        // Register child option sections so IOptions<T> resolution works for each sub-section.
        builder.Services.AddOptions<ApplicationOptions>()
            .Bind(builder.Configuration.GetSection("Ragnar:ApplicationOptions"))
            .ValidateDataAnnotations();

        builder.Services.AddOptions<EmbeddingOptions>()
            .Bind(builder.Configuration.GetSection("Ragnar:EmbeddingOptions"))
            .ValidateDataAnnotations();

        builder.Services.AddOptions<OllamaOptions>()
            .Bind(builder.Configuration.GetSection("Ragnar:OllamaOptions"))
            .ValidateDataAnnotations();

        builder.Services.AddOptions<FileLoadOptions>()
            .Bind(builder.Configuration.GetSection("Ragnar:FileLoadOptions"))
            .ValidateDataAnnotations();

        // Register the composite validator (uses child validators internally).
        builder.Services.AddSingleton<AbstractValidator<RagnarConfig>, RagnarConfigValidator>();
    }

    // ─────────────────────────────────────────────
    //  Dependency Injection
    // ─────────────────────────────────────────────

    /// <summary>Registers all application services into the DI container.</summary>
    private static void ConfigureServices(IServiceCollection services)
    {
        // Core infrastructure
        RegisterCoreServices(services);

        // Embedding & vector-store pipeline
        services.EmbeddingSetup();

        // Question sources (built-in + plugin DLLs)
        RegisterQuestionServices(services);

        // Pipeline stages
        RegisterPipelineStages(services);
    }

    private static void RegisterCoreServices(IServiceCollection services)
    {
        services.AddSingleton<IOutputWriter, AnsiOutputWriter>();
        services.AddSingleton<IApplicationHeader, ApplicationHeader>();
        services.AddSingleton<IFileValidator, DefaultFileValidator>();
        services.AddSingleton<IFileDiscoveryService, FileDiscoveryService>();
        services.AddSingleton<IFileParseFactory, CodeParserFactory>();
        services.AddSingleton<ISyntaxTreeChunker, RoslynSyntaxTreeChunker>();
        services.AddSingleton<IQdrantClient, QdrantClientFactory.Create>;
        services.AddSingleton<IOllamaClientFactory, OllamaClientFactory>();
        services.AddSingleton<IContextRetriever, QdrantContextRetriever>();
        services.AddSingleton<IRagOrchestrator, RagOrchestrator>();
        services.AddSingleton<IQuestionCatalogLoader, QuestionCatalogLoader>();
        services.AddSingleton<IQuestionBuilder, QuestionBuilder>();
        services.AddSingleton<IQuestionSourceBuilder, QuestionCombineBuilder>();
    }

    private static void RegisterQuestionServices(IServiceCollection services)
    {
        // Built-in question sources
        services.AddTransient<IQuestionSource, BuiltinQuestionSource>();

        // Discover and register plugin DLLs from Questions/Plugins/
        services.LoadQuestionPlugins();
    }

    private static void RegisterPipelineStages(IServiceCollection services)
    {
        // Embedding pipeline stages
        services.AddSingleton<IPipelineStage<EmbeddingContext>, BrandingStage>();
        services.AddSingleton<IPipelineStage<EmbeddingContext>, FileParseStage>();
        services.AddSingleton<IPipelineStage<EmbeddingContext>, EmbeddingWriteStage>();
        services.AddSingleton<IPipelineStage<EmbeddingContext>, ShowFileStage>();

        // Question pipeline stages
        services.AddSingleton<IPipelineStage<QuestionPipelineContext>, QuestionLoadStage>();
        services.AddSingleton<IPipelineStage<QuestionPipelineContext>, QuestionShowStage>();
        services.AddSingleton<IPipelineStage<QuestionPipelineContext>, QuestionExecutionStage>();
    }

    // ─────────────────────────────────────────────
    //  Pipeline Execution
    // ─────────────────────────────────────────────

    /// <summary>Assembles and executes both the embedding and question pipelines sequentially.</summary>
    private static async Task RunPipelineAsync(IHost host, IConfiguration configuration)
    {
        using var cts = new CancellationTokenSource();
        Console.CancelKeyPress += (_, e) =>
        {
            e.Cancel = true;
            cts.Cancel();
        };

        await RunEmbeddingPipelineAsync(host, cts.Token).ConfigureAwait(false);
        await RunQuestionPipelineAsync(host, cts.Token).ConfigureAwait(false);
    }

    /// <summary>Executes the embedding pipeline: discover → parse → embed → write to Qdrant.</summary>
    private static async Task RunEmbeddingPipelineAsync(IHost host, CancellationToken ct)
    {
        var logger = host.Services.GetRequiredService<Serilog.ILogger>();
        var runner = BuildEmbeddingPipeline(host.Services);
        var context = new EmbeddingContext();

        logger.Information("═══ Embedding Pipeline Start ═══");
        await runner.ExecuteAsync(context, ct).ConfigureAwait(false);
        logger.Information("═══ Embedding Pipeline Complete ═══");
    }

    /// <summary>Executes the question pipeline: load questions → optional display → RAG execution.</summary>
    private static async Task RunQuestionPipelineAsync(IHost host, CancellationToken ct)
    {
        var logger = host.Services.GetRequiredService<Serilog.ILogger>();
        var runner = BuildQuestionPipeline(host.Services);
        var context = new QuestionPipelineContext();

        logger.Information("═══ Question Pipeline Start ═══");
        await runner.ExecuteAsync(context, ct).ConfigureAwait(false);
        logger.Information("═══ Question Pipeline Complete ═══");
    }

    // ─────────────────────────────────────────────
    //  Pipeline Factories
    // ─────────────────────────────────────────────

    /// <summary>Constructs the ordered embedding pipeline with retry policies where appropriate.</summary>
    private static PipelineRunner<EmbeddingContext> BuildEmbeddingPipeline(IServiceProvider sp)
    {
        var logger = sp.GetRequiredService<Serilog.ILogger>();

        var runner = new PipelineRunner<EmbeddingContext>(logger);

        runner
            .AddStage(sp.GetRequiredService<BrandingStage>())
            .AddStage(sp.GetRequiredService<FileParseStage>(),
                RetryPolicy.ForHttp())
            .AddStage(sp.GetRequiredService<EmbeddingWriteStage>(),
                RetryPolicy.ForHttp())
            .AddStage(sp.GetRequiredService<ShowFileStage>());

        return runner;
    }

    /// <summary>Constructs the ordered question pipeline with retry policies where appropriate.</summary>
    private static PipelineRunner<QuestionPipelineContext> BuildQuestionPipeline(IServiceProvider sp)
    {
        var logger = sp.GetRequiredService<Serilog.ILogger>();

        var runner = new PipelineRunner<QuestionPipelineContext>(logger);

        runner
            .AddStage(sp.GetRequiredService<QuestionLoadStage>())
            .AddStage(sp.GetRequiredService<QuestionShowStage>())
            .AddStage(sp.GetRequiredService<QuestionExecutionStage>(),
                sp.GetRequiredService<QuestionExecutionStage>().RetryPolicy);

        return runner;
    }

    // ─────────────────────────────────────────────
    //  Error Rendering Helpers
    // ─────────────────────────────────────────────

    private static Markup RenderValidationErrors(ValidationException ex)
    {
        var table = new Table()
            .Title("[red]Configuration Validation Failed[/]")
            .BorderStyle(Style.Parse("red"))
            .AddColumn("Field")
            .AddColumn("Error");

        foreach (var err in ex.Errors)
            table.AddRow(err.PropertyName, err.ErrorMessage ?? "—");

        return table;
    }

    private static Markup RenderFatalError(Exception ex)
    {
        return new()
        {
            new Rule("[red]Unhandled Exception[/]"),
            new Text(ex.Message).Style(Style.Parse("bold red")),
            new Paragraph(ex.StackTrace ?? "").Style(Style.Parse("dim")),
            new NewLine(),
        };
    }
}

/// <summary>Factory for common Polly retry policies.</summary>
internal static class RetryPolicy
{
    /// <summary>Exponential-backoff retry for transient HTTP/timeout failures (3 attempts).</summary>
    public static AsyncRetryPolicy ForHttp() =>
        Polly.Policy
            .Handle<HttpRequestException>()
            .Or<TimeoutException>()
            .WaitAndRetryAsync(3, attempt => TimeSpan.FromSeconds(Math.Pow(2, attempt)));
}

## Key Improvements Over a Typical Monolithic `Program.cs`

| Concern | How it's addressed |
|---|---|
| **Readability** | The file is divided into clearly-labelled sections (`── Logging ──`, `── DI ──`, `── Pipeline ──`, etc.). Each method does *one* thing. |
| **Maintainability** | All DI registrations live in `ConfigureServices` → `RegisterCoreServices` / `RegisterQuestionServices` / `RegisterPipelineStages`. Adding a new service or stage is a one-line change in the right method. |
| **Separation of concerns** | Pipeline *assembly* (which stages, in what order, with what retry policy) is isolated in `BuildEmbeddingPipeline` / `BuildQuestionPipeline`. Swapping stage order or adding a stage never touches `Main`. |
| **Error handling** | `Main` is a thin try/catch that maps exception types to exit codes and user-friendly console output. No business logic is mixed in. |
| **CancellationToken propagation** | A single `CancellationTokenSource` is created in `RunPipelineAsync` and flows through both pipelines and the console cancel handler. |
| **Configuration** | All `IOptions<T>` binding and validation happens in `ConfigureConfiguration`, so the rest of the code simply resolves typed options. |
| **Extensibility** | New stages are registered in `RegisterPipelineStages` and added in the builder methods. New question sources plug in via `LoadQuestionPlugins` (already in your extension). New embedding backends only need a new `IEmbeddingService` registration. |
