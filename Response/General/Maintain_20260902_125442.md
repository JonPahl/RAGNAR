## 6. Inconsistent Null-Guarding Strategy

```csharp
// ✅ AFTER — one policy: GuardNull (or Guard.Against) at the top of every public API
using Guards;

public sealed class FileValidator : IFileValidator
{
    public bool IsValid(FileInfo file, in FileLoadOptions opts)
    {
        Guard.Against.Null(file, nameof(file));
        Guard.Against.Null(opts, nameof(opts));

        var dir   = file.DirectoryName ?? string.Empty;
        var name  = file.Name;
        var ext   = file.Extension;

        return opts.AllowedFileExtensions.Contains(ext, StringComparer.OrdinalIgnoreCase)
            && !opts.ExcludedFiles.Contains(name, StringComparer.OrdinalIgnoreCase)
            && !opts.ExcludedDirectories.Any(d =>
                  dir.Equals(d, StringComparison.OrdinalIgnoreCase) ||
                  dir.StartsWith($"{d}{Path.DirectorySeparatorChar}", StringComparison.OrdinalIgnoreCase));
    }
}
```

**Problem:** The `foreach` loop **returns on the first document**, silently skipping the rest, and swallows non-cancellation exceptions.

```csharp
// ✅ AFTER — process the whole batch; report partial failure
public async Task<UpdateResult> UpsertBatchAsync(CodeDocument[[]] docs, CancellationToken ct)
{
    Guard.Against.Null(docs);
    ct.ThrowIfCancellationRequested();

    var points = new List<PointStruct>();
    var failures = 0;

    foreach (var doc in docs)
    {
        ct.ThrowIfCancellationRequested();
        try
        {
            var vector = await _embeddingService.GenerateAsync(doc.ToEmbeddableText(), ct);
            points.AddRange(_generator.BuildPointStructs(doc.AsPoint(), vector.ToArray(), doc));
        }
        catch (Exception ex) when (ex is not OperationCanceledException)
        {
            logger.Error(ex, "Embedding failed for element {Name}. Skipping.", doc.ElementName);
            failures++;
        }
    }

    if (points.Count == 0)
        return new UpdateResult { Status = UpdateStatus.UnknownUpdateStatus };

    var result = await _qdrant.UpsertAsync(_appOptions.VectorStoreName, points, ct);

    return new UpdateResult
    {
        Status  = failures == 0 ? UpdateStatus.Completed : UpdateStatus.PartiallyCompleted,
        Details = failures > 0 ? $"{failures}/{docs.Length} documents failed." : null
    };
}
```
