// File: RagSession.cs
namespace Ragnar.Services;

/// <summary>
/// Stateful RAG conversation session with multi-turn context and confidence scoring.
/// </summary>
/// <param name="retriever">Service that performs similarity search against Qdrant.</param>
/// <param name="orchestrator">Service that composes prompts and invokes the LLM.</param>
/// <param name="options">Application configuration (vector store name, top-k, etc.).</param>
/// <param name="logger">Structured logger for diagnostic messages.</param>
/// <param name="writer">Console output writer for interactive rendering.</param>
public sealed class RagSession(
    IContextRetriever retriever,
    IRagOrchestrator orchestrator,
    IOptions<RagnarConfig> options,
    ILogger<RagSession> logger,
    IOutputWriter writer) : IRagSession
{
    private readonly List<ChatMessage> _history = [[]];
    private string? _collectionName;
    private int _maxTurns;
    private int _turnCount;

    /// <inheritdoc />
    public bool IsActive => _collectionName is not null;

    /// <inheritdoc />
    public IReadOnlyList<ChatMessage> History => _history.AsReadOnly();

    /// <inheritdoc />
    public Task BeginAsync(string vectorStoreName, int maxTurns = 10, CancellationToken cancellationToken = default)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(vectorStoreName);
        if (maxTurns < 1) throw new ArgumentOutOfRangeException(nameof(maxTurns), maxTurns, "Must be ≥ 1.");

        _collectionName = vectorStoreName;
        _maxTurns = maxTurns;
        _turnCount = 0;
        _history.Clear();

        // Seed the system message so the LLM knows its role.
        _history.Add(new ChatMessage(ChatRole.System,
            "You are Ragnar, a code-analysis assistant. Answer questions using ONLY the provided " +
            "retrieved context. If the context is insufficient, say so explicitly. Cite the source " +
            "file when referencing code. Keep responses under 400 words unless the user requests detail."));

        logger.LogInformation("Session started: collection='{Col}', maxTurns={MT}", vectorStoreName, maxTurns);
        return Task.CompletedTask;
    }

    /// <inheritdoc />
    public async Task<RagResponse> AskAsync(string question, QuestionCategory? category = null, CancellationToken cancellationToken = default)
    {
        Guard.Against.NullOrWhiteSpace(question);
        if (!IsActive) throw new InvalidOperationException("Session is not active. Call BeginAsync first.");

        cancellationToken.ThrowIfCancellationRequested();
        _turnCount++;

        // 1. Build the retrieval filter for the current question.
        Filter? filter = category.HasValue ? BuildCategoryFilter(category.Value) : null;

        // 2. Retrieve relevant document chunks.
        var stopwatch = System.Diagnostics.Stopwatch.StartNew();
        var context = await retriever
            .RetrieveContextAsync(_collectionName!, cancellationToken, filter)
            .ConfigureAwait(false);
        var retrievalMs = stopwatch.ElapsedMilliseconds;

        logger.LogDebug("Retrieved {Chars} chars of context in {Ms} ms", context.Length, retrievalMs);

        // 3. Compose the full prompt with history for multi-turn awareness.
        var prompt = BuildPromptWithHistory(question, context);

        // 4. Call the LLM via the orchestrator.
        var answer = await orchestrator
            .GenerateAsync(prompt, cancellationToken)
            .ConfigureAwait(false);

        // 5. Compute a simple confidence proxy based on retrieval length vs. question length.
        var confidence = ComputeConfidence(context, question);

        // 6. Record the turn.
        _history.Add(new ChatMessage(ChatRole.User, question));
        _history.Add(new ChatMessage(ChatRole.Assistant, answer));
        TrimHistory();

        return new RagResponse
        {
            Question = question,
            Answer = answer,
            RetrievedContext = context,
            Confidence = confidence,
            RetrievalDurationMs = retrievalMs,
            Category = category
        };
    }

    /// <inheritdoc />
    public Task EndAsync(string? exportPath = null, CancellationToken cancellationToken = default)
    {
        if (!IsActive) return Task.CompletedTask;

        if (exportPath is not null)
        {
            ExportTranscript(exportPath);
            logger.LogInformation("Transcript exported to {Path}", exportPath);
        }

        _collectionName = null;
        _history.Clear();
        _turnCount = 0;
        return Task.CompletedTask;
    }

    // ───────────────────────── Private helpers ─────────────────────────

    private string BuildPromptWithHistory(string question, string context)
    {
        var sb = new System.Text.StringBuilder();
        sb.AppendLine("## Retrieved Context");
        sb.AppendLine(context);
        sb.AppendLine();

        // Append recent turns (last 3) for conversational context.
        var recentTurns = _history.Where(m => m.Role != ChatRole.System).ToList();
        if (recentTurns.Count > 0)
        {
            sb.AppendLine("## Previous Conversation");
            foreach (var msg in recentTurns.TakeLast(6))
            {
                sb.AppendLine($"{msg.Role}: {msg.Content}");
            }

            sb.AppendLine();
        }

        sb.AppendLine("## Current Question");
        sb.AppendLine(question);
        return sb.ToString();
    }

    private void TrimHistory()
    {
        // Keep system message + last N*2 messages.
        var maxMessages = _maxTurns * 2 + 1; // +1 for system
        while (_history.Count > maxMessages)
        {
            // Remove the oldest non-system message.
            var idx = _history.FindIndex(m => m.Role != ChatRole.System);
            if (idx == -1) break;
            _history.RemoveAt(idx);
        }
    }

    private static double ComputeConfidence(string context, string question)
    {
        // Heuristic: ratio of unique word overlap between question and context.
        if (string.IsNullOrWhiteSpace(context)) return 0.0;
        var qWords = new HashSet<string>(question.Split(' ', StringSplitOptions.RemoveEmptyEntries), StringComparer.OrdinalIgnoreCase);
        var cWords = new HashSet<string>(context.Split(' ', StringSplitOptions.RemoveEmptyEntries), StringComparer.OrdinalIgnoreCase);
        var overlap = qWords.Count(w => cWords.Contains(w));
        return Math.Min(1.0, (double)overlap / Math.Max(1, qWords.Count));
    }

    private Filter BuildCategoryFilter(QuestionCategory category)
        => new(FieldCondition(Key: "Category", Match: new MatchValue(value: category.ToString())));

    private void ExportTranscript(string path)
    {
        var dir = System.IO.Path.GetDirectoryName(path);
        if (dir is not null && !Directory.Exists(dir)) Directory.CreateDirectory(dir);

        using var sw = new StreamWriter(path);
        sw.WriteLine("# Ragnar Session Transcript");
        sw.WriteLine($"Generated: {DateTime.UtcNow:O}");
        sw.WriteLine();
        foreach (var msg in History)
        {
            sw.WriteLine($"### {msg.Role}");
            sw.WriteLine(msg.Content);
            sw.WriteLine();
        }
    }
}

using System.Text;
using System.Text.Json;
using System.Text.Json.Serialization;

namespace Ragnar.Services;

/// <summary>
/// Renders RAG results into Markdown, HTML, or JSON report files.
/// </summary>
/// <param name="logger">Logger for export diagnostics.</param>
/// <param name="options">Application options supplying the app name and version for the report header.</param>
public sealed class ReportExporter(ILogger<ReportExporter> logger, IOptions<RagnarConfig> options) : IReportExporter
{
    private static readonly JsonSerializerOptions JsonOpts = new()
    {
        WriteIndented = true,
        Converters = { new JsonStringEnumConverter() }
    };

    /// <inheritdoc />
    public Task<string> ExportAsync(
        IReadOnlyList<RagResponse> results,
        ReportFormat format,
        string outputPath,
        CancellationToken cancellationToken = default)
    {
        Guard.Against.Null(results);
        Guard.Against.NullOrWhiteSpace(outputPath);

        cancellationToken.ThrowIfCancellationRequested();

        var dir = System.IO.Path.GetDirectoryName(outputPath);
        if (dir is not null && !Directory.Exists(dir))
            Directory.CreateDirectory(dir);

        string content = format switch
        {
            ReportFormat.Markdown => RenderMarkdown(results),
            ReportFormat.Html => RenderHtml(results),
            ReportFormat.Json => RenderJson(results),
            _ => throw new ArgumentOutOfRangeException(nameof(format), format, null)
        };

        File.WriteAllText(outputPath, content);
        logger.LogInformation("Report written: {Path} ({Format}, {Count} entries)", outputPath, format, results.Count);
        return Task.FromResult(Path.GetFullPath(outputPath));
    }

    // ───────────────────────── Renderers ─────────────────────────

    private string RenderMarkdown(IReadOnlyList<RagResponse> results)
    {
        var sb = new StringBuilder();
        sb.AppendLine($"# Ragnar Analysis Report");
        sb.AppendLine();
        sb.AppendLine($"| | |");
        sb.AppendLine($"|---|---|");
        sb.AppendLine($"|**Generated**|{DateTime.UtcNow:yyyy-MM-dd HH:mm:ss} UTC|");
        sb.AppendLine($"|**Questions**|{results.Count}|");
        sb.AppendLine($"|**Avg Confidence|{results.Average(r => r.Confidence):P1}|");
        sb.AppendLine();

        for (var i = 0; i < results.Count; i++)

            var r = results[[i]];
            sb.AppendLine($"## {i + 1}. {r.Question}");
            sb.AppendLine();
            sb.AppendLine($"- **Category:** {r.Category?.ToString() ?? "General"}");
            sb.AppendLine($"- **Confidence:** {r.Confidence:P1}");
            sb.AppendLine($"- **Retrieval time:** {r.RetrievalDurationMs} ms");
            sb.AppendLine();
            sb.AppendLine("### Answer");
            sb.AppendLine();
            sb.AppendLine(r.Answer);
            sb.AppendLine();
            sb.AppendLine("<details>");
            sb.AppendLine("<summary>Retrieved context</summary>");
            sb.AppendLine();
            sb.AppendLine("```");
            sb.AppendLine(r.RetrievedContext);
            sb.AppendLine("```");
            sb.AppendLine("</details>");
            sb.AppendLine();
            sb.AppendLine("---");
            sb.AppendLine();
        }

    }

    private string RenderHtml(IReadOnlyList<RagResponse> results)
    {
        var sb = new StringBuilder();
        sb.AppendLine("<!DOCTYPE html><html lang=\"en\"><head><meta charset=\"utf-8\">");
        sb.AppendLine($"<title>Ragnar Report – {DateTime.UtcNow:yyyy-MM-dd}</title>");

        sb.AppendLine(".high{background:#d4edda;color:#155724}.mid{background:#fff3cd;color:#856404}.low{background:#f8d7da;color:#721c24}");

        sb.AppendLine("details{margin:.5rem 0}");
        sb.AppendLine("</style></head><body>");
        sb.AppendLine($"<h1>Ragnar Analysis Report</h1>");
        sb.AppendLine($"<p>Generated {DateTime.UtcNow:yyyy-MM-dd HH:mm:ss} UTC · {results.Count} question(s)</p>");

        foreach (var r in results)
        {
            var level = r.Confidence >= 0.7 ? "high" : r.Confidence >= 0.4 ? "mid" : "low";
            sb.AppendLine($"<div class=\"question\">");
            sb.AppendLine($"<h2>{System.Net.WebUtility.HtmlEncode(r.Question)}</h2>");
            sb.AppendLine($"<span class=\"confidence {level}\">{r.Confidence:P1}</span> " +

            sb.AppendLine($"<p>{System.Net.WebUtility.HtmlEncode(r.Answer)}</p>");
            sb.AppendLine($"<details><summary>Context</summary><pre>{System.Net.WebUtility.HtmlEncode(r.RetrievedContext)}</pre></details>");
            sb.AppendLine("</div>");
        }

        sb.AppendLine("</body></html>");
        return sb.ToString();
    }

using Spectre.Console;

namespace Ragnar.Services;

/// <summary>
/// Renders a live Spectre.Console progress bar with ETA, percentage, and status text.
/// </summary>
/// <param name="writer">Console output writer for surrounding rules.</param>
/// <param name="logger">Logger for fallback diagnostics when ANSI is unavailable.</param>
public sealed class SpectreProgressTracker(IOutputWriter writer, ILogger<SpectreProgressTracker> logger) : IProgressTracker
{
    /// <inheritdoc />
    public Task ReportAsync(
        int total,
        CancellationToken cancellationToken,
        Func<ProgressReporter, Task> callback,
        string? label = null)
    {
        if (total <= 0) return Task.CompletedTask;

        var tcs = new TaskCompletionSource(TaskCreationOptions.RunContinuationsAsynchronously);

        var console = AnsiConsole.Create(new ConsoleSettings
        {
            Ansi = AnsiSupport.Detect,
            Profile = new ConsoleProfile
            {
                Width = Math.Min(120, Environment.SystemConsole?.Width ?? 80),
                Height = 40,
                SupportsAnsi = true
            }
        });

        var task = Task.Run(async () =>
        {
            await console.Progress()
                .Columns(
                    new TaskColumn(),
                    new BarColumn(),
                    new PercentageColumn(),
                    new RemainingTimeColumn(),
                    new TextColumn())
                .AutoClear(true)
                .StartAsync(async ctx =>
                {
                    var displayLabel = label ?? "Processing…";
                    var task = ctx.AddTask(displayLabel, total: total, AutoStart: true);

                    var reporter = new ProgressReporter((val, status) =>
                    {
                        task.Increment(val - task.Value);
                        if (status is not null) task.Description = status;
                    });

                    try
                    {
                        await callback(reporter);
                    }
                    finally
                    {
                        task.StopTask();
                    }
                });
        }, cancellationToken);

        await task;
        return Task.CompletedTask;
    }
}

// File: DeltaEmbedStage.cs (modified ExecuteAsync excerpt)
public sealed class DeltaEmbedStage(
    ILogger logger,
    IFileFingerprintStore fingerprintStore,
    IVectorStoreWriter vectorWriter,
    IOptions<RagnarConfig> options,
    IProgressTracker tracker) : IPipelineStage<EmbeddingContext>
{
    public string Name => "Embedding files…";
    public AsyncRetryPolicy? RetryPolicy => Policy.Handle<HttpRequestException>()
        .WaitAndRetryAsync(3, a => TimeSpan.FromSeconds(Math.Pow(2, a)));

    public bool ShouldRun(EmbeddingContext context) => context.Documents.Count > 0;

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        var chunks = context.Documents;
        var total = chunks.Count;
        var batchSize = Math.Max(1, options.Value.EmbeddingOptions.BatchSize ?? 64);

        logger.LogInformation("Embedding {Total} chunks (batch size {BS})", total, batchSize);

        await tracker.ReportAsync(total, cancellationToken, async reporter =>
        {
            var completed = 0;

            for (var i = 0; i < total; i += batchSize)
            {
                cancellationToken.ThrowIfCancellationRequested();
                var batch = chunks.Skip(i).Take(batchSize).ToList();

                var vectors = await vectorWriter.GenerateBatchAsync(
                    batch.Select(d => d.Code).ToList(), cancellationToken)
                    .ConfigureAwait(false);

                // Upsert each vector + payload to Qdrant.
                for (var j = 0; j < batch.Count; j++)
                {
                    await vectorWriter.UpsertAsync(
                        batch[[j]],
                        vectors.Content[[j]].Vector.ToArray(),
                        cancellationToken
                    ).ConfigureAwait(false);

                    completed++;
                }

                reporter.Report(completed, $"{completed}/{total} chunks embedded");
            }
        }, label: $"Embedding {total:N0} chunks");

        logger.LogInformation("Embedding complete: {Total} vectors upserted", total);
    }
}

    private static Command BuildEmbedCommand(IServiceCollection services)
    {
        var pathOption = new Option<string>("--path",

            Arity = OptionArity.ExactlyOne

        var batchSizeOption = new Option<int>("--batch-size",
            getDefaultValue: () => 64,
            description: "Number of chunks to embed per Ollama request (1–512).");

        var dryRunOption = new Option<bool>("--dry-run",
            getDefaultValue: () => false,
            description: "List files that would be embedded without actually embedding.");

        var cmd = new Command("embed",
            "Discovers, parses, and embeds source files into the vector store.");
        cmd.Add(pathOption);
        cmd.Add(collectionOption);
        cmd.Add(batchSizeOption);
        cmd.Add(dryRunOption);

        cmd.SetHandler(async (string path, string collection, int batchSize, bool dryRun, CancellationToken ct) =>
        {
            using var scope = services.CreateScope();
            var sp = scope.ServiceProvider;

            if (dryRun)
            {
                var files = Directory.EnumerateFiles(path, "*.*", SearchOption.AllDirectories)
                    .Where(f => !f.EndsWith(".min.js", StringComparison.OrdinalIgnoreCase))
                    .ToList();
                Console.WriteLine($"[[dry-run]] {files.Count} file(s) would be embedded into '{collection}'.");
                foreach (var f in files.Take(20)) Console.WriteLine($"  {f}");
                if (files.Count > 20) Console.WriteLine($"  … and {files.Count - 20} more.");
                return;
            }

            var writer = sp.GetRequiredService<IOutputWriter>();
            var pipeline = sp.GetRequiredService<IPipeline<EmbeddingContext>>();

            var ctx = new EmbeddingContext { SourceDirectory = path };
            writer.MarkupLine($"[[green]]Embedding[[/]] → [[blue]]{collection}[[/]]");

            await pipeline.ExecuteAsync(ctx, ct);
            writer.MarkupLine($"[[green bold]]✔[[/]] {ctx.Documents.Count} vectors stored in {ctx.TotalElapsed.TotalSeconds:F1}s");
        }, pathOption, collectionOption, batchSizeOption, dryRunOption, new System.CommandLine.Invocation.CancellationTokenOption());

        return cmd;
    }

    private static Command BuildAskCommand(IServiceCollection services)
    {
        var questionArg = new Argument<string?>("question",
            description: "The question to ask. Omit for interactive mode.");

        var interactiveOption = new Option<bool>("--interactive", "-i",
            description: "Enter interactive REPL mode (Ctrl+D / 'exit' to quit).");

        var categoryOption = new Option<string?>("--category",
            description: "Filter retrieval by question category (e.g. Performance, Security, Refactor).");

        var topKOption = new Option<int>("--top-k",
            getDefaultValue: () => 5,
            description: "Number of chunks to retrieve (1–50).");

        var cmd = new Command("ask",
            "Queries the knowledge base with a question or starts an interactive session.");
        cmd.Add(questionArg);
        cmd.Add(interactiveOption);
        cmd.Add(categoryOption);
        cmd.Add(topKOption);

        cmd.SetHandler(async (string? question, bool interactive, string? category, int topK, CancellationToken ct) =>
        {
            using var scope = services.CreateScope();
            var sp = scope.ServiceProvider;
            var session = sp.GetRequiredService<IRagSession>();
            var writer = sp.GetRequiredService<IOutputWriter>();

            QuestionCategory? cat = category is null
                ? null
                : (QuestionCategory?)Enum.Parse(typeof(QuestionCategory), category, ignoreCase: true);

            await session.BeginAsync("ragnar", maxTurns: interactive ? 20 : 3, ct);

            try
            {
                if (interactive)
                {
                    await RunInteractiveLoop(session, writer, cat, ct);
                }
                else if (question is not null)
                {
                    var response = await session.AskAsync(question, cat, ct);
                    writer.MarkupLine($"[[bold]]{response.Question}[[/]]");
                    writer.WriteRule();
                    writer.Markup(response.Answer);
                    writer.WriteLine();
                    writer.MarkupLine($"[[dim]]Confidence: {response.Confidence:P1} · {response.RetrievalDurationMs} ms[[/]]");
                }
                else
                {
                    writer.MarkupLine("[[red]]Provide a question or use --interactive.[[/]]");
                }
            }
            finally
            {
                await session.EndAsync();
            }
        }, questionArg, interactiveOption, categoryOption, topKOption, new System.CommandLine.Invocation.CancellationTokenOption());

        return cmd;
    }

    private static async Task RunInteractiveLoop(IRagSession session, IOutputWriter writer, QuestionCategory? category, CancellationToken ct)
    {
        writer.MarkupLine("[[cyan bold]]Interactive mode[[/]] — type a question, 'exit' to quit, 'clear' to reset history.");
        writer.WriteRule();

        while (true)
        {
            writer.WriteLine("[[green]]you >[[/]] ");
            var input = Console.ReadLine();

            if (input is null || input.Trim().ToLowerInvariant() is "exit" or "quit" or "q")
            {
                writer.MarkupLine("[[dim]]Session ended.[[/]]");
                break;
            }

            if (input.Trim().ToLowerInvariant() == "clear")
            {
                writer.MarkupLine("[[dim]]History cleared.[[/]]");
                continue;
            }

            if (string.IsNullOrWhiteSpace(input)) continue;

            try
            {
                var sw = System.Diagnostics.Stopwatch.StartNew();
                var response = await session.AskAsync(input, category, ct);
                sw.Stop();

                writer.WriteRule();
                writer.MarkupLine($"[[bold]]ragnar[[/]] ({sw.ElapsedMilliseconds} ms):");
                writer.Markup(response.Answer);
                writer.WriteLine();
                writer.MarkupLine($"[[dim]]  confidence {response.Confidence:P1} · category {category?.ToString() ?? "all"}[[/]]");
                writer.WriteRule();
            }
            catch (OperationCanceledException)
            {
                throw;
            }
            catch (Exception ex)
            {
                writer.MarkupLine($"[[red]]Error:[[/]] {ex.Message}");
            }
        }
    }

    private static Command BuildExportCommand(IServiceCollection services)
    {
        var formatOption = new Option<ReportFormat>("--format",
            getDefaultValue: () => ReportFormat.Markdown,
            description: "Output format: markdown, html, or json.");

        var outputOption = new Option<string>("--output",
            getDefaultValue: () => $"./reports/session_{DateTime.UtcNow:yyyyMMdd_HHmmss}.md",
            description: "Output file path.");

        var cmd = new Command("export", "Exports the most recent session transcript to a report file.");
        cmd.Add(formatOption);
        cmd.Add(outputOption);

        cmd.SetHandler(async (ReportFormat format, string output, CancellationToken ct) =>
        {
            using var scope = services.CreateScope();
            var exporter = scope.ServiceProvider.GetRequiredService<IReportExporter>();
            var session = scope.ServiceProvider.GetRequiredService<IRagSession>();

            if (!session.IsActive)
            {
                Console.Error.WriteLine("[[red]]No active session to export. Run 'ragnar ask' first.[[/]]");
                return;
            }

            var path = await exporter.ExportAsync(session.History.Select(HistoryToResponse).ToList(), format, output, ct);
            Console.WriteLine($"Report → {path}");
        }, formatOption, outputOption, new System.CommandLine.Invocation.CancellationTokenOption());

        return cmd;
    }

    private static RagResponse HistoryToResponse(ChatMessage msg)
        => new(msg.Content, string.Empty, string.Empty, 0, 0);
}

## Summary of Impact

| Feature | Problem Solved | Effort |
|---|---|---|
| Interactive REPL | No ad-hoc Q&A after one-shot run | M |
| Report Export | Results lost to stdout; no shareable artifact | S |
| Progress Tracking | Opaque multi-minute embed runs | S |
| CLI Commands | Cannot isolate operations; poor DX for scripting | M |

All four features compose cleanly with the existing `IPipeline<T>` / `IPipelineStage<T>` architecture—no existing code needs restructuring. The REPL and exporter plug in as new `IHostedService`-independent services resolved from the same `IServiceCollection`, and the progress tracker is simply injected into stages that already have access to `ILogger`.
