#### 1. Checkpoint Data Contract
```csharp
/// <summary>Immutable snapshot of pipeline progress for resumption.</summary>
/// <param name="Checksum">Deterministic hash of the serialized state to detect corruption.</param>
/// <param name="CreatedAt">Wall-clock timestamp of checkpoint creation.</param>
/// <param name="ProcessedStages">Zero-based indices of stages that completed successfully.</param>
/// <param name="Snapshot">Shallow copy of the current context state for restoration.</param>
public sealed record PipelineCheckpoint<TContext>(
    string Checksum,
    DateTimeOffset CreatedAt,
    IReadOnlyList<int> ProcessedStages,
    ReadOnlyMemory<byte> Snapshot)
    where TContext : class
{
    /// <summary>Serializes the context to JSON and computes a SHA-256 checksum.</summary>
    public static async Task<PipelineCheckpoint<TContext>> FromAsync(
        TContext context, 
        IOptions<RagnarConfig> options, 
        CancellationToken ct)
    {
        Guard.Against.Null(context);
        
        var json = JsonSerializer.Serialize(context, new JsonSerializerOptions 
        { 
            WriteIndented = false, 
            PropertyNamingPolicy = JsonNamingPolicy.CamelCase 
        });
        
        var snapshotBytes = Encoding.UTF8.GetBytes(json);
        var checksum = Convert.ToHexString(SHA256.HashData(snapshotBytes));
        
        return new PipelineCheckpoint<TContext>(checksum, DateTimeOffset.UtcNow, [[]], snapshotBytes) 
        { ProcessedStages = [[]] }; // Initialized to empty; middleware populates
    }

    /// <summary>Deserializes the checkpoint back into a context instance.</summary>
    public TContext Restore() => JsonSerializer.Deserialize<TContext>(Snapshot.Span)!;
}
```

```csharp
/// <summary>Provides checkpoint-aware extension members directly on stage execution context.</summary>
extension<TContext>(StageExecutionContext<TContext> ctx)
    where TContext : class
{
    /// <summary>Checks whether a checkpoint exists and matches the expected checksum.</summary>
    public bool HasValidCheckpoint(PipelineCheckpoint<TContext>? checkpoint, string[[]] requiredStages) 
    {
        if (checkpoint is null || !ctx.ProcessedStages.All(checkpoint.ProcessedStages.Contains))
            return false;

        // Verify checksum matches current stage sequence state
        var computed = BitConverter.ToString(Encoding.UTF8.GetBytes(string.Join("-", requiredStages)));
        return checkpoint.Checksum == computed;
    }

    /// <summary>Records completion of the current stage index into the pipeline progress tracking.</summary>
    public void MarkStageComplete() => ctx.ProcessedStages.Add(ctx.StageIndex);
    
    /// <summary>Resolves the configured checkpoint directory from application options.</summary>

        => Path.Join(config.Value.ApplicationOptions.SourceDirectory, ".ragnar", "checkpoints");
```

```csharp
/// <summary>Middleware that persists checkpoints after each stage and restores state on startup.</summary>
public sealed class ResumableMiddleware<TContext>(ILogger logger, IOptions<RagnarConfig> options)
    where TContext : class
{
    private readonly string _dir = PathResolver.ResolveCheckpointPath(options.Value.ApplicationOptions);
    private readonly SemaphoreSlim _lock = new(1, 1);

    public async Task Invoke(in StageExecutionContext<TContext> ctx, Func<Task> next, CancellationToken ct)
    {
        Guard.Against.Null(ctx);
        
        // 1. Attempt restoration if this is the first stage
        if (ctx.StageIndex == 0)
        {
            var restored = await TryRestoreAsync(ct).ConfigureAwait(false);
            if (restored)
            {
                logger.LogInformation("Resumed pipeline from checkpoint for {Stage}", ctx.Stage.Name);
                // Bypass middleware for restored stages in a real impl by tracking indices
            }
        }

        try
        {
            await next().ConfigureAwait(false);
            
            // 2. Persist checkpoint after successful stage execution
            if (ctx.HasValidCheckpoint(default!, Array.Empty<string>())) // Placeholder guard
                await SaveCheckpointAsync(ctx, ct).ConfigureAwait(false);
        }
        catch (OperationCanceledException) when (ct.IsCancellationRequested)
        {
            throw;
        }
        catch (Exception ex)
        {
            logger.LogCritical(ex, "Checkpoint save failed during stage [[{Index}]] {Name}", ctx.StageIndex, ctx.Stage.Name);
            throw;
        }
    }

    private async Task<bool> TryRestoreAsync(CancellationToken ct)
    {
        var latest = await GetLatestCheckpointAsync().ConfigureAwait(false);
        if (latest is null) return false;

        var context = latest.Restore();
        // Inject restored state back into pipeline runner (implementation detail varies by orchestrator)
        logger.LogInformation("Restored {StageCount} stages from checkpoint created at {Time}", 
            latest.ProcessedStages.Count, latest.CreatedAt);
        
        return true;
    }

    private async Task SaveCheckpointAsync(in StageExecutionContext<TContext> ctx, CancellationToken ct)
    {
        await _lock.WaitAsync(ct).ConfigureAwait(false);
        try
        {
            var snapshot = PipelineCheckpoint<TContext>.FromAsync(
                ctx.PipelineContext, options, ct).GetAwaiter().GetResult();
            
            snapshot.MarkStageComplete(); // Uses C# 14 extension member
            
            var path = Path.Join(_dir, $"checkpoint_{DateTimeOffset.UtcNow:yyyyMMdd_HHmmss}.json");
            Directory.CreateDirectory(_dir);
            await File.WriteAllTextAsync(path, JsonSerializer.Serialize(snapshot), ct).ConfigureAwait(false);
            
            logger.LogDebug("Checkpoint saved to {Path}", path);
        }
        finally 
        {
            _lock.Release();
        }
    }

    private static async Task<PipelineCheckpoint<TContext>?> GetLatestCheckpointAsync()
    {
        var files = Directory.EnumerateFiles(".ragnar/checkpoints", "*.json")
            .OrderByDescending(Path.GetFileName).ToList();
        
        if (files.Count == 0) return null;
        
        var content = await File.ReadAllTextAsync(files[[0]]).ConfigureAwait(false);
        return JsonSerializer.Deserialize<PipelineCheckpoint<EmbeddingContext>>(content);
    }
}
```
