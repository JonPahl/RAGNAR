using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Options;
using Ragnar.Core;
using Ragnar.Embedding;
using Ragnar.Questions;
using Serilog;

var builder = Host.CreateApplicationBuilder(args);

// 1. Infrastructure Setup (Config & Logging)
ConfigureInfrastructure(builder);

#region Infrastructure
static void ConfigureInfrastructure(HostApplicationBuilder builder)
{
    // Load configuration from appsettings.json & environment variables
    builder.Configuration.AddJsonFile("appsettings.json", optional: false, reloadOnChange: true);
    builder.Configuration.AddEnvironmentVariables();

    // Initialize Serilog before Host builds
    Log.Logger = new LoggerConfiguration()
        .ReadFrom.Configuration(builder.Configuration)
        .CreateLogger();

    builder.Host.UseSerilog();
}
#endregion

#region Dependency Injection
static void ConfigureServices(HostApplicationBuilder builder)
{
    // Bind & validate RagnarConfig
    builder.Services.Configure<RagnarConfig>(builder.Configuration.GetSection("RagnarConfig"));
    builder.Services.AddOptions<RagnarConfig>()
        .ValidateOnStart()
        .Validate(config => config.OllamaOptions is not null, "OllamaOptions must be configured");

    // Core application services
    builder.Services.AddSingleton<IOutputWriter, ConsoleOutputWriter>();
    builder.Services.AddSingleton<IAssemblyInfo, AssemblyInfoProvider>();
    builder.Services.AddScoped<EmbeddingPipeline>();
    builder.Services.AddScoped<ResponseWriter>();
    builder.Services.AddScoped<SystemPromptProvider>();
    builder.Services.AddKeyedSingleton<ISystemPromptProvider>("Summary", new SystemPromptProvider());

    // External clients (example: Qdrant & Ollama)
    var embeddingConfig = builder.Configuration.GetSection("RagnarConfig:EmbeddingOptions");
    builder.Services.AddSingleton<IQdrantClient>(sp =>
        new QdrantClient(new Uri($"http://{embeddingConfig["Host"]}:{embeddingConfig["Port"]}")));

    // Register validators (FluentValidation)
    builder.Services.AddValidatorsFromAssemblyContaining<OllamaOptionsValidator>();

    // Add other registrations as your app grows...
}
#endregion

#region Execution

    using var scope = services.CreateScope();

    var header = scope.ServiceProvider.GetRequiredService<IApplicationHeader>();

        header.RenderBranding();

        await pipeline.EnsureCollectionExistsAsync(CancellationToken.None);
        
        // TODO: Add main workflow calls here (e.g., question loading, LLM queries, response writing)
        
        logger.Information("Application completed successfully.");
    }
    catch (Exception ex)
    {
        Log.Fatal(ex, "Host terminated unexpectedly during execution.");
        throw;
    }
}
#endregion

### 🔑 Key Improvements & Why They Matter

| Area | Before (Typical) | After (Refactored) | Benefit |
|------|------------------|-------------------|---------|
| **Structure** | All config/DI/logic in one block | Separated into `Infrastructure`, `DI`, and `Execution` regions | Instant readability; each section can be modified independently |
| **Configuration** | Manual binding, no validation | `.ValidateOnStart()` + explicit section binding | Catches misconfigurations at startup instead of runtime crashes |
| **Service Registration** | Scattered or inline lambdas | Centralized `ConfigureServices()` with clear grouping | Easy to audit, swap implementations, or add new services |
| **Logging** | Mixed `Console.WriteLine` & Serilog calls | Dedicated infrastructure setup + structured logging in execution | Consistent telemetry; no log leakage across components |
| **Async Execution** | Blocking or unstructured async/await | Proper `IServiceProvider.CreateScope()` + try/catch with `Log.Fatal` | Prevents DI leaks, handles errors gracefully, aligns with .NET 10 host lifecycle |
