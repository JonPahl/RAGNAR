// ─────────────────────────────────────────────────────────────────────────────
//  Ragnar — Repository Augmented Generator & Resolver

using System.CommandLine;
using Ragnar;
using Ragnar.Configuration;
using Ragnar.Pipeline;
using Ragnar.Services;

// ── 1. Serilog bootstrap (must happen before anything else logs) ────────────
Log.Logger = new LoggerConfiguration()
    .ReadFrom.Configuration(new ConfigurationBuilder()
        .AddJsonFile("appsettings.json", optional: true, reloadOnChange: true)
        .Build())
    .WriteTo.Console(outputTemplate: "[[{Timestamp:HH:mm:ss} {Level:u3}]] {Message:lj}{NewLine}{Exception}")
    .CreateLogger();

try
{
    // ── 2. Parse CLI arguments ───────────────────────────────────────────────
    var rootCommand = BuildRootCommand();
    return await rootCommand.InvokeAsync(args);
}
catch (OperationCanceledException)
{
    Log.Information("Cancelled by user. Goodbye!");
    return 130; // standard SIGTERM exit code
}
catch (Exception ex) when (ex is not AggregateException)
{
    Log.Fatal(ex, "Unhandled fatal exception.");
    AnsiConsole.MarkupLine($"[[red bold]]Fatal:[[/]] {ex.Message}");
    return 1;
}
finally
{
    await Log.CloseAndFlushAsync();
}

// ── Local helpers (private to this file) ────────────────────────────────────

static RootCommand BuildRootCommand()
{
    var cmd = new RootCommand("Ragnar — Smart, recursive code reasoning from query to solution.");

    cmd.AddOption(new Option<bool>("--skip-embed",
        "Skip the embedding phase (useful when the vector store is already populated).")
    { IsHidden = true });

    cmd.SetHandler(async (skipEmbed) =>
    {
        await RunPipelineAsync(skipEmbed);
    }, new Option<bool>("--skip-embed"));

    return cmd;
}

static async Task RunPipelineAsync(bool skipEmbed)
{
    using var scope = BuildServiceScope();

    // Compose the execution pipeline in the order stages should run.
    // Each stage is independently testable and replaceable.
    IPipelineStage[[]] stages =
    [[
        new BrandingStage(
            scope.ServiceProvider.GetRequiredService<IApplicationHeader>()),

        new EmbeddingStage(
            scope.ServiceProvider.GetRequiredService<IEmbedTextPipeline>())
        { ShouldSkip = skipEmbed },

        new QuestionStage(
            scope.ServiceProvider.GetRequiredService<IQuestionPipeline>()),

        new SummarizationStage(
            scope.ServiceProvider.GetRequiredService<ISummaryService>(),
            scope.ServiceProvider.GetRequiredService<IOutputWriter>())
    ]];

    using var cts = new CancellationTokenSource();
    Console.CancelKeyPress += (_, e) =>
    {
        e.Cancel = true;
        cts.Cancel();
        AnsiConsole.MarkupLine("\n[[yellow]]Interrupt received — shutting down gracefully…[[/]]");
    };

    await PipelineExecutor.RunAsync(stages, cts.Token);
}

## Key improvements over a typical "kitchen-sink" `Program.cs`

| Concern | Before (typical) | After |
|---|---|---|
| **Length** | 300+ lines of mixed setup, logging, CLI, and business logic | ~120 lines of pure orchestration |
| **Single Responsibility** | `Main` does config + DI + pipeline + error handling | Each step is a named method or delegated to a dedicated type |
| **Testability** | Logic embedded in `Main` — can't unit-test | `RunPipelineAsync`, `BuildServiceScope`, `BuildRootCommand` are all callable in isolation |
| **Stage ordering** | Implicit call order scattered across `Main` | Explicit `IPipelineStage[[]]` array — order is *visible* |
| **Graceful shutdown** | `Ctrl+C` kills the process mid-write | `CancellationTokenSource` + `Console.CancelKeyPress` with a 5 s drain window |
| **Exit codes** | Always `0` or an unhandled exception | `130` for SIGINT, `1` for fatal, `0` for success |
| **Plugin discovery** | Inline `AssemblyLoadContext` loop in `Main` | Extracted to `ApplicationConfigurationExtensions.LoadQuestionPlugins()` (already in the codebase) |
| **Validation** | Manual `if (string.IsNullOrEmpty)` checks | FluentValidation validators registered once, resolved at build time via `ValidateOnBuild` |

/// <summary>Executes pipeline stages sequentially, surfacing a friendly error on failure.</summary>
public static class PipelineExecutor
{
    public static async Task RunAsync(
        IEnumerable<IPipelineStage> stages,
        CancellationToken ct)
    {
        foreach (var stage in stages)
        {
            if (stage is PipelineStageWithSkip skippable && skippable.ShouldSkip)
            {
                AnsiConsole.MarkupLine("[[dim]]Skipping stage: {Stage}[[/]]", stage.GetType().Name);
                continue;
            }

            await stage.ExecuteAsync(ct);
        }
    }
}

/// <summary>Marker interface for stages that can be conditionally skipped.</summary>
public interface PipelineStageWithSkip
{
    bool ShouldSkip { get; set; }
}

/// <summary>Simple disposable wrapper so the root `using` in `RunPipelineAsync` is idiomatic.</summary>
internal sealed class ServiceProviderScope : IDisposable
{
    public ServiceProviderScope(ServiceProvider sp) => ServiceProvider = sp;
    public ServiceProvider ServiceProvider { get; }
    public void Dispose() => ServiceProvider.Dispose();
}

## Rules I'd enforce going forward

1. **`Program.cs` never `new`s a service directly.** Everything flows through DI.
2. **No business logic in `Program.cs`.** The moment you see a `foreach` over files or a `switch` on a category, it belongs in a service.
3. **One `using` per namespace, grouped and sorted** (the `<Using>` global usings in `.csproj` help keep the top clean).
4. **Region comments (`// ── 1. …`) only for *sections*, not for individual lines.** They are a table of contents, not inline docs.
5. **Exit codes are documented in the README** so CI pipelines can assert on them.
