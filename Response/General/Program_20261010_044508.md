/// <summary>
/// Wires the full object graph, validates configuration, and executes
/// the appropriate pipeline stage set.

static async Task RunApplicationAsync()
{
    // 3a. Build configuration
    var config = BuildConfiguration();

    // 3b. Build the service collection
    var services = new ServiceCollection();
    ConfigureServices(services, config);

    // 3c. Build the provider and validate
    var provider = services.BuildServiceProvider();

    using var scope = provider.CreateScope();
    await ValidateConfigurationAsync(scope.ServiceProvider).ConfigureAwait(false);

    // 3d. Resolve and run the pipeline
    var pipeline = scope.ServiceProvider.GetRequiredService<PipelineRunner>();
    var context = new EmbeddingContext();

    using var cts = new CancellationTokenSource(TimeSpan.FromMinutes(30));
    await pipeline.ExecuteAsync(context, cts.Token).ConfigureAwait(false);
}

/// <summary>

{
    var path = Environment.GetEnvironmentVariable("RAGNAR_CONFIG")
               ?? "appsettings.json";

    return new ConfigurationBuilder()
        .SetBasePath(AppContext.BaseDirectory)

        .Build();

/// <summary>
/// Runs all FluentValidation validators; exits with a non-zero code on failure.
/// </summary>
static async Task ValidateConfigurationAsync(IServiceProvider sp)
{
    var config = sp.GetRequiredService<IOptions<RagnarConfig>>().Value;
    var validator = sp.GetRequiredService<IValidator<RagnarConfig>>();
    var logger = sp.GetRequiredService<Serilog.ILogger>();

    var result = await validator.ValidateAsync(config).ConfigureAwait(false);

    if (!result.IsValid)
    {
        var errors = string.Join(Environment.NewLine,
            result.Errors.Select(e => $"  • {e.PropertyName}: {e.ErrorMessage}"));

        AnsiConsole.MarkupLine($"[[red]]Configuration is invalid:[[/]]");
        Console.WriteLine(errors);
        Environment.ExitCode = 1;
        throw new InvalidOperationException("Configuration validation failed.");
    }

    logger.Information("Configuration validated successfully ({Version}).",
        Assembly.GetExecutingAssembly().InformationalVersion);
}

## What changed and why

| Area | Before (typical) | After | Benefit |
|---|---|---|---|
| **Top-level flow** | One 200-line `Main` mixing CLI, DI, validation, and pipeline | Three numbered regions: *Parse → Bootstrap → Run* | Read top-to-bottom in < 10 s |
| **DI registration** | All `AddSingleton` calls in one giant block | One private method per concern (`RegisterOllama`, `RegisterQdrant`, …) | Change the Qdrant URL without scrolling past Ollama code |
| **CLI handling** | Manual `args` parsing or buried in `Main` | `System.CommandLine` with named sub-commands (`embed`, `query`, `inspect`, `validate`) | Self-documenting; `ragnar --help` works for free |
| **Validation gate** | Inline `if/throw` after `GetRequiredService` | Dedicated `ValidateConfigurationAsync` that returns a clear error list | Easy to unit-test; non-zero exit code for CI |
| **CancellationToken** | Created ad-hoc inside `Main` | Single 30-min `CancellationTokenSource` passed through the pipeline | One place to adjust timeout |
| **Magic strings** | `"appsettings.json"`, `"ragnar"`, `"localhost"` scattered | All sourced from `IConfiguration` with safe defaults in `PostConfigureAll` | Config lives in one file; env-var override via `RAGNAR_` prefix |
| **Naming & docs** | `var x = …` with no intent | Every helper has an XML `<summary>` and a numbered region comment (`// 5.3  Ollama clients`) | IDE quick-nav and grep both work instantly |
| **Testability** | Static calls to `AnsiConsole` and `File.WriteAllText` | All I/O behind interfaces (`IOutputWriter`, `IResponseWriter`, `IQdrantClient`) already resolved via DI | Swap for fakes in xUnit without touching `Program.cs` |

### Quick-start checklist for a new developer

1. **Run** – `dotnet run -- query --category XML`
2. **Configure** – edit `appsettings.json` (or export `RAGNAR_APPLICATIONOPTIONS__VECTORSTORENAME=…`)
3. **Add a stage** – implement `IPipelineStage<EmbeddingContext>`, register it in `RegisterPipeline`, done
4. **Add a question source** – drop a DLL in `Questions/Plugins/`; `LoadQuestionPlugins` picks it up at startup
5. **Change the model** – update `OllamaOptions.Model` in config; `IOllamaClientFactory.ResolveClient` picks it up automatically

This layout keeps `Program.cs` under ~180 lines of *meaningful* code, with every region independently modifiable and unit-testable.
