## 1. Interactive REPL Mode

**Problem:** Ragnar currently runs as a fire-and-forget `IHostedService`. Users cannot iterate, ask follow-up questions, or explore the knowledge base interactively. The entire Q&A lifecycle executes once and terminates.

**Solution:** Add an interactive terminal mode where users can ask ad-hoc questions, browse stored files, and iterate without restarting the process.

### Implementation

```csharp
// File: InteractiveSession.cs
/// <summary>
/// Default interactive RAG session supporting multi-turn conversation,
/// streaming responses, and file browsing against the Qdrant collection.
/// </summary>
/// <param name="contextRetriever">Service for vector similarity search.</param>
/// <param name="promptBuilder">Fluent RAG prompt composer.</param>
/// <param name="chatClient">Ollama chat client for LLM generation.</param>
/// <param name="writer">Console output writer.</param>
/// <param name="logger">Structured logger.</param>
/// <param name="options">Application options (vector store name, model config).</param>
/// <param name="qdrant">Qdrant client for file browsing.</param>
public sealed class InteractiveSession(
    IContextRetriever contextRetriever,
    IRagPromptBuilder promptBuilder,
    IOllamaChatClient chatClient,
    IOutputWriter writer,
    ILogger<InteractiveSession> logger,
    IOptions<RagnarConfig> options,
    IQdrantClient qdrant) : IInteractiveSession
{
    private readonly List<ChatMessage> _history = [[]];
    private readonly string _vectorStore = options.Value.ApplicationOptions.VectorStoreName;

    /// <inheritdoc />
    public IReadOnlyList<ChatMessage> History => _history.AsReadOnly();

    /// <inheritdoc />
    public async IAsyncEnumerable<string> AskAsync(
        string question,
        [[EnumeratorCancellation]] CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        Guard.Against.NullOrWhiteSpace(question);

        // 1. Retrieve relevant context from the vector store
        var context = await contextRetriever
            .RetrieveContextAsync(_vectorStore, cancellationToken)
            .ConfigureAwait(false);

        // 2. Build the prompt including conversation history for multi-turn
        var prompt = promptBuilder
            .WithQuestion(new Question { Text = question, IsEnabled = true })
            .WithContext(context)
            .Build();

        // 3. Prepend history for conversational continuity
        var messages = _history
            .Concat([[new ChatMessage(ChatRole.User, prompt)]])
            .ToList();

        // 4. Stream the response from Ollama
        var fullResponse = new StringBuilder();
        await foreach (var chunk in chatClient
            .GetChatMessagesAsync(messages, cancellationToken: cancellationToken)
            .ConfigureAwait(false))
        {
            var text = chunk.ToString();
            fullResponse.Append(text);
            yield return text;
        }

        // 5. Persist to history
        _history.Add(new ChatMessage(ChatRole.User, prompt));
        _history.Add(new ChatMessage(ChatRole.Assistant, fullResponse.ToString()));

        logger.LogInformation(
            "Question answered. History length: {Count}, tokens approx: {Tokens}",
            _history.Count,
            fullResponse.Length / 4);
    }

    /// <inheritdoc />
    public async Task<IReadOnlyList<FileDescriptor>> BrowseFilesAsync(
        string searchTerm, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();
        var results = new List<FileDescriptor>();
        PointId? offset = null;

        do
        {
            var response = await qdrant.ScrollAsync(
                _vectorStore,
                limit: 50,
                offset: offset,
                filter: new Filter(
                    must: [[new FieldCondition(
                        key: "FileName",
                        match: new MatchText { Text = searchTerm })]]),
                cancellationToken: cancellationToken).ConfigureAwait(false);

            foreach (var point in response.Result)
            {
                var path = point.Payload[["FileName"]].StringValue;
                if (!results.Any(f => f.Path == path))
                {
                    results.Add(new FileDescriptor
                    {
                        Path = path,
                        ChunkCount = 1, // Will be accurate if we add a payload field
                        LastModified = point.Payload.TryGetValue("CreatedAt", out var ts)
                            ? ts.TimeValue
                            : default
                    });
                }
            }

            offset = response.NextPageOffset;
        } while (offset is not null);

        return results;
    }

    /// <inheritdoc />
    public Task ResetAsync(CancellationToken cancellationToken)
    {
        _history.Clear();
        logger.LogDebug("Session history cleared.");
        return Task.CompletedTask;
    }

    /// <inheritdoc />
    public Task ExitAsync(CancellationToken cancellationToken)
    {
        writer.MarkupLine("[[green]]Session ended. Goodbye![[/]]");
        return Task.CompletedTask;
    }

    /// <inheritdoc />
    public ValueTask DisposeAsync()
    {
        logger.LogInformation("InteractiveSession disposed. Final history: {Count}", _history.Count);
        return ValueTask.CompletedTask;
    }
}
```

```csharp
// File: InteractiveHostedService.cs
/// <summary>
/// Hosted service that runs the embedding pipeline first, then enters
/// an interactive terminal REPL for ad-hoc RAG queries.
/// </summary>
public sealed class InteractiveHostedService(
    RagPipelineRunner pipelineRunner,
    IInteractiveSession session,
    IOutputWriter writer,
    IApplicationHeader header,
    ILogger<InteractiveHostedService> logger) : IHostedService
{
    public async Task StartAsync(CancellationToken cancellationToken)
    {
        // Phase 1: Run the standard embedding + question pipeline
        await pipelineRunner.StartAsync(cancellationToken).ConfigureAwait(false);

        // Phase 2: Enter interactive mode
        header.RenderBranding();
        writer.WriteRule();
        writer.MarkupLine("[[bold cyan]]Interactive Mode[[/]] — Type a question, 'browse <term>', 'reset', or 'exit'.");
        writer.WriteRule();

        using var cts = CancellationTokenSource.CreateLinkedTokenSource(cancellationToken);
        Console.CancelKeyPress += (_, e) => { e.Cancel = true; cts.Cancel(); };

        while (!cts.IsCancellationRequested)
        {
            Console.Write("\n\u276F ");
            var input = Console.ReadLine()?.Trim();
            if (string.IsNullOrWhiteSpace(input)) continue;

            switch (input.ToLowerInvariant())
            {
                case "exit" or "quit" or "q":
                    await session.ExitAsync(cts.Token);
                    return;

                case "reset":
                    await session.ResetAsync(cts.Token);
                    writer.MarkupLine("[[dim]]History cleared.[[/]]");
                    break;

                case var cmd when cmd.StartsWith("browse ", StringComparison.OrdinalIgnoreCase):
                    var term = cmd[["browse ".Length..]].Trim();
                    var files = await session.BrowseFilesAsync(term, cts.Token);
                    foreach (var f in files)
                        writer.MarkupLine($"  [[blue]]{f.Path}[[/]]  ({f.ChunkCount} chunks)");
                    break;

                case "help":
                    writer.MarkupLine("[[dim]]Commands: <question> | browse <term> | reset | exit[[/]]");
                    break;

                default:
                    try
                    {
                        writer.WriteLine();
                        await foreach (var chunk in session.AskAsync(input, cts.Token))
                        {
                            Console.Write(chunk);
                        }

                        Console.WriteLine();
                        writer.WriteRule();
                    }
                    catch (OperationCanceledException) when (cts.IsCancellationRequested)
                    {
                        writer.MarkupLine("[[yellow]]Query cancelled.[[/]]");
                    }
                    catch (Exception ex)
                    {
                        writer.MarkupLine($"[[red]]Error: {ex.Message}[[/]]");
                        logger.LogError(ex, "Interactive query failed");
                    }
                    break;
            }
        }
    }

    public Task StopAsync(CancellationToken cancellationToken) => Task.CompletedTask;
}
```

## 2. Streaming LLM Responses

**Problem:** `OllamaEmbeddingGenerator` and the chat pipeline wait for the full response before displaying anything. For long answers (1000+ tokens), users stare at a blank console for 10–30 seconds, perceiving the app as hung.

**Solution:** Add a streaming abstraction to the Ollama chat layer so text is rendered token-by-token.

```csharp
/// <summary>
/// Streams Ollama chat completions by reading the SSE response stream
/// and deserializing each JSON delta into a string chunk.
/// </summary>
/// <param name="httpClient">Pre-configured HttpClient pointed at the Ollama endpoint.</param>
/// <param name="logger">Logger for streaming diagnostics.</param>
/// <param name="config">Ragnar config supplying model name and timeout.</param>
public sealed class OllamaStreamingChatClient(
    HttpClient httpClient,
    ILogger<OllamaStreamingChatClient> logger,
    IOptions<RagnarConfig> config) : IStreamingChatClient
{
    private static readonly JsonSerializerOptions JsonOpts = new(JsonSerializerDefaults.Web)
    {
        PropertyNameCaseInsensitive = true
    };

    private readonly string _model = config.Value.OllamaOptions.LlmModel;

    /// <inheritdoc />
    public async IAsyncEnumerable<string> StreamAsync(
        IReadOnlyList<ChatMessage> messages,
        GenerationOptions? options = null,
        [[EnumeratorCancellation]] CancellationToken cancellationToken = default)
    {
        Guard.Against.Null(messages);
        cancellationToken.ThrowIfCancellationRequested();

        var payload = new
        {
            model = _model,
            messages = messages.Select(m => new { role = m.Role.ToString().ToLowerInvariant(), content = m.Content }),
            stream = true,
            options = options is null ? null : new { temperature = options.Temperature, top_p = options.TopP }
        };

        using var request = new HttpRequestMessage(HttpMethod.Post, "/api/chat")
        {
            Content = JsonContent.Create(payload)
        };

        using var response = await httpClient
            .SendAsync(request, HttpCompletionOption.ResponseHeadersRead, cancellationToken)
            .ConfigureAwait(false);

        response.EnsureSuccessStatusCode();

        var stream = await response.Content.ReadAsStreamAsync(cancellationToken).ConfigureAwait(false);
        using var reader = new StreamReader(stream);

        while (!cancellationToken.IsCancellationRequested)
        {
            var line = await reader.ReadLineAsync(cancellationToken).ConfigureAwait(false);
            if (line is null) break; // Stream ended

            if (line.Length == 0) continue;

            // Ollama streams JSON lines: {"message":{"content":"..."},"done":false}
            var delta = ParseDelta(line);
            if (string.IsNullOrEmpty(delta)) continue;

            if (logger.IsEnabled(LogLevel.Trace))
                logger.LogTrace("Stream chunk ({Len} chars): {Preview}", delta.Length,
                    delta[[..Math.Min(60, delta.Length)]]);

            yield return delta;
        }

        logger.LogDebug("Stream complete.");
    }

    private static string ParseDelta(string jsonLine)
    {
        try
        {
            using var doc = JsonDocument.Parse(jsonLine);
            if (doc.RootElement.TryGetProperty("message", out var msg)
                && msg.TryGetProperty("content", out var content))
            {
                return content.GetString();
            }

            return null!;
        }
        catch (JsonException ex)
        {
            // Non-JSON line (e.g. keep-alive); ignore
            return null!;
        }
    }
}
```

/// <summary>
/// Renders a Ragnar session as a well-structured Markdown report
/// with YAML front-matter, per-question code blocks, and a summary appendix.
/// </summary>
/// <param name="logger">Logger for export diagnostics.</param>
/// <param name="clock">Abstraction over time for reproducible timestamps.</param>
public sealed class MarkdownReportExporter(
    ILogger<MarkdownReportExporter> logger,
    IClock clock) : IReportExporter
{
    private readonly JsonSerializerOptions _jsonOpts = new(JsonSerializerDefaults.Web)
    {
        WriteIndented = true,
        DefaultIgnoreCondition = System.Text.Json.Serialization.JsonIgnoreCondition.WhenWritingNull
    };

    /// <inheritdoc />
    public async Task<string> ExportMarkdownAsync(
        QuestionPipelineContext session,
        string outputPath,
        CancellationToken cancellationToken)
    {
        Guard.Against.NullOrWhiteSpace(outputPath);
        cancellationToken.ThrowIfCancellationRequested();

        var timestamp = clock.UtcNow.ToString("yyyy-MM-dd HH:mm:ss UTC");
        var sb = new StringBuilder();

        // ── YAML Front-Matter ──
        sb.AppendLine("---");
        sb.AppendLine($"title: Ragnar Session Report");
        sb.AppendLine($"generated: {timestamp}");
        sb.AppendLine($"version: {GetAssemblyVersion()}");
        sb.AppendLine($"question_count: {session.Questions.Count}");
        sb.AppendLine($"total_elapsed: {session.TotalElapsed:hh\\:mm\\:ss}");
        sb.AppendLine($"status: {session.HadFailures ? "partial" : "complete"}");
        sb.AppendLine("---");
        sb.AppendLine();
        sb.AppendLine("# Ragnar Code Analysis Report");
        sb.AppendLine();

        // ── Table of Contents ──
        sb.AppendLine("## Table of Contents");
        sb.AppendLine();
        for (var i = 0; i < session.Questions.Count; i++)
        {
            var q = session.Questions[[i]];
            var anchor = Slugify(q.Text);
            sb.AppendLine($"{i + 1}. [[{q.Text}]](#{anchor}) — `{q.Category}`");
        }

        sb.AppendLine();
        sb.AppendLine("---");
        sb.AppendLine();

        // ── Per-Question Sections ──
        for (var i = 0; i < session.Questions.Count; i++)
        {
            var q = session.Questions[[i]];
            var response = session.Responses.ElementAtOrDefault(i);
            var context = session.Contexts.ElementAtOrDefault(i);

            sb.AppendLine($"## {i + 1}. {q.Text}");
            sb.AppendLine();
            sb.AppendLine($"> **Category:** `{q.Category}`  ");
            sb.AppendLine($"> **Source:** `{q.Filename}`  ");
            sb.AppendLine($"> **Retrieved {clock.UtcNow:yyyy-MM-dd}");
            sb.AppendLine();

            // Retrieved context (collapsible)
            if (!string.IsNullOrWhiteSpace(context))
            {
                sb.AppendLine("<details>");
                sb.AppendLine("<summary><strong>Retrieved Context</strong> (click to expand)</summary>");
                sb.AppendLine();
                sb.AppendLine("```text");
                sb.AppendLine(context);
                sb.AppendLine("```");
                sb.AppendLine();
                sb.AppendLine("</details>");
                sb.AppendLine();
            }

            // LLM Response
            sb.AppendLine("### Response");
            sb.AppendLine();
            sb.AppendLine(response ?? "*No response generated.*");
            sb.AppendLine();
            sb.AppendLine("---");
            sb.AppendLine();
        }

        // ── Summary Appendix ──
        if (!string.IsNullOrWhiteSpace(session.Summary))
        {
            sb.AppendLine("## Appendix: Summary");
            sb.AppendLine();
            sb.AppendLine(session.Summary);
            sb.AppendLine();
        }

        // ── Write File ──
        var fullPath = Path.GetFullPath(outputPath);
        var dir = Path.GetDirectoryName(fullPath);
        if (!string.IsNullOrEmpty(dir))
            Directory.CreateDirectory(dir);

        await File.WriteAllTextAsync(fullPath, sb.ToString(), cancellationToken).ConfigureAwait(false);
        logger.LogInformation("Markdown report written to {Path} ({Size} bytes)", fullPath, sb.Length);

        return fullPath;
    }

    /// <inheritdoc />

        QuestionPipelineContext session,
        string outputPath,
        CancellationToken cancellationToken)
    {
        Guard.Against.NullOrWhiteSpace(outputPath);
        var fullPath = Path.GetFullPath(outputPath);
        var dir = Path.GetDirectoryName(fullPath);
        if (!string.IsNullOrEmpty(dir))
            Directory.CreateDirectory(dir);

        var doc = new
        {
            GeneratedAt = clock.UtcNow,
            Version = GetAssemblyVersion(),
            QuestionCount = session.Questions.Count,
            TotalElapsed = session.TotalElapsed,
            Questions = session.Questions.Zip(session.Responses ?? [[]], (q, r) => new
            {
                q.Text,
                q.Category,
                q.Filename,
                Response = r
            })
        };

        var json = JsonSerializer.Serialize(doc, _jsonOpts);
        await File.WriteAllTextAsync(fullPath, json, cancellationToken).ConfigureAwait(false);

        return fullPath;
    }

    private static string Slugify(string text)
    {
        var sb = new StringBuilder();
        foreach (var c in text.ToLowerInvariant())
            sb.Append(char.IsLetterOrDigit(c) ? c : '-');
        return sb.ToString().Trim('-').Replace("--", "-");
    }

    private static string GetAssemblyVersion() =>
        typeof(MarkdownReportExporter).Assembly.GetName().Version?.ToString() ?? "0.0.0";
}

## 4. Hybrid Search with BM25 Re-Ranking

**Problem:** The current `IContextRetriever` performs a pure cosine-similarity search against Qdrant. For code queries like *"fix the null reference in ParseAsync"*, keyword matching is often more precise than semantic similarity, especially when the target file contains the exact identifier. Pure vector search can miss the right chunk if the embedding model didn't weight the identifier heavily.

**Solution:** Add a two-stage retrieval: (1) Qdrant dense vector search for the top-K candidates, (2) a local BM25 re-ranker to score candidates by keyword overlap with the query, returning the top-N.

### Implementation

```csharp
// File: IBm25Ranker.cs
/// <summary>
/// Lightweight in-memory BM25 (Okapi) re-ranker for text chunks.
/// </summary>
public interface IBm25Ranker
{
    /// <summary>
    /// Scores a set of candidate documents against a query using BM25.
    /// </summary>
    /// <param name="query">The user's question or search term.</param>
    /// <param name="candidates">Pre-fetched candidate chunks to re-rank.</param>
    /// <returns>Candidates sorted by descending BM25 score.</returns>
    IReadOnlyList<ScoredChunk> Rank(string query, IReadOnlyList<CodeDocument> candidates);
}

// File: ScoredChunk.cs
/// <summary>A text chunk with its BM25 relevance score.</summary>
public sealed record ScoredChunk(CodeDocument Document, double Score)
{
    public string Text => Document.Code;
}

// File: Bm25Ranker.cs
/// <summary>
/// Okapi BM25 implementation for re-ranking vector-search candidates
/// by keyword overlap. Tunable k1 and b parameters.
/// </summary>
/// <param name="k1">Term frequency saturation parameter (default 1.5).</param>
/// <param name="b">Length normalization parameter (default 0.75).</param>
public sealed class Bm25Ranker(double k1 = 1.5, double b = 0.75) : IBm25Ranker
{
    private readonly double _k1 = k1;
    private readonly double _b = b;

    /// <inheritdoc />
    public IReadOnlyList<ScoredChunk> Rank(string query, IReadOnlyList<CodeDocument> candidates)
    {
        Guard.Against.Null(query);
        Guard.Against.Null(candidates);
        if (candidates.Count == 0) return [[]];

        var queryTerms = Tokenize(query).Where(t => t.Length > 1).ToHashSet();
        if (queryTerms.Count == 0)
            return candidates.Select(c => new ScoredChunk(c, 0.0)).ToList();

        // Pre-compute document statistics
        var docTerms = candidates.Select(c => Tokenize(c.Code)).ToList();
        var docLengths = docTerms.Select(terms => terms.Count).ToList();
        var avgDl = docLengths.Average();
        var df = new Dictionary<string, int>(); // Document frequency

        foreach (var terms in docTerms)
        {
            var unique = terms.ToHashSet();
            foreach (var t in unique)
                df[[t]] = df.GetValueOrDefault(t) + 1;
        }

        // Score each document
        var results = new List<ScoredChunk>(candidates.Count);
        for (var i = 0; i < candidates.Count; i++)
        {
            var score = ScoreDocument(queryTerms, docTerms[[i]], docLengths[[i]], avgDl, df, candidates.Count);
            results.Add(new ScoredChunk(candidates[[i]], score));
        }

        return results
            .OrderByDescending(r => r.Score)
            .ToList();
    }

    private double ScoreDocument(
        HashSet<string> queryTerms,
        List<string> docTerms,
        int docLength,
        double avgDl,
        Dictionary<string, int> df,
        int totalDocs)
    {
        var tf = docTerms.ToLookup(t => t).ToDictionary(g => g.Key, g => (double)g.Count());
        double score = 0.0;

        foreach (var term in queryTerms)
        {
            if (!tf.TryGetValue(term, out var freq)) continue;

            var n = df.GetValueOrDefault(term, 0);
            var idf = Math.Log(1 + (totalDocs - n + 0.5) / (n + 0.5));
            var numerator = freq * (_k1 + 1.0);
            var denominator = freq + _k1 * (1.0 - _b + _b * docLength / avgDl);

            score += idf * numerator / denominator;
        }

        return score;
    }

    private static List<string> Tokenize(string text) =>
    [[
        .. Regex.Split(text.ToLowerInvariant(), @"[[^\p{L}\p{N}]]+")
               .Where(t => t.Length > 0)
    ]];
}
```

### Enhanced Context Retriever

```csharp
// File: HybridContextRetriever.cs
/// <summary>
/// Two-stage context retrieval: dense vector search followed by BM25 re-ranking.
/// </summary>
/// <param name="qdrant">Qdrant client for the initial vector search.</param>
/// <param name="bm25">BM25 re-ranker for keyword-precision pass.</param>
/// <param name="options">Ragnar options (vector store name, top-K, top-N).</param>
/// <param name="logger">Structured logger.</param>
/// <param name="questionProvider">Supplies the current question text for BM25 scoring.</param>
public sealed class HybridContextRetriever(
    IQdrantClient qdrant,
    IBm25Ranker bm25,
    IOptions<RagnarConfig> options,
    ILogger<HybridContextRetriever> logger,
    IQuestionCatalogLoader questionProvider) : IContextRetriever
{
    private const int VectorTopK = 20;   // Initial vector candidates
    private const int FinalTopN = 5;     // Re-ranked results to include in prompt

    /// <inheritdoc />
    public async Task<string> RetrieveContextAsync(
        string vectorStoreName,
        CancellationToken cancellationToken,
        Filter? filter = null)
    {
        cancellationToken.ThrowIfCancellationRequested();

        // Stage 1: Dense vector search
        var questionText = questionProvider.LoadBuiltInQuestions().FirstOrDefault()?.Text ?? "";
        var searchResult = await SearchByVector(vectorStoreName, questionText, VectorTopK, filter, cancellationToken)
            .ConfigureAwait(false);

        logger.LogDebug("Vector search returned {Count} candidates", searchResult.Count);

        // Stage 2: BM25 re-rank
        var ranked = bm25.Rank(questionText, searchResult);
        var top = ranked.Take(FinalTopN).ToList();

        logger.LogDebug(
            "BM25 re-rank: top score {Top}, bottom {Bottom}",
            top.FirstOrDefault()?.Score ?? 0,
            top.LastOrDefault()?.Score ?? 0);

        // Compose context string
        return ComposeContext(top);
    }

    private async Task<IReadOnlyList<CodeDocument>> SearchByVector(
        string collection, string query, int limit, Filter? filter, CancellationToken ct)
    {
        var results = new List<CodeDocument>();
        PointId? offset = null;

        do
        {
            var response = await qdrant.ScrollAsync(
                collection,
                limit: (uint)limit,
                offset: offset,
                filter: filter,
                cancellationToken: ct).ConfigureAwait(false);

            foreach (var point in response.Result)
            {
                results.Add(new CodeDocument
                {
                    Code = point.Payload.TryGetValue("Code", out var code) ? code.StringValue : string.Empty,
                    FileName = point.Payload.TryGetValue("FileName", out var fn) ? fn.StringValue : string.Empty,
                    Elements = new Dictionary<string, string>()
                });
            }

            offset = response.NextPageOffset;
        } while (offset is not null && results.Count < limit);

        return results;
    }

    private static string ComposeContext(IReadOnlyList<ScoredChunk> chunks)
    {
        if (chunks.Count == 0) return "(no relevant context found)";

        var sb = new StringBuilder();
        sb.AppendLine("// Retrieved context (re-ranked by relevance):");
        sb.AppendLine();
        foreach (var chunk in chunks)
        {
            sb.AppendLine($"// ── {chunk.Document.FileName} (score: {chunk.Score:F3})");
            sb.AppendLine(chunk.Document.Code);
            sb.AppendLine();
        }

        return sb.ToString();
    }
}
```
