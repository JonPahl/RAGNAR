public sealed class RagPipelineRunner(
    EmbeddingPipelineDeps embedding,
    QuestionPipelineDeps question) : IHostedService
{
    // Build pipelines from the two dependency groups
    private readonly PipelineBuilder<EmbeddingContext> _embeddingPipeline = new(embedding.Log)
        .AddStage(new BrandingStage(embedding.Log, embedding.Header))
        .AddStage(new VectorStoreInitStage(embedding.VectorSetup, embedding.Writer))
        .AddStage(new DiscoveryStage(embedding.Options, embedding.FileService, embedding.Log, embedding.Writer))
        .AddStage(new ParsingStage(embedding.ParseFactory, embedding.Log))
        .AddStage(new DeltaEmbedStage(embedding.Log, embedding.FingerprintStore, embedding.VectorWriter, embedding.Options))
        .AddStage(new UpsertStage(embedding.VectorWriter, embedding.Log, embedding.Options, embedding.Writer))
        .AddStage(new StoredFileListStage(embedding.Options, embedding.QdrantClient, embedding.Log));

    private readonly PipelineBuilder<QuestionPipelineContext> _questionPipeline = new(question.Log)
        .UseMiddleware(new ExceptionHandlingMiddleware(question.Log).Invoke)
        .UseMiddleware(new StageTimingMiddleware(question.Log).Invoke)
        .UseMiddleware(new RetryMiddleware(question.Log).Invoke)
        .UseMiddleware(new ProgressMiddleware(question.Writer).Invoke)
        .AddStage(new QuestionLoadStage(question.QuestionSource, question.QuestionSourceBuilder, question.Log))
        .AddStage(new QuestionTableStage(question.Writer))
        .AddStage(new QuestionAnswerStage(question.RagOrchestrator, question.ContextRetriever, question.Clock, question.Log))
        .AddStage(new SummarizationStage(question.Summarizer, question.Writer));

    public async Task StartAsync(CancellationToken cancellationToken)
    {
        await _embeddingPipeline.Build().ExecuteAsync(embedding.Context, cancellationToken).ConfigureAwait(false);
        await _questionPipeline.Build().ExecuteAsync(question.Context, cancellationToken).ConfigureAwait(false);
    }

    public Task StopAsync(CancellationToken cancellationToken) => Task.CompletedTask;
}
