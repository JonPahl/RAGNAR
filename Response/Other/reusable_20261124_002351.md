### 📦 1. Composable Conditions via C# 14 Extension Members
Replace boolean flags with type-safe, chainable filters. C# 14's extension members let us add fluent composition directly on interfaces without inheritance bloat.

```csharp
public interface IPipelineCondition<TContext> where TContext : class
{
    bool Evaluate(in TContext context);
}

public static class PipelineConditions
{
    // C# 14 Extension Member: Adds fluent composition directly to the condition interface
    extension(IPipelineCondition<TContext> cond) where TContext : class
    {
        public static IPipelineCondition<TContext> MustSatisfy(in Func<TContext, bool> predicate) =>
            new CompositeCondition<TContext>(cond, predicate, (prev, pred, ctx) => prev && pred(ctx));

        public static IPipelineCondition<TContext> Or(in Func<TContext, bool> predicate) =>
            new CompositeCondition<TContext>(cond, predicate, (prev, pred, ctx) => prev || pred(ctx));

        public static IPipelineCondition<TContext> Never() => AlwaysFalse<TContext>.Instance;
        public static IPipelineCondition<TContext> Always() => AlwaysTrue<TContext>.Instance;
    }

    private record CompositeCondition<T>(IPipelineCondition<T> Previous, Func<T, bool> Predicate, 
        Func<bool, bool, in T, bool> Combine) : IPipelineCondition<T> where T : class
    {
        public bool Evaluate(in T ctx) => Combine(Previous.Evaluate(ctx), Predicate(ctx), ctx);
    }

    private record AlwaysTrue<T>() : IPipelineCondition<T> where T : class
    {
        public static readonly AlwaysTrue<T> Instance = new();
        public bool Evaluate(in T ctx) => true;
    }

    private record AlwaysFalse<T>() : IPipelineCondition<T> where T : class
    {
        public static readonly AlwaysFalse<T> Instance = new();
        public bool Evaluate(in T ctx) => false;
    }
}
```

### 📦 2. Registry-Driven Stage Definition & Auto-Discovery
Stages become reusable by decoupling execution logic from registration. Use attributes + DI scanning to auto-register and order stages.

```csharp
[[AttributeUsage(AttributeTargets.Class)]]
public sealed class PipelineStageAttribute(string name, int order = 0) : Attribute
{
    public string Name { get; } = name;
    public int Order { get; } = order;
}

// Reusable pipeline stage contract
public interface IReusabilityStage<TContext> where TContext : class
{
    string Name { get; }
    IPipelineCondition<TContext>? Condition => null; // Default: always run
    Task ExecuteAsync(in TContext context, CancellationToken ct);
}

// Example: Reusable Parsing Stage (no hardcoded paths)
[[PipelineStage("Parse Code", 10)]]
public sealed class ParsingStage(IFileParseFactory parseFactory, ILogger logger) : IReusabilityStage<EmbeddingContext>
{
    public string Name => "Parse Code";
    public IPipelineCondition<EmbeddingContext>? Condition => PipelineConditions.Always().MustSatisfy(ctx => ctx.DiscoveredFiles.Count > 0);

    public async Task ExecuteAsync(in EmbeddingContext context, CancellationToken ct)
    {
        // Use 'in' to avoid copying large readonly collections if needed
        Guard.Against.Null(context.DiscoveredFiles);
        
        var documents = new ConcurrentBag<CodeDocument>();
        var maxParallelism = Math.Min(Environment.ProcessorCount, 8);

        await AnsiConsole.Progress().AutoClear(true).StartAsync(async ctx =>
        {
            var task = ctx.AddTask(Name, maxValue: context.DiscoveredFiles.Count);
            await Parallel.ForEachAsync(context.DiscoveredFiles, new ParallelOptions { CancellationToken = ct, MaxDegreeOfParallelism = maxParallelism }, async (path, token) =>
            {
                var chunks = await parseFactory.ParseAsync(path, token).ConfigureAwait(false);
                foreach (var doc in chunks) documents.Add(doc);
                task.Increment(1);
            }).ConfigureAwait(false);
        }).ConfigureAwait(false);

        // Use 'out' to pass results without mutating the immutable context contract
        var updatedCtx = context with { Documents = [[..documents]] };
    }
}
```

### 📦 3. Dynamic Middleware Pipeline Builder
Allow middleware to be composed by name/key and reused across embedding/question pipelines.

```csharp
public static class MiddlewarePipeline
{
    public static IEnumerable<Func<TContext, Func<Task>, CancellationToken, Task>> 
        BuildFor<TContext>(IServiceCollection services, string[[]] groupNames) where TContext : class
    {
        var chain = new List<Func<TContext, Func<Task>, CancellationToken, Task>>();

        foreach (var name in groupNames)
        {
            switch (name)
            {
                case "Timing+Retry":
                    chain.Add(new StageTimingMiddleware(services.GetRequiredService<ILogger>()).Invoke);
                    chain.Add(new RetryMiddleware(services.GetRequiredService<ILogger>()).Invoke);
                    break;
                case "Progress+Exception":
                    chain.Add(new ProgressMiddleware(services.GetRequiredService<IOutputWriter>()).Invoke);
                    chain.Add(new ExceptionHandlingMiddleware(services.GetRequiredService<ILogger>()).Invoke);
                    break;
            }
        }

        return [[..chain]];
    }
}

// C# 14 Extension Member for pipeline composition
public static class PipelineExtensions
{
    extension<TContext>(IList<IReusabilityStage<TContext>> stages) where TContext : class
    {
        public IReadOnlyList<IReusabilityStage<TContext>> OrderedStages => 
            [[..stages.OrderBy(s => s.Name)]]; // Auto-ordering by name or attribute

        public void ExecuteAll(in TContext context, CancellationToken ct)
        {
            foreach (var stage in OrderedStages)
            {
                if (stage.Condition?.Evaluate(context) == false) continue;
                stage.ExecuteAsync(in context, ct).GetAwaiter().GetResult(); // or async pipeline runner
            }
        }
    }
}
```

### ✅ Why This Makes Your Pipeline & Stages Reusable

| Current Bottleneck | Reusable Solution | Benefit |
|-------------------|-------------------|---------|
| Manual `.AddStage()` wiring | `[[PipelineStage]]` + DI auto-discovery | Zero config registration; stages become plug-and-play modules |
| Rigid `ShouldRun(bool)` | Chainable `IPipelineCondition<T>` via C# 14 extensions | Compose filters declaratively (`Always().MustSatisfy(x => ...).Or(...)`) |
| Hardcoded middleware chains | Named groups (`"Timing+Retry"`) resolved at runtime | Swap middleware behaviors per environment (dev/prod) without code changes |
| Context mutation & allocations | `in` for read-only snapshots, `out` for diagnostics, immutable `with` records | GC pressure drops; thread safety improves in parallel stages |
| Tight coupling to `RagnarConfig` | Generic `IPipelineStage<TContext>` + DI options injection | Reuse same parsing/embedding logic in CLI, Web API, or Worker Service hosts |

### 🛠️ .NET 10 / C# 14 Alignment Notes
- **Extension Members**: Used to add composition methods (`MustSatisfy`, `ExecuteAll`) directly on interfaces/collections without inheritance chains.
- **Lambda Modifiers**: `in` prevents copying large `ReadOnlyCollection<T>` or span-like context snapshots in conditions/middleware delegates.
- **Records with `with`**: Used for safe, immutable state progression across pipeline stages (`context with { Documents = [[...]] }`).
- **Parallel/Async Best Practices**: `ConfigureAwait(false)`, `CancellationTokenSource.CreateLinkedTokenSource`, and `ConcurrentBag` ensure high-throughput, non-blocking execution.

This architecture transforms your pipeline from a monolithic workflow into a **composable engine** where stages, conditions, and middleware are fully interchangeable, testable, and reusable across .NET 10 workloads.
