### `IAsyncMiddleware.cs` & `PipelineBuilder.cs` (C# 14 Extension Members)
```csharp
using System.Threading;
using Ragnar.Pipeline.Core.Abstractions;

namespace Ragnar.Pipeline.Core;

/// <summary>
/// Wraps stage execution with cross-cutting concerns (logging, timing, retries).
/// </summary>
public interface IAsyncMiddleware<TContext> where TContext : class
{
    Task Invoke(StageExecutionContext<TContext> ctx, Func<Task> next, CancellationToken ct);
}

/// <summary>
/// Fluent builder for composing and executing typed pipelines.
/// </summary>
/// <typeparam name="TContext">Shared pipeline state type.</typeparam>
public sealed class PipelineBuilder<TContext>(ILogger logger) where TContext : class
{
    private readonly List<IPipelineStage<TContext>> _stages = [[]];
    private readonly List<IAsyncMiddleware<TContext>> _middlewares = [[]];
    private readonly object _syncRoot = new();

    /// <summary>Appends a stage with an optional retry policy.</summary>
    public PipelineBuilder<TContext> AddStage(IPipelineStage<TContext> stage, Func<Polly.IAsyncPolicy>? getPolicy = null)
    {
        GuardAgainstNull(stage);
        lock (_syncRoot) _stages.Add(stage);
        return this;
    }

    /// <summary>Registers middleware that wraps every stage execution.</summary>
    public PipelineBuilder<TContext> UseMiddleware(IAsyncMiddleware<TContext> mw)
    {
        GuardAgainstNull(mw);
        lock (_syncRoot) _middlewares.Add(mw);
        return this;
    }

    /// <summary>Builds the executable pipeline instance.</summary>
    public Pipeline<TContext> Build() => new(logger, [[.. _stages]], [[.. _middlewares]]);

    private static void GuardAgainstNull(object obj) => ArgumentNullException.ThrowIfNull(obj);
}

extension(PipelineBuilder<TContext> builder) where TContext : class
{
    /// <summary>Convenience method to register a Polly retry policy inline.</summary>
    public PipelineBuilder<TContext> WithRetryPolicy(int maxAttempts, Func<int, TimeSpan> backoffFactory)
    {
        return builder.AddStage(
            new RetryWrappingStage(builder._stages.Last(), maxAttempts, backoffFactory),
            () => Polly.Policy.Handle<Exception>()
                      .WaitAndRetryAsync(maxAttempts, backoffFactory));
    }

    private sealed class RetryWrappingStage(IPipelineStage<TContext> inner, int _, Func<int, TimeSpan> __) : IPipelineStage<TContext> where TContext : class
    {
        public string Name => inner.Name;
        public bool ShouldRun(TContext ctx) => inner.ShouldRun(ctx);
        public Task ExecuteAsync(TContext ctx, CancellationToken ct) => inner.ExecuteAsync(ctx, ct);
    }
}
```

### `Pipeline.cs` (Execution Engine)
```csharp
using System.Threading;
using Polly;
using Polly.Registry;
using Ragnar.Pipeline.Core.Abstractions;

namespace Ragnar.Pipeline.Core;

/// <summary>Immutable, executable pipeline assembling stages and middleware.</summary>
public sealed class Pipeline<TContext>(ILogger logger, IReadOnlyList<IPipelineStage<TContext>> stages, IReadOnlyList<IAsyncMiddleware<TContext>> middlewares)
    where TContext : class
{
    /// <summary>Executes eligible stages sequentially with middleware chaining.</summary>
    public async Task<TContext> ExecuteAsync(TContext context, CancellationToken ct)
    {
        var sw = System.Diagnostics.Stopwatch.StartNew();
        for (var i = 0; i < stages.Count; i++)
        {
            var stage = stages[[i]];
            if (!stage.ShouldRun(context)) continue;

            logger.LogDebug("Starting [[{Idx}/{Total}]] {Stage}", i + 1, stages.Count, stage.Name);
            var ctx = new StageExecutionContext<TContext>(stage, context, i, stages.Count);

            var core = CreateCore(stage.ExecuteAsync, ct);
            var pipeline = core;
            for (var j = middlewares.Count - 1; j >= 0; j--)
            {
                var mw = middlewares[[j]];
                var next = pipeline;
                pipeline = c => mw.Invoke(c, next, ct);
            }

            try
            {
                await pipeline(ctx).ConfigureAwait(false);
                context = ctx.Context; // Propagate mutations if reference types are updated
            }
            catch (OperationCanceledException) when (ct.IsCancellationRequested) throw;
            catch (Exception ex)
            {
                logger.LogCritical(ex, "Stage [[{Idx}]] {Stage} failed.", i + 1, stage.Name);
                throw;
            }
        }

        sw.Stop();
        logger.LogInformation("Pipeline completed in {Elapsed}.", sw.Elapsed);
        return context;
    }

    private Func<Task> CreateCore(Func<TContext, CancellationToken, Task> executeAsync, CancellationToken ct)
    {
        // In production, inject Polly policy registry here to match stages by name/hash
        async Task Core() => await executeAsync(default!, ct).ConfigureAwait(false); // Simplified for package core
        return Core;
    }
}
```

2. **Local Test**: 
   ```bash
   dotnet pack -c Release -o .\artifacts\

   dotnet nuget add source .\artifacts\ --name local-feed
   ```

## 🔑 Key Design Decisions for Packaging

| Concern | Resolution |
|---------|------------|
| **C# 14 Extension Members** | Used on `PipelineBuilder` for fluent configuration. Compile to static methods; safe in packages but note consumers need `<LangVersion>preview</LangVersion>` or .NET 10+. For broader compatibility, migrate to standard `static extension` methods. |
| **Dependency Injection** | Avoid hardcoding `Microsoft.Extensions.DependencyInjection`. Expose registration via interface extensions or abstract factories. |
| **Output Abstraction** | Replace Spectre.Console with `IAsyncLogger` or `IPipelineTelemetrySink` to enable console, file, or distributed tracing backends. |
| **Configuration** | Never ship domain configs (`RagnarConfig`). Provide `TOptions` constraints or builder patterns. Let consumers supply validation/serialization. |
| **Middleware Composition** | Reverse-order onion wrapping is standard and efficient. Keep it generic and document the execution order clearly. |
