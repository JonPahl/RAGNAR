# Extracting a Reusable Pipeline NuGet Package

Looking at your codebase, the **pipeline framework** (runner, middleware, stage contracts, execution context, metrics) is cleanly separated from the **application-specific stages** (branding, delta-embed, question filtering). That's exactly the boundary you want your package along.

---

## 1. Identify the Boundary

| **Go into the NuGet package** (framework) | **Stay in your app** (consumers) |

### `PipelineRunner.cs` (moved from your app, unchanged logic)

```csharp
using Polly;

namespace Ragnar.Pipeline;

/// <summary>
/// Orchestrates sequential execution of pipeline stages with retry,
/// middleware wrapping, and structured metrics.
/// </summary>
public sealed class PipelineRunner<TContext>(Microsoft.Extensions.Logging.ILogger logger)
    : BasePipeline<TContext> where TContext : class
{
    private readonly List<Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task>> _middlewares = [[]];
    private readonly List<StageEntry<TContext>> _stages = [[]];
    private readonly Lock _gate = new();

    // ── Middleware ───────────────────────────────────────────────
    public override IPipeline<TContext> UseMiddleware(
        Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task> middleware)
    {
        Guard.Against.Null(middleware);
        lock (_gate) _middlewares.Add(middleware);
        return this;
    }

    public IPipeline<TContext> UseTiming(StageTimingMiddleware mw)
        => UseMiddleware((ctx, next, ct) => mw.Invoke(ctx, next, ct));

    // ── Stage registration ───────────────────────────────────────
    public override IPipeline<TContext> AddStage(IPipelineStage<TContext> stage, IAsyncPolicy? policy = null)
    {
        Guard.Against.Null(stage);
        lock (_gate)
        {
            if (_stages.Any(x => x.PipelineStage.Name == stage.Name))
                throw new ArgumentException($"Stage '{stage.Name}' is already registered.", nameof(stage));
            _stages.Add(new StageEntry<TContext>(stage, policy));
        }
        return this;
    }

    public override IPipeline<TContext> RemoveStage(string name)
    {
        lock (_gate) _stages.RemoveAll(s => s.PipelineStage.Name == name);
        return this;
    }

    public override IPipeline<TContext> Reorder(string fromName, int newIndex)
    {
        lock (_gate)
        {
            var e = _stages.FirstOrDefault(s => s.PipelineStage.Name == fromName)
                    ?? throw new KeyNotFoundException(fromName);
            _stages.Remove(e);
            _stages.Insert(Math.Clamp(newIndex, 0, _stages.Count), e);
        }
        return this;
    }

    public IPipeline<TContext> ReplaceStage(string oldName, IPipelineStage<TContext> newStage, IAsyncPolicy? policy = null)
    {
        var idx = _stages.FindIndex(s => s.PipelineStage.Name == oldName);
        if (idx < 0) throw new KeyNotFoundException(oldName);
        RemoveStage(oldName);
        AddStage(newStage, policy);
        return Reorder(newStage.Name, idx);
    }

    public IPipeline<TContext> ClearStages()
    {
        lock (_gate) _stages.Clear();
        return this;
    }

    // ── Introspection ────────────────────────────────────────────
    public IReadOnlyList<string> StageNames
    {
        get { lock (_gate) return _stages.ConvertAll(s => s.PipelineStage.Name).AsReadOnly(); }
    }

    // ── Execution ────────────────────────────────────────────────
    public override async Task<TContext> ExecuteAsync(TContext context, CancellationToken ct)
    {
        var totalSw = System.Diagnostics.Stopwatch.StartNew();
        List<StageEntry<TContext>> snapshot;
        lock (_gate) snapshot = [[.._stages]];

        for (var i = 0; i < snapshot.Count; i++)
        {
            var entry = snapshot[[i]];
            var stage = entry.PipelineStage;

            if (!stage.ShouldRun(context))
            {
                logger.LogInformation("Skipping stage [[{Index}]]: {Name}", i, stage.Name);
                continue;
            }

            logger.LogInformation("Starting stage [[{Index}/{Total}]]: {Name}", i + 1, snapshot.Count, stage.Name);
            var execCtx = new StageExecutionContext<TContext>(stage, context, i, snapshot.Count);

            async Task Core()
            {
                if (entry.AsyncPolicy is null)
                    await stage.ExecuteAsync(context, ct).ConfigureAwait(false);
                else
                    await entry.AsyncPolicy.ExecuteAsync(
                        async () => await stage.ExecuteAsync(context, ct).ConfigureAwait(false)
                    ).ConfigureAwait(false);
            }

            try
            {
                await ExecuteStageWithMiddlewareAsync(execCtx, Core, ct).ConfigureAwait(false);
            }
            catch (OperationCanceledException) when (ct.IsCancellationRequested) { throw; }
            catch (Exception ex)
            {
                logger.LogCritical(ex, "Stage [[{Index}]] {Name} failed", i, stage.Name);
                throw new PipelineStageException(stage.Name, ex);
            }
        }

        logger.LogInformation("Pipeline finished in {Total}", totalSw.Elapsed);
        return context;
    }

    // ── Private ──────────────────────────────────────────────────
    private async Task ExecuteStageWithMiddlewareAsync(
        StageExecutionContext<TContext> ctx, Func<Task> core, CancellationToken ct)
    {
        List<Func<StageExecutionContext<TContext>, Func<Task>, CancellationToken, Task>> snap;
        lock (_gate) snap = [[.._middlewares]];

        var pipeline = core;
        for (var i = snap.Count - 1; i >= 0; i--)
        {
            var mw = snap[[i]];
            var nextRef = pipeline;
            pipeline = () => mw(ctx, nextRef, ct);
        }
        await pipeline().ConfigureAwait(false);
    }
}
```

### `StageTimingMiddleware.cs`

```csharp
using System.Diagnostics;

namespace Ragnar.Pipeline;

/// <summary>Measures wall-clock duration of a stage and populates <see cref="StageMetric"/>.</summary>
public sealed class StageTimingMiddleware(ILogger logger)
{
    public async Task Invoke<TContext>(
        StageExecutionContext<TContext> context,
        Func<Task> next,
        CancellationToken cancellationToken
    ) where TContext : class
    {
        ArgumentNullException.ThrowIfNull(context);
        ArgumentNullException.ThrowIfNull(next);

        var sw = Stopwatch.StartNew();
        var ok = false;
        string? err = null;
        try
        {
            await next().ConfigureAwait(false);
            ok = true;
        }
        catch (Exception ex)
        {
            err = ex.Message;
            throw;
        }
        finally
        {
            sw.Stop();
            context.Metric = new StageMetric
            {
                StageName  = context.Stage.Name,
                Duration   = sw.Elapsed,
                Success    = ok,
                RetryCount = 0,
                Error      = err
            };
            logger.LogInformation(
                "Stage [[{Idx}/{Tot}]] {Name} → {Elapsed} ({Ok})",
                context.StageIndex + 1, context.StageTotal,
                context.Stage.Name, sw.Elapsed, ok);
        }
    }
}
```

```csharp
using System.Diagnostics;

namespace Ragnar.Pipeline;

/// <summary>Prints a live progress line per stage (start, success, failure, cancellation).</summary>
public sealed class ProgressMiddleware(IOutputWriter writer)
{
    public async Task Invoke<TContext>(
        StageExecutionContext<TContext> context,
        Func<Task> next,
        CancellationToken cancellationToken
    ) where TContext : class
    {
        var label = $"[[{context.StageIndex + 1}/{context.StageTotal}]] {context.Stage.Name}";
        writer.WriteLine($"{label} …");
        var sw = Stopwatch.StartNew();
        try
        {
            await next().ConfigureAwait(false);
            writer.WriteLine($"✓ {sw.Elapsed:hh\\:mm\\:ss}");
        }
        catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
        {
            writer.WriteLine($"✗ cancelled after {sw.Elapsed:hh\\:mm\\:ss}");
            throw;
        }
        catch (Exception ex)
        {
            writer.WriteLine($"✗ {ex.GetType().Name} after {sw.Elapsed:hh\\:mm\\:ss}");
            throw;
        }
        writer.WriteLine();
    }
}
```

### `PipelineServiceExtensions.cs` (DI sugar)

```csharp
using Microsoft.Extensions.DependencyInjection;

namespace Ragnar.Pipeline.Extensions;

/// <summary>Registers the pipeline runner and default middleware in the DI container.</summary>
public static class PipelineServiceExtensions
{
    /// <summary>
    /// Registers a <see cref="PipelineRunner{TContext}"/> and the default
    /// timing + progress middleware for the given context type.
    /// </summary>
    public static IServiceCollection AddPipeline<TContext>(this IServiceCollection services)
        where TContext : class
    {
        services.AddSingleton<IPipeline<TContext>>(sp =>
        {
            var logger = sp.GetRequiredService<ILogger<PipelineRunner<TContext>>>();
            var writer = sp.GetRequiredService<IOutputWriter>();

            return new PipelineRunner<TContext>(logger)
                .UseTiming(new StageTimingMiddleware(logger))
                .UseMiddleware((ctx, next, ct) => new ProgressMiddleware(writer).Invoke(ctx, next, ct));
        });
        return services;
    }
}
```

## 5. Consumer-Side: A Trivial Stage

```csharp
using Ragnar.Pipeline;

public sealed class GreetingStage : IPipelineStage<MyContext>
{
    public string Name => "Greeting";
    public Polly.IAsyncPolicy? RetryPolicy => null;
    public bool ShouldRun(MyContext ctx) => true;

    public Task ExecuteAsync(MyContext ctx, CancellationToken ct)
    {
        ctx.Messages.Add("Hello from a custom stage!");
        return Task.CompletedTask;
    }
}

public sealed class MyContext
{
    public List<string> Messages { get; } = [[]];
}
```

```csharp
// Program.cs  (or any host)
var builder = Host.CreateApplicationBuilder(args);

builder.Services.AddSingleton<IOutputWriter, ConsoleOutputWriter>();

pipeline.AddStage(new GreetingStage());

var ctx = new MyContext();
await pipeline.ExecuteAsync(ctx, CancellationToken.None);
```

```bash
# Build + pack a local .nupkg
dotnet pack Src/Ragnar.Pipeline -c Release -o ./nupkg

# Publish to a feed
```

In your **app** project:

```xml
<PackageReference Include="Ragnar.Pipeline" Version="0.1.0" />
```

Then delete the pipeline source files from `Src/Ragnar/` and add a `<ProjectReference>` to the package (or the NuGet ref above). Your app keeps only the concrete stages, contexts, and DI registrations.

## TL;DR Checklist

1. **Create** `Src/Ragnar.Pipeline/` as a class-library project.
2. **Move** the generic framework types (interfaces, runner, middleware, records) in.
3. **Remove** any app-specific references (Qdrant, Ollama, Serilog, FluentValidation) from the package.
4. **Add** NuGet metadata to the `.csproj`, a `README.md`, and a `LICENSE`.
5. **Pack** with `dotnet pack` → consume via `<PackageReference>`.
6. **Test** the package in isolation with stub stages.
7. **Iterate** – your app is now the first real consumer; any friction you feel is a packaging smell worth fixing.
