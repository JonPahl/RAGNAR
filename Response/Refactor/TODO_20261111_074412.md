```csharp
/// <summary>
/// Middleware that wraps stage execution in a Polly retry policy and
/// records the final retry count into <see cref="StageMetric.RetryCount"/>.
/// </summary>
/// <example>
/// <![[CDATA[[
/// runner.UseMiddleware(async (ctx, next, ct) =>
/// {
///     var mw = new StageRetryMiddleware(logger, policy);
///     await mw.Invoke(ctx, next, ct);
/// });
/// ]]]]>
/// </example>
public sealed class StageRetryMiddleware(
    ILogger logger,
    Func<Exception, int, Task> onRetry = null)
{
    /// <summary>
    /// Wraps the next link with a retry loop (up to <paramref name="maxAttempts"/>),
    /// populating <see cref="StageMetric.RetryCount"/> on the context.
    /// </summary>
    /// <param name="context">The stage execution context.</param>
    /// <param name="next">The downstream middleware / core stage delegate.</param>
    /// <param name="cancellationToken">Token to abort.</param>
    /// <param name="maxAttempts">Total attempts including the first (default 3).</param>
    public async Task Invoke<TContext>(
        StageExecutionContext<TContext> context,
        Func<Task> next,
        CancellationToken cancellationToken,
        int maxAttempts = 3)
        where TContext : class
    {
        ArgumentNullException.ThrowIfNull(context);
        ArgumentNullException.ThrowIfNull(next);

        var lastEx = default(Exception);

        for (var attempt = 1; attempt <= maxAttempts; attempt++)
        {
            try
            {
                await next().ConfigureAwait(false);
                lastEx = null;
                break; // success
            }
            catch (Exception ex) when (attempt < maxAttempts)
            {
                lastEx = ex;
                var delay = TimeSpan.FromSeconds(Math.Pow(2, attempt - 1));
                logger.LogWarning(
                    "Stage {Name} attempt {Attempt}/{Max} failed: {Msg}. Retrying in {Delay}…",
                    context.Stage.Name, attempt, maxAttempts, ex.Message, delay);

                onRetry?.Invoke(ex, attempt);
                await Task.Delay(delay, cancellationToken).ConfigureAwait(false);
            }
            catch (Exception ex)
            {
                lastEx = ex;
                break; // final attempt failed
            }
        }

        // Record retry count so the timing middleware (running outer) can pick it up
        if (context.Metric is not null)
            context.Metric.RetryCount = lastEx is null ? 0 : maxAttempts;
        else if (lastEx is not null)
            throw lastEx; // let the timing middleware catch & rethrow

        // If we exhausted retries the last exception must still propagate.
        // (The timing middleware will rethrow it.)
        if (lastEx is not null)
            throw lastEx;
    }
}
```

If a parser supports multiple extensions (e.g. `.md` **and** `.markdown`), only the first is indexed. Fix:

```csharp
public class CodeParserFactory(
    ILogger logger,
    IOptions<RagnarConfig> options,
    IEnumerable<IFileParser> parsers) : IFileParseFactory
{
    // Map EVERY supported extension → its parser
    private readonly Dictionary<string, IFileParser> _byExt;

    public CodeParserFactory(
        ILogger logger,
        IOptions<RagnarConfig> options,
        IEnumerable<IFileParser> parsers) : this(logger, options)
    {
        _byExt = new Dictionary<string, IFileParser>(StringComparer.OrdinalIgnoreCase);
        foreach (var parser in parsers)
            foreach (var ext in parser.SupportedExtensions)
                _byExt[[ext]] = parser;   // last one wins on duplicates
    }

    // keep the primary ctor for DI
    private CodeParserFactory(ILogger logger, IOptions<RagnarConfig> options)
    {
        _logger   = logger;
        _options  = options;
        _byExt    = new(StringComparer.OrdinalIgnoreCase);
    }

    private readonly ILogger _logger;
    private readonly IOptions<RagnarConfig> _options;

    public async Task<IEnumerable<CodeDocument>> ParseAsync(
        string file, CancellationToken cancellationToken)
    {
        var ext = Path.GetExtension(file);          // e.g. ".cs"

        IFileParser parser;
        if (!_byExt.TryGetValue(ext, out parser))
        {
            parser = new MarkdownFileParser(_options, _logger);   // fallback
            _logger.LogWarning("No parser for '{Ext}'; falling back to Markdown", ext);
        }

        _logger.LogDebug("Dispatching '{Ext}' → {Parser}", ext, parser.GetType().Name);
        return await parser.ParseAsync(file, cancellationToken).ConfigureAwait(false);
    }
}
```
