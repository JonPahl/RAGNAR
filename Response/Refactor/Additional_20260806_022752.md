        Guard.Against.NullOrEmpty(filter.AllowedFileExtensions);

        var directoryName = file.DirectoryName;
        var extensionMatch = filter.AllowedFileExtensions.Any(ext => 
            ext.Equals(file.Extension, StringComparison.OrdinalIgnoreCase));
        var nameExcluded = filter.ExcludedFiles.Contains(file.Name);
        var dirExcluded = filter.ExcludedDirectories.Any(dir => 
            directoryName?.Contains(dir, StringComparison.OrdinalIgnoreCase) == true);

        return extensionMatch && !nameExcluded && !dirExcluded;
    }
}

public static class ChunkBySyntaxTree
{
    private const string UnknownElementName = Constants.UnknownElementName;

    public static IList<CodeDocument>? ChunkSourceFile(string filename, string codeText)
    {
        var response = new List<CodeDocument>();
        var tree = CSharpSyntaxTree.ParseText(codeText);
        if (tree.GetRoot() is not CompilationUnitSyntax root)
            return null;

        foreach (var node in root.DescendantNodes())
        {
            if (node is ClassDeclarationSyntax classDecl)
                response.Add(ExtractClassDocument(filename, classDecl));
        }

        return response;
    }

    private static CodeDocument ExtractClassDocument(string filename, ClassDeclarationSyntax classDefine)
    {
        var category = InferCategoryFromPath(filename);
        return CreateCodeDocument(
            filename,
            classDefine,
            classDefine.Kind().ToString(),
            classDefine.Identifier.ValueText ?? UnknownElementName,
            classDefine.GetLeadingTrivia(),
            category);
    }

    private static string InferCategoryFromPath(string path)
    {
        var fileName = Path.GetFileNameWithoutExtension(path).ToLowerInvariant();
        return fileName switch
        {
            _ when Regex.IsMatch(fileName, "^test", RegexOptions.IgnoreCase, TimeSpan.FromSeconds(5)) => "Testing",
            _ when Regex.IsMatch(fileName, "^plugin", RegexOptions.IgnoreCase, TimeSpan.FromSeconds(5)) => "Plugin",
            _ when Regex.IsMatch(fileName, "^embedding", RegexOptions.IgnoreCase, TimeSpan.FromSeconds(5)) => "Embedding",
            _ when Regex.IsMatch(fileName, "^core", RegexOptions.IgnoreCase, TimeSpan.FromSeconds(5)) => "Core",
            _ => "Other"
        };
    }

    private static CodeDocument CreateCodeDocument(string filename, SyntaxNode node, string elementType, string elementName, SyntaxTriviaList leadingTrivia, string category)
    {
        var comments = LocateComments(leadingTrivia);
        return new CodeDocument(
            filename,
            elementType,
            elementName,
            Comment: string.Join(Environment.NewLine, comments),
            CommentLength: string.Join(Environment.NewLine, comments).CharacterCount(),
            Code: node.NormalizeWhitespace().ToFullString(),
            Category: category);
    }

    private static List<string> LocateComments(SyntaxTriviaList? leadingTrivia)
    {
        if (leadingTrivia == null)
            return [];

        return leadingTrivia.Value
            .Where(t => t.IsKind(SyntaxKind.SingleLineDocumentationCommentTrivia) ||
                        t.IsKind(SyntaxKind.MultiLineDocumentationCommentTrivia) ||
                        t.IsKind(SyntaxKind.SingleLineCommentTrivia))
            .Select(t => t.ToString().Trim())
            .ToList();
    }
}

public static class StringExtensions
{
    public static int CharacterCount(this ReadOnlySpan<char> xmlComment)
    {
        var count = 0;
        var inTag = false;

        foreach (var c in xmlComment)
        {
            if (c == '<')
            {
                inTag = true;
                continue;
            }

            if (c == '>')
            {
                inTag = false;
                continue;
            }

            if (!inTag && c != '/')
                count++;
        }

        return count;
    }

    public static ReadOnlySpan<char> GetLastFolder(this string path)
    {
        if (string.IsNullOrEmpty(path))
            return ReadOnlySpan<char>.Empty;

        var span = path.AsSpan().TrimEnd(Path.DirectorySeparatorChar);
        var lastSlash = span.LastIndexOf(Path.DirectorySeparatorChar);
        return lastSlash < 0 ? span : span.Slice(lastSlash + 1);
    }
}
