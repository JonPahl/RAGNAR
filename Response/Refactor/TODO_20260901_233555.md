```csharp
// ── IFileValidator.cs (unchanged contract) ──────────────────────────
public interface IFileValidator
{
    bool IsValid(FileInfo file, in FileLoadOptions options);
}

// ── FileValidationResult.cs (new value object for the validator) ────
public sealed record FileValidationInput(
    string Directory,
    string FileName,
    string Extension,
    in FileLoadOptions Options);

// ── FileValidationRules.cs (FluentValidation validator) ─────────────
public sealed class FileValidationRules : AbstractValidator<FileValidationInput>
{
    public FileValidationRules()
    {
        RuleFor(x => x.Extension)
            .NotEmpty().WithMessage("File extension is required.")
            .Must((input, ext) =>
                input.Options.AllowedFileExtensions.Any(a => a.Equals(ext, StringComparison.OrdinalIgnoreCase)))
            .WithMessage("File extension '{Extension}' is not in the allowed list.");

        RuleFor(x => x.FileName)
            .Must((input, name) =>
                !input.Options.ExcludedFiles.Contains(name, StringComparer.OrdinalIgnoreCase))
            .WithMessage("File '{FileName}' is excluded by configuration.");

        RuleFor(x => x.Directory)
            .Must((input, dir) =>
                !input.Options.ExcludedDirectories.Any(excl =>
                    dir.Equals(excl, StringComparison.OrdinalIgnoreCase) ||
                    dir.StartsWith($"{excl}{Path.DirectorySeparatorChar}", StringComparison.OrdinalIgnoreCase)))
            .WithMessage("Directory '{Directory}' is excluded by configuration.");
    }
}

// ── FileValidator.cs (thin wrapper, DI-friendly) ────────────────────
public sealed class FileValidator(IValidator<FileValidationInput> fileValidator) : IFileValidator
{
    public bool IsValid(FileInfo file, in FileLoadOptions options)
    {
        var input = new FileValidationInput(
            file.DirectoryName ?? string.Empty,
            file.Name,
            file.Extension,
            in options);

        var result = fileValidator.Validate(input);
        return result.IsValid;
    }
}

// ── ServiceCollection registration ──────────────────────────────────
services.AddSingleton<IValidator<FileValidationInput>, FileValidationRules>();
services.AddSingleton<IFileValidator, FileValidator>();
```

        }
        catch (Exception ex)
        {
            _logger.Error(ex, ex.Message);

```csharp
// ── ParseDocumentsStage.cs ──────────────────────────────────────────
public sealed class ParseDocumentsStage(
    DiscoverFilesStage discoverStage,
    IFileParseFactory parseFactory,
    ILogger logger) : IPipelineStage
{
    private readonly ConcurrentBag<CodeDocument> _documents = [[]];

    public IReadOnlyList<CodeDocument> ParsedDocuments { get; private set; } = [[]];

    public async Task ExecuteAsync(CancellationToken cancellationToken)
    {
        if (discoverStage.DiscoveredFiles.Count == 0) return;

        await AnsiConsole.Progress().AutoClear(true).StartAsync(async ctx =>
        {
            var task = ctx.AddTask("Parsing files…", maxValue: discoverStage.DiscoveredFiles.Count);
            await Parallel.ForEachAsync(
                discoverStage.DiscoveredFiles,
                new ParallelOptions { CancellationToken = cancellationToken, MaxDegreeOfParallelism = Math.Min(Environment.ProcessorCount, 8) },
                async (filePath, token) =>
                {
                    var elements = await parseFactory.ParseAsync(filePath, token);
                    foreach (var e in elements) _documents.Add(e);
                    task.Increment(1);
                });
        });

        ParsedDocuments = [[.. _documents]];
    }
}
```

```csharp
// ── UpsertEmbeddingsStage.cs ────────────────────────────────────────
public sealed class UpsertEmbeddingsStage(
    ParseDocumentsStage parseStage,
    IVectorStoreRepository repository,
    ILogger logger) : IPipelineStage
{
    private const int BatchSize = 1;

    public async Task ExecuteAsync(CancellationToken cancellationToken)
    {
        var docs = parseStage.ParsedDocuments;
        if (docs.Count == 0) return;

        await AnsiConsole.Progress().AutoClear(true).StartAsync(async ctx =>
        {
            var task = ctx.AddTask("Embedding & upserting…", maxValue: docs.Count);
            foreach (var batch in docs.Chunk(BatchSize))
            {
                cancellationToken.ThrowIfCancellationRequested();
                await repository.UpsertBatchAsync(batch, cancellationToken);
                task.Increment(batch.Length);
                ctx.Refresh();
            }
        });
    }
}
```
