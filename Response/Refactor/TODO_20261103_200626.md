## 1. Retry Middleware (comment in `StageTimingMiddleware.cs`)

> `RetryCount = 0, // populate via a separate retry middleware if needed`

The timing middleware hard-codes `RetryCount` to `0`. A dedicated retry middleware is expected to track how many times a stage was actually retried before succeeding (or failing).

```csharp
// ─────────────────────────────────────────────
// RetryMiddleware.cs
// ─────────────────────────────────────────────
using Polly;
using Serilog;

/// <summary>
/// Middleware that wraps a stage in a Polly retry policy and
/// populates <see cref="StageMetric.RetryCount"/> on the context.
/// </summary>
/// <param name="logger">Serilog logger for diagnostic messages.</param>
public sealed class RetryMiddleware(Serilog.ILogger logger)
{
    /// <summary>
    /// Invokes the next middleware, counting each retry attempt
    /// and recording the final count on the stage metric.
    /// </summary>
    public async Task Invoke<TContext>(
        StageExecutionContext<TContext> context,
        Func<Task> next,
        CancellationToken cancellationToken)
        where TContext : class
    {
        var attempt = 0;
        Exception? lastEx = null;

        // Use a simple retry policy as a demonstration.
        // In production you would pull the policy from the StageEntry.
        var policy = Policy
            .Handle<TransientHttpErrorCondition>()
            .Or<TimeoutException>()
            .WaitAndRetryAsync(
                retries: 3,
                retryCount => TimeSpan.FromSeconds(Math.Pow(2, retryCount)),
                onRetry: (ex, ts) =>
                {
                    attempt++;
                    logger.Warning(
                        "Stage {Stage} attempt {Attempt} failed: {Ex}. Retrying in {Ts}",
                        context.Stage.Name, attempt, ex.Message, ts);
                });

        try
        {
            await policy.ExecuteAsync(async () =>
            {
                cancellationToken.ThrowIfCancellationRequested();
                await next().ConfigureAwait(false);
            });
        }
        catch (Exception ex)
        {
            lastEx = ex;
            throw;
        }
        finally
        {
            // Stamp the retry count onto the metric that the
            // timing middleware (or a later middleware) will persist.
            if (context.Metric is not null)
                context.Metric = context.Metric with { RetryCount = attempt };
            else
                context.Metric = new StageMetric
                {
                    StageName = context.Stage.Name,
                    RetryCount = attempt,
                    Success   = lastEx is null,
                    Error     = lastEx?.Message,
                };
        }
    }
}
```

```csharp
public void Render()
{
    if (_numbered)
    {
        inner.Render();
        return;
    }

    _numbered = true;
    var table = inner.Table;

    // 1. Ensure the index column exists
    if (!table.Columns.Contains(columnHeader, StringComparer.OrdinalIgnoreCase))
        table.AddColumns(columnHeader);

    // 2. Snapshot existing rows, then CLEAR and re-add with index
    var existing = table.Rows
                       .Select(r => r.Values.Select(v => v.ToString() ?? string.Empty).ToList())
                       .ToList();

    // Clear all rows
    while (table.Rows.Count > 0)
        table.Rows.RemoveAt(table.Rows.Count - 1);

    // Re-add with 1-based index prepended
    for (var i = 0; i < existing.Count; i++)
    {
        var cells = new List<TableCell> { new(columnHeader) { Value = (i + 1).ToString() } };
        cells.AddRange(existing[[i]].Select(c => new TableCell(c)));
        table.AddRow(cells);
    }

    // 3. Delegate final rendering to inner
    inner.Render();
}
```

```csharp
public async Task<IEnumerable<CodeDocument>> ParseAsync(string file, CancellationToken cancellationToken)
{
    cancellationToken.ThrowIfCancellationRequested();
    var fileInfo = new FileInfo(file);

    var ext = fileInfo.Extension.ToLowerInvariant();

    if (ext is not (".cs" or ".md" or ".markdown" or ".txt" or ".xml" or ".json" or ".yaml" or ".yml"))
    {
        logger.Warning(
            "File {File} has unrecognised extension '{Ext}'. Falling back to generic parser.",
            file, ext);
    }

    var docs = ext switch
    {
        ".cs"            => await GetCodeDocumentsAsync(file, cancellationToken).ConfigureAwait(false),
        _                => await ParseFile(file, cancellationToken).ConfigureAwait(false),
    };

    logger.Debug("Parse dispatched for {File}, produced {Count} documents", file, docs.Count());
    return docs;
}
```
