### 🔴 1. Broken Document-to-Vector Mapping in Batch Upserting
**Impact:** High | **Remediation Effort:** Medium

#### Problem
In `VectorStoreWriter.UpsertBatchAsync`, documents are mapped to embeddings using index-based correlation:
```csharp
var texts = batch.Select(d => $"Context: {d.ElementName}\nCode:\n{d.Code}").ToList();
var generated = await embeddingService.GenerateBatchAsync(texts, ct);
var points = batch.Select((d, idx) => generatorService.BuildPointStructs(d.AsPoint(), generated[idx].Vector.ToArray(), d))
                  .SelectMany(p => p).ToList(); // ❌ BREAKS 1:N CORRELATION
```
`BuildPointStructs` returns `IReadOnlyList<PointStruct>`, but `SelectMany` flattens the collection, losing the index alignment. Payloads will mismatch vectors, corrupting Qdrant's similarity search and creating orphaned points.

#### C# 14 Fix (Leveraging `in` & Span for Zero-Allocation Safety)
```csharp
extension(IVectorStoreWriter self)
{
    public async Task<UpdateResult> UpsertBatchFixedAsync(IEnumerable<CodeDocument> codeDocuments, in CancellationToken cancellationToken)
    {
        Guard.Against.Null(codeDocuments);
        var docs = codeDocuments as List<CodeDocument> ?? [..codeDocuments];
        var batchSize = options?.Value?.EmbeddingOptions?.BatchSize ?? 32;
        
        await Parallel.ForEachAsync(docs.Chunk(batchSize), new ParallelOptions { CancellationToken = cancellationToken }, async (batch, token) =>
        {
            // Efficient string span construction to avoid allocations
            ReadOnlySpan<char> promptSpan = $"Context: {d.ElementName}\nCode:\n{d.Code}".AsSpan();
            
            var embeddings = await embeddingService.GenerateBatchAsync(batch.Select(d => promptSpan.ToString()).ToArray(), token);
            
            // Preserve 1:N relationship safely
            var points = new List<PointStruct>(batch.Length * 2);
            foreach (var (doc, idx) in batch.Select((d, i) => (d, i)))
            {
                if (idx >= embeddings.Length) continue;
                
                var structList = generatorService.BuildPointStructs(doc.AsPoint(), embeddings[idx].Vector.Span, doc);
                points.AddRange(structList);
            }
            
            await qdrantClient.UpsertAsync(_appOptions.VectorStoreName, points, token).ConfigureAwait(false);
        }).ConfigureAwait(false);
        
        return new UpdateResult { Status = UpdateStatus.Completed };
    }
}
```

### 🟠 2. Fragile & Monolithic DI Registration
**Impact:** High | **Remediation Effort:** High

#### Problem
`RagPipelineHostBuilder.AddCoreServices()` manually wires ~15 services in a single method, overrides default `ILogger` resolution, and calls `AddHttpClients()` twice. This breaks modularity, makes unit testing difficult, and introduces registration-order dependencies.

        services.AddSingleton<IRagPromptBuilder, RagPromptBuilder>();
        services.AddSingleton<IOutputWriter, AnsiConsoleOutputWriter>();
        services.AddSingleton<IClock, SystemClock>();
        services.AddSingleton<IPathResolver, PathResolver>();

        // ✅ Async factory for Qdrant (avoids blocking host startup)
        services.AddSingleton(sp =>
        {
            var opt = sp.GetRequiredService<IOptions<RagnarConfig>>().Value;
            return new QdrantClient(opt.EmbeddingOptions.Host.Host, opt.EmbeddingOptions.Host.Port, https: false);
        });

        // ✅ Keyed & named service registration for LLM providers
        services.AddKeyedSingleton<IChatPromptProvider>("Common", sp => new PromptTemplateProvider());

                    BreakDuration = TimeSpan.FromMinutes(1),
                    MinimumThroughput = 3,

            });
        }

        services.AddHttpClient(nameof(OllamaServiceType.Ollama)).ConfigureHttpClient(c => Configure(c, nameof(OllamaServiceType.Ollama)));
        services.AddHttpClient(nameof(OllamaServiceType.Embedding)).ConfigureHttpClient(c => Configure(c, nameof(OllamaServiceType.Embedding)));
        return services;

### 🟡 3. Unsafe Console Live Rendering & Stream Coupling
**Impact:** Medium-High | **Remediation Effort:** Medium

#### Problem
`OllamaChatService.GenerateResponse` directly mutates `AnsiConsole.Live` targets inside a token-stream loop without synchronization, cancellation checkpoints, or backpressure control. Rapid LLM tokens cause console thrashing, potential deadlocks, and memory leaks if the task is cancelled mid-refresh.

#### C# 14 Fix (Channel-Based Buffering + Safe Refresh)
```csharp
public class SafeLiveConsoleWriter : IOutputWriter
{
    private readonly Channel<string> _buffer = Channel.CreateBounded<string>(new BoundedChannelOptions(64) { FullMode = BoundedChannelFullMode.DropOldest });
    private Task _processor;

    public SafeLiveConsoleWriter()
    {
        _processor = ProcessBufferAsync(CancellationToken.None);
    }

    extension(IOutputWriter self)
    {
        public void QueueUpdate(in string content, in Style style)
        {
            var markup = new Markup(content, style.EnsureValidStyle());
            _buffer.Writer.TryWrite(markup.ToString());
        }

        public async Task FlushAsync(out CancellationToken ct)
        {
            ct = new CancellationTokenSource(TimeSpan.FromSeconds(2)).Token;
            // Wait safely for buffer drain without blocking the LLM thread
            await _processor.WaitAsync(ct).ConfigureAwait(false);
        }
    }

    private async Task ProcessBufferAsync(CancellationToken cancellationToken)
    {
        await foreach (var line in _buffer.Reader.ReadAllAsync(cancellationToken).ConfigureAwait(false))
        {
            if (cancellationToken.IsCancellationRequested) break;
            AnsiConsole.MarkupLine(line, new Style(Color.Grey));
            // Throttle rapid updates to prevent UI thread starvation
            await Task.Delay(16, cancellationToken).ConfigureAwait(false);
        }
    }
}
```
