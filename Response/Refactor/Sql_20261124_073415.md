// File: SqliteQuestionSource.cs
using Microsoft.EntityFrameworkCore;

namespace YourNamespace.Data;

/// <summary>Loads questions directly from a SQLite database instead of CSV files.</summary>
public sealed class SqliteQuestionSource(IOptions<RagnarConfig> options, ILogger<SqliteQuestionSource> logger) : IQuestionSource
{
    private readonly string _defaultDbPath = Path.Join(AppContext.BaseDirectory, "Data", "questions.db");

    public string SourceName => "SQLite Database";

    /// <summary>
    /// Loads enabled questions. <paramref name="fileName"/> is used as the SQLite database path.
    /// </summary>
    public async Task<IEnumerable<Question>> LoadQuestionsAsync(string fileName, CancellationToken cancellationToken)
    {
        // Use 'in' modifier for zero-copy string handling where appropriate
        var dbPath = string.IsNullOrWhiteSpace(fileName) ? _defaultDbPath : Path.GetFullPath(fileName);
        
        using var context = new QuestionsDbContext(new DbContextOptionsBuilder<QuestionsDbContext>()
            .UseSqlite($"Data Source={dbPath};Cache=Shared;Pooling=True")
            .EnableSensitiveDataLogging(false)
            .Options);

        try
        {
            var entities = await context.Questions
                .AsNoTracking()
                .Where(q => q.IsEnabled)
                .OrderBy(q => q.Category)
                .ThenBy(q => q.Filename)
                .ToListAsync(cancellationToken)
                .ConfigureAwait(false);

            // C# 14 extension member invocation for clean domain mapping
            return entities.Select(e => e.ToDomain()).ToArray();
        }
        catch (Exception ex) when (ex is not OperationCanceledException)
        {
            logger.LogError(ex, "Failed to load questions from SQLite at {Path}", dbPath);
            // Fail gracefully to prevent pipeline stall
            return [[]];
        }
    }

    /// <summary>Validates DB existence and returns success state via 'out' parameter.</summary>
    public static bool TryGetDbPath(in string configSource, out string resolvedPath)
    {
        var path = Path.Join(AppContext.BaseDirectory, "Data", "questions.db");
        if (string.IsNullOrWhiteSpace(configSource))
        {
            resolvedPath = path;
            return File.Exists(path);
        }

        resolvedPath = Path.GetFullPath(configSource);
        return File.Exists(resolvedPath);
    }
}
