```csharp
public static class QdrantRepositoryExtensions { }

extension(IUpstserRepository repository)
{
    /// <summary>Upserts a batch of code documents with deterministic point IDs.</summary>
    public async Task<UpdateResult> UpsertAsync(in ReadOnlySpan<CodeDocument> documents, CancellationToken ct)
    {
        Guard.Against.Null(repository);
        var chunks = documents.Chunk(32);
        var failed = 0;

        foreach (var batch in chunks)
        {
            try
            {
                await repository.ProvisionAsync(batch.AsSpan(), ct).ConfigureAwait(false);
            }
            catch (Exception ex) when (ex is not OperationCanceledException)
            {
                failed++;
            }
        }

        return new UpdateResult
        {
            Status = failed == 0 ? UpdateStatus.Completed : UpdateStatus.UnknownUpdateStatus,
            UpsertedCount = documents.Length - failed
        };
    }
}
```

#### 3. Optimized Prompt & RAG Builder with `ref`/`out` Semantics

```csharp
public interface IRagPromptBuilder
{
    IRagPromptBuilder WithQuestion(Question question);
    IRagPromptBuilder WithContext(in ReadOnlySpan<char> context);
    string Build();
}

public sealed class RagPromptBuilder : IRagPromptBuilder
{
    private readonly StringBuilder _prompt = new(512);
    private Question? _question;
    private ReadOnlyMemory<char> _context;

    public IRagPromptBuilder WithQuestion(Question question)
    {
        _question = question ?? throw new ArgumentNullException(nameof(question));
        return this;
    }

    public IRagPromptBuilder WithContext(in ReadOnlySpan<char> context)
    {
        // 'in' avoids string copy for large RAG contexts
        _context = context.ToArray();
        return this;
    }

    public string Build()
    {
        var sb = _prompt;
        sb.Clear();
        sb.AppendLine("Context:");
        sb.Append(_context.Span);
        sb.AppendLine("\n\nQuestion:");
        sb.AppendLine(_question?.Text ?? "N/A");
        sb.AppendLine("\nAnswer:");
        
        return sb.ToString(0, sb.Length); // Zero-copy span to string for final output
    }
}
```
