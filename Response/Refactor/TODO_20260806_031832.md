```csharp
public static int CharacterCount(this ReadOnlySpan<char> xml)
{
    var count = 0;
    var inTag = false;
    var i = 0;
    while (i < xml.Length)
    {
        if (xml[i] == '<')
        {
            inTag = true;
            // Skip comment delimiters <!-- -->
            if (i + 3 < xml.Length && xml.Slice(i, 4).ToString() == "<!--")
            {
                i += 4;
                while (i + 2 < xml.Length && !xml.Slice(i, 3).ToString().Equals("-->"))
                    i++;
                i += 3;
                continue;
            }
        }
        else if (xml[i] == '>')
        {
            inTag = false;
        }

        if (!inTag && xml[i] != '/')
            count++;

        i++;
    }
    return count;
}
```

```csharp
public static string ValidateHost(this string host)
{
    Guard.Against.NullOrEmpty(host);
    if (!Uri.TryCreate(host, UriKind.RelativeOrAbsolute, out var uri))
        throw new ArgumentException("Invalid URI.", nameof(host));
```
