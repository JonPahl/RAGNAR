        var sb = new StringBuilder();

public interface IStreamResponseWriter
{
    /// <summary>Streams token chunks directly to a file without full in-memory buffering.</summary>
    Task WriteAsync(in IAsyncEnumerable<string> tokenStream, in string outputPath, in CancellationToken ct);
}

// StreamResponseWriter.cs
public sealed class StreamResponseWriter : IStreamResponseWriter
{
    public async Task WriteAsync(in IAsyncEnumerable<string> tokenStream, in string outputPath, in CancellationToken ct)
    {
        var dir = Path.GetDirectoryName(outputPath);
        if (!string.IsNullOrEmpty(dir)) Directory.CreateDirectory(dir);

        await using var fs = new FileStream(outputPath, FileMode.Append, FileAccess.Write, FileShare.None, bufferSize: 4096, useAsync: true);
        await foreach (var chunk in tokenStream.WithCancellation(ct).ConfigureAwait(false))
        {
            var bytes = Encoding.UTF8.GetBytes(chunk);
            await fs.WriteAsync(bytes.AsMemory(), ct).ConfigureAwait(false);
        }
    }
}

// Extension property for safe path resolution (C# 14)
public static class PathExtensions
{
    extension(string path)
    {
        public string GetSafeDirectory(in IPathResolver resolver) => 
            string.IsNullOrEmpty(Path.GetDirectoryName(path)) ? Directory.GetCurrentDirectory() : resolver.ResolveResponseDirectory(null);
    }
}
