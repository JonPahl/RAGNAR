```csharp
public static class CategoryInferencer
{
    private static readonly Regex[] _categoryPatterns =
    {
        new("^test", RegexOptions.IgnoreCase, TimeSpan.FromSeconds(5)),
        new("^plugin", RegexOptions.IgnoreCase, TimeSpan.FromSeconds(5)),
        new("^embedding", RegexOptions.IgnoreCase, TimeSpan.FromSeconds(5)),
        new("^core", RegexOptions.IgnoreCase, TimeSpan.FromSeconds(5))
    };

    private static readonly string[] _categoryNames = ["Testing", "Plugin", "Embedding", "Core"];

    public static string InferFromPath(string path)
    {
        var fileName = Path.GetFileNameWithoutExtension(path).ToLowerInvariant();
        for (int i = 0; i < _categoryPatterns.Length; i++)
        {
            if (_categoryPatterns[i].IsMatch(fileName))
                return _categoryNames[i];
        }
        return "Other";
    }
}
```

### 2. **CommentExtractor** *(SRP)*
```csharp
public static class CommentExtractor
{
    public static List<string> ExtractDocumentationComments(SyntaxTriviaList? triviaList)
    {
        if (triviaList is null) return [];

        return triviaList
            .Where(t => t.IsKind(SyntaxKind.SingleLineDocumentationCommentTrivia) ||
                        t.IsKind(SyntaxKind.MultiLineDocumentationCommentTrivia) ||
                        t.IsKind(SyntaxKind.SingleLineCommentTrivia))
            .Select(t => t.ToString().Trim())
            .ToList();
    }
}
```

### 3. **DocumentExtractor** *(SRP, DRY)*
```csharp
public static class DocumentExtractor
{
    public static CodeDocument ExtractFromSyntaxNode(
        string filePath,
        SyntaxNode node,
        string elementType,
        string elementName,
        SyntaxTriviaList? leadingTrivia = null)
    {
        var category = CategoryInferencer.InferFromPath(filePath);
        var comments = CommentExtractor.ExtractDocumentationComments(leadingTrivia);
        var commentText = string.Join(Environment.NewLine, comments);

        return new CodeDocument(
            fileName: filePath,
            elementType: elementType,
            elementName: elementName,
            comment: commentText,
            commentLength: commentText.CharacterCount(),
            code: node.NormalizeWhitespace().ToFullString(),
            category: category);
    }

    public static CodeDocument ExtractClassDocument(string filePath, ClassDeclarationSyntax classDecl)
        => ExtractFromSyntaxNode(filePath, classDecl, "Class", classDecl.Identifier.ValueText ?? "Unknown", classDecl.GetLeadingTrivia());

    public static CodeDocument ExtractMethodDocument(string filePath, MethodDeclarationSyntax method)
        => ExtractFromSyntaxNode(filePath, method, "Method", method.Identifier.ValueText ?? "Unknown", method.GetLeadingTrivia());
}
```

```csharp
public class FileParserSelector : IFileParserSelector
{
    private readonly BaseFileParser _csParser = new ParseCSharpFile();
    private readonly BaseFileParser _genericParser;

    public FileParserSelector(IOptions<AppConfiguration> options)
        => _genericParser = new FileParser(options);

    public ValueTask<CodeDocument[]> ParseAsync(string filePath, CancellationToken ct)
    {
        var ext = Path.GetExtension(filePath);
        return ext == ".cs" ? _csParser.ParseFileAsync(filePath, ct) : _genericParser.ParseFileAsync(filePath, ct);
    }
```

```csharp
public static class HostBuilderExtensions
{
    public static IServiceCollection RegisterAllOptions(this IServiceCollection services, HostBuilderContext context)
    {
        services.AddOptions<RagOptions>()
            .Bind(context.Configuration.GetSection("RagOptions"))
            .ValidateDataAnnotations()
            .ValidateOnStart()
            .Validate(rag => !string.IsNullOrWhiteSpace(rag.SourceDirectory), "RagOptions.SourceDirectory must not be empty")
            .Validate(rag => Directory.Exists(rag.SourceDirectory.ExpandDirectory()), "RagOptions.SourceDirectory does not exist");

        services.AddOptions<EmbeddingOptions>()
            .Bind(context.Configuration.GetSection("EmbeddingOptions"))
            .ValidateDataAnnotations()
            .ValidateOnStart();

        services.AddOptions<FileLoadOptions>()
            .Bind(context.Configuration.GetSection("FileLoadOptions"))
            .ValidateDataAnnotations()
            .ValidateOnStart();

        services.AddOptions<OllamaOptions>()
            .Bind(context.Configuration.GetSection("OllamaOptions"))
            .ValidateDataAnnotations()
            .ValidateOnStart()
            .PostConfigure(opt =>
            {
                opt.Host = opt.Host.ValidateHost();
                opt.Port = opt.Port.ValidatePort();
            });

        services.Configure<AppConfiguration>(context.Configuration)
            .PostConfigure(opts => opts.RagOptions.SourceDirectory = opts.RagOptions.SourceDirectory.ExpandDirectory());

        return services;
    }

    public static IServiceCollection RegisterEmbeddingGenerator(this IServiceCollection services, HostBuilderContext context)
    {
        services.AddSingleton(sp =>
        {
            var embeddingOpt = context.Configuration.GetSection("EmbeddingOption").Get<EmbeddingOptions>();
            var ollamaOpt = context.Configuration.GetSection("OllamaOption").Get<OllamaOptions>();
            var httpClient = sp.GetRequiredService<IHttpClientFactory>().CreateClient(nameof(OllamaServiceType.Ollama));

            httpClient.BaseAddress = new Uri($"{ollamaOpt.Host}:{ollamaOpt.Port}");
            httpClient.Timeout = ollamaOpt.Timeout;

            return new OllamaApiClient(httpClient)
            {
                SelectedModel = embeddingOpt.EmbeddingModel ?? throw new ArgumentException("Embedding Model is required")
            }.AsEmbeddingGenerator();
        });

        return services;
    }

    public static IServiceCollection AddResilientHttpClient(this IServiceCollection services, string name, Action<HttpClient> configureClient, Action<ResiliencePipelineBuilder<HttpResponseMessage>> configureResilience)
    {
        services.AddHttpClient(name, configureClient)
            .AddStandardResilienceHandler(configureResilience);

        return services;
    }
}
```
