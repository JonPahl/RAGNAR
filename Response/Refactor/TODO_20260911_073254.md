```csharp
namespace Core.PipelineStages;

/// <summary>Discovers source files matching configured load options.</summary>
public sealed class DiscoveryStage(
    IOptions<RagnarConfig> options,
    ICustomFileLoader fileLoader,
    ILogger logger,
    IOutputWriter writer) : IPipelineStage<EmbeddingContext>
{
    public string Name => "Discovering files…";

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken ct)
    {
        var sourceDir = context.SourceDirectory
            ?? options.Value.ApplicationOptions.SourceDirectory;

        if (!Directory.Exists(sourceDir))
        {
            logger.Warning("Source directory not found: {Dir}", sourceDir);
            return;
        }

        var files = await fileLoader
            .GetFilesAsync(sourceDir, options.Value.FileLoadOptions, ct)
            .ToListAsync(ct);

        writer.MarkupLine($"[[green]]Discovered {files.Count} source file(s).[[/]]");

        // Let the caller optionally render a live table (decorator – see TODO 2)
        context.DiscoveredFiles = files;
    }
}
```

```csharp
// Core/PipelineStages/ParsingStage.cs

namespace Core.PipelineStages;

/// <summary>Parses discovered files into CodeDocument segments in parallel.</summary>
public sealed class ParsingStage(
    IFileParseFactory parseFactory,
    ILogger logger,
    IOutputWriter writer) : IPipelineStage<EmbeddingContext>
{
    public string Name => "Parsing files…";

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken ct)
    {
        var files = context.DiscoveredFiles;
        if (files.Count == 0)
        {
            logger.Information("No files to parse. Skipping.");
            return;
        }

        var documents = new ConcurrentBag<CodeDocument>();
        int maxParallelism = Math.Min(Environment.ProcessorCount, 8);

        await AnsiConsole.Progress()
            .AutoClear(true)
            .StartAsync(async ctx =>
            {
                var task = ctx.AddTask(Name, maxValue: files.Count);

                await Parallel.ForEachAsync(
                    files,
                    new ParallelOptions
                    {
                        CancellationToken = ct,
                        MaxDegreeOfParallelism = maxParallelism
                    },
                    async (filePath, token) =>
                    {
                        var elements = await parseFactory.ParseAsync(filePath, token);
                        foreach (var element in elements)
                            documents.Add(element);
                        task.Increment(1);
                    });
            });

        context.Documents = [[.. documents]];
        logger.Information("Parsed {Count} code document(s).", context.Documents.Count);
    }
}
```

```csharp
// Core/PipelineStages/UpsertStage.cs

namespace Core.PipelineStages;

/// <summary>Embeds and upserts parsed documents into the Qdrant collection in batches.</summary>
public sealed class UpsertStage(
    IVectorStoreRepository repository,
    ILogger logger,
    IOutputWriter writer) : IPipelineStage<EmbeddingContext>
{
    private const int BatchSize = 32;
    public string Name => "Embedding & upserting…";

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken ct)
    {
        var documents = context.Documents;
        if (documents.Count == 0)
        {
            logger.Information("No documents to upsert. Skipping.");
            return;
        }

        UpdateResult lastResult = new() { Status = UpdateStatus.Completed };

        await AnsiConsole.Progress()
            .AutoClear(true)
            .StartAsync(async ctx =>
            {
                var task = ctx.AddTask(Name, maxValue: documents.Count);

                foreach (var batch in documents.Chunk(BatchSize))
                {
                    ct.ThrowIfCancellationRequested();
                    lastResult = await repository.UpsertBatchAsync(batch, ct);

                    if (lastResult.Status != UpdateStatus.Completed)
                        logger.Error("Upsert failed for batch: {Status}", lastResult.Status);

                    task.Increment(batch.Length);
                    ctx.Refresh();
                }
            });

        context.UpsertResult = lastResult;
        writer.MarkupLine(lastResult.Status == UpdateStatus.Completed
            ? "[[green]]Upsert complete.[[/]]"
            : "[[red]]Upsert finished with errors.[[/]]");
    }
}
```

### Step 4 — The Pipeline Orchestrator

This **replaces** the old `EmbedTextPipeline` class. It composes stages and threads the context through them.

```csharp
// Core/PipelineStages/EmbeddingPipelineOrchestrator.cs

namespace Core.PipelineStages;

/// <summary>
/// Orchestrates the full embed-and-upsert pipeline by executing
/// <see cref="IPipelineStage{EmbeddingContext}"/> instances in order.
/// </summary>
/// <remarks>
/// Replaces the monolithic <c>EmbedTextPipeline</c>. Each concern is now a
/// discrete, testable stage that reads/writes only the <see cref="EmbeddingContext"/>
/// fields it owns.
/// </remarks>
public sealed class EmbeddingPipelineOrchestrator(
    IReadOnlyList<IPipelineStage<EmbeddingContext>> stages,
    ILogger logger)
{
    /// <summary>Executes every registered stage in registration order.</summary>
    /// <param name="context">The shared context pre-populated (e.g. SourceDirectory).</param>
    /// <param name="cancellationToken">Token to abort the entire pipeline.</param>
    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        var sw = Stopwatch.StartNew();

        foreach (var stage in stages)
        {
            cancellationToken.ThrowIfCancellationRequested();

            logger.Information("Stage started: {Stage}", stage.Name);
            var stageSw = Stopwatch.StartNew();

            await stage.ExecuteAsync(context, cancellationToken);

            logger.Information("Stage finished: {Stage} ({Elapsed:F1}s)",
                stage.Name, stageSw.Elapsed.TotalSeconds);
        }

        context.TotalElapsed = sw.Elapsed;
        logger.Information("Pipeline complete in {Elapsed:F1}s", context.TotalElapsed.TotalSeconds);
    }
}
```

### What Changed vs. the Original

| Aspect | Before (monolith) | After (stages + context) |
|---|---|---|
| **Testability** | Entire `RunAsync` must be exercised together | Each stage tested in isolation with a pre-built context |
| **Extensibility** | New step = edit `RunAsync` | New step = new `IPipelineStage<EmbeddingContext>` class + DI registration |
| **Reusability** | Tied to one flow | Stages can be composed into different pipelines (e.g., re-parse only) |

```csharp
// Core/Rendering/ITableRenderer.cs

using Spectre.Console;

namespace Core.Rendering;

/// <summary>
/// Base interface for rendering tabular data to the console.
/// </summary>
public interface ITableRenderer
{
    /// <summary>Builds and displays the table.</summary>
    void Render();

    /// <summary>The underlying table being built.</summary>
    ITableBuilder Table { get; }
}

/// <summary>
/// Mutable builder for a Spectre.Console table.
/// </summary>
public sealed class ITableBuilder
{
    private readonly List<string> _columns = new();
    private readonly List<List<string>> _rows = new();
    public bool ShowRowSeparators { get; set; }
    public bool Expand { get; set; }

    public ITableBuilder AddColumns(params string[[]] columns)
    {
        _columns.AddRange(columns);
        return this;
    }

    public ITableBuilder AddRow(params string[[]] cells)
    {
        _rows.Add(cells.ToList());
        return this;
    }

    public IReadOnlyList<string> Columns => _columns;
    public IReadOnlyList<List<string>> Rows => _rows;

    /// <summary>Materialises a Spectre.Console <see cref="Table"/> from accumulated state.</summary>
    public Table ToTable()
    {
        var table = new Table().Expand(Expand).ShowRowSeparators(ShowRowSeparators);
        table.AddColumns(_columns);
        foreach (var row in _rows)
            table.AddRow(row);
        return table;
    }
}
```

```csharp
// Core/Rendering/Decorators/LiveProgressDecorator.cs

namespace Core.Rendering.Decorators;

/// <summary>
/// Decorator that renders the table inside a Spectre.Console Live context,
/// refreshing after every row is added (for progressive / animated display).
/// </summary>
public sealed class LiveProgressDecorator : ITableRenderer
{
    private readonly ITableRenderer _inner;
    private readonly Func<List<string>, int> _rowFactory; // returns count after adding a row

    /// <param name="inner">The renderer to decorate.</param>
    /// <param name="rowFactory">
    ///   A delegate that, given the current row list, appends one row and returns
    ///   the total row count (used as the progress max).
    /// </param>
    public LiveProgressDecorator(ITableRenderer inner, Func<List<string>, int> rowFactory)
    {
        _inner  = inner;
        _rowFactory = rowFactory;
    }

    public ITableBuilder Table => _inner.Table;

    public void Render()
    {
        var table = Table.ToTable();
        int total = _rowFactory(null); // get the count first (or pre-computed)

        AnsiConsole.Live(table).Start(ctx =>
        {
            // The row data is already in Table.Rows from the builder.
            // This decorator is useful when rows are added *during* rendering:
            for (int i = 0; i < Table.Rows.Count; i++)
            {
                // If the table was pre-built, just refresh to show animation
                ctx.Refresh();
            }
        });
    }
}
```

### Step 4 — A Factory / Fluent Builder for Convenience

```csharp
// Core/Rendering/TableRendererFactory.cs

namespace Core.Rendering;

/// <summary>
/// Fluent factory that composes decorators around a base table renderer.
/// </summary>
/// <example>
/// <![[CDATA[[
/// TableRendererFactory.Create()
///     .WithColumns("cnt", "Path")
///     .WithRows(files.Select(f => new[[]] { f }))
///     .StyledBorder(Color.Green, "Source Files")
///     .ShowRowSeparators()
///     .Render();
/// ]]]]>
/// </example>
public sealed class TableRendererFactory
{
    private readonly ITableBuilder _builder = new();
    private ITableRenderer? _renderer;

    public static TableRendererFactory Create() => new();

    public TableRendererFactory WithColumns(params string[[]] columns)
    {
        _builder.AddColumns(columns);
        return this;
    }

    public TableRendererFactory WithRows(IEnumerable<IEnumerable<string>> rows)
    {
        foreach (var row in rows)
            _builder.AddRow([[.. row]]);
        return this;
    }

    public TableRendererFactory ShowRowSeparators()
    {
        _builder.ShowRowSeparators = true;
        return this;
    }

    public TableRendererFactory Expand()
    {
        _builder.Expand = true;
        return this;
    }

    public TableRendererFactory StyledBorder(Color color, string? header = null)
    {
        _renderer = new StyledBorderDecorator(BuildBase(), color, header);
        return this;
    }

    public TableRendererFactory WithLiveProgress()
    {
        _renderer = new LiveProgressDecorator(BuildBase(), _ => _builder.Rows.Count);
        return this;
    }

    public void Render()
    {
        (_renderer ?? BuildBase()).Render();
    }

    private ITableRenderer BuildBase() => new ConsoleTableRenderer(_builder);
}
```
