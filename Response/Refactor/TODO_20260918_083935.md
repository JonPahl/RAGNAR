```csharp
// Ragnar\Stages\LoadQuestionsStage.cs
using Ragnar.Abstractions;
using Ragnar.Core.Model;
using Ragnar.Extensions;
using Serilog;
using Spectre.Console;

namespace Ragnar.Stages;

/// <summary>
/// Loads and filters questions from CSV plugin files into the shared context.
/// </summary>
/// <param name="builder">Fluent builder that discovers and filters question sources.</param>
/// <param name="logger">Serilog logger for diagnostics.</param>
/// <param name="writer">Console output writer for the question table display.</param>
/// <example><![[CDATA[[await stage.ExecuteAsync(ctx, ct);]]]]></example>
public sealed class LoadQuestionsStage(
    IQuestionSourceBuilder builder,
    ILogger logger,
    IOutputWriter writer) : IPipelineStage<EmbeddingContext>
{
    /// <inheritdoc/>
    public string Name => "Loading questions…";

    /// <inheritdoc/>
    public bool ShouldRun => true;

    /// <summary>
    /// Discovers CSV files, applies category filters, and stores sorted
    /// <see cref="Question"/> objects in <see cref="EmbeddingContext.Questions"/>.
    /// </summary>
    /// <param name="context">Shared pipeline context that receives the question list.</param>
    /// <param name="cancellationToken">Token to abort the loading operation.</param>
    /// <example><![[CDATA[[await stage.ExecuteAsync(ctx, ct);]]]]></example>
    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        var pluginDir = Path.Join(AppContext.BaseDirectory, "Plugins");

        // Fluent chain: categories → file config → CSV discovery → filter → build
        builder
            .GetCategories()
            .GetFileConfig();

        await builder
            .GetCsvFilesAsync(pluginDir, cancellationToken)
            .ConfigureAwait(false);

        builder.WithCategoryFilter();
        var sorted = builder.Build();

        // Store in context for downstream stages
        context.Questions = sorted;

        // Render a quick console table (mirrors the old ShowTable helper)
        RenderQuestionTable(sorted, writer);

        logger.Information("Loaded {Count} question(s) from plugin directory '{Dir}'.",
            sorted.Count, pluginDir);
    }

    /// <summary>
    /// Renders a Spectre.Console table of the loaded questions.
    /// </summary>
    private static void RenderQuestionTable(IReadOnlyList<Question> questions, IOutputWriter writer)
    {
        if (questions.Count == 0) return;

        var table = new Table()
            .Expand()
            .AddColumns("#", "Category", "Text")
            .ShowRowSeparators()
            .RoundedBorder()
            .BorderStyle(Styles.Green);

        for (var i = 0; i < questions.Count; i++)
        {
            table.AddRow(
                (i + 1).ToString(),
                questions[[i]].Category.ToString(),
                questions[[i]].Text);
        }

        writer.Write(table);
        writer.WriteLine();
    }
}
```

```csharp
// Ragnar\Stages\QuestionExecutionStage.cs  (refactored)
public sealed class QuestionExecutionStage(
    IRagOrchestrator rag,
    IOptions<RagnarConfig> config,
    IOutputWriter writer,
    IQuestionEmbedding questionEmbedding,
    ILogger logger,
    // IQuestionSourceBuilder is NO LONGER needed here
    ) : IPipelineStage<EmbeddingContext>
{
    public string Name => "Ask Questions Stage";
    public bool ShouldRun => true;

    public async Task ExecuteAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        var questions = context.Questions;   // ← read from context, not from builder
        if (questions.Count == 0)
        {
            logger.Information("No questions to process. Skipping.");
            return;
        }

        for (var i = 0; i < questions.Count; i++)
        {
            writer.MarkupLine($"{i + 1} of {questions.Count}", Styles.Cyan);

            var q = questions[[i]];
            var ctx = await questionEmbedding
                .RetrieveContextAsync(config.Value.ApplicationOptions.VectorStoreName, cancellationToken)
                .ConfigureAwait(false);

            await rag.ExecuteAsync(q, ctx, cancellationToken).ConfigureAwait(false);
        }
    }
}
```
