```csharp
public static class PathUtilities
{
    /// <summary>Expands environment variables and validates directory existence.</summary>
    public static string ExpandAndValidate(string path)
    {
        var expanded = Environment.ExpandEnvironmentVariables(path);
        var fullPath = System.IO.Path.GetFullPath(expanded);
        
        if (!Directory.Exists(fullPath))
            throw new DirectoryNotFoundException($"Directory not found: '{fullPath}'");
            
        return fullPath;
    }

    /// <summary>Constructs and ensures the target response directory exists.</summary>
    public static string BuildResponsePath(string sourceDir, string? category = null)
    {
        var baseDir = string.IsNullOrWhiteSpace(category) ? AppDefaults.UncategorizedCategory : category;
        var responseDir = System.IO.Path.Join(sourceDir, AppDefaults.ResponseDirectoryName);
        return Directory.CreateDirectory(System.IO.Path.Join(responseDir, baseDir)).FullName;
    }

    /// <summary>Ensures a directory exists and returns its full path.</summary>
    public static string EnsureDirectoryExists(string path) => Directory.CreateDirectory(path).FullName;
}
```

```csharp
public interface IDirectoryContentLoader
{
    Task<string> LoadAndWrapAsync(string folderPath, CancellationToken ct);
}

public class DirectoryContentLoader : IDirectoryContentLoader
{
    public async Task<string> LoadAndWrapAsync(string folderPath, CancellationToken ct)
    {
        var files = await Directory.GetFilesAsync(folderPath, "*", new EnumerationOptions { IgnoreInaccessible = true }, ct);
        if (files.Length == 0) return "No items to summarize. Please ignore.";

        var contents = new List<string>(files.Length);
        await Parallel.ForEachAsync(files, new ParallelOptions { CancellationToken = ct }, async (file, token) =>
        {
            using var reader = File.OpenText(file);
            var pooledBuffer = ArrayPool<char>.Shared.Rent(8192);
            try
            {
                var text = new StringBuilder();
                int bytesRead;
                while ((bytesRead = await reader.ReadAsync(pooledBuffer.AsMemory(0, pooledBuffer.Length), token)) > 0)
                    text.Append(pooledBuffer.AsSpan(0, bytesRead));
                
                contents.Add($"---\n[{AppDefaults.FileMarkerStart}]{Path.GetFileName(file)}[{AppDefaults.FileMarkerEnd}]\n{text}\n");
            }
            finally { ArrayPool<char>.Shared.Return(pooledBuffer); }
        });

        return $"{AppDefaults.CodeBlockStart}{string.Concat(contents)}{AppDefaults.CodeBlockEnd}";
    }
}
```
