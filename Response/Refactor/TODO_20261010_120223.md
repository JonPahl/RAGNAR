```csharp
// SummarizeSaveResponse.cs
using System.Globalization;
using System.Text;

/// <summary>
/// Writes a summarized RAG response to disk as a Markdown file.
/// </summary>
/// <example><![[CDATA[[var path = await responseWriter.WriteResponseAsync(record, ct);]]]]></example>
public sealed class SummarizeSaveResponse : IResponseWriter
{
    private readonly IOutputWriter _writer;

    // Cached so we only create the directory once per process lifetime.
    private string? _responseDir;
    private readonly object _dirLock = new();

    public SummarizeSaveResponse(IOutputWriter writer)
    {
        _writer = writer;
    }

    /// <summary>
    /// Persists the full response record to a Markdown file under a Response subfolder.
    /// </summary>
    /// <param name="details">The response record containing question, answer, elapsed time, and metadata.</param>
    /// <param name="cancellationToken">Token to cancel the write operation.</param>
    /// <returns>The full file path of the written summary.</returns>
    /// <example><![[CDATA[[await writer.WriteResponseAsync(record, ct);]]]]></example>
    public async Task<string> WriteResponseAsync(ResponseRecord details, CancellationToken cancellationToken)
    {
        Guard.Against.Null(details, nameof(details));
        Guard.Against.Null(details.Question, nameof(details.Question));

        // 1. Determine the output directory (co-located with the source file when possible).
        var sourceDir = Path.GetDirectoryName(details.Question.Filename)
                        ?? Directory.GetCurrentDirectory();
        _responseDir ??= Directory
            .CreateDirectory(Path.Join(sourceDir, AppDefaults.ResponseDirectoryName))
            .FullName;

        // 2. Build a deterministic file name.
        var safeStem = Path.GetFileNameWithoutExtension(details.Question.Filename)
                      .Replace(' ', '_')
                      .Replace('\\', '_')
                      .Replace('/', '_');
        var fileName = $"{safeStem}_summary.md";
        var summaryPath = Path.Join(_responseDir, fileName);

        // 3. Compose the Markdown document using every available field.
        var sb = new StringBuilder();
        sb.AppendLine("# RAG Response Summary");
        sb.AppendLine();
        sb.AppendLine($"| Field        | Value |");
        sb.AppendLine($"|--------------|-------|");
        sb.AppendLine($"| **Question** | {details.Question.Text} |");
        sb.AppendLine($"| **Category** | {details.Question.Category} |");
        sb.AppendLine($"| **File**     | {details.Question.Filename} |");
        sb.AppendLine($"| **Elapsed**  | {details.ElapsedTime} |");
        sb.AppendLine($"| **Generated**| {DateTime.UtcNow:O} |");
        sb.AppendLine();

        sb.AppendLine("## Answer");
        sb.AppendLine();
        sb.AppendLine(details.Answer ?? string.Empty);

        // Optionally include the original prompt if the flag is set
        // (RagOrchestrator appends it before calling this method when enabled).
        if (!string.IsNullOrWhiteSpace(details.OriginalPrompt))
        {
            sb.AppendLine();
            sb.AppendLine($"***");
            sb.AppendLine($"{AppDefaults.OriginalPromptLabel}");
            sb.AppendLine(details.OriginalPrompt);
            sb.AppendLine($"{AppDefaults.OriginalPromptLabelEnd}");
            sb.AppendLine($"***");
        }

        // 4. Write to disk.
        await File.WriteAllTextAsync(summaryPath, sb.ToString(), cancellationToken)
            .ConfigureAwait(false);

        return summaryPath;
    }
}
```

### 2. Create a `ToastNotifier` service

```csharp
// ToastNotifier.cs
using Windows.AppNotifications;

/// <summary>
/// Sends Windows toast notifications (Windows App SDK) for pipeline events.
/// </summary>
public sealed class ToastNotifier
{
    /// <summary>
    /// Shows a toast indicating that a single question has been processed.
    /// </summary>
    /// <param name="questionIndex">1-based index of the question.</param>
    /// <param name="total">Total number of questions in the batch.</param>
    /// <param name="questionText">Truncated question text for display.</param>
    public void ShowQuestionComplete(int questionIndex, int total, string questionText)
    {
        var truncated = questionText.Length > 80 ? questionText[[..77]] + "…" : questionText;

        var notification = new AppNotificationBuilder()
            .AddArgument("action", "viewItem")
            .AddArgument("index", questionIndex.ToString())
            .AddText($"Question {questionIndex}/{total} complete")
            .AddText(truncated)
            .AddButton(new AppNotificationButton("Acknowledge")
                .AddArgument("action", "acknowledge")
                .AddArgument("index", questionIndex.ToString()))
            .BuildNotification();

        AppNotificationManager.Default.Show(notification);
    }

    /// <summary>
    /// Shows a toast indicating the entire question batch has finished.
    /// </summary>
    /// <param name="total">Number of questions processed.</param>
    /// <param name="elapsed">Total elapsed time.</param>
    public void ShowBatchComplete(int total, string elapsed)
    {
        var notification = new AppNotificationBuilder()
            .AddArgument("action", "viewSummary")
            .AddText("Ragnar – All questions complete")
            .AddText($"{total} question(s) answered in {elapsed}")
            .AddButton(new AppNotificationButton("Open Results")
                .AddArgument("action", "openResults"))
            .BuildNotification();

        AppNotificationManager.Default.Show(notification);
    }
}
```

### 4. Wire it into `QuestionExecutionStage`

```csharp
// QuestionExecutionStage.cs  (updated)
public class QuestionExecutionStage(
    IRagOrchestrator    ragOrchestrator,
    IOptions<RagnarConfig> ragnarConfig,
    IOutputWriter       writer,
    IContextRetriever   questionEmbedding,
    Serilog.ILogger     logger,
    ToastNotifier       toastNotifier)          // ← new dependency
    : IPipelineStage<QuestionPipelineContext>
{
    public string Name => "Ask Questions Stage";

    public bool ShouldRun(QuestionPipelineContext context) => context.Questions.Any();

    private readonly string _collectionName =
        ragnarConfig.Value.ApplicationOptions.VectorStoreName;

    public async Task ExecuteAsync(QuestionPipelineContext context, CancellationToken cancellationToken)
    {
        var questions     = context.Questions!;
        var processedCount = 0;
        var total          = questions.Count;
        var batchSw        = Stopwatch.StartNew();

        foreach (var question in questions)
        {
            cancellationToken.ThrowIfCancellationRequested();

            var contextText = await questionEmbedding
                .RetrieveContextAsync(_collectionName, cancellationToken, question.Filter)
                .ConfigureAwait(false);

            writer.WriteRule();
            writer.WriteLine();
            writer.MarkupLine($"[[blue]]Question: {Environment.NewLine}{Markup.Escape(question.Text)} [[/]]");
            writer.WriteLine();
            writer.Write(new Rule());

            logger.Information(
                "Processing question [[{QuestionId}]] from user [[{UserId}]] with {ContextLength} chars",
                question.Text, Environment.UserName, contextText.Length.ToString("N0"));

            var qSw = Stopwatch.StartNew();
            await ragOrchestrator.ExecuteAsync(question, contextText, cancellationToken)
                .ConfigureAwait(false);
            qSw.Stop();

            processedCount++;

            // ── TODO (now resolved): fire a toast per question ──────────────
            toastNotifier.ShowQuestionComplete(processedCount, total, question.Text);

            writer.MarkupLine($"{processedCount} of {total}", Styles.Cyan);
        }

        batchSw.Stop();

        // ── Final "batch complete" toast ────────────────────────────────────
        toastNotifier.ShowBatchComplete(total, batchSw.FormatElapsedTime());
    }
}
```
