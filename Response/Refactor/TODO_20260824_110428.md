        // Register parsers dynamically via DI or reflection
        var extensions = config.Value.FileLoadOptions.AllowedFileExtensions ?? Array.Empty<string>();
        
        foreach (var ext in extensions)
        {
            _parserMap[ext] = GetParserType(ext);
        }
    }

    public async Task<ParseResult> ParseAsync(string filePath, CancellationToken Ct)
    {
        var extension = Path.GetExtension(filePath).TrimStart('.');
        if (!_parserMap.TryGetValue(extension, out var parserType))
            throw new NotSupportedException($"Unsupported file type: .{extension}");

        var parser = (IFileParser)Activator.CreateInstance(parserType)!;
        return await parser.ParseAsync(filePath, Ct);
    }

    private static Type GetParserType(string extension) => extension switch

        "json" => typeof(JsonConfigParser),
        "editorconfig" => typeof(EditorConfigParser),
