// Ragnar.Core/Io/CsvLoader.cs
public interface ICsvLoader<T> where T : class
{
    Task<IEnumerable<T>> LoadAsync(string filePath, CancellationToken ct);
}

public class CsvLoader<T> : ICsvLoader<T> where T : class
{
    private readonly CsvConfiguration _config = new(CultureInfo.InvariantCulture)
    {
        HasHeaderRecord = true, TrimOptions = TrimOptions.Trim
    };

    public async Task<IEnumerable<T>> LoadAsync(string filePath, CancellationToken ct)
    {
        using var reader = new StreamReader(filePath);
        using var csv = new CsvReader(reader, _config);
        return await csv.GetRecordsAsync<T>().ToListAsync(ct);
    }
}

// Ragnar.Core/Io/DirectoryFileLoader.cs
public interface IDirectoryFileLoader
{
    Task<string> LoadAllContentsAsync(string folderPath, CancellationToken ct);
}

public class DirectoryFileLoader : IDirectoryFileLoader
{
    private const string FileMarker = "### FILE: {0}\n";
    private const string CodeBlockStart = "```csharp\n";
    private const string CodeBlockEnd = "\n```\n";

    public async Task<string> LoadAllContentsAsync(string folderPath, CancellationToken ct)
    {
        var files = Directory.GetFiles(folderPath);
        if (files.Length == 0) return "No items to summarize.";

        var contents = await Task.WhenAll(files.Select(async file =>
        {
            using var reader = File.OpenText(file);
            var text = await reader.ReadToEndAsync(ct);
            return string.Format(FileMarker, Path.GetFileName(file)) + text;
        }));
        return $"{CodeBlockStart}{string.Concat(contents)}{CodeBlockEnd}";
    }
}

// Ragnar.Core/Writers/ResponseWriter.cs
public sealed class ResponseWriter : IResponseWriter
{
    private const string MarkdownHeaderTemplate = "{0}\n> **Date Generated**: {1:G}\n> ## Question:\n> {2}\n> **Method Call Duration**: {3}";
    private const string MarkdownResponseTemplate = "\n## Response:\n{4}";

    public async Task<string> WriteResponseAsync(SaveDetails details, CancellationToken ct)
    {
        var sourceDir = Config.Value.ApplicationOptions.SourceDirectory.ExpandDirectory();
        var categoryDir = Path.Join(sourceDir, "Response", details.Question.Category.ToString());
        Directory.CreateDirectory(categoryDir);

        var fileName = $"{details.Question.Filename}_{DateTime.Now:yyyyMMdd_HHmmss}.md";
        var filePath = Path.Join(categoryDir, fileName);

        var content = string.Format(MarkdownHeaderTemplate,
            details.Question.MarkdownHeader, DateTime.Now, details.Question.Text, details.Duration) +
            string.Format(MarkdownResponseTemplate, details.Response);

        await File.WriteAllTextAsync(filePath, content, ct);
        return filePath;
    }
}
