/// EF Core context for the SQLite-backed question store.

/// <example><![[CDATA[[await using var db = new QuestionDbContext(options);]]]]></example>
public class QuestionDbContext(DbContextOptions<QuestionDbContext> options)
    : DbContext(options)
{
    /// <summary>Set of all questions.</summary>
    /// <example><![[CDATA[[await db.Questions.ToListAsync(ct);]]]]></example>
    public DbSet<Question> Questions => Set<Question>();

    /// <inheritdoc/>
    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        modelBuilder.Entity<Question>(entity =>
        {
            entity.HasKey(q => q.Id);
            entity.Property(q => q.Text).IsRequired().HasMaxLength(8192);
            entity.Property(q => q.Category).HasConversion<int>();
            entity.Property(q => q.Tags).HasMaxLength(512);
            entity.HasIndex(q => q.Category);
            entity.HasIndex(q => q.IsEnabled);
        });
    }
}

using Microsoft.EntityFrameworkCore;

/// <summary>
/// EF Core / SQLite implementation of <see cref="IQuestionRepository"/>.
/// </summary>
/// <param name="factory">Context factory that builds SQLite-backed contexts.</param>
/// <param name="logger">Diagnostic logger.</param>
/// <example><![[CDATA[[var repo = new SqliteQuestionRepository(factory, log);]]]]></example>
public sealed class SqliteQuestionRepository(
    QuestionDbContextFactory factory,
    Microsoft.Extensions.Logging.ILogger logger) : IQuestionRepository
{
    /// <inheritdoc/>
    public async Task<IReadOnlyList<Question>> GetAllAsync(CancellationToken cancellationToken = default)
    {
        await using var db = factory.CreateContext();
        var list = await db.Questions
            .OrderBy(q => q.Category)
            .ThenBy(q => q.SortOrder)
            .ThenBy(q => q.Id)
            .ToListAsync(cancellationToken)
            .ConfigureAwait(false);
        return list.AsReadOnly();
    }

    /// <inheritdoc/>
    public async Task<IReadOnlyList<Question>> GetEnabledAsync(CancellationToken cancellationToken = default)
    {
        await using var db = factory.CreateContext();
        var list = await db.Questions
            .Where(q => q.IsEnabled)
            .OrderBy(q => q.Category)
            .ThenBy(q => q.SortOrder)
            .ThenBy(q => q.Id)
            .ToListAsync(cancellationToken)
            .ConfigureAwait(false);
        return list.AsReadOnly();
    }

    /// <inheritdoc/>
    public async Task<IReadOnlyList<Question>> GetByCategoryAsync(
        IEnumerable<QuestionCategory>? categories,
        CancellationToken cancellationToken = default)
    {
        if (categories is null or { Count: 0 })
            return await GetAllAsync(cancellationToken).ConfigureAwait(false);

        var set = categories.ToHashSet();
        await using var db = factory.CreateContext();
        var list = await db.Questions
            .Where(q => set.Contains(q.Category))
            .OrderBy(q => q.SortOrder)
            .ThenBy(q => q.Id)
            .ToListAsync(cancellationToken)
            .ConfigureAwait(false);
        return list.AsReadOnly();
    }

    /// <inheritdoc/>
    public async Task<Question?> GetByIdAsync(Guid id, CancellationToken cancellationToken = default)
    {
        await using var db = factory.CreateContext();
        return await db.Questions.FirstOrDefaultAsync(q => q.Id == id, cancellationToken).ConfigureAwait(false);
    }

    /// <inheritdoc/>
    public async Task<Question> AddAsync(Question question, CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(question);
        await using var db = factory.CreateContext();
        db.Questions.Add(question);
        await db.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
        logger.LogDebug("Inserted question {Id}", question.Id);
        return question;
    }

    /// <inheritdoc/>
    public async Task<Question> UpsertAsync(Question question, CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(question);
        await using var db = factory.CreateContext();
        var existing = await db.Questions.FirstOrDefaultAsync(q => q.Id == question.Id, cancellationToken).ConfigureAwait(false);
        if (existing is null)
        {
            db.Questions.Add(question);
        }
        else
        {
            existing.Text = question.Text;
            existing.Category = question.Category;
            existing.IsEnabled = question.IsEnabled;
            existing.Tags = question.Tags;
            existing.SortOrder = question.SortOrder;
            existing.ModifiedUtc = DateTime.UtcNow;
        }
        await db.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
        logger.LogDebug("Upserted question {Id}", question.Id);
        return existing ?? question;
    }

    /// <inheritdoc/>
    public async Task<bool> DeleteAsync(Guid id, CancellationToken cancellationToken = default)
    {
        await using var db = factory.CreateContext();
        var existing = await db.Questions.FirstOrDefaultAsync(q => q.Id == id, cancellationToken).ConfigureAwait(false);
        if (existing is null) return false;
        db.Questions.Remove(existing);
        await db.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
        logger.LogDebug("Deleted question {Id}", id);
        return true;
    }

    /// <inheritdoc/>
    public async Task BulkUpsertAsync(IEnumerable<Question> questions, CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(questions);
        await using var db = factory.CreateContext();
        await db.Database.OpenConnectionAsync(cancellationToken).ConfigureAwait(false);
        await using var tx = await db.Database.BeginTransactionAsync(cancellationToken).ConfigureAwait(false);
        try
        {
            var list = questions.ToList();
            var existingIds = await db.Questions
                .Where(q => list.Select(l => l.Id).Contains(q.Id))
                .ToDictionaryAsync(q => q.Id, cancellationToken)
                .ConfigureAwait(false);

            foreach (var q in list)
            {
                if (existingIds.TryGetValue(q.Id, out var existing))
                {
                    existing.Text = q.Text;
                    existing.Category = q.Category;
                    existing.IsEnabled = q.IsEnabled;
                    existing.Tags = q.Tags;
                    existing.SortOrder = q.SortOrder;
                    existing.ModifiedUtc = DateTime.UtcNow;
                }
                else
                {
                    db.Questions.Add(q);
                }
            }

            await db.SaveChangesAsync(cancellationToken).ConfigureAwait(false);
            await tx.CommitAsync(cancellationToken).ConfigureAwait(false);
            logger.LogInformation("Bulk upsert completed: {Count} questions", list.Count);
        }
        catch
        {
            await tx.RollbackAsync(cancellationToken).ConfigureAwait(false);
            throw;
        }
    }

    /// <inheritdoc/>
    public async Task<int> CountAsync(CancellationToken cancellationToken = default)
    {
        await using var db = factory.CreateContext();
        return await db.Questions.CountAsync(cancellationToken).ConfigureAwait(false);
    }

    /// <inheritdoc/>
    public async Task SeedAsync(CancellationToken cancellationToken = default)
    {
        var count = await CountAsync(cancellationToken).ConfigureAwait(false);
        if (count > 0)
        {
            logger.LogDebug("Seed skipped: {Count} questions already present", count);
            return;
        }

        logger.LogInformation("Seeding default questions into SQLite…");
        await BulkUpsertAsync(DomainDefaults.SeedQuestions, cancellationToken).ConfigureAwait(false);
    }
}

            var factory = sp.GetRequiredService<QuestionDbContextFactory>();
            var logger = sp.GetRequiredService<Microsoft.Extensions.Logging.ILogger<SqliteQuestionRepository>>();
            return new SqliteQuestionRepository(factory, logger);

/// <summary>
/// Pipeline stage that loads questions from the SQLite database
/// and populates the <see cref="QuestionPipelineContext"/>.
/// Replaces the former CSV file reader.
/// </summary>
/// <param name="logger">Diagnostic logger.</param>
/// <param name="repo">SQLite-backed question repository.</param>
/// <example><![[CDATA[[runner.AddStage(new QuestionLoadStage(log, repo));]]]]></example>
public sealed class QuestionLoadStage(
    Microsoft.Extensions.Logging.ILogger logger,
    IQuestionRepository repo) : IPipelineStage<QuestionPipelineContext>
{
    /// <inheritdoc/>
    public string Name => "Load Questions";

    /// <inheritdoc/>
    public Polly.AsyncRetryPolicy? RetryPolicy => null;

    /// <inheritdoc/>
    public bool ShouldRun(QuestionPipelineContext context) => context.Questions is null;

    /// <summary>
    /// Queries the SQLite database for enabled questions and stores them in the context.
    /// </summary>
    /// <param name="context">Shared pipeline context to populate.</param>
    /// <param name="cancellationToken">Token to abort.</param>
    /// <returns>Completed task once questions are loaded.</returns>
    /// <example><![[CDATA[[await stage.ExecuteAsync(ctx, ct);]]]]></example>
    public async Task ExecuteAsync(QuestionPipelineContext context, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();

        logger.LogInformation("Loading questions from SQLite…");

        var questions = await repo.LoadActiveAsync(cancellationToken).ConfigureAwait(false);

        if (questions.Count == 0)
        {
            logger.LogWarning("No enabled questions found in the database.");
        }
        else
        {
            logger.LogInformation("Loaded {Count} enabled questions from SQLite", questions.Count);
        }

        context.SetQuestions(questions);
    }
}

builder.Services.AddOptions<RagnarConfig>()
    .Bind(builder.Configuration.GetSection("Ragnar"))
    .ValidateDataAnnotations()
    .ValidateOnStart();

// ── Question store (SQLite) ─────────────────────────────────────────
var config = builder.Services.BuildServiceProvider()
              .GetRequiredService<IOptions<RagnarConfig>>();

builder.Services.QuestionDbSetup(config);

// ── Ensure DB exists & seed ─────────────────────────────────────────
var sp = builder.Services.BuildServiceProvider();
var initializer = sp.GetRequiredService<QuestionDbInitializer>();
await initializer.RunAsync();

// ── Embedding pipeline ──────────────────────────────────────────────
builder.Services.EmbeddingSetup();

// ── Pipeline runner ─────────────────────────────────────────────────
builder.Services.AddSingleton<IPipelineStage<QuestionPipelineContext>, QuestionLoadStage>();

var host = builder.Build();
await host.RunAsync();

## Summary of Changes

| Concern | Before (CSV) | After (SQLite) |
|---------|-------------|----------------|
| **Read** | `CsvHelper` parses `questions.csv` | `SqliteQuestionRepository.GetAllAsync()` |
| **Write / Seed** | Append CSV rows at startup | `BulkUpsertAsync` inside a transaction |
| **Filter** | In-memory `Where` after full load | `EF Core` LINQ → SQL `WHERE` |
| **Concurrency** | File locks | SQLite shared-cache + EF transactions |
| **Migrations** | None (schemaless) | `EnsureCreated` / future EF migrations |
| **DI** | CSV path in config | `QuestionDbPath` in `ApplicationOptions` |
| **Extensions** | Same surface (`ActiveOnly`, `WithCategory`) | Unchanged + async repo helpers |

All existing call-sites that consume `IReadOnlyList<Question>` (e.g. `QuestionPipelineContext.SetQuestions`, `QuestionsExtensions.ActiveOnly`, `QuestionFilterSelector`) continue to work without modification.
