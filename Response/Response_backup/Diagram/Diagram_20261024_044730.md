    %% Stage 1 – Load Questions
    Runner->>Loader: ExecuteAsync(ctx, ct)
    Loader->>Loader: Load & filter questions
    Loader-->>Runner: ctx.Questions populated

    %% Stage 2 – Per-Question Execution
    loop For each Question in ctx.Questions
        Runner->>Selector: FindFilter(question.Category, threshold)
        Selector->>XmlStrat: CreateFilter(sizeThreshold)
        XmlStrat->>QBuilder: AddRange("CommentLength", t).AddExceptKeywords("Category", "Test", ...)
        QBuilder-->>XmlStrat: Filter
        XmlStrat-->>Selector: Filter
        Selector-->>Runner: Filter

        Runner->>Orchestrator: ExecuteAsync(question, contextText, ct)

        Orchestrator->>Cfg: Value.ApplicationOptions.IncludeOriginalPrompt
        Orchestrator->>Factory: ResolveClient(OllamaServiceType.Ollama)
        Factory-->>Orchestrator: IOllamaApiClient

        Orchestrator->>Orchestrator: Build GenerateRequest (Model, Prompt, System)
        Orchestrator->>Prompt: System
        Prompt-->>Orchestrator: systemPrompt string

        Orchestrator->>OllamaGen: GenerateResponse(request, ct)
        OllamaGen->>OllamaGen: HTTP POST /api/generate
        OllamaGen-->>Orchestrator: response string

        alt IncludeOriginalPrompt == true
            Orchestrator->>Orchestrator: response += ShowPrompt() + system
        end

        Orchestrator->>Writer: WriteResponseAsync(new ResponseRecord(q, response, elapsed), ct)
        Writer->>Writer: Format via IContentFormatter
        Writer->>Writer: File.WriteAllText(path, content)
        Writer-->>Orchestrator: savedPath
        Orchestrator->>CLI: writer.MarkupLine("[red underline]path[/]")

        Orchestrator-->>Runner: Task complete
    end

    Runner-->>CLI: Enriched context (Responses, Timings)
