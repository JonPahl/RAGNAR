# Test Folder Reorganization Script

Here's a PowerShell script that automatically reorganizes all test files into a logical folder structure based on the feature area they cover:

```powershell
<#
.SYNOPSIS
    Reorganizes test files into a feature-based folder layout.

.DESCRIPTION
    Scans the test project for *.cs test files, determines the target
    subfolder based on a class-name-to-folder mapping, creates the
    directories, moves the files, and updates the namespace declaration.

.EXAMPLE
    .\Reorganize-Tests.ps1 -TestProjectPath ".\Ragnar.Tests"
    .\Reorganize-Tests.ps1 -TestProjectPath ".\Ragnar.Tests" -DryRun
#>

[[CmdletBinding()]]
param(
    [[Parameter(Mandatory = $true)]]
    [[string]] $TestProjectPath,

    [[switch]] $DryRun
)

# ──────────────────────────────────────────────
# 1.  Mapping: test class name → target folder
# ──────────────────────────────────────────────
$folderMap = [[ordered]]@{
    # ── Pipeline orchestration ──
    'PipelineRunnerTests'        = 'Pipeline'
    'StageTimingMiddlewareTests' = 'Pipeline'
    'StageMetricTests'           = 'Pipeline'
    'ShowFileStageTests'         = 'Pipeline'
    'QuestionShowStageTests'     = 'Pipeline'
    'BrandingStageTests'         = 'Pipeline'

    # ── Vector store / Qdrant ──
    'VectorStoreRepositoryTests' = 'VectorStore'
    'FilterBuilderTests'         = 'VectorStore'
    'CollectionInfoTests'        = 'VectorStore'
    'QdrantClientTests'          = 'VectorStore'

    # ── Embeddings ──
    'EmbeddingSetupTests'        = 'Embeddings'
    'OllamaEmbeddingTests'       = 'Embeddings'
    'EmbeddingOptionsTests'      = 'Embeddings'

    # ── File handling / discovery ──
    'FileLoadOptionsValidatorTests'   = 'FileHandling'
    'FileDiscoveryServiceTests'       = 'FileHandling'
    'CodeParserFactoryTests'          = 'FileHandling'
    'IFileEligibilityValidatorTests'  = 'FileHandling'
    'FileSystemEntryExtensionsTests'  = 'FileHandling'

    # ── Validation / options ──
    'GenerationOptionsTests'     = 'Validation'
    'FileLoadOptionsTests'       = 'Validation'
    'EmbeddingOptionsTests'      = 'Validation'

    # ── Infrastructure / misc ──
    'AssemblyExtensionsTests'    = 'Infrastructure'
    'OutputPathHelperTests'      = 'Infrastructure'
    'NumberedRowDecoratorTests'  = 'Infrastructure'
    'EnumConverterTests'         = 'Infrastructure'
}

# Fallback: any test class not in the map goes here
$defaultFolder = 'Uncategorized'

# ──────────────────────────────────────────────
# 2.  Resolve the test project root
# ──────────────────────────────────────────────
$testRoot = (Resolve-Path $TestProjectPath).Path

if (-not (Test-Path (Join-Path $testRoot '*.csproj'))) {
    Write-Warning "No .csproj found in $testRoot – proceeding anyway."
}

# ──────────────────────────────────────────────
# 3.  Discover all test files
# ──────────────────────────────────────────────
$allTestFiles = Get-ChildItem -Path $testRoot -Filter '*Tests.cs' -Recurse |
    Where-Object { $_.Directory.FullName -eq $testRoot }   # top-level only (first pass)

# Also pick up files already in subfolders (idempotent re-run)
$subTestFiles = Get-ChildItem -Path $testRoot -Filter '*Tests.cs' -Recurse |
    Where-Object { $_.Directory.FullName -ne $testRoot }

$testFiles = @($allTestFiles) + @($subTestFiles)

Write-Host "`n═══ Found $($testFiles.Count) test file(s) ═══" -ForegroundColor Cyan

# ──────────────────────────────────────────────
# 4.  Determine target folder for each file
# ──────────────────────────────────────────────
$namespace = $null
$nsMatch  = $null

foreach ($file in $testFiles) {
    $content = Get-Content $file.FullName -Raw

    # Extract the test class name (heuristic: first "public class <Name>Tests")
    $classMatch = [[regex]]::Match($content, 'public\s+(?:sealed\s+)?(?:class|record)\s+(\w+Tests)')
    $className  = if ($classMatch.Success) { $classMatch.Groups[[1]].Value } else { $null }

    if (-not $className) {
        # Fallback: derive from file name
        $className = [[IO.Path]]::GetFileNameWithoutExtension($file.Name)
    }

    # Look up target folder
    $targetFolder = $folderMap[[$className]]
    if (-not $targetFolder) {
        # Try partial / fuzzy match (e.g. "FooTests" → key starts with "Foo")
        $targetFolder = ($folderMap.Keys | Where-Object { $className -like "$_*" } | Select-Object -First 1)
    }
    if (-not $targetFolder) {
        $targetFolder = $defaultFolder
    }

    $targetDir = Join-Path $testRoot $targetFolder
    $targetFile = Join-Path $targetDir "$($file.BaseName).cs"

    # Skip if already in the right place
    if ($file.FullName -eq $targetFile) {
        Write-Host "  ✓ $($file.Name)  →  already in $($targetFolder)/" -ForegroundColor DarkGray
        continue
    }

    # ── Dry-run mode ──
    if ($DryRun) {
        Write-Host "  [[DRY]] $($file.Name)  →  $($targetFolder)/$($file.Name)" -ForegroundColor Yellow
        continue
    }

    # ── Execute ──
    # Create target directory
    if (-not (Test-Path $targetDir)) {
        New-Item -Path $targetDir -ItemType Directory -Force | Out-Null
    }

    # Update namespace before moving
    $nsMatch2 = [[regex]]::Match($content, 'namespace\s+([[\w.]]+)')
    if ($nsMatch2.Success) {
        $rootNs = $nsMatch2.Groups[[1]].Value
        $currentFolder = $nsMatch2.Groups[[1]].Value -replace '^.*\.', ''
        # Build new namespace: <RootNs>.<TargetFolder>
        $newNs = "$rootNs.$targetFolder"
        $updatedContent = $content -replace [[regex]]::Escape($nsMatch2.Value), "namespace $newNs"
        Set-Content -Path $file.FullName -Value $updatedContent -NoNewline
    }

    # Move the file
    Move-Item -Path $file.FullName -Destination $targetFile -Force
    Write-Host "  ✓ $($file.Name)  →  $($targetFolder)/" -ForegroundColor Green
}

# ──────────────────────────────────────────────
# 5.  (Optional) clean up empty sub-folders
# ──────────────────────────────────────────────
if (-not $DryRun) {
    Get-ChildItem -Path $testRoot -Directory | ForEach-Object {
        $files = Get-ChildItem -Path $_.FullName -Filter '*.cs' -Recurse
        if (-not $files) {
            Remove-Item -Path $_.FullName -Recurse -Force
            Write-Host "  ✗ Removed empty folder: $($_.Name)/" -ForegroundColor DarkGray
        }
    }
}

# ──────────────────────────────────────────────
# 6.  Summary
# ──────────────────────────────────────────────
Write-Host "`n═══ Resulting layout ═══" -ForegroundColor Cyan
Get-ChildItem -Path $testRoot -Filter '*Tests.cs' -Recurse |
    Sort-Object { $_.FullName.Substring($testRoot.Length) } |
    ForEach-Object {
        $rel = $_.FullName.Substring($testRoot.Length + 1)
        Write-Host "  $rel" -ForegroundColor White
    }

Write-Host "`nDone." -ForegroundColor Green
```

## Key design decisions

| Concern | Approach |
|---|---|
| **Idempotency** | Re-running the script is safe – files already in the correct folder are skipped. |
| **Namespace sync** | The script rewrites the `namespace` declaration to include the new sub-folder (e.g. `Ragnar.Tests.Pipeline`). |
| **Fuzzy matching** | If a test class name isn't in the map, a prefix match is attempted before falling back to `Uncategorized/`. |
| **Extensibility** | Add new entries to `$folderMap` as your test suite grows; any unmapped class lands in `Uncategorized/` so nothing is lost. |
| **Dry-run** | `-DryRun` prints the planned moves without touching disk, so you can review before committing. |

After running the script, build the solution (`dotnet build`) to confirm all namespaces resolve correctly. If you use a solution-level `.gitignore` or VS solution folder grouping, you may want to add the new sub-folders to the `.csproj` `<ItemGroup>` (though SDK-style projects auto-glob `**/*.cs`, so usually no change is needed).
