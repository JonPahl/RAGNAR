    public async Task ExecuteAsync_CancelledToken_ShouldThrowOperationCanceled()
    {
        using var cts = new CancellationTokenSource();
        await cts.CancelAsync();

        var stage = new Mock<IPipelineStage<QuestionPipelineContext>>();
        stage.Setup(s => s.Name).Returns("CancelStage");
        stage.Setup(s => s.ShouldRun(It.IsAny<QuestionPipelineContext>())).Returns(true);
        stage.Setup(s => s.ExecuteAsync(It.IsAny<QuestionPipelineContext>(), It.IsAny<CancellationToken>()))

        _sut.AddStage(stage.Object);

        await Assert.ThrowsAsync<OperationCanceledException>(

    }

        var @new = new Mock<IPipelineStage<QuestionPipelineContext>>();

        var names = _sut.StageNames;

        Assert.Equal("New", names[[0]]);
    }

    {
        var mock = new Mock<IPipelineStage<QuestionPipelineContext>>();

        mock.Setup(s => s.ShouldRun(It.IsAny<QuestionPipelineContext>())).Returns(true);
        mock.Setup(s => s.ExecuteAsync(It.IsAny<QuestionPipelineContext>(), It.IsAny<CancellationToken>()))
            .Callback(execute)
            .Returns(Task.CompletedTask);
        return mock.Object;
    }
