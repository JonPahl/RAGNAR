    class IFilterStrategy {

    class Condition {
        +FieldCondition Field

        +string Key

        +Match Match
    }

    class Range {
        +int Lt
        +int LtEq
        +int Gt
        +int GtEq
    }

    class Match {
        +RepeatedStrings ExceptKeywords
    }

    class RepeatedStrings {
        +List~string~ Strings
    }

    class XmlCommentFilterStrategy {
        +SupportedCategory QuestionCategory.XML
        +CreateFilter(threshold) Filter
        note "Excludes:\n- Comments shorter than threshold\n- Test-related keywords\n  (Tests, Test, tests, test, Testing)"
    }

    IFilterStrategy <|-- XmlCommentFilterStrategy
    Filter *-- Condition
    Condition *-- FieldCondition
    FieldCondition o-- Range
    FieldCondition o-- Match
    Match o-- RepeatedStrings
    RepeatedStrings o-- Strings
