public sealed class CSharpFileParser(Serilog.ILogger logger) : BaseFileParser(logger)
{
    public async Task<IEnumerable<CodeDocument>> ParseAsync(string filePath, CancellationToken ct)
    {
        var content = await ReadFileAsync(filePath, ct);
        var parser  = CSharpSyntaxTree.ParseText(content).GetRoot();

        return parser.DescendantNodes()
                     .OfType<MethodDeclarationSyntax>()
                     .Concat(parser.DescendantNodes().OfType<ClassDeclarationSyntax>())
                     .Concat(parser.DescendantNodes().OfType<InterfaceDeclarationSyntax>())
                     .Select(node => new CodeDocument
                     {
                         ElementName = node.Identifier.ToString(),
                         Code        = node.ToString(),
                         FilePath    = filePath,
                         Category    = InferCategory(node)
                     });
    }

    private static CodeDocument.Category InferCategory(SyntaxNode node)
        => node is ClassDeclarationSyntax { BaseList.Types.Count > 0 }
             && node.BaseList!.Types.Any(t => t.Type.ToString().Contains("Tests", StringComparison.OrdinalIgnoreCase))
            ? CodeDocument.Category.Test
            : CodeDocument.Category.Source;
}

        return this;
    }

    public IQuestionSourceAggregator GetFileConfig() => this; // no-op placeholder

    public async Task<IQuestionSourceAggregator> GetCsvFilesAsync(
        string directory, CancellationToken ct)
    {
        foreach (var provider in providers)
        {
            var csvFiles = Directory.GetFiles(directory, "*.csv");
            foreach (var file in csvFiles)
            {
                var loaded = await provider.LoadQuestionsAsync(file, ct);
                _questions.AddRange(loaded);
            }
        }
        return this;
    }

    public IQuestionSourceAggregator WithCategoryFilter()
    {
        if (_filter.HasValue)
            _questions.RemoveAll(q => q.Category != _filter);
        return this;
    }

    public IReadOnlyList<Core.Model.Question> Build()
        => _questions
           .Where(q => q.IsActive)
           .OrderBy(q => q.Category)
           .ThenBy(q => q.Text)
           .ToList();
}

public sealed class FileParseFactory(
    IEnumerable<IFileParser> parsers,
    IOptions<RagnarConfig>  config) : IFileParseFactory
{
    public async Task<IEnumerable<CodeDocument>> ParseAsync(string filePath, CancellationToken ct)
    {
        var ext  = Path.GetExtension(filePath).ToLowerInvariant();
        var parser = parsers.FirstOrDefault(p => Supports(p, ext))
            ?? throw new NotSupportedException($"No parser for extension '{ext}'.");

        return await parser.ParseAsync(filePath, ct);
    }

    private static bool Supports(IFileParser parser, string ext)
        => parser.GetType().Name.Contains(ext.TrimStart('.').ToUpperInvariant(), StringComparison.Ordinal);
}
