```csharp
namespace Ragnar.Stages;

/// <summary>
/// Abstract base for all embedding-pipeline stages, enforcing the common
/// Name / ShouldRun / ExecuteAsync contract and centralising timing + error
/// logging so concrete stages stay focused on domain logic.
/// </summary>
/// <typeparam name="TContext">The shared pipeline context type.</typeparam>
public abstract class BasePipelineStage<TContext>(
    Serilog.ILogger logger,
    IOutputWriter writer) : IPipelineStage<TContext>
{
    public abstract string Name { get; }
    public bool ShouldRun => true;

    protected Serilog.ILogger Logger { get; } = logger;
    protected IOutputWriter Writer { get; } = writer;

    /// <summary>
    /// Template method that wraps <see cref="ExecuteCoreAsync"/> with
    /// stopwatch timing and structured error logging.
    /// </summary>
    public async Task ExecuteAsync(TContext context, CancellationToken cancellationToken)
    {
        var sw = System.Diagnostics.Stopwatch.StartNew();
        try
        {
            await ExecuteCoreAsync(context, cancellationToken).ConfigureAwait(false);
        }
        finally
        {
            Logger.Information("Stage {Name} completed in {Elapsed}", Name, sw.ElapsedTimeString());
        }
    }

    /// <summary>Override with the concrete stage logic.</summary>
    protected abstract Task ExecuteCoreAsync(TContext context, CancellationToken cancellationToken);
}
```

```csharp
namespace Ragnar.Stages;

/// <summary>
/// Pipeline stage that loads questions and runs RAG queries against the vector store.
/// </summary>
/// <param name="ragOrchestrator">Orchestrator for retrieval-augmented generation.</param>
/// <param name="ragnarConfig">Application configuration (collection name, etc.).</param>
/// <param name="writer">Console output writer for status and markup lines.</param>
/// <param name="questionEmbedding">Service for embedding questions and retrieving context.</param>
/// <param name="logger">Serilog logger for per-question diagnostics.</param>
/// <param name="builder">Fluent builder for combining question sources.</param>
/// <param name="tableRenderer">Renderer for tabular question display.</param>
/// <example><![[CDATA[[await stage.ExecuteAsync(context, ct);]]]]></example>
public sealed class QuestionExecutionStage(
    IRagOrchestrator ragOrchestrator,
    IOptions<RagnarConfig> ragnarConfig,
    IOutputWriter writer,
    IQuestionEmbedding questionEmbedding,
    Serilog.ILogger logger,
    IQuestionSourceAggregator builder,
    ITableRenderer tableRenderer) : BasePipelineStage<EmbeddingContext>(logger, writer)
{
    public override string Name => "Ask Questions Stage";

    private readonly string _collectionName = ragnarConfig.Value.ApplicationOptions.VectorStoreName;

    protected override async Task ExecuteCoreAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        var questions = await LoadQuestionsAsync(cancellationToken).ConfigureAwait(false);
        if (questions.Count == 0)
        {
            Logger.Information("No questions to process. Skipping.");
            return;
        }

        RenderQuestionTable(questions);

        var contextText = await questionEmbedding.GetContext(_collectionName, cancellationToken).ConfigureAwait(false);
        var processedCount = 0;
        var total = questions.Count;

        foreach (var question in questions)
        {
            ConsoleSectionWriter.WriteHeader(Writer, question.Text);

            Logger.Information(
                "Processing question [[{Question}]] for user [[{User}]] with {CtxLen} chars of context",
                question.Text, Environment.UserName, contextText.Length.ToString("N0"));

            await ragOrchestrator.ExecuteAsync(question, contextText, cancellationToken).ConfigureAwait(false);

            processedCount++;
            Writer.MarkupLine($"{processedCount} of {total}", Styles.Cyan);
        }
    }

    private async Task<IReadOnlyList<Core.Model.Question>> LoadQuestionsAsync(CancellationToken cancellationToken)
    {
        var sorted = await builder
            .GetCategories()
            .GetFileConfig()
            .GetCsvFilesAsync(Path.Join(AppContext.BaseDirectory, "Plugins"), cancellationToken)
            .WithCategoryFilter()
            .Build()
            .ConfigureAwait(false);

        return sorted;
    }

    /// <summary>Renders a Spectre.Console table of discovered questions via the shared table infrastructure.</summary>
    private void RenderQuestionTable(IReadOnlyList<Core.Model.Question> questions)
    {
        var b = tableRenderer.Table
            .AddColumns("#", "Category", "Text")
            .WithRows(questions.Select((q, i) => new[[]] { (i + 1).ToString(), q.Category.ToString(), q.Text }));

        b.ShowRowSeparators = true;
        b.Expand = true;

        tableRenderer.Render();
    }
}
```

```csharp
namespace Ragnar.Stages;

/// <summary>Parses discovered files into CodeDocument segments in parallel.</summary>
public sealed class ParsingStage(
    IFileParseFactory parseFactory,
    Serilog.ILogger logger,
    IOutputWriter writer) : BasePipelineStage<EmbeddingContext>(logger, writer)
{
    public override string Name => "Parsing files…";

    protected override async Task ExecuteCoreAsync(EmbeddingContext context, CancellationToken cancellationToken)
    {
        var files = context.DiscoveredFiles;
        if (files.Count == 0)
        {
            Logger.Information("No files to parse. Skipping.");
            return;
        }

        var documents = new ConcurrentBag<CodeDocument>();
        var maxParallelism = Math.Min(Environment.ProcessorCount, 8);

        await AnsiConsole.Progress().AutoClear(true).StartAsync(async ctx =>
        {
            var task = ctx.AddTask(Name, maxValue: files.Count);
            await Parallel.ForEachAsync(
                files,
                new ParallelOptions { CancellationToken = cancellationToken, MaxDegreeOfParallelism = maxParallelism },
                async (filePath, token) =>
                {
                    var elements = await parseFactory.ParseAsync(filePath, token).ConfigureAwait(false);
                    foreach (var element in elements)
                        documents.Add(element);
                    task.Increment(1);
                }).ConfigureAwait(false);
        }).ConfigureAwait(false);

        context.Documents = [[.. documents]];
        Logger.Information("Parsed {Count} code document(s).", context.Documents.Count);
    }
}
```

## 5. New: `BatchExecutor.cs`

```csharp
namespace Ragnar.Infrastructure;

/// <summary>
/// Generic helper that splits a collection into fixed-size batches and invokes
/// an async handler per batch, continuing on per-batch failures.
/// Eliminates the repeated "for i = 0; i &lt; count; i += batchSize" boilerplate.
/// </summary>
public static class BatchExecutor
{
    /// <summary>
    /// Processes <paramref name="items"/> in batches of <paramref name="batchSize"/>.
    /// </summary>
    /// <typeparam name="T">Element type.</typeparam>
    /// <param name="items">Source collection.</param>
    /// <param name="batchSize">Maximum items per batch (≤ 0 defaults to 32).</param>
    /// <param name="handler">Async callback receiving each batch and its 0-based start index.</param>
    /// <param name="onBatchFailure">Optional callback for per-batch errors (index + exception).</param>
    /// <param name="cancellationToken">Cancellation token.</param>
    /// <returns>Number of items that were successfully processed.</returns>
    public static async Task<int> ProcessAsync<T>(
        IEnumerable<T> items,
        int batchSize,
        Func<IReadOnlyList<T>, int, CancellationToken, Task> handler,
        Action<int, Exception>? onBatchFailure = null,
        CancellationToken cancellationToken = default)
    {
        var list = items.ToList();
        if (batchSize <= 0) batchSize = 32;

        var succeeded = 0;

        for (var i = 0; i < list.Count; i += batchSize)
        {
            cancellationToken.ThrowIfCancellationRequested();
            var batch = list.Skip(i).Take(batchSize).ToList();

            try
            {
                await handler(batch, i, cancellationToken).ConfigureAwait(false);
                succeeded += batch.Count;
            }
            catch (OperationCanceledException) when (cancellationToken.IsCancellationRequested)
            {
                throw;
            }
            catch (Exception ex)
            {
                onBatchFailure?.Invoke(i, ex);
                // continue with next batch
            }
        }

        return succeeded;
    }
}
```

```csharp
namespace Ragnar.Infrastructure;

/// <summary>Persists embedding vectors to Qdrant in configured batches.</summary>
/// <param name="logger">Serilog logger for fatal upsert-error tracking.</param>
/// <param name="embeddingService">Service that generates float vectors from text.</param>
/// <param name="qdrantClient">Qdrant gRPC client for collection operations.</param>
/// <param name="generatorService">Builder for Qdrant point structures.</param>
/// <param name="config">Application-wide Ragnar configuration wrapper.</param>
public sealed class VectorStoreRepository(
    Serilog.ILogger logger,
    IEmbeddingService embeddingService,
    IQdrantClient qdrantClient,
    IPointStructFactory generatorService,
    IOptions<RagnarConfig> config) : IVectorStoreRepository
{
    private readonly ApplicationOptions _appOptions = config.Value.ApplicationOptions;
    private readonly int _batchSize = config.Value.EmbeddingOptions?.BatchSize ?? 32;

    public async Task<UpdateResult> UpsertBatchAsync(IEnumerable<CodeDocument> codeDocuments, CancellationToken cancellationToken)
    {
        cancellationToken.ThrowIfCancellationRequested();

        var failed = new List<(int Index, Exception Ex)>();

        var succeeded = await BatchExecutor.ProcessAsync(
            codeDocuments,
            _batchSize,
            handler: async (batch, start, ct) =>
            {
                var texts = batch.ConvertAll(d => $"Context: {d.ElementName}\nCode:\n{d.Code}");
                var generated = await embeddingService.GenerateBatchAsync(texts, ct).ConfigureAwait(false);
                var points = batch
                    .Select((d, idx) => generatorService.BuildPointStructs(d.AsPoint(), generated[[idx]].Vector.ToArray(), d))
                    .SelectMany(p => p)
                    .ToList();
                await qdrantClient.UpsertAsync(_appOptions.VectorStoreName, points, cancellationToken: ct).ConfigureAwait(false);
            },
            onBatchFailure: (index, ex) =>
            {
                logger.Error(ex, "Batch starting at {Index} upsert failed.", index);
                failed.Add((index, ex));
            },
            cancellationToken).ConfigureAwait(false);

        return new UpdateResult
        {
            Status = failed.Count == 0 ? UpdateStatus.Completed : UpdateStatus.UnknownUpdateStatus
        };
    }
}
```

```csharp
/// <summary>Generates vector embeddings using the Ollama client factory.</summary>
public class OllamaEmbeddingService(
    Serilog.ILogger logger,
    IOllamaClientFactory clientFactory,
    IOptions<RagnarConfig> config) : IEmbeddingService
{
    private readonly IEmbeddingGenerator<string, Embedding<float>> _generator =
        clientFactory.ResolveClient(OllamaServiceType.Embedding).AsEmbeddingGenerator();

    {
        Guard.Against.NullOrWhiteSpace(input);
        using var cts = CreateLinkedCts(ct);

        return result.Vector;

    public async Task<GeneratedEmbeddings<Embedding<float>>> GenerateBatchAsync(IReadOnlyCollection<string> inputs, CancellationToken ct)

    {
        var cts = CancellationTokenSource.CreateLinkedTokenSource(parent);
        cts.CancelAfter(_timeout);
        return cts;
    }

        }
```
```csharp
    }
```

## 15. Refactored: `PipelineRunner.cs` (leverages base-class timing)

```csharp
namespace Ragnar.Pipeline;

/// <summary>
/// Executes a sequence of pipeline stages in order, with per-stage
/// timing, structured logging, and cancellation propagation.
/// </summary>
public sealed class PipelineRunner(IOutputWriter writer, Serilog.ILogger logger)
{
    private readonly List<IPipelineStage<EmbeddingContext>> _stages = [[]];

    public PipelineRunner AddStage(IPipelineStage<EmbeddingContext> stage)
    {
        _stages.Add(stage);
        return this;
    }

    public async Task<EmbeddingContext> ExecuteAsync(EmbeddingContext context, CancellationToken ct)
    {
        var total = _stages.Count;
        var sw = System.Diagnostics.Stopwatch.StartNew();

        for (var i = 0; i < total; i++)
        {
            var stage = _stages[[i]];

            if (!stage.ShouldRun)
            {
                logger.Information("Skipping stage [[{Index}/{Total}]]: {Name}", i + 1, total, stage.Name);
                continue;
            }

            logger.Information("Starting stage [[{Index}/{Total}]]: {Name}", i + 1, total, stage.Name);

            try
            {
                await stage.ExecuteAsync(context, ct).ConfigureAwait(false);
            }
            catch (OperationCanceledException) when (ct.IsCancellationRequested)
            {
                throw;
            }
            catch (Exception ex)
            {
                logger.Fatal(ex, "Stage [[{Index}]] {Name} failed.", i, stage.Name);
                throw new PipelineStageException(stage.Name, ex);
            }
        }

        logger.Information("Pipeline finished in {Total}", sw.ElapsedTimeString());
        return context;
    }
}
```

## Summary of DRY gains

| Before | After | Files touched |
|--------|-------|---------------|
| 2+ stages each repeat Name/ShouldRun/timing/log boilerplate | `BasePipelineStage<T>` owns the template | `QuestionExecutionStage`, `ParsingStage`, `PipelineRunner` |
| `ShowTable` builds a Spectre table inline | Reuses `DataTableBuilder` + `ITableRenderer` | `QuestionExecutionStage` |
| 4+ test classes hand-build `RagnarConfig` | `TestConfigFactory.Create(...)` | `AskQuestionsStageTests`, `VectorStoreBuilderTests`, `OllamaClientFactoryTests` |
| Builder mock chain duplicated in every test method | `SetupQuestions()` / `SetupContextRetrieval()` helpers | `AskQuestionsStageTests` |
| Batch loop + try/catch repeated in `VectorStoreRepository` | `BatchExecutor.ProcessAsync<T>` | `VectorStoreRepository` |
| Rule + header + rule written ad-hoc per question | `ConsoleSectionWriter.WriteHeader` | `QuestionExecutionStage` |
| Timeout re-read from `IOptions` on every call | Captured once into a field | `OllamaEmbeddingService` |

All public contracts (`IPipelineStage<T>`, `IFileParser`, `IEmbeddingService`, `IVectorStoreRepository`, `ITableRenderer`, etc.) remain unchanged, so no downstream consumer needs modification.
