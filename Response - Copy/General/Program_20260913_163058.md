using Ragnar.Core;

using Ragnar.Infrastructure;

// ═══════════════════════════════════════════════════════════
// Top-level helpers

static void ValidateConfiguration(IServiceProvider services)
{
    var config = services.GetRequiredService<IOptions<RagnarConfig>>().Value;
    var validator = new OllamaOptionsValidator();

    var result = validator.TestValidate(config.OllamaOptions);
    if (result.HasValidationErrors)
    {
        throw new InvalidOperationException(
            "Configuration validation failed:\n" +
            string.Join("\n", result.Errors.Select(e => $"  • {e.PropertyName}: {e.ErrorMessage}")));
    }
}

static async Task RunPipelineAsync(IHost host)
{
    using var scope = host.Services.CreateScope();
    var sp = scope.ServiceProvider;

    var header = sp.GetRequiredService<IApplicationHeader>();
    var writer = sp.GetRequiredService<IOutputWriter>();
    var runner = sp.GetRequiredService<PipelineRunner>();
    var context = new EmbeddingContext();
    var cts = new CancellationTokenSource();

    // Graceful shutdown on Ctrl+C
    Console.CancelKeyPress += (_, e) =>
    {
        e.Cancel = true;
        cts.Cancel();
    };

    try
    {
        header.RenderBranding();
        await runner.ExecuteAsync(context, cts.Token).ConfigureAwait(false);
    }
    catch (OperationCanceledException) when (cts.Token.IsCancellationRequested)
    {
        writer.MarkupLine("[[yellow]]Pipeline cancelled by user.[[/]]");
    }
    catch (PipelineStageException ex)
    {
        writer.MarkupLine($"[[red]]Pipeline failed at stage: {ex.StageName}[[/]]");
        Serilog.Log.Fatal(ex, "Pipeline aborted.");
    }
}

// ═══════════════════════════════════════════════════════════
// DI Extension Methods (one per concern)
// ═══════════════════════════════════════════════════════════

file static class DependencyInjectionExtensions
{
    // ── Logging ─────────────────────────────────────────────
    public static IServiceCollection AddRagnarLogging(this IServiceCollection services)
    {
        services.AddLogging(logging =>
        {
            logging.AddSerilog(dispose: true);
        });
        return services;
    }

    // ── Configuration ───────────────────────────────────────
    public static IServiceCollection AddRagnarConfiguration(this IServiceCollection services)
    {
        services.AddOptions<RagnarConfig>()
            .Bind(services.GetRequiredService<IConfiguration>().GetSection("RagnarConfig"))
            .ValidateDataAnnotations()
            .ValidateOnStart();

        return services;
    }

    // ── Infrastructure (HTTP, Qdrant, Ollama) ──────────────
    public static IServiceCollection AddRagnarInfrastructure(this IServiceCollection services, IConfiguration config)
    {
        var ollamaOpts = config.GetSection("RagnarConfig:OllamaOptions").Get<OllamaOptions>() ?? new OllamaOptions();
        var embedOpts = config.GetSection("RagnarConfig:EmbeddingOptions").Get<EmbeddingOptions>() ?? new EmbeddingOptions();

        services.AddHttpClient("ollama", client =>
        {
            client.BaseAddress = new Uri($"http://{ollamaOpts.Host}:{ollamaOpts.Port}");
            client.Timeout = ollamaOpts.Timeout;
        });

        services.AddHttpClient("embedding", client =>
        {
            client.BaseAddress = new Uri($"http://{embedOpts.Host}:{embedOpts.Port}");
            client.Timeout = embedOpts.Timeout;
        });

        services.AddSingleton<IQdrantClient>(sp =>
        {
            var storeName = sp.GetRequiredService<IOptions<RagnarConfig>>().Value.ApplicationOptions.VectorStoreName;
            return new QdrantClient(storeName);
        });

        return services;
    }

    // ── Core Services ───────────────────────────────────────
    public static IServiceCollection AddRagnarServices(this IServiceCollection services)
    {
        // Ollama client factory (cached per service type)
        services.AddSingleton<IOllamaClientFactory, OllamaClientFactory>();

        // Embedding
        services.AddScoped<IEmbeddingService, OllamaEmbeddingService>();
        services.AddScoped<IQuestionEmbedding, QuestionEmbeddingService>();
        services.AddScoped<IVectorSearchService, VectorSearchService>();

        // RAG orchestration
        services.AddScoped<IRagOrchestrator, RagOrchestrator>();

        // Vector store persistence
        services.AddScoped<IPointStructFactory, PointStructFactory>();
        services.AddScoped<IVectorStoreRepository, VectorStoreRepository>();
        services.AddSingleton<VectorStoreBuilder>();

        // File parsing
        services.AddScoped<IFileParser, CSharpFileParser>();
        services.AddScoped<IFileParseFactory, FileParseFactory>();

        // Question loading
        services.AddSingleton<IRecordParser<QuestionRecord>, CsvRecordParser>();
        services.AddSingleton<IQuestionProvider, CsvFileQuestionProvider>();
        services.AddSingleton<IQuestionSourceAggregator, QuestionSourceAggregator>();

        // Output & formatting
        services.AddSingleton<IOutputWriter, ConsoleOutputWriter>();
        services.AddSingleton<IWriter, FileWriter>();
        services.AddSingleton<IPathResolver, PathResolver>();
        services.AddSingleton<IOutputFormatter, MarkdownFormatter>();
        services.AddScoped<IResponseWriter, ResponseWriter>();
        services.AddScoped<ISummaryService, SummaryService>();
        services.AddSingleton<IChatPromptProvider, SummarizePromptProvider>();

        // UI
        services.AddSingleton<IApplicationHeader, ApplicationHeader>();

        return services;
    }

    // ── Pipeline Stages ─────────────────────────────────────
    public static IServiceCollection AddRagnarPipeline(this IServiceCollection services)
    {
        services.AddSingleton<PipelineRunner>();

        services.AddSingleton<IPipelineStage<EmbeddingContext>>(sp =>
            new ParsingStage(
                sp.GetRequiredService<IFileParseFactory>(),
                sp.GetRequiredService<ILogger>(),
                sp.GetRequiredService<IOutputWriter>()));

        services.AddSingleton<IPipelineStage<EmbeddingContext>>(sp =>
            new QuestionExecutionStage(
                sp.GetRequiredService<IRagOrchestrator>(),
                sp.GetRequiredService<IOptions<RagnarConfig>>(),
                sp.GetRequiredService<IOutputWriter>(),
                sp.GetRequiredService<IQuestionEmbedding>(),
                sp.GetRequiredService<Serilog.ILogger>(),
                sp.GetRequiredService<IQuestionSourceAggregator>()));

        return services;
    }
}

## Key Improvements

| Area | Before (typical) | After |
|------|-----------------|-------|
| **Structure** | Monolithic 200+ line `Main` with mixed concerns | Top-level statements + 5 focused extension methods |
| **DI Registration** | Inline `services.AddSingleton<…>()` scattered in `Main` | Grouped by concern: `AddRagnarInfrastructure`, `AddRagnarServices`, `AddRagnarPipeline` |
| **Config Validation** | Deferred / missing | Fail-fast `ValidateConfiguration` before pipeline starts |
| **Error Handling** | Bare `try/catch` swallowing context | Typed catches (`PipelineStageException`, `OperationCanceledException`) with user-facing messages |
| **Graceful Shutdown** | `Console.ReadKey()` or no handler | `CancellationTokenSource` wired to `Ctrl+C` |
| **Readability** | Nested lambdas, long DI chains | Each registration ≤ 5 lines; helper methods for HTTP clients |
| **Maintainability** | Adding a service means editing a 100-line block | Add one line inside the relevant `Add*` extension |

## How to Extend

- **New pipeline stage** → add one `services.AddSingleton<IPipelineStage<EmbeddingContext>>(…)` line inside `AddRagnarPipeline`.
- **New question source** → add one `services.AddSingleton<IQuestionProvider, …>()` inside `AddRagnarServices`.
- **New configuration section** → add a `ValidateDataAnnotations` rule in `AddRagnarConfiguration`; the fail-fast check picks it up automatically.
