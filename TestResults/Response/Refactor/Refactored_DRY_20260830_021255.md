public static class PathUtilities
{
    public static string ExpandAndValidateDirectory(string path)
    {
        Guard.Against.NullOrWhiteSpace(path);
        var expanded = Environment.ExpandEnvironmentVariables(path);
        var fullPath = Path.GetFullPath(expanded);
        
        if (!Directory.Exists(fullPath))
            throw new DirectoryNotFoundException($"Resolved directory does not exist: '{fullPath}'. Original config value: '{path}'");
            
        return fullPath;
    }

    public static string ResolveResponseDirectory(string sourceDir, QuestionCategory? category)
    {
        var baseDir = string.IsNullOrWhiteSpace(category?.ToString()) 
            ? nameof(QuestionCategory.Uncategorized) 
            : category.ToString();
            
        return Path.Join(sourceDir, "Response", baseDir);
    }
}

public static class AssemblyExtensions
{
    public static string GetInformationalVersion(this Assembly asm) =>
        asm.GetCustomAttribute<AssemblyInformationalVersionAttribute>()?.InformationalVersion ?? "1.0.0";
}

{
    public static Style EnsureValidStyle(this Style? style) => style ?? Spectre.Console.Style.Plain;

public sealed class ConsoleStreamWriter : IConsoleStreamWriter
{
    private readonly OllamaApiClient _client;
    
    public ConsoleStreamWriter(IOllamaClientFactory factory) =>
        _client = factory.FindClient(OllamaServiceType.Ollama);

    public async Task<string> StreamResponseAsync(ChatRequest request, CancellationToken ct = default)
    {
        var options = new RequestOptions
        {
            NumPredict = 8192,
            NumCtx = 16384,
            Temperature = 0.6f
        };

        var responseMarkup = new Markup("", StylesExtensions.EnsureValidStyle(Styles.Yellow));
        var spinnerRow = new Columns(new SpinnerWidget(Spinner.Known.Dots), new Text(" Thinking..."));
        var liveContainer = new Rows(responseMarkup, spinnerRow);
        var table = new Table { AddColumn("..."), AddRow(responseMarkup), AddRow(spinnerRow) };

        string completeText = "";
        
        await AnsiConsole.Live(liveContainer).StartAsync(async ctx =>
        {
            await foreach (var token in _client.ChatAsync(request, ct))
            {
                var msg = token.Message.Content;
                if (string.IsNullOrEmpty(msg)) continue;

                completeText += Markup.Escape(msg);
                responseMarkup.Update(completeText, StylesExtensions.EnsureValidStyle(Styles.Yellow));
                table.UpdateCell(0, 0, responseMarkup);
                ctx.Refresh();
            }
            
            table.RemoveRow(1);
            ctx.Refresh();
        });

        AnsiConsole.WriteLine();
        AnsiConsole.Write(new Rule().RuleStyle("grey"));
        AnsiConsole.WriteLine();
        return completeText;
    }
}

public interface IDirectoryContentReader
{
    Task<string> ReadAllAsync(string folder, CancellationToken ct = default);
}

public sealed class DirectoryContentReader : IDirectoryContentReader
{
    public async Task<string> ReadAllAsync(string folder, CancellationToken ct = default)
    {
        var files = Directory.EnumerateFiles(folder).ToList();
        if (files.Count == 0) return "No items to summarize. Please ignore.";

        var builder = new StringBuilder();
        foreach (var file in files)
        {
            await using var reader = File.OpenText(file);
            var text = await reader.ReadToEndAsync(ct);
            builder.AppendLine($"---\n{AppDefaults.FILE_MARKER_START}{Path.GetFileName(file)}{AppDefaults.FILEMARKEREND}\n{text}\n");
        }
        return $"{AppDefaults.CODE_BLOCK_START} {builder} {AppDefaults.CODE_BLOCK_END}";
    }
}

{
    private readonly IQdrantClient _qdrantClient;
    private readonly Serilog.ILogger _logger;

    }

    public string VectorStoreName { get => _vectorStoreName; set => _vectorStoreName = value; }

    public async Task<bool> BuildAsync(CancellationToken ct = default)

        return IsExisting;
    }

    public async ValueTask<IVectorStoreBuilder> ExistsAsync(CancellationToken ct = default)
    {
        IsExisting = await _qdrantClient.CollectionExistsAsync(VectorStoreName, ct);
        return this;
    }

    public async ValueTask<IVectorStoreBuilder> CreateAsync(CancellationToken ct = default)
    {
        await _qdrantClient.CreateCollectionAsync(VectorStoreName, new VectorParams { Size = Dimension, Distance = Distance }, cancellationToken: ct);
        _logger.Information("New collection {Name} created.", VectorStoreName);
        IsExisting = true;
        return this;
    }

    public async ValueTask<IVectorStoreBuilder> MakeIndexAsync(string indexName, PayloadSchemaType schemaType, CancellationToken ct = default)
    {
        await _qdrantClient.CreatePayloadIndexAsync(VectorStoreName, fieldName: indexName, schemaType: schemaType, cancellationToken: ct);
        return this;
    }
}

public sealed class VectorStoreRepository : IVectorStoreRepository
{
    private readonly IQdrantClient _qdrantClient;
    private readonly Serilog.ILogger _logger;
    private readonly string _vectorStoreName;
    private readonly IEmbeddingGenerator<string, Embedding<float>> _generator;

    public VectorStoreRepository(
        Serilog.ILogger logger, 
        IOllamaClientFactory clientFactory, 
        IQdrantClient qdrantClient, 
        IOptions<RagnarConfig> ragnarOptions)
    {
        _qdrantClient = qdrantClient;
        _logger = logger;
        _vectorStoreName = ragnarOptions.Value.ApplicationOptions.VectorStoreName;
        var embeddingClient = clientFactory.FindClient(OllamaServiceType.Embedding);
        _generator = embeddingClient.AsEmbeddingGenerator();
    }

    public async Task<UpdateResult> UpsertBatchAsync(CodeDocument[[]] codeDocuments, CancellationToken ct = default)
    {
        ct.ThrowIfCancellationRequested();
        
        var embeddingGroup = new List<PointStruct>();
        var tasks = codeDocuments.Select(async doc =>
        {
            var textToEmbed = $"Context: {doc.ElementName}\nCode:\n{doc.Code}";
            return (doc, vector: await GenerateEmbeddingAsync(textToEmbed, ct));
        }).ToList();

        await Parallel.ForEachAsync(tasks, new ParallelOptions { MaxDegreeOfParallelism = 4 }, async (task, cts) =>
        {
            var result = await task;
            embeddingGroup.Add(new PointStruct
            {
                Id = result.doc.AsPoint(),
                Vectors = result.vector.ToArray(),
                Payload = { result.doc.Dictionary }
            });
        });

        if (embeddingGroup.Count == 0) return new UpdateResult();

        try
        {
            return await _qdrantClient.UpsertAsync(_vectorStoreName, embeddingGroup, cancellationToken: ct);
        }
        catch (Exception ex) when (ex is not OperationCanceledException)
        {
            _logger.Fatal(ex, "Failed to upsert embeddings batch to Qdrant.");
            return new UpdateResult { Status = UpdateStatus.UnknownUpdateStatus };
        }
    }

    private async Task<float[[]]> GenerateEmbeddingAsync(string chunk, CancellationToken ct)
    {
        var embedding = await _generator.GenerateAsync(chunk, cancellationToken: ct).ConfigureAwait(false);
        return embedding.Vector.ToArray();
    }
}
