public async Task<string> GenerateResponse(GenerateRequest Request, CancellationToken Ct)
{
    RequestOptions options = new()
    {
        NumPredict = 8192,
        NumCtx = 16384,
        Temperature = 0.6f,
    };

    var chatRequest = new ChatRequest
    {
        Messages = [[new Message(ChatRole.System, Request.System), new Message(ChatRole.User, Request.Prompt)]],
        Options = options
    };

    // 1. Check capability before streaming
    bool supportsThinking = ModelSupportsThinking(Request.LlmModel ?? "unknown");
    
    var responseMarkup = new Markup("");
    var spinnerRow = new Columns(new SpinnerWidget(Spinner.Known.Dots), new Text(" Thinking..."));
    var liveContainer = new Rows(responseMarkup, spinnerRow);
    string completeText = "";

    // 2. Safely subscribe to OnThink if supported
    IDisposable? thinkSubscription = null;
    if (supportsThinking)

    }

    try
    {
        await AnsiConsole.Live(liveContainer).StartAsync(async ctx =>
        {
            var table = new Table();
            table.AddColumn("...");
            table.AddRow(responseMarkup);
            table.AddRow(spinnerRow);

            await foreach (var token in _ollamaClient.ChatAsync(chatRequest, Ct))
            {
                var msg = token.Message.Content;
                if (string.IsNullOrEmpty(msg)) continue;

                completeText += Markup.Escape(msg);
                responseMarkup = new Markup(completeText, Styles.Yellow);
                table.UpdateCell(0, 0, responseMarkup);
                ctx.Refresh();
            }

            table.RemoveRow(1);
            ctx.Refresh();
        });
    }
    finally
    {
        // 3. Always unsubscribe to prevent memory leaks
        thinkSubscription?.Dispose();
    }

    AnsiConsole.WriteLine();
    AnsiConsole.Write(new Rule().RuleStyle("grey"));
    AnsiConsole.WriteLine();
    
    return completeText;
}
