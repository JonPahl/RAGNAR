        {

        };
        var expectedVector = new float[[]] { 0.1f, 0.2f, 0.3f };
        _mockEmbeddingGen.Setup(x => x.GenerateAsync(It.IsAny<string>(), It.IsAny<CancellationToken>()))
            .ReturnsAsync(new Embedding<float>(expectedVector));
        _mockQdrantClient.Setup(x => x.UpsertAsync("test_store", It.IsAny<IEnumerable<PointStruct>>(), It.IsAny<CancellationToken>()))
            .ReturnsAsync(new UpdateResult { Status = UpdateStatus.Success });

        // Assert
        Assert.Equal(UpdateStatus.Success, result.Status);

        // Assert
