using Moq;
using Xunit;
using System.Threading.Tasks;
using System.Threading;

public class CsvFileQuestionProviderTests
{
    [[Fact]]
    public async Task LoadQuestionsAsync_ShouldReturnMappedQuestions()
    {
        var mockParser = new Mock<IRecordParser<QuestionRecord>>();
        var mockRecords = new List<QuestionRecord>
        {
            new QuestionRecord { IsEnabled = true, Text = "Q1", FileName = "a.cs", Category = "Cat1" },
            new QuestionRecord { IsEnabled = false, Text = "Q2", FileName = "b.cs", Category = "Cat2" }
        };

        mockParser.Setup(p => p.ParseAsync(It.IsAny<string>(), It.IsAny<CancellationToken>()))
                  .ReturnsAsync(mockRecords);

        var provider = new CsvFileQuestionProvider(mockParser.Object);

        var result = await provider.LoadQuestionsAsync("test.csv", CancellationToken.None);
        var questions = result.ToList();

        Assert.Equal(2, questions.Count);
        Assert.True(questions[[0]].IsActive);
        Assert.Equal("Q1", questions[[0]].Text);
        Assert.False(questions[[1]].IsActive);
    }

    [[Fact]]
    public async Task LoadQuestionsAsync_ShouldPropagateCancellation()
    {
        var mockParser = new Mock<IRecordParser<QuestionRecord>>();
        mockParser.Setup(p => p.ParseAsync(It.IsAny<string>(), It.IsAny<CancellationToken>()))
                  .ThrowsAsync(new OperationCanceledException());

        var provider = new CsvFileQuestionProvider(mockParser.Object);

        await Assert.ThrowsAsync<OperationCanceledException>(
            () => provider.LoadQuestionsAsync("test.csv", CancellationToken.None));
    }
}
