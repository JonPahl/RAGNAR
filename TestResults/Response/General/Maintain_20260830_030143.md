#### 1. Fix Thread Safety & Async Blocking in `VectorStoreRepository`
**Problem:** `List<T>.Add()` is not thread-safe. `.Result` blocks threads and can cause deadlocks.
**Fix:** Use a thread-safe collection, properly await tasks, and separate embedding generation from point construction.

```csharp
public async Task<UpdateResult> UpsertBatchAsync(CodeDocument[[]] codeDocuments, CancellationToken ct)
{
    ct.ThrowIfCancellationRequested();

    // 1. Generate embeddings asynchronously (safe parallelism)
    var embeddingTasks = codeDocuments.Select(async doc =>
    {
        var textToEmbed = $"Context: {doc.ElementName}\nCode:\n{doc.Code}";
        var vector = await _generator.GenerateAsync(textToEmbed, ct).ConfigureAwait(false);
        return (doc, vector: vector.Vector.ToArray());
    });

    // 2. Await all embeddings concurrently
    var embeddingResults = await Task.WhenAll(embeddingTasks).ConfigureAwait(false);

    // 3. Build points safely (single-threaded now)
    var points = new List<PointStruct>(codeDocuments.Length);
    foreach (var result in embeddingResults)
    {
        points.Add(new PointStruct
        {
            Id = result.doc.AsPoint(),
            Vectors = result.vector,
            Payload = { result.doc.Dictionary }
        });
    }

    if (points.Count == 0) return new UpdateResult();

    try
    {
        return await _qdrantClient.UpsertAsync(_applicationOption.VectorStoreName, points, ct).ConfigureAwait(false);
    }
    catch (OperationCanceledException) when (ct.IsCancellationRequested)
    {
        throw; // Re-throw cancellation explicitly
    }
    catch (Exception ex)
    {
        _logger.Error(ex, "Failed to upsert embeddings batch to Qdrant.");
        return new UpdateResult { Status = UpdateStatus.UnknownUpdateStatus };
    }
}
```

    public async Task<string> GenerateResponse(GenerateRequest request, CancellationToken ct)

            NumCtx = 16384,
            Temperature = 0.6f,
        };

        var chatRequest = new ChatRequest
        {
            Messages = [[new Message(ChatRole.System, request.System), new Message(ChatRole.User, request.Prompt)]],
            Options = options
        };

        var sb = new StringBuilder(); // ✅ O(1) appends
        var liveContainer = new Rows(new Markup(""), new Columns(new SpinnerWidget(Spinner.Known.Dots), new Text(" Thinking...")));
        
        await AnsiConsole.Live(liveContainer).StartAsync(async ctx =>
        {
            await foreach (var token in _ollamaClient.ChatAsync(chatRequest, ct).ConfigureAwait(false))

                var markup = new Markup(sb.ToString(), Styles.Yellow);

                ctx.Refresh();
