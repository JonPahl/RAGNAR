### 🔹 1. Split Service Registration into Feature-Specific Extensions
Instead of one giant `ConfigureServices` block, group registrations by domain. This makes it obvious what each part of the system does.

```csharp
// Infrastructure/Extensions/OllamaExtensions.cs
public static class OllamaExtensions
{
    public static IServiceCollection AddOllamaIntegration(this IServiceCollection services, IConfiguration config)
    {
        services.Configure<OllamaOptions>(config.GetSection("OllamaOptions"));
        services.AddSingleton<IOllamaClientProvider, OllamaClientProvider>();
        return services;
    }

// Infrastructure/Extensions/VectorStoreExtensions.cs
public static class VectorStoreExtensions
{
    public static IServiceCollection AddVectorStore(this IServiceCollection services, IConfiguration config)
    {
        services.Configure<EmbeddingOptions>(config.GetSection("EmbeddingOptions"));
        services.AddSingleton<IQdrantClient>(sp =>
        {
            var opts = sp.GetRequiredService<IOptions<EmbeddingOptions>>().Value;
            return new QdrantClient(opts.Host, opts.Port, https: false);
        });
        return services;
    }
}
```

### 🔹 3. Centralize & Validate Configuration Early
Avoid manual `GetSection().Get<T>()` inside DI factories. Bind configuration once at startup and let DI resolve it via `IOptions<T>`.

```csharp
// Instead of this in DI:
var ollamaOption = context.Configuration.GetSection("OllamaOption").Get<OllamaOptions>();

// Do this:
services.AddOptions<OllamaOptions>()
    .Bind(builder.Configuration.GetSection("OllamaOptions"))
    .ValidateDataAnnotations() // or FluentValidation
    .ValidateOnStart();
```
✅ Benefits: Fail-fast on misconfiguration, strongly-typed access everywhere, easier to mock in tests.

### 🔹 4. Extract Plugin Loading & Resilience Setup
Your `RegisterQuestionPlugins()` and `AddHttpClients()` methods are doing heavy lifting. Move them to dedicated extension classes or use a plugin manifest approach if the directory structure grows.

```csharp
// Infrastructure/Extensions/PluginLoader.cs
public static class PluginLoaderExtensions
{
    public static IServiceCollection LoadQuestionPlugins(this IServiceCollection services)
    {
        var pluginDir = Path.Combine(AppContext.BaseDirectory, "Questions", "Plugins");
        if (!Directory.Exists(pluginDir)) return services;

        foreach (var dll in Directory.EnumerateFiles(pluginDir, "*.dll"))
        {
            try
            {
                var assembly = AssemblyLoadContext.Default.LoadFromAssemblyPath(dll);
                var types = assembly.GetTypes()
                    .Where(t => typeof(IQuestionProvider).IsAssignableFrom(t) 
                             && t.IsClass && !t.IsAbstract 
                             && t.GetConstructor(Type.EmptyTypes) != null);
                
                foreach (var type in types)
                    services.AddTransient(typeof(IQuestionProvider), type);
            }
            catch (Exception ex) when (ex is BadImageFormatException or ReflectionTypeLoadException)
            {
                // Log gracefully, don't crash the host
                Console.WriteLine($"⚠️ Skipped plugin {dll}: {ex.Message}");
            }
        }
        return services;
    }
}
```
